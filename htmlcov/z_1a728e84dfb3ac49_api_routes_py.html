16"""Define the JSON API router for the Dipper plugin. 

17 

18Mounted at ``/api/apps/dipper/`` via ``apps_router`` in 

19``app/extensions/api/router.py``. Authentication is enforced at the ``api_router`` 

20mount level (``IsApiAuthenticated``) and by ``RequireBearerForUnsafeMethods`` 

21on the ``apps_router``. Route layout: 

22 

23* ``GET /schema`` — static plugin schema 

24* ``GET /`` — Dipper execution history 

25* ``GET /form-schema`` — context-specific execute form schema 

26* ``GET /script-preview`` — preview the collector script for a service 

27* ``POST /`` — execute a collector script against a service 

28"""

87@router.get("/") 

88async def dipper_api_list( 

89 tasks_api: TaskAPI, pagination: PaginationDep 

90) -> PaginatedResponse[ArbitraryMapping]: 

91 """Return Dipper execution history rows. 

92 

93 :param tasks_api: Async client for the tasks sub-app. 

94 :param pagination: Validated offset/limit forwarded to the upstream history call. 

95 :return: A paginated envelope of task-history rows filtered to Dipper runs. Because 

96 the ``dipper/`` filter runs client-side, ``total`` is the filtered count of the 

97 current page, not a global count across all pages. 

98 """ 

99 response = await tasks_api.get( 

100 "/history/", params={"offset": pagination.offset, "limit": pagination.limit} 

101 ) 

102 items = [ 

103 item 

104 for item in response.get("items", []) 

105 if (_snippet_filename_for(item) or "").startswith("dipper/") 

106 ] 

107 return build_proxied_page(items, response, pagination, client_side_filtered=True)

110@router.get( 

111 "/form-schema", 

112 response_model_by_alias=True, 

113) 

114async def dipper_api_form_schema( 

115 service_id: int, 

116 collector_type: CollectorTypeEnum, 

117 inventory_api: InventoryAPI, 

118 executor_hosts: ExecutorHosts, 

119 pmm_api: PMMAPIDep, 

120) -> AppSchema: 

121 """Return the selected Dipper payload's dynamic execution schema. 

122 

123 :param service_id: Inventory ID of the database service. 

124 :type service_id: int 

125 :param collector_type: Which collector variant to render. 

126 :type collector_type: CollectorTypeEnum 

127 :param inventory_api: Async client for the inventory sub-app. 

128 :type inventory_api: RemoteAPI 

129 :param executor_hosts: Available executor hosts keyed by hostname. 

130 :type executor_hosts: dict 

131 :param pmm_api: The configured PMM API client, or ``None`` when PMM is not 

132 configured (injected via ``PMMAPIDep``). 

133 :type pmm_api: PMMRemoteAPI | None 

134 :return: Context-specific schema including payload parameters. 

135 """ 

136 service_data = await inventory_api.get(f"/services/{service_id}") 

137 service = CreatedService.model_validate(service_data) 

138 try: 

139 script = await load_dipper_script(service, collector_type, update_meta=True) 

140 except HTTPNotFoundException as exc: 

141 raise HTTPUnprocessableEntityException( 

142 detail=( 

143 f"No {collector_type.value!r} collector script is available" 

144 f" for {service.type.value!r} services." 

145 ) 

146 ) from exc 

147 defaults = None 

148 node_options: list[str] = [] 

149 service_options: list[str] = [] 

150 if collector_type == CollectorTypeEnum.PMM: 

151 defaults = get_pmm_form_defaults( 

152 resolve_pmm_executor_host(executor_hosts), 

153 service.name, 

154 service.node.name if service.node else "", 

155 ) 

156 node_options, service_options = await fetch_pmm_node_service_names(pmm_api) 

157 return build_dipper_form_schema( 

158 script, 

159 service.id, 

160 collector_type.value, 

161 defaults=defaults, 

162 node_options=node_options, 

163 service_options=service_options, 

164 )

201@router.post( 

202 "/", 

203 status_code=http_status.HTTP_201_CREATED, 

204) 

205async def dipper_api_execute( 

206 request: Request, 

207 body: DipperExecuteWrite, 

208 inventory_api: InventoryAPI, 

209 tasks_api: TaskAPI, 

210) -> DipperExecutionResponse: 

211 """Execute a Dipper collector script against a database service. 

212 

213 Delegates all script resolution, PMM defaults merging, argument 

214 validation, and metadata assembly to ``build_dipper_execution_meta``. 

215 

216 :param request: The HTTP request (used to derive the artifact download URL). 

217 :type request: Request 

218 :param body: Validated JSON request body. 

219 :type body: DipperExecuteWrite 

220 :param inventory_api: Async client for the inventory sub-app. 

221 :type inventory_api: RemoteAPI 

222 :param tasks_api: Async client for the tasks sub-app. 

223 :type tasks_api: RemoteAPI 

224 :return: Execution response with the created task identifier. 

225 :rtype: DipperExecutionResponse 

226 :raises HTTPNotFoundException: When the service does not exist. 

227 :raises HTTPUnprocessableEntityException: When the script/args are invalid. 

228 """ 

229 service_data = await inventory_api.get(f"/services/{body.service_id}") 

230 service = CreatedService.model_validate(service_data) 

231 execution_meta, execution_task_name = await build_dipper_execution_meta( 

232 service, body, request 

233 ) 

234 logger.info( 

235 "Executing [%s] dipper script %r for service %d", 

236 execution_meta.interpreter, 

237 execution_meta.snippet_filename, 

238 service.id, 

239 ) 

240 task_id = await post_task_execution(tasks_api, execution_task_name, execution_meta) 

241 return DipperExecutionResponse( 

242 task_id=task_id, 

243 task_name=execution_task_name, 

244 snippet_filename=execution_meta.snippet_filename, 

245 service_id=service.id, 

246 collector_type=body.collector_type, 

247 )
