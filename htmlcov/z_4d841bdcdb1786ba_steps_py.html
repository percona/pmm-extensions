16"""Define the canonical Nomad task/step names used inside Nomad job specs. 

17 

18This module is a pure leaf: it imports nothing beyond ``enum`` so callers can 

19name a step without constructing the Nomad executor. Wire values are spelled 

20literally (no ``auto()``) so the Nomad task ``Name`` is never implicit. 

21""" 

22 

23from enum import StrEnum 

24 

25 

26class NomadStep(StrEnum): 

27 """Represent Nomad task ``Name`` values used by PMM Extensions job-spec templates. 

28 

29 ``STEP1`` is legacy — only historical task logs / the frozen Alembic 

30 migration still emit it. Do not wire new job specs to ``step1``. 

31 """ 

32 

33 RUN_SCRIPT = "run-script" 

34 PREPARE_ENV = "prepare-env" 

35 CLEAN_UP = "clean-up" 

36 CHECK_STALENESS = "check-staleness" 

37 CHECK_LAUNCHABLE = "check-launchable" 

38 LOG_CAPTURE_HOLD = "log-capture-hold" 

39 STEP1 = "step1" 

40 

41 @classmethod 

42 def anonymized(cls) -> frozenset["NomadStep"]: 

43 """Return the steps whose log output is PII-anonymized. 

44 

45 :return: The frozen set of steps classified as anonymized. 

46 """ 

47 return frozenset( 

48 step for step, anonymize in NOMAD_STEP_ANONYMIZE.items() if anonymize 

49 ) 

50 

51 @classmethod 

52 def is_persistable(cls, step: str) -> bool: 

53 """Return whether PMM Extensions persists ``step``'s streams and gates release on it. 

54 

55 Tests against :data:`NON_PERSISTABLE_STEPS`, which is also what the 

56 capture-status aggregate filters on in SQL. Sharing one definition is 

57 what keeps a second non-producing step from being excluded here and 

58 silently counted there. 

59 

60 Takes a raw name rather than a member, and admits every name outside 

61 that set: a Nomad allocation's task states are keyed by whatever the job 

62 spec declared, so a step this enum does not know is an unrecognized 

63 *producer* — excluding it would silently drop its output from the live 

64 viewer and leave its capture unclassified. 

65 

66 :param step: The Nomad task (step) name to classify. 

67 :return: ``True`` for every step PMM Extensions treats as a log producer. 

68 """ 

69 return step not in NON_PERSISTABLE_STEPS 

70 

71 

72#: Allocation-relative output-files directory of every job spec that pins its 

73#: :attr:`NomadStep.RUN_SCRIPT` task's ``work_dir`` to 

74#: ``${NOMAD_TASK_DIR}/output_files`` (``run-python``, ``exec-artifact``, 

75#: ``exec-python-artifact``). It is the 

76#: :attr:`~app.tasks.models.TaskBase.output_files_path` those specs run under, 

77#: so a payload's working directory and the path PMM Extensions reads its files back from 

78#: are the same place. ``run-command`` pins no ``work_dir`` and so has no 

79#: output-files path. Derived from the step name rather than spelled out, so 

80#: renaming the step cannot leave the path behind; ``local`` is 

81#: ``${NOMAD_TASK_DIR}``. 

82RUN_SCRIPT_OUTPUT_FILES_PATH = f"{NomadStep.RUN_SCRIPT}/local/output_files" 

83 

84#: Steps that emit no task output of their own. ``LOG_CAPTURE_HOLD`` exists only 

85#: to keep the allocation readable, so draining it would gate release on a stream 

86#: that only ends when the hold itself does. Read by both 

87#: :meth:`NomadStep.is_persistable` and the capture-status aggregate's SQL 

88#: filter, so the two cannot diverge. 

89NON_PERSISTABLE_STEPS: frozenset[str] = frozenset({NomadStep.LOG_CAPTURE_HOLD}) 

90 

91NOMAD_STEP_ANONYMIZE: dict[NomadStep, bool] = { 

92 NomadStep.RUN_SCRIPT: True, 

93 NomadStep.PREPARE_ENV: False, 

94 NomadStep.CLEAN_UP: False, 

95 NomadStep.CHECK_STALENESS: False, 

96 NomadStep.CHECK_LAUNCHABLE: False, 

97 NomadStep.LOG_CAPTURE_HOLD: False, 

98 NomadStep.STEP1: True, 

99} 

100 

101#: Exit code :attr:`NomadStep.CHECK_LAUNCHABLE` aborts with when a command in 

102#: the launch chain does not resolve on the executor node, mapped to 

103#: ``UNLAUNCHABLE`` by the Nomad executor the way 75 is mapped to ``STALE``. 

104#: ``EX_CONFIG`` from ``sysexits.h``: the allocation was misconfigured for the 

105#: node it landed on. Shared by the seeded job spec that emits it and the 

106#: executor that reads it back, so the two cannot drift apart. 

107LAUNCH_CHECK_EXIT_CODE = 78 

108 

109#: Hold duration a job dispatched without the meta key falls back to, in 

110#: seconds. Spans three 30 s sync cadences so a beat delayed by worker backlog 

111#: still lands inside the hold. 

112LOG_CAPTURE_HOLD_DEFAULT_SECONDS = 90
