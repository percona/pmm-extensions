258 def _split_lines_from_buffer( 

259 buffer: bytearray, data: bytes, encoding: str = "utf-8" 

260 ) -> Generator[str]: 

261 """Append data to buffer and yield complete lines. 

262 

263 Each call consumes the newlines ``data`` introduced and drops what 

264 precedes them, so ``buffer`` holds no newline when the next call 

265 arrives. Searching only the arriving bytes therefore finds exactly what 

266 a search from the front finds, at a cost proportional to ``data`` rather 

267 than to the remainder behind it. 

268 

269 The narrowing holds only while ``buffer`` is newline-free on entry, so 

270 this call empties it of terminators before it yields anything: a 

271 consumer that abandons the generator part-way cannot leave one behind 

272 for the next call to search past. An abandoned generator therefore drops 

273 the lines it had not yet yielded instead of corrupting the buffer. 

274 Nothing but this function may mutate ``buffer`` between calls. 

275 

276 :param buffer: A bytearray buffer to hold incomplete line data. Must 

277 hold no newline on entry. 

278 :param data: Incoming byte data to append to the buffer. 

279 :param encoding: The character encoding to use for decoding lines. Defaults to 

280 `"utf-8"`. 

281 :yield: Each complete line decoded from the buffer, terminator stripped; 

282 empty lines are dropped. 

283 """ 

284 if not data: 

285 return 

286 # Two cursors, not one: the terminator can only be in the arriving 

287 # bytes, but the line it ends begins at the front of the buffer, where 

288 # the remainder carried from earlier calls sits. 

289 search_from = len(buffer) 

290 buffer.extend(data) 

291 line_start = 0 

292 lines: list[bytearray] = [] 

293 while True: 

294 newline_pos = buffer.find(b"\n", search_from) 

295 if newline_pos == -1: 

296 break 

297 lines.append(buffer[line_start:newline_pos]) 

298 line_start = search_from = newline_pos + 1 

299 del buffer[:line_start] 

300 for line_bytes in lines: 

301 if line_bytes: 

302 yield line_bytes.decode(encoding)
