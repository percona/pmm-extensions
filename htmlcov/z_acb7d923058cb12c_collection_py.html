16"""Delete the inventory tombstones no persisted reference resolves any more. 

17 

18A tombstoned inventory entity may be deleted when all three of these hold: 

19 

201. **Age.** It has been retired for at least ``COLLECTION_RETENTION``. 

212. **No live referent.** No artifact that can still *resolve or re-emit* its id 

22 survives. For a service that means: no ``MysqlBackupRun`` row, no ``Task.data`` 

23 meta (soft-deleted tasks included), no in-flight ``TaskHistory``, and no 

24 Celery-beat ``PeriodicTask.kwargs`` naming it. Nodes, schemas and tables have 

25 no external persisted holder at all, so this clause is vacuous for them. 

263. **Subtree safety.** Every descendant in its retirement subtree is itself 

27 collectible — enforced by the Inventory API, which owns the walk. 

28 

29Purely *historical* references do not block collection. A terminal 

30``TaskHistory.execution_request`` meta is the only such holder: it can never 

31create a new dangling reference, and it already carries ``_service_name`` 

32unconditionally. The one path that resolves its id — the legacy form backfill — 

33reads inventory through the retired-hiding default manager, so a retired 

34service's id stops matching there the moment it is tombstoned and deleting the 

35row later changes nothing. An *in-flight* history is not historical and does 

36block, because it can still complete and write a new ``MysqlBackupRun`` row 

37naming the id. 

38 

39The scan and the delete are not one transaction. For every holder that already 

40existed when the scan ran the window is closed: ``app/tasks/run_result.py`` 

41resolves a recorder off the live ``Task`` row, so a recorder firing mid-window 

42belongs to a task the scan already read — which is why the ``Task`` scan reads 

43every row rather than only the active ones. 

44 

45A holder persisted *during* the window is not covered. The Tasks API takes 

46arbitrary envelope meta, so a task or beat schedule created between a batch's 

47dry run and its delete can name an id that batch removes, and a later run of it 

48would record a backup run pointing at a service the catalog can no longer 

49resolve. Fencing that off needs a lease or epoch spanning four databases; it is 

50left open instead, bounded by how narrow it is — the id has to have been 

51tombstoned for at least ``COLLECTION_RETENTION``, and the write has to land 

52between two consecutive calls of one batch. 

53"""

91def _positive_ids(values: Iterable[Any]) -> set[int]: 

92 """Coerce raw JSON-extracted values to the positive integers among them. 

93 

94 The extraction crosses a dialect boundary — PostgreSQL hands back text while 

95 SQLite hands back the native scalar — and the source is a free-form JSON 

96 blob, so anything that is not a positive integer is a malformed envelope 

97 rather than an error to raise on. 

98 

99 :param values: The raw values read out of a ``meta`` key. 

100 :return: The positive integers among them. 

101 """ 

102 ids: set[int] = set() 

103 for value in values: 

104 try: 

105 entity_id = int(value) 

106 except (TypeError, ValueError): 

107 logger.debug( 

108 "Ignoring non-integer %s meta value %r", SERVICE_ID_META_KEY, value 

109 ) 

110 continue 

111 if entity_id > 0: 

112 ids.add(entity_id) 

113 return ids

116def _beat_kwargs_service_ids(payloads: Iterable[str | None]) -> set[int]: 

117 """Extract the service ids the beat schedules still carry. 

118 

119 ``PeriodicTask.kwargs`` is a JSON *string* column declared by 

120 ``sqlalchemy-celery-beat``, so the payload is parsed in Python rather than by 

121 a JSON operator. Every level is ``isinstance``-guarded: the column is 

122 operator-populated free text, so a malformed row is skipped rather than 

123 raised on. The extracted leaves are gathered into a list rather than a set 

124 for the same reason — the value is unconstrained JSON and may be unhashable, 

125 and judging it is :func:`_positive_ids`'s job. 

126 

127 Only the rows :class:`PeriodicTaskManager` surfaces reach here, which is the 

128 ``execute_task_by_name`` indirection. That is the sole beat shape carrying an 

129 ``execution_data`` envelope, so a row pointing straight at a Celery function 

130 cannot name a service id in the first place. 

131 

132 :param payloads: The raw ``kwargs`` strings of the beat schedules. 

133 :return: The service ids named by their envelopes. 

134 """ 

135 ids: list[Any] = [] 

136 for payload in payloads: 

137 if not payload: 

138 continue 

139 try: 

140 decoded = json.loads(payload) 

141 except json.JSONDecodeError: 

142 logger.warning("Skipping periodic-task kwargs that is not valid JSON") 

143 continue 

144 if not isinstance(decoded, dict): 

145 continue 

146 execution_data = decoded.get("execution_data") 

147 meta = execution_data.get("meta") if isinstance(execution_data, dict) else None 

148 if isinstance(meta, dict) and SERVICE_ID_META_KEY in meta: 

149 ids.append(meta[SERVICE_ID_META_KEY]) 

150 return _positive_ids(ids)

153async def collect_task_envelope_service_ids() -> set[int]: 

154 """Return every inventory service id a task envelope can still re-emit. 

155 

156 Three reads, in two databases. The first two are what closes the scan-to- 

157 delete window; the third covers a schedule that has not fired yet. 

158 

159 A read that cannot complete propagates rather than returning what it managed 

160 to gather: a holder that could not be read is not a holder that is empty, and 

161 treating it as empty would collect the entities it names. 

162 

163 :return: The service ids the task envelopes still name. 

164 :raises Exception: Whatever a database this reads raises, unchanged. 

165 """ 

166 tasks_session_maker = get_tasks_session_maker() 

167 async with tasks_session_maker() as session: 

168 service_ids = _positive_ids( 

169 await TaskManager.envelope_meta_values(session, SERVICE_ID_META_KEY) 

170 ) 

171 service_ids |= _positive_ids( 

172 await TaskHistoryManager.in_flight_meta_values(session, SERVICE_ID_META_KEY) 

173 ) 

174 beat_session_maker = get_beat_session_maker() 

175 async with beat_session_maker() as session: 

176 service_ids |= _beat_kwargs_service_ids( 

177 await PeriodicTaskManager.values_list(session, ["kwargs"]) 

178 ) 

179 return service_ids

182async def collect_referenced_entities() -> dict[RetirableEntityName, set[int]]: 

183 """Return every inventory id PMM Extensions still holds a resolvable reference to. 

184 

185 Any failure propagates: issuing the delete with a partial retained set would 

186 collect an entity something still resolves, which is exactly the outcome this 

187 job exists to prevent. 

188 

189 :return: The referenced ids, keyed by inventory entity type. 

190 :raises Exception: Whatever a declared provider or the task-envelope scan 

191 raises, unchanged. 

192 """ 

193 referenced = { 

194 RetirableEntityName.SERVICE: await collect_task_envelope_service_ids() 

195 } 

196 extensions_session_maker = get_extensions_session_maker() 

197 async with extensions_session_maker() as session: 

198 for provider in collect_inventory_reference_providers(): 

199 for entity_name, entity_ids in (await provider(session)).items(): 

200 referenced.setdefault(entity_name, set()).update(entity_ids) 

201 return referenced

204async def clear_absence_ledger( 

205 collected: Mapping[RetirableEntityName, list[int]], 

206) -> None: 

207 """Drop the missing-grace ledger rows of the entities about to be deleted. 

208 

209 Cleared *before* the delete rather than after, because that is the only 

210 order with no crash window, and it is safe: the syncer short-circuits on an 

211 already-retired entity before it can record a fresh absence, so an early 

212 clear cannot be undone, and clearing a superset is harmless. 

213 

214 Cleared for every syncer rather than the configured ones, which is the same 

215 licence read the other way: a ledger row outlives the configuration that 

216 wrote it, so scoping the delete to ``EXTENSIONS.SYNCERS`` would leave the rows of a 

217 syncer since removed or renamed behind permanently, with the entity they 

218 name gone and nothing left that could ever reach them. 

219 

220 :param collected: The ids the dry run reported, per entity type. 

221 """ 

222 extensions_session_maker = get_extensions_session_maker() 

223 async with extensions_session_maker() as session: 

224 for entity_name, entity_ids in collected.items(): 

225 if not entity_ids: 

226 continue 

227 await SyncEntityAbsenceManager.clear( 

228 session, 

229 None, 

230 _LEDGER_ENTITY_TYPES[entity_name], 

231 *entity_ids, 

232 )

235def _parse_batch( 

236 payload: Any, 

237) -> tuple[dict[RetirableEntityName, list[int]], bool]: 

238 """Read one collection response into its collected ids and remaining flag. 

239 

240 Parsed through the response model the Inventory API answers with, so every 

241 level of the body is checked rather than only its outermost type. 

242 

243 :param payload: The decoded JSON body the Inventory API answered with. 

244 :return: The collected ids per entity type, and whether more are waiting. 

245 :raises ValidationError: If the body is not the documented object, or names 

246 an entity type this service does not know. 

247 """ 

248 parsed = InventoryCollectResponse.model_validate(payload) 

249 return parsed.deleted, parsed.remaining

252async def run_inventory_collection(api_key: str) -> None: 

253 """Collect tombstoned inventory entities no live reference resolves. 

254 

255 Each batch calls the Inventory API twice with the same ``retired_before`` and 

256 retained set: a dry run naming the ids, then — once their ledger rows are 

257 gone — the real delete. The cutoff is computed once for the whole run so 

258 successive batches cannot drift. 

259 

260 :param api_key: The internal token authenticating the Inventory API calls. 

261 :raises HTTPException: Whatever :meth:`RemoteAPI.post` raises for a non-2xx 

262 answer from the Inventory API — an expired token or a restarting service 

263 is the likeliest runtime failure of this job. 

264 :raises ValidationError: If the Inventory API answers a collect call with 

265 something other than the documented object, or names an unknown entity 

266 type. 

267 :raises Exception: Whatever computing the retained set raises, unchanged — 

268 deleting against a partial retained set is the failure this prevents. 

269 """ 

270 retired_before = utc_now() - inventory_app_settings.COLLECTION_RETENTION 

271 keep = { 

272 name.value: sorted(entity_ids) 

273 for name, entity_ids in (await collect_referenced_entities()).items() 

274 } 

275 client = await get_inventory_api_standalone() 

276 body = { 

277 "retired_before": retired_before.isoformat(), 

278 "keep": keep, 

279 "limit": inventory_app_settings.COLLECTION_BATCH_SIZE, 

280 "dry_run": False, 

281 } 

282 with client.auth(api_key): 

283 async with client.hold(): 

284 for batch in range(inventory_app_settings.COLLECTION_MAX_BATCHES): 

285 candidates, _ = _parse_batch( 

286 await client.post(COLLECT_PATH, json={**body, "dry_run": True}) 

287 ) 

288 await clear_absence_ledger(candidates) 

289 deleted, remaining = _parse_batch( 

290 await client.post(COLLECT_PATH, json=body) 

291 ) 

292 logger.info( 

293 "Inventory collection batch %s deleted %s entities", 

294 batch + 1, 

295 sum(len(ids) for ids in deleted.values()), 

296 ) 

297 if not remaining: 

298 return 

299 logger.info( 

300 "Inventory collection stopped at its %s-batch cap; the next run continues.", 

301 inventory_app_settings.COLLECTION_MAX_BATCHES, 

302 )

305async def run_scheduled_inventory_collection() -> None: 

306 """Run inventory collection using the configured internal token. 

307 

308 :raises HTTPException: Whatever the Inventory API's non-2xx answers raise. 

309 :raises ValidationError: If the Inventory API answers a collect call with 

310 something other than the documented object. 

311 :raises Exception: Whatever computing the retained set raises, unchanged. 

312 Every failure is logged and re-raised so the Celery run records it and 

313 the task's failure alert fires; aborting deletes nothing, except in the 

314 window between clearing a batch's ledger rows and its delete. 

315 """ 

316 try: 

317 await run_inventory_collection(get_internal_token()) 

318 except Exception: 

319 logger.exception("Scheduled inventory collection failed") 

320 raise
