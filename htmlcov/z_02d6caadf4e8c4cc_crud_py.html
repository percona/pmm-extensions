37def _meta_text(key: str) -> ColumnElement: 

38 """Return the ``meta`` JSON value at ``key`` as a text expression. 

39 

40 One definition of "read this meta key" shared by the sort allowlist, the search 

41 set, the service-type filter, and the whole-dataset facet, so those four can 

42 never disagree on what a padded or absent value means. SQLAlchemy renders it per 

43 dialect (``->>`` on PostgreSQL, ``JSON_EXTRACT`` on SQLite), which also keeps it 

44 usable in a class-level spec where no session — and so no engine name — exists. 

45 

46 :param key: The ``meta`` JSON key to read. 

47 :return: A text-typed column expression over ``Snippet.meta``. 

48 """ 

49 return Snippet.meta[key].as_string()

52class SnippetManager(BaseSQLModelManager): 

53 """Manage Snippet operations, including retrieval, listing, and deletion. 

54 

55 :cvar Model: The SQLModel class this manager is responsible for (`Snippet`). 

56 :vartype Model: type[Snippet] 

57 :cvar list_query_spec: The request-boundary sort/search allowlist backing the 

58 derived list route, and the single authority for the default ordering. 

59 First-class columns sort and search directly; ``title`` and ``service_type`` 

60 resolve through :func:`_meta_text`. ``-approved_at`` keeps the approved-first

71 "title": _meta_text(META_KEY_TITLE), 

72 "service_type": _meta_text(META_KEY_SERVICE_TYPE), 

73 }, 

74 default_sort="-approved_at", 

75 tie_breaker=col(Snippet.id),

78 _meta_text(META_KEY_TITLE), 

79 _meta_text(META_KEY_DESCRIPTION),

83 @staticmethod 

84 def _service_type_exprs() -> tuple[ColumnElement, ColumnElement]: 

85 """Return the raw and trimmed ``meta.service_type`` expressions. 

86 

87 The whole-dataset facet and the list filter normalise the free-form service 

88 type through this one definition, so a padded value groups identically on 

89 both paths instead of the facet and the predicate disagreeing on what counts 

90 as blank. 

91 

92 :return: The raw meta-text expression and its ``TRIM``-normalised form. 

93 """ 

94 raw = _meta_text(META_KEY_SERVICE_TYPE) 

95 return raw, func.trim(raw)

97 @classmethod 

98 def _list_query_filters( 

99 cls, list_query: SnippetListQuery 

100 ) -> list[ColumnExpressionArgument[bool]]: 

101 """Build the filter predicates for a snippets list query. 

102 

103 Sort and search are Core's, resolved into ``list_query.core``; what is built 

104 here is the snippets-specific approval and service-type restriction. They stay 

105 separate predicates so the applier folds them into the one clause set feeding 

106 both the count and the data query. 

107 

108 :param list_query: The validated list-query selections. 

109 :return: The list of predicates (empty when nothing is filtered). 

110 """ 

111 filters: list[ColumnExpressionArgument[bool]] = [] 

112 

113 if list_query.approval is SnippetApprovalFilter.APPROVED: 

114 filters.append(col(Snippet.approved_at).is_not(None)) 

115 elif list_query.approval is SnippetApprovalFilter.NOT_APPROVED: 

116 filters.append(col(Snippet.approved_at).is_(None)) 

117 

118 service_type_raw, service_type_trimmed = cls._service_type_exprs() 

119 if list_query.uncategorized: 

120 filters.append(or_(service_type_raw.is_(None), service_type_trimmed == "")) 

121 elif list_query.service_type is not None: 

122 # Strip only spaces on the filter value to mirror SQL ``TRIM``'s default, 

123 # so the comparison agrees with the facet for tab/newline-padded values. 

124 filters.append(service_type_trimmed == list_query.service_type.strip(" ")) 

125 

126 return filters

134 pagination: Pagination, 

135 ) -> PaginatedResponse[Snippet]: 

136 """Return a page of snippets with server-side search, filters, and sorting. 

137 

138 The snippets filters compose with the Core-resolved sort and search in one 

139 whereclause set, so the paginated ``total`` matches the visible, filtered 

140 result set and the ordering carries the spec's unique tie-breaker. 

141 

142 :param session: The SQLAlchemy asynchronous session to use for query execution. 

143 :param list_query: The validated sort/search/filter selections. 

144 :param pagination: The validated offset/limit window for this page. 

145 :return: A paginated response over the filtered, ordered snippets. 

146 :raises sqlalchemy.exc.SQLAlchemyError: When a count or data query fails to 

147 execute. 

148 """ 

149 return await cls.list_query_paginated( 

150 session, 

151 *cls._list_query_filters(list_query), 

152 list_query=list_query.core, 

153 pagination=pagination,

181 @classmethod 

182 async def list_service_types(cls, session: AsyncSession) -> tuple[list[str], bool]: 

183 """Return the distinct service types across the whole snippets table. 

184 

185 Backs the list page's service-type filter so its options reflect the 

186 complete dataset rather than the loaded page. Blank and absent values are 

187 folded into the ``has_uncategorized`` flag rather than emitted as selectable 

188 values. The trimming shares :meth:`_service_type_exprs` with the list 

189 filter, so a value the facet omits as blank is the same one the filter 

190 treats as uncategorized. 

191 

192 :param session: The SQLAlchemy asynchronous session to use for query execution. 

193 :return: A tuple of the sorted distinct non-blank service types and whether 

194 any snippet has an absent or blank service type. 

195 :raises sqlalchemy.exc.SQLAlchemyError: When the query fails to execute. 

196 """ 

197 _, service_type_trimmed = cls._service_type_exprs() 

198 values = ( 

199 await cls._exec(session, select(service_type_trimmed).distinct()) 

200 ).all() 

201 return sorted({value for value in values if value}), any( 

202 not value for value in values

205 @classmethod 

206 async def get_or_create( 

207 cls, 

208 session: AsyncSession, 

209 instance_create: Snippet, 

210 filter_include: set[str] | None = None, 

211 **extra_fields: Any, 

212 ) -> tuple[Snippet, bool]: 

213 """Retrieve an existing Snippet instance or create a new one if none exists. 

214 

215 This method overrides the default `get_or_create` method to generate the `meta` 

216 field of the Snippet instance based on the snippet's path. 

217 

218 :param session: The SQLAlchemy asynchronous session to use for database 

219 operations. 

220 :type session: AsyncSession 

221 :param instance_create: The data used to filter and possibly create the 

222 instance. 

223 :type instance_create: Snippet 

224 :param filter_include: The set of fields of `instance_create` to be included in 

225 the search filter. Use None (default) for all fields. 

226 :param extra_fields: Additional fields to be set on the created instance. 

227 :type extra_fields: Any 

228 :return: The existing or newly created instance of `cls.Model`, and a bool 

229 specifying whether a new instance was created. 

230 :rtype: tuple[Snippet, bool] 

231 """ 

232 existent_instance = await cls.first( 

233 session, **instance_create.model_dump(include=filter_include) 

234 ) 

235 if existent_instance: 

236 return existent_instance, False 

237 await instance_create.update_meta() 

238 return await cls.create(session, instance_create, **extra_fields), True
