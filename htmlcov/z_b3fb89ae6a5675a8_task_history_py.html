16"""Define the ``/api/extensions/task-history/`` JSON endpoints proxying task history. 

17 

18Expose task-history reads through the PMM Extensions gateway so the React frontend never 

19calls the Tasks sub-app directly. ``GET /`` either lists all history 

20(passthrough) or merges ``GET /{task}/history/`` across the supplied task names; 

21``POST /{id}/stop/`` proxies a stop request for a single history row. 

22"""

53@router.get("/", responses=UPSTREAM_TASKS_502_RESPONSE) 

54async def list_merged_task_history( 

55 tasks_api: TaskAPI, 

56 pagination: PaginationDep, 

57 *, 

58 task_names: Annotated[list[str] | None, Query()] = None, 

59 task_status: Annotated[TaskHistoryStatusEnum | None, Query(alias="status")] = None, 

60 exclude_internal: Annotated[bool, Query()] = False, 

61) -> PaginatedResponse[ExtensionsTaskHistoryResponse]: 

62 """Return task-history rows, listing all of them or merging selected names. 

63 

64 Three-way on ``task_names``: 

65 

66 * **omitted** (``None``) -- proxy the upstream ``GET /history/`` list, already 

67 paginated, forwarding the ``status`` filter, ``exclude_internal`` flag, and 

68 client ``offset`` / ``limit``. 

69 * **provided, at least one non-blank name** -- query each name independently 

70 against the Tasks API, then merge, sort newest-first, and paginate globally. 

71 * **provided, every name blank after trimming** -- reject with ``422``. 

72 

73 :param tasks_api: The Tasks API client used to fetch upstream history. 

74 :param pagination: Validated offset/limit query parameters. 

75 :param task_names: Zero or more task names (repeat the query param); omit to 

76 list all history. 

77 :param task_status: Optional exact status filter forwarded upstream. 

78 :param exclude_internal: When ``True``, forward the filter to the upstream 

79 list-all path so internal maintenance tasks are excluded before pagination. 

80 Not forwarded on the ``task_names`` merge path. Defaults to ``False``. 

81 :return: Paginated task history, either the upstream list or the merged set, 

82 with every actor identifier resolved to the name a reader should see. 

83 :raises HTTPUnprocessableEntityException: When ``task_names`` is supplied but 

84 every value is empty after trimming. 

85 :raises HTTPBadGatewayException: For an upstream server error (status >= 500) 

86 or a connection-level ``OSError``, on either the list-all passthrough or 

87 the merged-history fan-out. 

88 """ 

89 if task_names is None: 

90 params = { 

91 "offset": pagination.offset, 

92 "limit": pagination.limit, 

93 **({"status": task_status.value} if task_status is not None else {}), 

94 **({"exclude_internal": "true"} if exclude_internal else {}), 

95 } 

96 with reraise_upstream_tasks_errors(): 

97 payload = await tasks_api.get("/history/", params=params) 

98 page = PaginatedResponse[ExtensionsTaskHistoryResponse].model_validate(payload) 

99 else: 

100 if not normalize_task_history_names(task_names): 

101 raise HTTPUnprocessableEntityException( 

102 "task_names must contain at least one non-empty name" 

103 ) 

104 with reraise_upstream_tasks_errors(): 

105 page = await fetch_merged_task_history( 

106 tasks_api, 

107 task_names, 

108 status=task_status, 

109 pagination=pagination, 

110 ) 

111 return resolve_task_history_actors(page, await get_username_mapping()) 

112

114@router.post( 

115 "/{task_history_id}/stop/", 

116 dependencies=[RequireBearerForUnsafeMethods], 

117 responses=UPSTREAM_TASKS_502_RESPONSE, 

118) 

119async def stop_task_history( 

120 task_history_id: int, tasks_api: TaskAPI 

121) -> ArbitraryMapping: 

122 """Dispatch a stop request for a single task-history row to the Tasks API. 

123 

124 :param task_history_id: The id of the task-history row to stop. 

125 :param tasks_api: The Tasks API client used to issue the stop request. 

126 :return: The stopped task-history row as returned by the Tasks API. 

127 :raises HTTPException: Re-raised unchanged for an upstream client error 

128 (status < 500), e.g. a ``400`` when the task is not running. 

129 :raises HTTPBadGatewayException: For an upstream server error (status >= 500) 

130 or a connection-level ``OSError``. 

131 """ 

132 with reraise_upstream_tasks_errors(): 

133 return ArbitraryMapping( 

134 as_json_object(await tasks_api.post(f"/history/{task_history_id}/stop/")) 

135 )
