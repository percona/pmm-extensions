86async def authenticate_bearer_token(token: str) -> BaseUser: 

87 """Return the authenticated user from an OAuth2 token. 

88 

89 When ``settings.EXTENSIONS_INTERNAL_TOKEN`` is non-empty and the incoming Bearer 

90 token matches it (constant-time comparison), return a synthetic non-admin 

91 "service principal" user instead of contacting the OAuth provider. This 

92 allows PMM Extensions internal service-to-service calls (e.g. scheduled inventory 

93 sync) to authenticate with a stable deployment-level secret rather than a 

94 short-lived personal access token. 

95 

96 Otherwise the token is validated through ``User.from_bearer``, which accepts 

97 the credential types the active provider honors on its Bearer surface — for 

98 Grafana, an access assertion or a short-lived session-exchange assertion. The 

99 session-cookie surface stays narrower and validates through ``from_jwt``. 

100 

101 :param token: The OAuth2 token to authenticate the user. 

102 :return: The authenticated user. 

103 :raises HTTPUnauthorizedException: If the token is invalid and authentication fails. 

104 :raises InactiveUserException: If authentication succeeds but the user is not 

105 active. 

106 :raises BaseAuthProviderException: If the auth provider errors while 

107 validating the credential. 

108 """ 

109 secret = settings.EXTENSIONS_INTERNAL_TOKEN.get_secret_value() 

110 if secret and secrets.compare_digest(token, secret): 

111 set_log_context(user=SERVICE_PRINCIPAL.username) 

112 return _build_service_principal(secret) 

113 try: 

114 user = await User.from_bearer(token) 

115 except ValidationError: 

116 logger.exception("Failed to authenticate user") 

117 raise HTTPUnauthorizedException from None 

118 if not user.is_active: 

119 raise InactiveUserException 

120 set_log_context(user=user.username) 

121 return user

124#: Scope key under which a request's resolved users are cached. Kept out of 

125#: ``scope["state"]``, which a request inherits as a shallow copy of the ASGI 

126#: lifespan state: a cache under a key anything publishes there would be shared 

127#: process-wide. Entries key on a digest of the credential rather than the 

128#: credential itself, so no structure the request carries grows a key that is a 

129#: secret. 

130_RESOLVED_USERS_KEY: Final = "app.api.deps.resolved_users"

133async def get_current_user(request: Request, token: AuthToken) -> BaseUser: 

134 """Return the authenticated user, resolving each credential once per request. 

135 

136 The cache is keyed on a digest of the credential, so a resolution is never 

137 served to a caller presenting a different one, and holds successes only, so a 

138 refused credential is re-derived rather than remembered. It lives under a 

139 private key in the request scope, which the ASGI server builds per request, so 

140 it cannot outlive the request that created it. 

141 

142 A hit re-establishes the log identity the resolution it replaces would have 

143 set, since that identity is a context variable the last resolution wrote 

144 rather than something the returned user carries. 

145 

146 :param request: The incoming HTTP request, whose scope holds the cache. 

147 :param token: The OAuth2 token to authenticate the user. 

148 :return: The authenticated user, resolved here or on an earlier call. 

149 :raises HTTPUnauthorizedException: If the token is invalid and authentication fails. 

150 :raises InactiveUserException: If authentication succeeds but the user is not 

151 active. 

152 :raises BaseAuthProviderException: If the auth provider errors while 

153 validating the credential. 

154 """ 

155 resolved: dict[bytes, BaseUser] = request.scope.setdefault(_RESOLVED_USERS_KEY, {}) 

156 key = sha256(token.encode()).digest() 

157 if (user := resolved.get(key)) is not None: 

158 set_log_context(user=user.username) 

159 return user 

160 resolved[key] = user = await authenticate_bearer_token(token) 

161 return user

239class ServicePrincipalWriteRoute(APIRoute): 

240 """Build a route admitting only the service principal when any method is unsafe. 

241 

242 Set as a router's ``route_class`` so a write route added to it is 

243 syncer-only by default, and one a human may call has to say so by declaring 

244 :data:`ExemptFromServicePrincipalDep`. The restriction is added as a normal 

245 route dependency rather than checked in a router-level dependency's body, so 

246 it keeps advertising the Bearer requirement in the schema, is resolved from 

247 FastAPI's per-request cache, and runs ahead of every other dependency and the 

248 path lookups — a refused caller gets 403 before any 404 could tell it which 

249 identifiers exist. 

250 

251 ``include_router`` rebuilds a route with the class it was declared with, not 

252 the including router's, so a router nested into one built on this class has 

253 to set it as its ``route_class`` too, or its writes skip the default. 

254 """ 

255

299def require_minimum_role(role: UserRole) -> Callable[[F], F]: 

300 """Return a decorator registering the minimum role one route requires. 

301 

302 The registration keys on the object FastAPI stores as 

303 ``APIRoute.endpoint``, so this belongs below any decorator that returns a 

304 *new* function: a wrapper between the two leaves the registry keyed on a 

305 callable no request reaches, and the route falls back to 

306 :data:`DEFAULT_MINIMUM_ROLE` with nothing raised to say so. Order against 

307 the route decorator itself is immaterial — that one returns the endpoint 

308 unchanged. Registering a rank below ``ADMIN`` opens a surface the gate would 

309 otherwise close, so it belongs on a route whose operation the named rank is 

310 trusted with — and ``UserRole.NONE`` on one whose method is unsafe but whose 

311 operation is a read. 

312 

313 :param role: The lowest rank the route admits. 

314 :return: A decorator registering the endpoint it receives, unchanged. 

315 """ 

316 

317 def register(endpoint: F) -> F: 

318 _ROUTE_MINIMUM_ROLES[endpoint] = role 

319 return endpoint 

320 

321 return register

324def minimum_role_for(route: object | None) -> UserRole: 

325 """Return the minimum role a matched route requires. 

326 

327 The single home of the unregistered-route default, read by the gate and by 

328 the test that classifies PMM Extensions' unsafe surface, so the two cannot disagree 

329 about what an unregistered route resolves to. A request no route matched 

330 carries no ``route`` in its scope and resolves the same way. 

331 

332 :param route: The matched route from the request scope, if any. Typed 

333 ``object`` because the scope value is untyped and need not be an 

334 ``APIRoute``; anything without an ``endpoint`` takes the default. 

335 :return: The registered minimum, or :data:`DEFAULT_MINIMUM_ROLE`. 

336 """ 

337 endpoint = getattr(route, "endpoint", None) 

338 if endpoint is None: 

339 return DEFAULT_MINIMUM_ROLE 

340 return _ROUTE_MINIMUM_ROLES.get(endpoint, DEFAULT_MINIMUM_ROLE)

343async def require_minimum_role_for_unsafe_methods(request: Request) -> None: 

344 """Require each mutating HTTP method's route-registered minimum role. 

345 

346 The authorization sibling of ``require_bearer_for_unsafe_methods``. Safe 

347 methods pass untouched, keeping whatever authentication they already carry; 

348 every other method — ``POST``, ``PUT``, ``PATCH`` and ``DELETE``, and 

349 equally anything no route declares — resolves the caller and admits it only 

350 from :func:`minimum_role_for` upwards, ranks comparing on ``UserRole``'s 

351 declared order rather than on equality. 

352 

353 A ``UserRole.NONE`` minimum is answered before the credential is looked at, 

354 so the gate imposes no authentication of its own on a route it waives and 

355 that route's own ``dependencies`` stay the only thing authenticating it. 

356 

357 The user is resolved in the body rather than through a sub-dependency: a 

358 sub-dependency resolves eagerly and would force authentication onto the 

359 unauthenticated ``GET /health`` that every service exposes. The request is 

360 passed along so the route's own authentication dependency is served from this 

361 resolution rather than repeating it. 

362 

363 ``EXTENSIONS_INTERNAL_TOKEN``'s service principal is admitted by identity so 

364 scheduled inventory sync and scheduled execution keep working. It holds 

365 ``UserRole.VIEWER`` and so would fail every minimum above it; the bypass 

366 stays scoped to this gate, leaving every pre-existing ``IsApiAdmin`` / 

367 ``IsAdminDep`` check refusing it as before. 

368 

369 :param request: The incoming HTTP request. 

370 :raises HTTPUnauthorizedException: When the method is unsafe and the request 

371 carries no Bearer credential, or the credential does not validate. 

372 :raises HTTPForbiddenException: When the resolved user ranks below the 

373 route's minimum, or is inactive. 

374 :raises BaseAuthProviderException: When the auth provider errors while 

375 validating the credential. 

376 """ 

377 if request.method in SAFE_HTTP_METHODS: 

378 return 

379 minimum = minimum_role_for(request.scope.get("route")) 

380 if minimum is UserRole.NONE: 

381 return 

382 if not is_bearer_authenticated(request): 

383 raise HTTPUnauthorizedException 

384 user = await get_current_user(request, await oauth2_scheme(request)) 

385 if user.id == SERVICE_PRINCIPAL_ID: 

386 return 

387 if user.role < minimum: 

388 raise HTTPForbiddenException
