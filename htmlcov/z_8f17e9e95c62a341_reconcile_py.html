16"""Reconcile diagnostics runs whose outcome the recorder hook never observed. 

17 

18Two gaps make this sweep load-bearing rather than belt-and-braces, and both are 

19permanent: ``maybe_record_run`` documents three terminal transitions it is not 

20invoked for (a run stopped via the stop route, one that failed before dispatch, 

21one the connectivity probe drives terminal), and every execution dispatched before 

22ATW owned a proxy task predates the recorder entirely. 

23 

24There is deliberately **no time cutoff**: an older incident showing an 

25authoritative-looking zero is exactly the wrong answer for the screen this 

26feature exists to fix. 

27"""

51class _UpstreamRun(BaseModel): 

52 """Read just the outcome fields off an upstream task-history payload. 

53 

54 Validating rather than reading raw keys is what coerces the serialized status 

55 to its enum member and the ISO timestamp to an aware ``datetime``, so an 

56 unrecognized status surfaces as a rejection instead of reaching the column. 

57 

58 :param status: The run's current status. 

59 :param finished_at: When the run finished, if it has. 

60 """ 

61 

62 status: TaskHistoryStatusEnum 

63 finished_at: UTCDatetime | None = None

66async def _resolve_outcome( 

67 tasks_api: RemoteAPI, execution_id: UUID, task_history_id: int 

68) -> dict[str, Any]: 

69 """Return the outcome columns to write for one execution, if any are known. 

70 

71 The two failure branches are the load-bearing pair. A genuine ``404`` means the 

72 upstream row is gone and no outcome will ever exist, so the row is retired 

73 permanently and never re-queried. Any other HTTP or transport error is 

74 transient and must leave the row eligible — reading a 503 as "gone" would 

75 permanently retire a live execution. 

76 

77 This is why the sweep does **not** reuse ``fetch_task_history``, which degrades 

78 every ``HTTPException | OSError`` to an empty mapping: correct for a listing 

79 that wants to blank one row, catastrophic here. 

80 

81 :param tasks_api: The authenticated Tasks API client. 

82 :param execution_id: The ATW execution row being reconciled, for logging. 

83 :param task_history_id: The upstream history row to read the outcome from. 

84 :return: The columns to write, or an empty mapping to leave the row unresolved. 

85 """ 

86 try: 

87 payload = as_json_object(await tasks_api.get(f"/history/{task_history_id}")) 

88 except HTTPNotFoundException: 

89 logger.info( 

90 "Task history %s no longer resolves; retiring ATW execution %s as having " 

91 "no recoverable outcome.", 

92 task_history_id, 

93 execution_id, 

94 ) 

95 return {"outcome_unrecoverable": True} 

96 except (HTTPException, OSError): 

97 logger.warning( 

98 "Could not reach task history %s; leaving ATW execution %s unresolved for " 

99 "a later tick.", 

100 task_history_id, 

101 execution_id, 

102 exc_info=True, 

103 ) 

104 return {} 

105 try: 

106 run = _UpstreamRun.model_validate(payload) 

107 except ValidationError: 

108 logger.warning( 

109 "Task history %s reported an unreadable outcome; leaving ATW execution %s " 

110 "unresolved.", 

111 task_history_id, 

112 execution_id, 

113 exc_info=True, 

114 ) 

115 return {} 

116 if not run.status.is_terminal(): 

117 return {} 

118 return {"terminal_status": run.status, "finished_at": run.finished_at}

121async def _reconcile_one( 

122 tasks_api: RemoteAPI, execution_id: UUID, task_history_id: int 

123) -> None: 

124 """Write one execution's resolved outcome, stamping the attempt either way. 

125 

126 ``reconcile_attempted_at`` is stamped on every path, including both failures, so 

127 a row the sweep could not resolve still yields its batch slot to the rows behind 

128 it — which is what makes each tick a round-robin rather than a re-examination of 

129 the same head of the queue. 

130 

131 Each row is written on its own session rather than one shared across the batch: 

132 under PostgreSQL a failed statement aborts the whole transaction, so a shared 

133 session would turn one bad row into a lost tick. The sessions are cheap — they 

134 come from the same engine's pool. 

135 

136 :param tasks_api: The authenticated Tasks API client. 

137 :param execution_id: The ATW execution row to update. 

138 :param task_history_id: The upstream history row to read the outcome from. 

139 """ 

140 outcome = await _resolve_outcome(tasks_api, execution_id, task_history_id) 

141 async with get_async_session_maker()() as session: 

142 await AtwIncidentExecutionManager.update_where( 

143 session, 

144 {"reconcile_attempted_at": utc_now(), **outcome}, 

145 id=execution_id, 

146 )

149async def reconcile_executions(batch_size: int) -> None: 

150 """Resolve one bounded batch of executions with no recorded outcome. 

151 

152 Selection is least-recently-attempted first, so each tick is a round-robin over 

153 the unresolved set and a row that is legitimately still running cannot starve 

154 the rows behind it. 

155 

156 The batch is claimed before its first upstream request: the selecting session 

157 stamps every selected row's attempt cursor. One request may take minutes against 

158 a slow tasks service, so a tick can outlive the schedule interval, and without 

159 the claim the next tick would re-select the rows this one is still working 

160 through and request each of them a second time. 

161 

162 :param batch_size: The most executions this tick may examine. 

163 """ 

164 async with get_async_session_maker()() as session: 

165 rows = await AtwIncidentExecutionManager.unresolved_batch(session, batch_size) 

166 # Read inside the session that selected them: the loop below runs after it 

167 # closes, so it must not touch an ORM attribute. 

168 targets = [(row.id, row.task_history_id) for row in rows] 

169 if targets: 

170 await AtwIncidentExecutionManager.update_where( 

171 session, 

172 {"reconcile_attempted_at": utc_now()}, 

173 col(AtwIncidentExecution.id).in_( 

174 [execution_id for execution_id, _ in targets] 

175 ), 

176 ) 

177 if not targets: 

178 return 

179 client = await get_tasks_api() 

180 with client.auth(get_internal_token()) as tasks_api: 

181 for execution_id, task_history_id in targets: 

182 await _reconcile_one(tasks_api, execution_id, task_history_id) 

183 logger.info("Examined %d unresolved ATW execution(s).", len(targets))
