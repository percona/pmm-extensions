137 @model_validator(mode="before") 

138 @classmethod 

139 def _derive_role(cls, data: Any) -> Any: 

140 """Derive the ordered role from Casdoor's admin flag. 

141 

142 Casdoor exposes no role concept, so the payload's admin boolean is the 

143 only signal. An unset flag resolves to ``VIEWER`` rather than the lowest 

144 role, keeping the read access a non-admin Casdoor user has today; a flag 

145 that is present is validated as a boolean rather than tested for 

146 truthiness, so a spelled-out ``"false"`` still resolves to ``VIEWER``. 

147 

148 Both wire spellings are read because ``model_config`` accepts either for 

149 every field on this model: Casdoor's API sends ``isAdmin``, and 

150 ``populate_by_name`` admits the snake-case form. 

151 

152 :param data: The raw model input. 

153 :return: The input with a ``role`` filled in, or ``data`` unchanged. 

154 :raises ValueError: If the admin flag is not a value Pydantic accepts as 

155 a boolean. 

156 """ 

157 if not isinstance(data, dict) or "role" in data: 

158 return data 

159 for key in ("isAdmin", "is_admin"): 

160 if key in data: 

161 return {**data, "role": cls._role_from_admin_flag(data[key])} 

162 return {**data, "role": UserRole.VIEWER}

182 @staticmethod 

183 async def get_oauth_token( 

184 code: str | None = None, 

185 username: str | None = None, 

186 password: str | None = None, 

187 refresh_token: str | None = None, 

188 ) -> OAuthToken: 

189 """Retrieve an OAuth token for the user. 

190 

191 Get an OAuth token for the user from Casdoor's SDK. Either ``code``, 

192 ``refresh_token``, or ``username`` and ``password`` must be set. 

193 

194 :param code: An authorization code used to obtain the token. 

195 :param username: The username of the user. 

196 :param password: The password of the user. 

197 :param refresh_token: A refresh token used to obtain a new access token. 

198 :return: The OAuth token for the user. 

199 """ 

200 casdoor = _active_casdoor_sdk() 

201 if refresh_token: 

202 oauth_data = await casdoor.refresh_token_request(refresh_token) 

203 else: 

204 oauth_data = await casdoor.get_access_token(code, username, password) 

205 return OAuthToken(**oauth_data)
