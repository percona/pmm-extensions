184class BackupEntry(BaseModel): 

185 """A single backup record. 

186 

187 :param id: Unique identifier for this backup entry. 

188 :type id: str 

189 :param alias: Human-readable alias for the backup schedule. 

190 :type alias: str 

191 :param name: Node name where the backup was taken. 

192 :type name: str 

193 :param type: Backup type (e.g. ``"physical"``, ``"logical"``). 

194 :type type: str 

195 :param status: Result status of the backup. 

196 :type status: BackupStatus 

197 :param size: Raw backup size value from the metric label, or ``"0"`` if unknown. 

198 :type size: str 

199 :param estimated_data: ``True`` when the time period or size data is estimated. 

200 :type estimated_data: bool 

201 :param enabled: Whether the backup schedule is currently enabled, if available. 

202 :type enabled: bool | None 

203 :param encryption: Encryption state label reported by PMM. 

204 :type encryption: str 

205 :param period: Dictionary with ``start``, ``end``, and ``duration`` of the backup. 

206 :type period: dict[str, Any] 

207 """ 

208 

209 model_config = ConfigDict(extra="allow") 

210 

211 id: str 

212 alias: str = "" 

213 name: str = "" 

214 type: str = "" 

215 status: BackupStatus = BackupStatus.UNKNOWN 

216 size: str = "0" 

217 estimated_data: bool = True 

218 enabled: bool | None = None 

219 encryption: str = "Unknown" 

220 period: dict[str, Any] = Field( 

221 default_factory=dict, json_schema_extra=ARBITRARY_ARGS_SCHEMA 

222 )

363class ReportData(BaseModel): 

364 """Complete report payload ready for rendering. 

365 

366 :param full: When ``True``, all check results and full backup history are included. 

367 :type full: bool 

368 :param refresh: When ``True``, advisor checks were refreshed before data collection. 

369 :type refresh: bool 

370 :param metadata: Report title, generation timestamp, and period labels. 

371 :type metadata: ReportMetadata 

372 :param monitored: Summary counts of monitored nodes and services. 

373 :type monitored: MonitoredSummary 

374 :param advisors: Advisor check results grouped by family. 

375 :type advisors: AdvisorSection 

376 :param alerts: Alert annotation history and aggregations. 

377 :type alerts: AlertSection 

378 :param backups: Backup status entries and aggregations. 

379 :type backups: BackupSection 

380 :param storage: Disk usage entries per node mountpoint. 

381 :type storage: StorageSection 

382 :param uptime: Service uptime entries for the report period. 

383 :type uptime: UptimeSection 

384 :param inventory: Monitored service inventory with connectivity status. 

385 :type inventory: InventorySection 

386 """ 

387 

388 full: bool = True 

389 refresh: bool = False 

390 metadata: ReportMetadata 

391 monitored: MonitoredSummary = Field(default_factory=MonitoredSummary) 

392 advisors: AdvisorSection = Field(default_factory=AdvisorSection) 

393 alerts: AlertSection = Field(default_factory=AlertSection) 

394 backups: BackupSection = Field(default_factory=BackupSection) 

395 storage: StorageSection = Field(default_factory=StorageSection)
