16"""Serve the estate OM holds, and the sweeps that filled it. 

17 

18Every read here follows from one constraint: **the caller must never wait for a Nomad 

19job.** PMM assembles its topology document on the request path in about a tenth of a 

20second; a probe sweep takes tens of seconds. So nothing on this router probes. It 

21serves rows, each carrying when it was last collected, and leaves the caller to 

22decide — which it can, because the age travels with the data and the consumer merges 

23by precedence rather than by trust. 

24 

25``GET /services`` is the contract with pmm-managed. It replaced ``GET /facts``, which 

26served the last sweep's output as a flat fact list: an estate that is *upserted* 

27answers "what is running on this service" even when the most recent sweep never 

28reached it, where the last sweep's output answered only "what did the last sweep 

29see". 

30 

31Auth is applied at the mount level: ``/api/apps`` carries the ``IsApiAuthenticated`` 

32router guard, and unsafe methods additionally require a bearer and the rank 

33:func:`app.api.deps.require_minimum_role` registers per route. An unregistered route 

34resolves to ``DEFAULT_MINIMUM_ROLE`` (admin), so every unsafe route here registers its 

35rank explicitly rather than inheriting one: the estate deletes and the configuration 

36writes are admin, and refreshing is **editor**, because a refresh runs a fixed payload 

37and is the button beside a row — requiring admin would put the routine question behind 

38the rarest role. pmm-managed's service principal is admitted by identity at that gate, 

39so PMM reaches all of these with its deployment token whatever rank a human needs. 

40"""

112def _redact_config(responses: list[SettingResponse]) -> list[SettingResponse]: 

113 """Redact fields too sensitive for this route's viewer-readable audience. 

114 

115 ``GET /config`` is gated only by ``IsApiAuthenticated`` — any logged-in PMM Extensions 

116 user — because that is what the principal behind the pmm-managed 

117 ``--extensions-token`` flag needs to reach the rest of this settings class. 

118 ``CREDENTIALS_PATH`` is the one field that same audience must not see: it names 

119 a file a MongoDB driver reads as a URI, and the settings class docstring already 

120 calls it too sensitive to make writable, let alone world-readable to any 

121 signed-in viewer. 

122 

123 The row still comes back, just with ``value`` forced to ``None``, rather than 

124 being dropped: the class default is already ``None``, so a redacted read and a 

125 genuinely unset deployment are indistinguishable, which is the honest answer, 

126 and no consumer has to special-case a missing row. 

127 

128 :param responses: Every field's response, as collected for this settings class. 

129 :return: The same responses, with redacted keys' ``value`` set to ``None``. 

130 """ 

131 return [ 

132 response.model_copy(update={"value": None}) 

133 if response.key in _REDACTED_CONFIG_KEYS 

134 else response 

135 for response in responses 

136 ]

161def _executor_usable(host: OmHost) -> bool: 

162 """Return whether a payload can actually be dispatched to this host. 

163 

164 Reads ``observed.executor``, which :func:`~app.extensions.apps.om_inventory.crud.

166 probed — unlike ``executor_host``, which is set the moment *any* known executor

171 :return: ``True`` when the host's executor is reachable and driver-healthy. 

172 """ 

173 executor = (host.observed or {}).get("executor") or {} 

174 return bool(executor.get("reachable")) and bool(executor.get("driver_healthy"))

234@router.get("/hosts") 

235async def list_estate_hosts( 

236 session: SessionDep, 

237 pagination: PaginationDep, 

238 has_service: bool | None = Query( 

239 default=None, 

240 description="True for hosts running a MongoDB service, False for those with " 

241 "none. Omit for all of them.", 

242 ), 

243 failing: bool | None = Query( 

244 default=None, description="Restrict to hosts that are, or are not, failing." 

245 ), 

246 executor: bool | None = Query( 

247 default=None, 

248 description="True for hosts a payload can run on, False for those with no " 

249 "executor.", 

250 ), 

251) -> PaginatedResponse[HostResponse]: 

252 """Return every host OM holds, each with its services. 

253 

254 ``has_service=false`` is the question this table exists to answer: which machines 

255 carry a PMM client and no database. It is a filter rather than an endpoint of its 

256 own so there is one list contract to learn, and because the same list with the 

257 filter inverted is the ordinary estate view. 

258 

259 Counts describe the *tables*, not the last run. A scoped refresh must not make the 

260 estate look one host wide. 

261 

262 ``total`` counts every host matching the filters, not the page, so a caller can 

263 tell "this is the whole estate" from "this is the first twenty of it". 

264 

265 :param session: The database session. 

266 :param pagination: The offset/limit window for the page. 

267 :param has_service: Filter on whether a MongoDB service is registered here. 

268 :param failing: Filter on whether the host is currently failing. 

269 :param executor: Filter on whether an executor serves it. 

270 :return: One page of hosts, by name, with the matching total. 

271 """ 

272 clauses = OmHostManager.estate_clauses(has_service=has_service, failing=failing) 

273 

274 if executor is None: 

275 page = await OmHostManager.list_paginated( 

276 session, 

277 *clauses, 

278 pagination=pagination, 

279 order_by=[col(OmHost.name), col(OmHost.node_id)], 

280 ) 

281 hosts, total = page.items, page.total 

282 else: 

283 # executor reads observed.executor, a JSON sub-document — not something 

284 # worth a dialect-specific path expression for SQLite and PostgreSQL 

285 # each, unlike has_service and failing above. Still bounded by 

286 # those two when given, rather than always reading the whole table: this 

287 # filter is the rare, deliberate query, not the default estate browse. 

288 candidates = await OmHostManager.list( 

289 session, *clauses, order_by=[col(OmHost.name), col(OmHost.node_id)] 

290 ) 

291 matching = [host for host in candidates if _executor_usable(host) is executor] 

292 total = len(matching) 

293 hosts = pagination.slice(matching) 

294 

295 node_ids = [host.node_id for host in hosts] 

296 services = ( 

297 await OmServiceManager.list( 

298 session, 

299 col(OmService.node_id).in_(node_ids), 

300 order_by=[col(OmService.name)], 

301 ) 

302 if node_ids 

303 else [] 

304 ) 

305 by_node: dict[str, list[OmService]] = {} 

306 for service in services: 

307 by_node.setdefault(service.node_id, []).append(service) 

308 

309 items = [_host_response(host, by_node.get(host.node_id, [])) for host in hosts] 

310 return PaginatedResponse( 

311 items=items, total=total, offset=pagination.offset, limit=pagination.limit 

312 )

390 """Forget one host, and by cascade its services. 

391 

392 For rows PMM no longer has, which is not hypothetical: restarting a node's 

393 pmm-agent runs ``setup --force``, which *replaces* the node and mints a new id, so

408 :param session: The database session. 

409 :raises HTTPNotFoundException: When OM holds no such host.

493@router.post( 

494 "/runs", 

495 status_code=http_status.HTTP_202_ACCEPTED, 

496) 

497@require_minimum_role(UserRole.EDITOR) 

498async def trigger_probe( 

499 session: SessionDep, request: TriggerRequest | None = None 

500) -> ProbeRunAccepted: 

501 """Start a probe sweep, over the whole estate or over named hosts. 

502 

503 A scoped refresh exists because the two questions are different sizes. "What does 

504 the estate look like" is a sweep of everything and costs a Nomad job per executor 

505 host — a minute and a half in this sandbox. "I just did something to this host, 

506 is it healthy now" should not cost that, and it is the question PMM's UI will ask 

507 after every action it grows. 

508 

509 The scope is node ids, which is what PMM already holds, so its trigger passes them 

510 through untranslated — the payoff for keying these tables on PMM's own ids rather 

511 than minting OM ones. 

512 

513 Conflict is judged **per host**, not globally. A refresh of one host has no reason 

514 to be blocked by a refresh of another, and blocking it would make the scoped 

515 trigger useless exactly when the estate is busiest. Two runs collide only when 

516 they would touch the same host; a full refresh collides with everything, including 

517 another full refresh. 

518 

519 :param session: The database session. 

520 :param request: The optional scope. Absent, or an empty list, means everything. 

521 :raises HTTPServiceUnavailableException: When PMM's Operations for MongoDB 

522 switch has ``ENABLED`` off. 

523 :raises HTTPNotFoundException: When a requested node id is not in the estate. 

524 :raises HTTPConflictException: When a requested host is already being refreshed. 

525 :return: The queued sweep. 

526 """ 

527 if not om_inventory_settings.ENABLED: 

528 raise HTTPServiceUnavailableException(detail=SWITCHED_OFF_DETAIL) 

529 

530 node_ids = list(dict.fromkeys(request.node_ids)) if request else [] 

531 

532 # An id OM does not hold is answered by name rather than by running a refresh 

533 # that would quietly do nothing. PMM Extensions' inventory copy can lag PMM's, so this is a 

534 # real case rather than a typo guard. 

535 for node_id in node_ids: 

536 if await get_host(session, node_id) is None: 536 ↛ anywhereline 536 didn't jump anywhere: it always raised an exception.

537 raise HTTPNotFoundException(detail=f"Host {node_id} not found") 

538 

539 # The same check the sweep itself makes, so a caller and the schedule cannot 

540 # disagree about what counts as a conflict. 

541 blocking = await conflicting_run( 

542 session, node_ids, stale_after=om_inventory_settings.STALE_RUN_AFTER 

543 ) 

544 if blocking is not None: 

545 raise HTTPConflictException(detail=conflict_detail(blocking, node_ids)) 

546 

547 run = await ProbeRunManager.save(session, ProbeRun(scope=node_ids or None)) 

548 run_om_probe.delay(str(run.id), node_ids or None) 

549 return ProbeRunAccepted( 

550 run_id=run.id, 

551 status=str(run.status), 

552 started_at=run.started_at, 

553 scope=run.scope, 

554 )

598 session: SessionDep, 

599 actor: ApiCurrentUser, 

600) -> list[SettingResponse]: 

601 """Update this app's configuration at runtime. 

602 

603 The batch is atomic: a single bad key rejects all of it with a per-key 422 and 

604 writes nothing, so a caller never has to work out how far a partial apply got. 

605 

606 Only ``hot_field`` fields are accepted. ``CREDENTIALS_PATH`` is deliberately 

607 not one: it names a file the payload reads on every database *host* and hands 

608 to a driver as a URI, so making it settable here would widen "configure this 

609 app" into "read a chosen file across the estate". 

610 

611 An ``ENABLED`` or ``SCHEDULE`` change lands without a restart - 

612 ``periodic_task_schedules`` is a thunk re-read on registry rebuild - but beat 

613 runs as a forked side-car process, which reaches the new value through its own 

614 settings refresher rather than through this request. 

615

617 route with the credential of the pmm-managed ``--extensions-token`` flag (see 

618 ``require_minimum_role``'s service-principal bypass): it flips independently of 

619 ``SCHEDULE``, so the configured cadence survives Operations for MongoDB being 

620 turned off and back on rather than being overwritten each time. 

621

623 callbacks fired for the keys this changed. 

624 :param body: The ``{key: value, ...}`` batch. 

625 :param session: The database session. 

626 :param actor: The calling principal, whose username is recorded on every 

627 row written. Not ``ApiAdminUsername``: this endpoint admits any 

628 authenticated caller, PMM's non-admin service principal included. 

629 :return: One row per applied key, in input order. 

630 """

637 body=body, 

638 actor=actor.username, 

639 )
