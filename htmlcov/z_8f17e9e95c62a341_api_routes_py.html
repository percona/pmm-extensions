250@router.get("/") 

251async def atw_api_list(session: SessionDep) -> list[ATWCategoryListing]: 

252 """List ATW-tagged snippets grouped by category. 

253 

254 Categories with no matching snippets are omitted to keep the payload small; 

255 the ATW enum still defines the full taxonomy for validation (plugin schema). 

256 

257 :param session: The database session. 

258 :return: One listing row per category that has at least one approved snippet. 

259 """ 

260 snippets = await SnippetManager.list(session, col(Snippet.approved_at).is_not(None)) 

261 snippets_by_cell = defaultdict(list) 

262 for snippet in snippets: 

263 root = derive_category_root(snippet.meta.get(META_KEY_SERVICE_TYPE)) 

264 tags: list[str] = [] 

265 for key in (META_KEY_DIAGNOSTIC_CATEGORIES, META_KEY_ATW): 

266 raw_categories = snippet.meta.get(key, _MISSING_META) 

267 if raw_categories is _MISSING_META: 

268 continue 

269 validated = _validated_category_tags(key, raw_categories, snippet.filename) 

270 if validated is not None: 

271 tags = validated 

272 break 

273 for tag in dict.fromkeys(tags): 

274 snippets_by_cell[(root, tag)].append(snippet) 

275 

276 grouped = [] 

277 for root_label in CATEGORY_ROOT_LABELS.values(): 

278 for category in ATWCategory: 

279 cell_snippets = snippets_by_cell.get((root_label, category.name), []) 

280 if not cell_snippets: 

281 continue 

282 category_snippets = [_build_summary(snippet) for snippet in cell_snippets] 

283 grouped.append( 

284 ATWCategoryListing( 

285 category_root=root_label, 

286 parent_category=category.parent.name, 

287 parent_category_label=category.parent.value, 

288 category=category.name, 

289 category_label=category.value, 

290 snippet_count=len(category_snippets), 

291 snippets=category_snippets, 

292 ) 

293 ) 

294 

295 return grouped

326def _last_activity_at( 

327 incident: AtwIncident, last_execution_at: datetime | None 

328) -> datetime: 

329 """Resolve an incident's last-activity time from its own timestamps and its runs. 

330 

331 Computed in Python rather than SQL because SQLAlchemy ships no portable 

332 construct for a row-wise maximum: PostgreSQL spells it ``GREATEST`` and SQLite 

333 spells it as a multi-argument ``max()``. Every candidate is normalized first — 

334 mixing a naive value in raises ``TypeError`` on comparison, and whether the 

335 aggregate's timestamp arrives timezone-aware is dialect-dependent. 

336 

337 ``created_at`` is always set, so the result is never ``None`` — an incident 

338 with no runs reports its own last change and the client needs no empty state. 

339 

340 Delivery attempts are deliberately not a source: ``atw_send_log`` rows are real 

341 activity on an incident, but folding a third table into the aggregate is out of 

342 scope, so a send does not move this timestamp. 

343 

344 :param incident: The incident whose own timestamps take part. 

345 :param last_execution_at: The aggregated execution timestamp, when it has runs. 

346 :return: The latest of the incident's creation, update, and execution times. 

347 """ 

348 candidates = (incident.created_at, incident.updated_at, last_execution_at) 

349 return max( 

350 make_datetime_utc(candidate) 

351 for candidate in candidates 

352 if candidate is not None 

353 )

356def _build_incident_responses( 

357 incidents: Iterable[AtwIncident], 

358 aggregates: Mapping[UUID4, IncidentRunAggregate], 

359) -> list[AtwIncidentResponse]: 

360 """Render incidents with their run totals attached. 

361 

362 Shared by all six routes that return :class:`AtwIncidentResponse`, so none of 

363 them serves a defaulted ``0`` for an incident that actually has runs. An 

364 incident absent from ``aggregates`` has no executions and is zeroed here. 

365 

366 :param incidents: The incidents to render, in the order to return them. 

367 :param aggregates: Run totals keyed by incident id, as produced by 

368 :meth:`AtwIncidentExecutionManager.aggregate_by_incident`. 

369 :return: One response per incident, in the given order. 

370 """ 

371 responses: list[AtwIncidentResponse] = [] 

372 for incident in incidents: 

373 aggregate = aggregates.get(incident.id) 

374 responses.append( 

375 AtwIncidentResponse.model_validate(incident).model_copy( 

376 update={ 

377 "run_count": 0 if aggregate is None else aggregate.run_count, 

378 "failed_run_count": ( 

379 0 if aggregate is None else aggregate.failed_run_count 

380 ), 

381 "last_activity_at": _last_activity_at( 

382 incident, 

383 None if aggregate is None else aggregate.last_execution_at, 

384 ), 

385 } 

386 ) 

387 ) 

388 return responses

518@router.get("/execution-schema/") 

519async def atw_execution_schema( 

520 snippet_filename: Annotated[ 

521 list[str], 

522 Query( 

523 min_length=1, 

524 max_length=MAX_BATCH_SNIPPETS, 

525 description="Snippet filenames to build a batch form for.", 

526 ), 

527 ], 

528) -> ATWMergedSchemaResponse: 

529 """Build one execution form covering several snippets, merging common parameters. 

530 

531 Unknown or unsafe filenames fail the whole request — a form the caller cannot 

532 fill for every selected snippet is not a partial success. 

533 

534 :param snippet_filename: The selected snippet filenames, repeated per snippet 

535 and deduplicated order-preserving. 

536 :return: The shared section followed by each snippet's remaining fields. 

537 :raises HTTPBadRequestException: When a filename is unsafe or malformed, failing 

538 the whole request before any item is dispatched. 

539 :raises HTTPNotFoundException: When a filename matches no snippet. 

540 """ 

541 unique = list(unique_everseen(snippet_filename)) 

542 resolved = await resolve_snippets(unique) 

543 missing = next((filename for filename in unique if filename not in resolved), None) 

544 if missing is not None: 

545 raise HTTPNotFoundException(detail=snippet_not_found_detail(missing)) 

546 scripts = [resolved[filename] for filename in unique] 

547 fields_per_script = [parameter_fields(script) for script in scripts] 

548 

549 declarations = defaultdict(list) 

550 for fields in fields_per_script: 

551 for field in fields: 

552 declarations[field.name].append(field) 

553 shared_names = shared_field_names(declarations) 

554 

555 shared = batch_execution_fields(scripts) 

556 shared += [ 

557 fields[0] for name, fields in declarations.items() if name in shared_names 

558 ] 

559 return ATWMergedSchemaResponse( 

560 shared=shared, 

561 per_snippet=[ 

562 ATWSnippetSchema( 

563 snippet_filename=script.filename, 

564 fields=[field for field in fields if field.name not in shared_names], 

565 ) 

566 for script, fields in zip(scripts, fields_per_script, strict=True) 

567 ], 

568 )

583@router.post( 

584 "/incidents/{incident_id}/executions/", status_code=status.HTTP_201_CREATED 

585) 

586async def atw_batch_execute( 

587 session: SessionDep, 

588 incident: OpenAtwIncidentDep, 

589 body: ATWBatchExecuteWrite, 

590 tasks_api: TaskAPI, 

591) -> ATWBatchExecuteResponse: 

592 """Execute several snippets against one incident, reporting each item separately. 

593 

594 A malformed selection fails the whole request up front: ``resolve_snippets`` 

595 runs the traversal guard over every filename before dispatch, so an unsafe 

596 filename raises before any item runs. Past that guard, one failing item never 

597 blocks the rest: each is dispatched and recorded inside its own guard, and the 

598 response always carries an entry per requested item — an unresolved filename 

599 becomes that item's error. A failed attempt produces no task-history row to 

600 reference, so it is reported here rather than persisted. 

601 

602 The row-write guard rolls the shared session back before the loop continues, 

603 or one failed write would leave the transaction aborted and doom every later 

604 item on PostgreSQL; the dispatch guard rolls back defensively, having written 

605 nothing itself. ``incident.id`` is read once up front because both a commit and 

606 a rollback expire the instance, and re-reading it would trigger a lazy load. 

607 

608 ATW's proxy is resolved once per distinct interpreter before the loop, so a batch 

609 of twenty items costs the same upstream traffic as one item. That resolution has 

610 its own guard: it cannot fail for one item and not another, so a failure degrades 

611 the whole batch to unwrapped dispatch instead of failing a request whose 

612 dispatches may still succeed. 

613 

614 :param session: The database session. 

615 :param incident: The incident resolved from the ``incident_id`` path parameter. 

616 :param body: The batch payload. 

617 :param tasks_api: The authenticated Tasks API client. 

618 :return: One outcome entry per requested item, in request order. 

619 :raises HTTPBadRequestException: When any filename is unsafe or malformed, 

620 failing the whole request before any item is dispatched. 

621 """ 

622 incident_id = incident.id 

623 resolved = await resolve_snippets([item.snippet_filename for item in body.items]) 

624 try:

626 script.execution_task_name for script in resolved.values() 

627 ) 

628 except (HTTPException, OSError, RuntimeError):

630 # way — so it degrades the whole batch to unwrapped dispatch rather than 

631 # failing a request whose dispatches may well succeed. Outcomes then come 

632 # from the reconciliation sweep instead of the recorder. 

633 logger.warning( 

634 "Could not resolve ATW's proxy tasks; dispatching this batch unwrapped.", 

635 exc_info=True, 

636 ) 

637 proxies = {} 

638 items = [] 

639 for item in body.items: 

640 script = resolved.get(item.snippet_filename) 

641 if script is None: 

642 items.append( 

643 ATWBatchExecuteItemResponse( 

644 snippet_filename=item.snippet_filename, 

645 error=snippet_not_found_detail(item.snippet_filename), 

646 ) 

647 ) 

648 continue 

649 try: 

650 dispatched = await dispatch_batch_item( 

651 body, 

652 item, 

653 script, 

654 tasks_api, 

655 execution_task_name=proxies.get(script.execution_task_name), 

656 ) 

657 except (HTTPException, OSError) as exc: 

658 await session.rollback() 

659 items.append( 

660 ATWBatchExecuteItemResponse( 

661 snippet_filename=item.snippet_filename, 

662 error=_failure_detail(exc), 

663 ) 

664 ) 

665 continue 

666 if dispatched.task_id is None: 

667 items.append( 

668 ATWBatchExecuteItemResponse( 

669 snippet_filename=item.snippet_filename, 

670 task_name=dispatched.task_name, 

671 error=NO_TASK_ID_ERROR, 

672 ) 

673 ) 

674 continue 

675 try: 

676 await AtwIncidentExecutionManager.save( 

677 session, 

678 AtwIncidentExecution( 

679 incident_id=incident_id, 

680 task_history_id=dispatched.task_id, 

681 snippet_filename=dispatched.snippet_filename, 

682 ), 

683 ) 

684 except (HTTPException, OSError) as exc: 

685 await session.rollback() 

686 items.append( 

687 ATWBatchExecuteItemResponse( 

688 snippet_filename=item.snippet_filename, 

689 task_name=dispatched.task_name, 

690 task_history_id=dispatched.task_id, 

691 error=f"{UNRECORDED_EXECUTION_ERROR}: {_failure_detail(exc)}", 

692 ) 

693 ) 

694 continue 

695 items.append( 

696 ATWBatchExecuteItemResponse( 

697 snippet_filename=item.snippet_filename, 

698 task_name=dispatched.task_name, 

699 task_history_id=dispatched.task_id, 

700 ) 

701 ) 

702 return ATWBatchExecuteResponse(items=items)

745def _args_unreadable_upstream(history: Mapping[str, Any]) -> bool: 

746 """Return whether the tasks service could not read an execution's arguments. 

747 

748 Read off the documented ``unreadable_request_leaves`` field rather than 

749 inferred from the value: the service serialises a leaf it could not decrypt

755 :return: Whether the recorded arguments are unreadable upstream. 

756 """

758 return isinstance(unreadable, list) and ARGS_LEAF in unreadable

761def _execution_args( 

762 history: dict[str, Any], script: SnippetScript | None 

763) -> tuple[str | None, bool]: 

764 """Return an execution's masked argument string and whether it was withheld. 

765 

766 An empty ``history`` is what an upstream lookup failure degrades to, so it 

767 withholds rather than reporting the empty state -- the arguments exist, they 

768 just could not be read. A payload whose nesting is not shaped as expected 

769 withholds for the same reason (see :func:`_execution_meta`). 

770 

771 Masking derives what is sensitive from the snippet's *current* frontmatter, 

772 while the arguments were rendered from whatever it said at execution time. The 

773 two only coincide while the file is unchanged, so the digest recorded alongside 

774 the arguments is compared against the snippet's current one first: an edit that 

775 dropped, renamed, or de-sensitised a parameter would otherwise leave its value 

776 unrecognised and rendered in the clear. A digest mismatch withholds, which also 

777 means a snippet edit blanks the arguments of executions recorded before it -- 

778 the price of not persisting the metadata each run was masked against. 

779 

780 Deriving that metadata can also fail outright on a malformed or stale ``meta`` 

781 column -- an ``arg_format`` that does not tokenise, a parameter definition the 

782 model rejects, a value that will not serialise. Any of those withholds this row 

783 alone, never the page. ``ValidationError`` and ``PydanticUserError`` need no 

784 separate arm: they arrive as the ``ValueError`` and ``TypeError`` they 

785 respectively subclass. 

786 

787 A row the tasks service stored encrypted and could not read back also 

788 withholds. It is recognised from the documented ``unreadable_request_leaves`` 

789 field, never by inspecting the value: the service already serialises such a 

790 leaf as ``null``, which is otherwise indistinguishable from an execution that 

791 recorded no arguments. 

792 

793 :param history: The upstream task-history payload, empty when unavailable. 

794 :param script: The resolved snippet, or ``None`` when its filename no longer 

795 resolves and the parameter metadata masking needs is unavailable. 

796 :return: The masked argument string paired with the withheld flag; a ``None`` 

797 string and a false flag mean the execution recorded no arguments. 

798 """ 

799 if not history or _args_unreadable_upstream(history): 

800 return None, True 

801 if (meta := _execution_meta(history)) is None: 

802 return None, True 

803 if not (raw := meta.get("args")): 

804 return None, False 

805 if script is None or meta.get("md5_checksum") != script.snippet.md5_digest: 

806 return None, True 

807 try: 

808 masked = mask_snippet_args(raw, script.get_execution_model()) 

809 except (TypeError, ValueError, KeyError, AttributeError): 

810 logger.warning(ATW_ARG_MASKING_WARNING, script.filename, exc_info=True) 

811 return None, True 

812 return masked, masked is None
