29class TopologyCollectWrite(BaseModel): 

30 """Describe the request body for ``POST /collect``. 

31 

32 :param shards: Number of executor hosts to dispatch in parallel. Hosts 

33 are split round-robin across the chosen executors. Capped at 

34 :data:`MAX_TOPOLOGY_SHARDS`. 

35 :param executor_host: Optional explicit executor. Must be used with 

36 ``shards=1`` because it selects a single-shard run. 

37 :param connect_timeout: Per-host MySQL TCP connect timeout (seconds). 

38 :param read_timeout: Per-host MySQL read/write timeout (seconds). 

39 """ 

40 

41 shards: int = Field(default=1, ge=1, le=MAX_TOPOLOGY_SHARDS) 

42 executor_host: str | None = None 

43 connect_timeout: int = Field(default=5, ge=1, le=60) 

44 read_timeout: int = Field(default=10, ge=1, le=120) 

45 

46 @model_validator(mode="after") 

47 def _reject_executor_with_multiple_shards(self) -> Self: 

48 if self.executor_host and self.shards != 1: 

49 raise ValueError("executor_host requires shards=1") 

50 return self

53class TopologyCollectResponse(BaseModel): 

54 """Represent the response body for ``POST /collect``. 

55 

56 ``task_history_ids`` lists the dispatched ``run-python`` tasks the 

57 frontend then polls (``/result``) to assemble the topology graph. 

58 ``targets`` echoes the executor hosts the work was sharded across so 

59 the UI can surface where the collection ran. 

60 

61 :param task_history_ids: Created task history ids, one per shard. 

62 :param targets: Executor hosts selected for topology collection. 

63 :param host_count: Number of MySQL hosts included in the collection. 

64 :param shard_count: Number of dispatched topology shards. 

65 """ 

66 

67 task_history_ids: list[int] 

68 targets: list[str] 

69 host_count: int 

70 shard_count: int

228class TopologyResultResponse(BaseModel): 

229 """Represent the response body for ``GET /result``. 

230 

231 ``status`` is ``running`` while any of the underlying tasks is 

232 still pending, ``ok`` once every task has finished, and ``failed`` 

233 when at least one task failed and produced no usable output. 

234 ``graph`` is the merged React-Flow graph; ``pending_task_ids`` 

235 lists the still-running tasks for the UI's progress chip, and 

236 ``failed_task_ids`` lets the UI warn when only some shards failed. 

237 

238 :param status: Aggregate topology collection status. 

239 :param graph: Merged React-Flow graph when collection output is ready. 

240 :param pending_task_ids: Task ids still pending or running. 

241 :param failed_task_ids: Terminal task ids that did not finish successfully. 

242 """ 

243 

244 status: Literal["running", "ok", "failed"] 

245 graph: TopologyGraph | None = None 

246 pending_task_ids: list[int] = Field(default_factory=list) 

247 failed_task_ids: list[int] = Field(default_factory=list)
