34class AuthProviderEnum(Enum): 

35 """Define an enumeration for authentication providers. 

36 

37 Built-in members map a provider name to its class. ``CUSTOM`` is a sentinel 

38 that selects an out-of-tree provider by importable class path 

39 (``PROVIDER_CLASS``) instead of a built-in class. 

40 """ 

41 

42 CASDOOR = CasdoorAuthProvider 

43 GRAFANA = GrafanaAuthProvider 

44 CUSTOM = "custom"

47class _LegacyCasdoorSettings(BaseYamlSettings):

72class AuthSettings(BaseYamlSettings): 

73 """Define authentication configuration. 

74 

75 Auth has exactly one active provider, so ``PROVIDER`` is a single-entry map 

76 ``{<name>: <config>}`` keyed by the provider name (canonical env 

77 ``AUTH__PROVIDER__<NAME>__*``). The name resolves to a provider class via 

78 :class:`AuthProviderEnum` -- a built-in member, or ``CUSTOM`` carrying an 

79 importable ``PROVIDER_CLASS``. 

80 

81 :cvar SETTINGS_PREFIXES: The settings prefixes for auth configuration. 

82 :param PROVIDER: The single active provider, keyed by name. 

83 """ 

84 

85 model_config = SettingsConfigDict( 

86 **{**BaseYamlSettings.model_config, "arbitrary_types_allowed": True} 

87 ) 

88 SETTINGS_PREFIXES: ClassVar[list[str]] = ["AUTH"] 

89 PROVIDER: dict[str, BaseAuthProvider] = {}

92 @classmethod 

93 def _resolve_provider(cls, data: Any) -> Any: 

94 """Fold in the legacy ``CASDOOR`` config and resolve provider classes. 

95 

96 Run a fixed sequence -- no reliance on cross-validator ordering: 

97 

98 1. When the deprecated top-level ``CASDOOR`` config is present, log a 

99 deprecation warning and reconcile it with ``PROVIDER``: merge it under 

100 the ``casdoor`` entry (legacy values fill gaps; explicit 

101 ``AUTH.PROVIDER`` values win on conflict), adding the entry when no 

102 provider is configured. If a *non-casdoor* provider is configured 

103 instead, ignore the legacy config. 

104 2. Drop any entry whose value is ``None`` -- an overlay can set 

105 ``<name>: null`` to remove a provider the ``default`` settings block 

106 merged in. The drop runs after the legacy fold above, so an explicit 

107 ``null`` wins over a legacy ``CASDOOR`` resurrection. 

108 3. Resolve each remaining entry's class via :class:`AuthProviderEnum` (a 

109 built-in member, or ``CUSTOM`` via ``import_var(PROVIDER_CLASS)``) and 

110 construct the provider. 

111 

112 :param data: The raw settings input. 

113 :return: The input with ``PROVIDER`` mapped to constructed providers. 

114 :raises ValueError: Propagates from ``_resolve_entry`` for an 

115 unresolvable provider entry. 

116 """

187 @model_validator(mode="after") 

188 def _exactly_one_provider(self) -> Self: 

189 """Require exactly one configured provider. 

190 

191 :return: The validated settings instance. 

192 :raises ValueError: When zero or more than one provider is configured. 

193 """ 

194 if len(self.PROVIDER) != 1: 

195 raise ValueError( 

196 "Exactly one auth provider must be configured; got " 

197 f"{len(self.PROVIDER)} ({sorted(self.PROVIDER)})."

233def detect_removed_auth_user_model() -> None: 

234 """Reject non-default ``AUTH_USER_MODEL`` overrides and warn on the removal. 

235 

236 ``Settings`` uses ``extra="ignore"``, so a stale ``AUTH_USER_MODEL`` in the 

237 environment or YAML is silently dropped. Read it explicitly: emit a 

238 deprecation warning when the (default) value is present, and fail fast when 

239 it names a non-default class -- a deliberate override that would otherwise be 

240 ignored. 

241 

242 :raises ValueError: When ``AUTH_USER_MODEL`` names a non-default class. 

243 """ 

244 legacy = _LegacyAuthUserModelSettings().AUTH_USER_MODEL 

245 if not legacy: 

246 return 

247 if legacy != _REMOVED_AUTH_USER_MODEL_DEFAULT: 

248 raise ValueError( 

249 f"AUTH_USER_MODEL is removed but is set to a non-default class " 

250 f"{legacy!r}. Select a provider via AUTH__PROVIDER__* instead " 

251 f"(e.g. AUTH__PROVIDER__CUSTOM__PROVIDER_CLASS)." 

252 ) 

253 logger.warning( 

254 "AUTH_USER_MODEL is removed and its value is ignored; select a provider " 

255 "via AUTH__PROVIDER__* (AUTH.PROVIDER)."
