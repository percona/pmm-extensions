93def parse_backup_namespaces(namespaces: str) -> list[str]: 

94 """Parse a comma-separated selective namespace list into tokens. 

95 

96 PBM splits each token at the first ``.``, so collection names may contain 

97 additional dots (for example ``db.orders.archive``). Empty comma-separated 

98 entries and database wildcards such as ``*.users`` are rejected. 

99 

100 :param namespaces: Raw ``db.collection`` / ``db.*`` list from the form. 

101 :return: Non-empty stripped namespace tokens, preserving order. 

102 :raises ValueError: When any token is empty after split or fails the 

103 ``db.collection`` / ``db.*`` shape. 

104 """ 

105 tokens = [part.strip() for part in namespaces.split(",")] 

106 if any(not token for token in tokens): 

107 raise ValueError("Backup namespaces must not contain empty entries") 

108 invalid = [] 

109 for token in tokens: 

110 if "." not in token: 

111 invalid.append(token) 

112 continue 

113 database, collection = token.split(".", 1) 

114 if not _NAMESPACE_DB_RE.fullmatch(database) or not collection: 

115 invalid.append(token) 

116 if invalid: 

117 raise ValueError( 

118 "Backup namespaces must be comma-separated db.collection or db.* " 

119 f"entries; invalid: {', '.join(invalid)}" 

120 ) 

121 return tokens

124def validate_selective_users_and_roles( 

125 namespaces: str | None, 

126 *, 

127 with_users_and_roles: bool, 

128) -> None: 

129 """Reject ``--with-users-and-roles`` unless every namespace is database-level. 

130 

131 :param namespaces: Comma-separated selective namespaces, or ``None`` when unset. 

132 :param with_users_and_roles: Whether the opt-in users/roles flag is enabled. 

133 :raises ValueError: When the flag is set without namespaces, or any namespace 

134 is a single collection rather than ``db.*``. 

135 """ 

136 if not with_users_and_roles: 

137 return 

138 if not namespaces or not str(namespaces).strip(): 

139 raise ValueError( 

140 "Include users and roles requires backup namespaces set to database-level " 

141 "db.* entries" 

142 ) 

143 tokens = parse_backup_namespaces(str(namespaces)) 

144 non_db_level = [ 

145 token for token in tokens if not _DB_LEVEL_NAMESPACE_RE.fullmatch(token) 

146 ] 

147 if non_db_level: 

148 raise ValueError( 

149 "Include users and roles is valid only with database-level db.* " 

150 f"namespaces; invalid: {', '.join(non_db_level)}" 

151 )

203def parse_backup_priority(priority_str: str) -> dict[str, float]: 

204 """Parse the Node Priority YAML into a node -> priority mapping. 

205 

206 Shared by the request-model validator (create-time) and the spec builder so the 

207 parse rules cannot drift and a present priority is never silently dropped. 

208 

209 :param priority_str: Raw YAML from the Node Priority field. 

210 :return: Mapping of node address to numeric priority. 

211 :raises ValueError: On invalid YAML, a non-mapping result, an empty mapping, 

212 or a non-numeric priority value. 

213 """ 

214 try: 

215 parsed = yaml.safe_load(priority_str) 

216 except yaml.YAMLError as exc: 

217 raise ValueError(f"Node priority is not valid YAML: {exc}") from exc 

218 if not isinstance(parsed, dict): 

219 # Raise ValueError, not TypeError, so Pydantic surfaces a 422 instead of a 500. 

220 raise ValueError( # noqa: TRY004 

221 "Node priority must be a YAML mapping of node address to priority number" 

222 ) 

223 if not parsed: 

224 # A present-but-empty mapping would be dropped as falsy in the spec builder; 

225 # reject it so a present field always takes effect. 

226 raise ValueError("Node priority mapping is empty; provide at least one node") 

227 result = {} 

228 for node, value in parsed.items(): 

229 # Reject booleans explicitly — they would otherwise coerce to 1.0 / 0.0. 

230 if isinstance(value, bool) or not isinstance(value, int | float): 

231 raise ValueError( # noqa: TRY004 

232 f"Priority for {node!r} must be a number, got {value!r}" 

233 ) 

234 result[str(node)] = float(value) 

235 return result

252# Storage backends PMM Extensions builds a PBM config for; the ``azure`` backend has no 

253# builder support and is not offered in the form, so it is rejected at create time. 

254_SUPPORTED_STORAGE_TYPES = frozenset( 

255 {StorageType.S3.value, StorageType.FILESYSTEM.value} 

256) 

257 

258# DNS-compliant S3 bucket names: 3-63 chars, dot-separated labels of lowercase 

259# letters, digits and hyphens, each label starting and ending alphanumeric -- so no 

260# consecutive dots and no dot-adjacent hyphens (``a..b``, ``a.-b``). IP-address-formatted 

261# names are rejected separately below. A well-formed but non-existent bucket is caught 

262# later, when the config is applied. 

263_S3_BUCKET_RE = re.compile( 

264 r"^(?=.{3,63}$)[a-z0-9](?:[a-z0-9-]*[a-z0-9])?" 

265 r"(?:\.[a-z0-9](?:[a-z0-9-]*[a-z0-9])?)*$" 

266) 

267# S3 bucket names must not be formatted as an IPv4 address (e.g. ``192.168.5.4``). 

268_S3_BUCKET_IP_RE = re.compile(r"^\d+(?:\.\d+){3}$") 

269# AWS region format (e.g. ``us-east-1``, ``us-gov-east-1``). Enforced only for native 

270# AWS (no custom endpoint); with a custom endpoint the region is provider-defined. 

271_AWS_REGION_RE = re.compile(r"^[a-z]{2}-[a-z-]+-\d+$")

279def _validate_s3_bucket_name(value: str) -> str: 

280 """Validate a DNS-compliant, non-IP-formatted S3 bucket name. 

281 

282 A value-only check carried on the field annotation so a malformed bucket is 

283 reported against its own input (a non-empty ``loc``, which the flash-message 

284 helper needs to name the field) rather than the whole model. 

285 

286 :param value: The (already-stripped) S3 bucket name.

294 "(3-63 chars: lowercase letters, digits, dots, hyphens; " 

295 "no consecutive dots and not an IP address)" 

296 )

304def _validate_s3_storage( 

305 *, 

306 bucket: str | None, 

307 region: str | None, 

308 endpoint_url: str | None, 

309 filesystem_path: str | None, 

310) -> None: 

311 """Validate the cross-field rules of the ``s3`` branch of :func:`validate_storage_config`. 

312 

313 Value-only checks (DNS-compliant bucket, ``http(s)`` endpoint URL) live on the 

314 field annotations; this helper covers only the rules that need cross-field 

315 context: the other backend's fields being unset, the required fields being 

316 present, and the AWS-region format being enforced only when no custom endpoint 

317 marks the storage as S3-compatible (non-AWS). 

318 

319 :param bucket: The S3 bucket; required. 

320 :param region: The S3 region; required, and a valid AWS region unless a custom 

321 endpoint marks the storage as S3-compatible (non-AWS). 

322 :param endpoint_url: The optional S3 endpoint URL; its presence relaxes the 

323 AWS-region format check. 

324 :param filesystem_path: The filesystem path, which must not be set for S3. 

325 :raises ValueError: When a required field is missing or the region is malformed. 

326 """ 

327 if not _is_blank(filesystem_path): 

328 raise ValueError("Filesystem path must not be set for S3 storage") 

329 if _is_blank(bucket): 

330 raise ValueError("S3 storage requires a non-empty bucket") 

331 if _is_blank(region): 

332 raise ValueError("S3 storage requires a non-empty region") 

333 if _is_blank(endpoint_url) and not _AWS_REGION_RE.match(region): 

334 raise ValueError( 

335 f"S3 region {region!r} is not a valid AWS region (e.g. us-east-1); " 

336 "set an endpoint URL for S3-compatible (non-AWS) storage" 

337 )

341 *, 

342 path: str | None, 

343 s3_bucket: str | None, 

344 s3_region: str | None, 

345 s3_prefix: str | None, 

346 s3_endpoint_url: str | None,

348 """Validate the ``filesystem`` branch of :func:`validate_storage_config`. 

349 

350 :param path: The filesystem path; required and non-blank. 

351 :param s3_bucket: The S3 bucket, which must not be set for filesystem storage. 

352 :param s3_region: The S3 region, which must not be set for filesystem storage.

364 )

366 ]

372 raise ValueError("Filesystem storage requires a non-empty path")

375def validate_storage_config( 

376 storage_type: str | None, 

377 *, 

378 s3_bucket: str | None, 

379 s3_region: str | None, 

380 s3_prefix: str | None, 

381 s3_endpoint_url: str | None, 

382 filesystem_path: str | None, 

383) -> None: 

384 """Validate a per-task PBM storage configuration at create time. 

385 

386 Ensure the selected backend is one PMM Extensions supports, that its required fields are 

387 present, and that fields belonging to the *other* backend are not set — so an 

388 incomplete, malformed, or cross-wired storage config is rejected at create time 

389 (422 on the JSON API, a flash-redirect on the form path) rather than silently 

390 accepted (and then partly ignored by the spec builder) and never applied. The 

391 value-only field checks (DNS-compliant bucket, ``http(s)`` endpoint URL) run on 

392 the field annotations; this helper covers the cross-field rules only. The checks 

393 are structural; bucket reachability is surfaced later, when the config is applied 

394 to PBM. 

395 

396 :param storage_type: The selected storage backend (``s3`` or ``filesystem``). 

397 :param s3_bucket: The S3 bucket; required (non-blank) when ``storage_type`` is 

398 ``s3``, forbidden otherwise. 

399 :param s3_region: The S3 region; required when ``storage_type`` is ``s3`` (and 

400 a valid AWS region unless a custom endpoint is set), forbidden otherwise. 

401 :param s3_prefix: The S3 key prefix; forbidden unless ``storage_type`` is ``s3``. 

402 :param s3_endpoint_url: The S3 endpoint URL; its presence relaxes the AWS-region 

403 format check, forbidden unless ``storage_type`` is ``s3``. 

404 :param filesystem_path: The filesystem path; required when ``storage_type`` is 

405 ``filesystem``, forbidden otherwise. 

406 :raises ValueError: When the storage type is unsupported, a required field is 

407 missing, the region is malformed, or a field of the other backend is set. 

408 """ 

409 if storage_type not in _SUPPORTED_STORAGE_TYPES: 

410 raise ValueError( 

411 "storage_type must be one of " 

412 f"{sorted(_SUPPORTED_STORAGE_TYPES)}, got {storage_type!r}" 

413 ) 

414 if storage_type == StorageType.S3.value: 

415 _validate_s3_storage( 

416 bucket=s3_bucket, 

417 region=s3_region, 

418 endpoint_url=s3_endpoint_url, 

419 filesystem_path=filesystem_path, 

420 ) 

421 else: 

422 _validate_filesystem_storage( 

423 path=filesystem_path, 

424 s3_bucket=s3_bucket, 

425 s3_region=s3_region, 

426 s3_prefix=s3_prefix, 

427 s3_endpoint_url=s3_endpoint_url, 

428 )
