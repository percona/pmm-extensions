79def _is_grafana_verdict(exc: HTTPException, status_code: int) -> bool: 

80 """Return whether Grafana itself answered ``exc`` with ``status_code``. 

81 

82 A non-JSON error body keeps its status but is stamped with 

83 ``UPSTREAM_NON_JSON_HEADER``: it came from something in front of Grafana, 

84 such as a proxy's HTML page, and says nothing about the credential. 

85 

86 :param exc: The error a Grafana call raised. 

87 :param status_code: The status that would be Grafana's verdict. 

88 :return: ``True`` when the status matches and the body was Grafana's JSON. 

89 """ 

90 return exc.status_code == status_code and not (exc.headers or {}).get( 

91 UPSTREAM_NON_JSON_HEADER

112class GrafanaSDK(RemoteAPI): 

113 """Interact with Grafana's authentication and user-management APIs. 

114 

115 The ``GrafanaSDK`` class extends ``RemoteAPI`` to authenticate human logins 

116 against Grafana's password-login endpoint and to read user and org-user data 

117 with a service-account bearer token. It sets no default ``Authorization`` 

118 header: the human-login flow authenticates via Grafana's session cookie, 

119 while the programmatic calls inject the service-account bearer per request. 

120 

121 :param endpoint: The base URL for the Grafana API endpoint. 

122 :param verify_ssl: Whether to verify SSL certificates. Defaults to True. 

123 :param service_account_token: The Grafana service-account token used as a 

124 bearer credential for programmatic user reads. 

125 :param access_token_max_age: How long a minted access assertion stays valid 

126 (the per-request Bearer credential). Defaults to 1 hour. 

127 :param refresh_token_max_age: How long a minted refresh assertion stays valid 

128 (the SPA's ``HttpOnly`` refresh cookie). Defaults to 7 days. 

129 :param exchange_token_max_age: How long a minted session-exchange assertion 

130 stays valid (the embedded UI's in-memory bearer). Defaults to 5 minutes. 

131 This value alone bounds how long a signed-out browser keeps embedded 

132 access, and how long a Grafana role change takes to take effect, so it 

133 deliberately does not fall back to ``access_token_max_age``. A 

134 non-positive value expires every assertion at mint time and is rejected 

135 at config load rather than silently disabling embedded-UI auth. 

136 :param service_account_bearer_revocation_window: How long a validated or 

137 refused service-account token verdict is reused, i.e. how long a 

138 revoked, disabled or re-roled service account keeps its previous 

139 verdict on new requests. ``0`` re-verifies on every request. Defaults to 

140 5 minutes. 

141 :param error_detail_key: The key Grafana uses for error details. Defaults to 

142 "message". 

143 :param session_cookie_name: The name of the cookie Grafana sets on a 

144 successful password login. Defaults to ``grafana_session``. 

145 """ 

146 

147 model_config = ConfigDict(ignored_types=(_LRUCacheWrapper,)) 

148 logger_name: str = __name__ 

149 service_account_token: SecretStr 

150 access_token_max_age: TimedeltaSeconds = timedelta(hours=1) 

151 refresh_token_max_age: TimedeltaSeconds = timedelta(days=7) 

152 exchange_token_max_age: Annotated[TimedeltaSeconds, Gt(timedelta(0))] = timedelta( 

153 minutes=5 

154 ) 

155 service_account_bearer_revocation_window: Annotated[ 

156 TimedeltaSeconds, Ge(timedelta(0)) 

157 ] = timedelta(minutes=5) 

158 error_detail_key: NonEmptyStr = "message" 

159 session_cookie_name: NonEmptyStr = "grafana_session"

161 def _cookie_jar(self) -> AbstractCookieJar: 

162 """Return a jar that keeps no cookies. 

163 

164 Grafana authenticates a request by its session cookie when the bearer 

165 it carries fails, so a ``grafana_session`` a login stored would answer 

166 for every later call's credential: a refused service-account token 

167 would come back as the human who logged in. :meth:`login` reads the 

168 cookie off its own response, and the session-bound reads pass theirs 

169 explicitly, so nothing needs a stored one. 

170 

171 :return: A jar that stores nothing. 

172 """ 

173 return DummyCookieJar()

197 async def login(self, username: str, password: str) -> str: 

198 """Authenticate a user against Grafana and return the session cookie. 

199 

200 Post the credentials to Grafana's JSON login endpoint and return the 

201 established session cookie value. The cookie is used only to read the 

202 logging-in user's identity and is never persisted. 

203 

204 :param username: The Grafana username. 

205 :param password: The Grafana password. 

206 :return: The value of the established ``grafana_session`` cookie. 

207 :raises HTTPUnauthorizedException: If Grafana rejects the credentials 

208 (HTTP 401). 

209 :raises GrafanaException: If Grafana returns an unexpected status, 

210 establishes no session, or cannot be reached. 

211 """ 

212 try: 

213 async with self._request( 

214 "POST", 

215 "/login", 

216 json={"user": username, "password": password}, 

217 ) as response: 

218 if response.status == status.HTTP_401_UNAUTHORIZED: 

219 raise HTTPUnauthorizedException("Invalid username or password") 

220 if response.status != status.HTTP_200_OK: 

221 raise GrafanaException( 

222 detail=f"Grafana login failed (HTTP {response.status})." 

223 ) 

224 session_cookie = response.cookies.get(self.session_cookie_name) 

225 if session_cookie is None: 

226 raise GrafanaException( 

227 detail="Grafana did not establish a session." 

228 ) 

229 return session_cookie.value 

230 except ClientConnectionError: 

231 self.logger.exception("Failed to connect to Grafana.") 

232 raise GrafanaException( 

233 detail=f"Cannot connect to Grafana at {self.endpoint}" 

234 ) from None

274 @alru_cache(ttl=300) 

275 async def get_service_accounts(self) -> list[dict[str, Any]]: 

276 """Read every service account in PMM Extensions' org via the service-account token. 

277 

278 Pages are read until the collected rows reach the listing's 

279 ``totalCount`` or a page comes back empty. Only the pagination fields 

280 are checked here; the rows are validated by their reader. 

281 

282 :return: The ``serviceAccounts`` rows of every page. 

283 :raises HTTPException: Whatever a page request raised, including the 502 

284 for a page that is not a JSON object. 

285 :raises TimeoutError: If a page request times out. 

286 :raises aiohttp.ClientError: If a page request fails in transport. 

287 :raises ValueError: If a page body cannot be decoded. 

288 :raises GrafanaException: If Grafana cannot be reached, or a page carries 

289 no integer ``totalCount`` or no ``serviceAccounts`` list. 

290 """ 

291 accounts: list[dict[str, Any]] = [] 

292 page = 1 

293 with self.auth(self.service_account_token.get_secret_value()): 

294 while True: 

295 payload = as_json_object( 

296 await self.get( 

297 "/api/serviceaccounts/search", 

298 params={"perpage": _SERVICE_ACCOUNTS_PAGE_SIZE, "page": page}, 

299 ) 

300 ) 

301 rows = payload.get("serviceAccounts") 

302 total = payload.get("totalCount") 

303 if not isinstance(rows, list) or not isinstance(total, int): 

304 raise GrafanaException( 

305 detail="Grafana returned an unreadable service-account list." 

306 ) 

307 accounts.extend(rows) 

308 if not rows or len(accounts) >= total: 

309 return accounts 

310 page += 1

327 async def verify_service_account_token( 

328 self, token: str 

329 ) -> GrafanaServiceAccountRecord | None: 

330 """Verify a Grafana service-account token presented to PMM Extensions. 

331 

332 A verdict (the account record, or a refusal Grafana itself answered) 

333 is reused for ``service_account_bearer_revocation_window``, keyed on the 

334 token's SHA-256 digest so the cache never holds the secret. The window 

335 is counted from when the check started rather than from when Grafana 

336 answered, since the record is read after that moment, so a slow answer 

337 cannot stretch a verdict past the window. An upstream failure is never 

338 reused. 

339 

340 :param token: The ``glsa_`` token the caller presented. 

341 :return: The ``/api/serviceaccounts/{id}`` record, or ``None`` when 

342 Grafana rejected the token or the account is not in PMM Extensions' org. 

343 :raises GrafanaException: If Grafana could not decide: unreachable, 

344 slow, erroring, refusing PMM Extensions' own credential, or answering off 

345 contract. 

346 """ 

347 cache = self._service_account_verdicts 

348 key = (sha256(token.encode()).hexdigest(),) 

349 started = monotonic() 

350 with cache.lock, suppress(KeyError): 

351 return cache.get(key, started) 

352 verdict = await self._fetch_service_account_verdict(token) 

353 with cache.lock: 

354 cache.set(key, verdict, started) 

355 cache.evict_if_needed(monotonic()) 

356 return verdict

358 async def _fetch_service_account_verdict( 

359 self, token: str 

360 ) -> GrafanaServiceAccountRecord | None: 

361 """Ask Grafana who ``token`` belongs to, then read that account's record. 

362 

363 ``/api/user`` proves the token and names the account in ``uid``, but it 

364 reports every service account as enabled, so role and disabled flag are 

365 read from the account record through PMM Extensions' own credential. That call is 

366 org-scoped by Grafana, so a 404 covers an account in another org as well 

367 as one deleted in between. 

368 

369 :param token: The presented service-account token. 

370 :return: The account record, or ``None`` for a Grafana-answered refusal. 

371 :raises GrafanaException: For every outcome that is not a verdict. 

372 """ 

373 identity = await self._read_for_verification( 

374 "/api/user", token, refusal=status.HTTP_401_UNAUTHORIZED 

375 ) 

376 if identity is None: 

377 return None 

378 uid, login = identity.get("uid"), identity.get("login") 

379 match = ( 

380 _SERVICE_ACCOUNT_UID_PATTERN.fullmatch(uid) 

381 if isinstance(uid, str) 

382 else None 

383 ) 

384 if match is None or not isinstance(login, str) or not login.strip(): 

385 raise GrafanaException( 

386 detail="Grafana returned an unreadable service-account identity." 

387 ) 

388 account_id = int(match.group(1)) 

389 payload = await self._read_for_verification( 

390 f"/api/serviceaccounts/{account_id}", 

391 self.service_account_token.get_secret_value(), 

392 refusal=status.HTTP_404_NOT_FOUND, 

393 ) 

394 if payload is None: 

395 return None 

396 try: 

397 record = SERVICE_ACCOUNT_RECORD.validate_python(payload) 

398 except ValidationError: 

399 raise GrafanaException( 

400 detail="Grafana returned an unreadable service-account record." 

401 ) from None 

402 if record["id"] != account_id or record["login"] != login: 

403 raise GrafanaException( 

404 detail="Grafana returned an unreadable service-account record." 

405 ) 

406 return record

408 async def _read_for_verification( 

409 self, path: str, credential: str, *, refusal: int 

410 ) -> dict[str, Any] | None: 

411 """Read one verification record, separating a verdict from a failure. 

412 

413 :param path: The Grafana path to read. 

414 :param credential: The bearer token to read it with. 

415 :param refusal: The status that, answered by Grafana in JSON, is a 

416 refusal rather than a failure. 

417 :return: The JSON object, or ``None`` for a Grafana-answered refusal. 

418 :raises GrafanaException: For any other status, a timeout, a transport 

419 error, or a body that is not a JSON object. 

420 """ 

421 try: 

422 with self.auth(credential): 

423 payload = await self.get(path, timeout=_SERVICE_ACCOUNT_CALL_TIMEOUT) 

424 except GrafanaException: 

425 raise 

426 except HTTPException as exc: 

427 if _is_grafana_verdict(exc, refusal): 

428 return None 

429 self.logger.exception( 

430 "Grafana answered HTTP %s to %s while verifying a service-account " 

431 "token.", 

432 exc.status_code, 

433 path, 

434 ) 

435 raise GrafanaException( 

436 detail=f"Grafana could not verify the service-account token " 

437 f"(HTTP {exc.status_code})." 

438 ) from None 

439 except (TimeoutError, ClientError, ValueError): 

440 self.logger.warning( 

441 "Grafana did not answer %s while verifying a service-account token.", 

442 path, 

443 exc_info=True, 

444 ) 

445 raise GrafanaException( 

446 detail="Grafana could not verify the service-account token." 

447 ) from None 

448 if not isinstance(payload, dict): 

449 raise GrafanaException( 

450 detail=f"Grafana returned an unreadable {path} response." 

451 ) 

452 return payload
