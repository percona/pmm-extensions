164def exception_for_status( 

165 status_code: int, *, detail: Any, headers: dict[str, str] | None = None 

166) -> HTTPException: 

167 """Return the project exception mapped to ``status_code``, else a bare HTTPException. 

168 

169 Fall back to a bare :class:`fastapi.HTTPException` when no project class is 

170 mapped. Every mapped class accepts a ``headers`` kwarg, so headers are always 

171 preserved. 

172 

173 A non-JSON error body (marked with ``UPSTREAM_NON_JSON_HEADER``) signals a 

174 proxy/gateway failure rather than an app-level status, so a non-JSON 404 stays 

175 a bare HTTPException -- callers narrowing to ``except HTTPNotFoundException`` 

176 must not mistake an upstream infra failure for a real resource-absent 

177 response. The other statuses keep their mapping on non-JSON bodies, matching 

178 how they already behave for JSON bodies. 

179 

180 :param status_code: The upstream HTTP error status to translate. 

181 :param detail: The error detail payload to attach to the exception. 

182 :param headers: Optional response headers to preserve (e.g. ``X-Error-Code``). 

183 :return: The mapped project exception, or a bare HTTPException. 

184 """ 

185 exc_class = _HTTP_EXCEPTION_BY_STATUS.get(status_code) 

186 is_non_json = bool(headers) and UPSTREAM_NON_JSON_HEADER in headers 

187 if exc_class is None or (is_non_json and exc_class is HTTPNotFoundException): 

188 return HTTPException(status_code=status_code, detail=detail, headers=headers) 

189 return exc_class(detail, headers=headers)

264def _sanitize_request_kwargs( 

265 kwargs: dict[str, Any], 

266 *, 

267 extra_sensitive_headers: frozenset[str] = frozenset(), 

268 extra_sensitive_body_fields: frozenset[str] = frozenset(), 

269) -> dict[str, Any]: 

270 """Return a shallow copy of request kwargs with credentials redacted. 

271 

272 The auth context injects an ``Authorization`` header into ``kwargs`` before 

273 the request is logged, and some endpoints post credentials in the request 

274 body (a password-login payload, for example); this scrubs both 

275 credential-bearing headers and known-sensitive body fields so secrets never 

276 reach the debug log. Only the returned copy is masked -- the outgoing 

277 request keeps the real values. 

278 

279 :param kwargs: The request keyword arguments about to be logged. 

280 :param extra_sensitive_headers: Additional lowercase header names to mask, 

281 beyond the always-masked credential headers. 

282 :param extra_sensitive_body_fields: Additional lowercase body field names to 

283 mask, beyond the always-masked credential fields. 

284 :return: A copy safe to log, with sensitive header and body values masked. 

285 """ 

286 sensitive_headers = _SENSITIVE_HEADERS | extra_sensitive_headers 

287 sensitive_body_fields = _SENSITIVE_BODY_FIELDS | extra_sensitive_body_fields 

288 safe = {**kwargs} 

289 headers = kwargs.get("headers") 

290 if headers: 

291 safe["headers"] = { 

292 key: (_REDACTED_VALUE if key.lower() in sensitive_headers else value) 

293 for key, value in headers.items() 

294 } 

295 for body_key in ("json", "data"): 

296 body = kwargs.get(body_key) 

297 if isinstance(body, dict): 

298 safe[body_key] = { 

299 key: ( 

300 _REDACTED_VALUE if key.lower() in sensitive_body_fields else value 

301 ) 

302 for key, value in body.items() 

303 } 

304 return safe

321async def _iter_lines_from_chunks( 

322 chunks: AsyncIterator[bytes], path: str 

323) -> AsyncGenerator[bytes, None]: 

324 """Yield newline-terminated lines from an async iterator of byte chunks. 

325 

326 Buffer chunks from ``chunks`` and yield each newline-terminated slice with 

327 the trailing newline byte preserved. Flush any remaining partial line at 

328 end-of-stream so callers receive the final unterminated chunk. Reject any 

329 line larger than ``_MAX_STREAM_LINE_BYTES`` to protect consumers from a 

330 runaway producer. 

331 

332 Every chunk consumes the newlines it introduced and drops what precedes 

333 them, so the carried remainder is newline-free when the next chunk arrives. 

334 Searching only the arriving bytes therefore finds exactly what a search from 

335 the front finds, at a cost proportional to the chunk rather than to the 

336 remainder behind it. 

337 

338 :param chunks: An async iterator producing byte chunks (e.g. from 

339 ``aiohttp`` ``StreamReader.iter_any()``). 

340 :param path: The stream path, included in the error message when a single 

341 line exceeds the cap. 

342 :yield: Each line as ``bytes`` with its trailing newline preserved; the 

343 final unterminated chunk is also yielded when the stream ends without 

344 a newline. 

345 :raises ValueError: If a single line exceeds ``_MAX_STREAM_LINE_BYTES``. 

346 """ 

347 buffer = bytearray() 

348 async for chunk in chunks: 

349 if not chunk: 

350 continue 

351 # Two cursors, not one: the terminator can only be in the arriving 

352 # bytes, but the line it ends begins at the front of the buffer, where 

353 # the remainder carried from earlier chunks sits. 

354 search_from = len(buffer) 

355 buffer.extend(chunk) 

356 line_start = 0 

357 while True: 

358 newline_pos = buffer.find(b"\n", search_from) 

359 if newline_pos == -1: 

360 break 

361 line_end = newline_pos + 1 

362 line_size = line_end - line_start 

363 if line_size > _MAX_STREAM_LINE_BYTES: 

364 _raise_stream_line_too_big(line_size, path) 

365 yield bytes(buffer[line_start:line_end]) 

366 line_start = search_from = line_end 

367 if line_start: 

368 del buffer[:line_start] 

369 if len(buffer) > _MAX_STREAM_LINE_BYTES: 

370 _raise_stream_line_too_big(len(buffer), path)

408class PendingCloses: 

409 """Track clients whose :meth:`BaseRemoteAPI.close_when_idle` deferred a close. 

410 

411 Each owning component keeps its own instance so a shutdown sweep only 

412 reaches clients that owner itself retired. A client that drains normally 

413 before shutdown is removed here and is not force-closed again. 

414 

415 Clients are keyed by identity: :class:`BaseRemoteAPI` compares by field 

416 values, so a value-keyed set would collapse two retired instances that 

417 shared an endpoint. 

418 

419 :meth:`seal` (and :meth:`force_close`) permanently refuse new deferrals so a 

420 rebind that races shutdown cannot register a client after the sweep has 

421 already run. :meth:`add` returns ``False`` when sealed; the caller must 

422 close immediately. Callers that open a *replacement* client during 

423 shutdown also check :attr:`sealed` and discard the replacement instead of 

424 publishing it (the active slot is owned by teardown once sealing starts). 

425 """ 

426 

427 def __init__(self) -> None: 

428 self._clients: dict[int, BaseRemoteAPI] = {} 

429 self._sealed = False 

430 

431 @property 

432 def sealed(self) -> bool: 

433 """Return whether shutdown has sealed this collection against new deferrals.""" 

434 return self._sealed 

435 

436 def seal(self) -> None: 

437 """Reject further deferrals; late retirements must close immediately.""" 

438 self._sealed = True 

439 

440 def add(self, client: "BaseRemoteAPI") -> bool: 

441 """Remember ``client`` until it drains or :meth:`force_close` runs. 

442 

443 :param client: The client whose close was deferred. 

444 :return: ``True`` when registered for deferred close, ``False`` when 

445 this collection is already sealed and the caller must close 

446 ``client`` now (use :meth:`track` if that immediate close must 

447 stay discoverable on failure). 

448 """ 

449 if self._sealed: 

450 return False 

451 self._clients[id(client)] = client 

452 return True 

453 

454 def track(self, client: "BaseRemoteAPI") -> None: 

455 """Remember ``client`` until a successful close removes it. 

456 

457 Unlike :meth:`add`, this registers even when sealed so an immediate 

458 close that fails or is cancelled stays discoverable for a later 

459 :meth:`force_close`. Callers that discard a late replacement use this 

460 before awaiting :meth:`~BaseRemoteAPI.close`. 

461 

462 :param client: The client to keep tracked until a close succeeds. 

463 """ 

464 self._clients[id(client)] = client 

465 

466 def discard(self, client: "BaseRemoteAPI") -> None: 

467 """Drop ``client`` after a normal deferred close. 

468 

469 :param client: The client that closed (or was force-closed). 

470 """ 

471 self._clients.pop(id(client), None) 

472 

473 async def force_close(self) -> None: 

474 """Seal, then close every still-deferred client. 

475 

476 Sealing happens before any ``await`` so a concurrent rebind that tries 

477 to register after this returns cannot reintroduce a leak. 

478 Only the hold-triggered close flag is cleared before ``close`` so a 

479 concurrent :meth:`BaseRemoteAPI.hold` finally will not start another 

480 close from ``_close_when_idle``. The client stays registered here until 

481 a *successful* session close removes it, so a failed or cancelled close 

482 remains discoverable for a later sweep. If that hold already began 

483 tearing down the session, :meth:`BaseRemoteAPI.close` joins the 

484 in-progress operation so this sweep still waits for the socket to 

485 finish closing. Safe to call when empty. 

486 """ 

487 self._sealed = True 

488 clients = list(self._clients.values()) 

489 if not clients: 

490 return 

491 for client in clients: 

492 client.clear_hold_triggered_close() 

493 await _close_logging_failures(clients, action="force-closing")
