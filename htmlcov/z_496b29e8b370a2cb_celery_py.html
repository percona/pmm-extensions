16"""Define the Operations Inventory Celery entry point. 

17 

18``@owned_by("om_inventory")`` tags the task so the app-drain reconciler counts it 

19toward this app rather than treating it as a core task. The module is included in the 

20worker's ``include`` list because the app is registered in ``EXTENSIONS.APPS`` and the 

21registry derives the Celery module path from the package's ``celery.py``. 

22""" 

23 

24import logging 

25from uuid import UUID 

26 

27from app.celery import celery 

28from app.extensions.app_drain import owned_by 

29from app.extensions.apps.om_inventory.service import run_probe 

30 

31logger = logging.getLogger(__name__) 

32 

33 

34@owned_by("om_inventory") 

35@celery.task 

36def run_om_probe( 

37 execution_id: str | None = None, node_ids: list[str] | None = None 

38) -> str: 

39 """Run one probe sweep and return its id. 

40 

41 Scheduled by this app's ``periodic_task_schedules`` contribution, and invoked by 

42 the trigger endpoint with an already-created id so the caller can be answered 

43 before the Nomad work begins. 

44 

45 :param execution_id: An already-created run's id, or ``None`` to mint one. 

46 :param node_ids: The hosts to refresh, or ``None`` for the whole estate. The 

47 scheduled sweep passes nothing, which is what keeps it a full refresh. 

48 :return: The run's id, as a string. 

49 """ 

50 # ty-attr-ok: Celery installs `loop` at runtime. 

51 resolved = celery.loop.run_until_complete( 

52 run_probe(UUID(execution_id) if execution_id else None, node_ids) 

53 ) 

54 return str(resolved)
