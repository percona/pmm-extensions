58async def _reconcile_nomad(_: SnapshotChange) -> None: 

59 """Rebind the live Nomad executor when its override changed. 

60 

61 Registered as the ``(TASKS_SETTINGS, NOMAD)`` rebind callback by 

62 :func:`tasks_lifespan`. :meth:`NomadLifecycle.__aexit__` clears the holder 

63 before the override refresher task is cancelled, so a refresh cycle racing 

64 shutdown can find ``tasks_app.state.nomad_lifecycle`` already gone; the 

65 rebind is skipped in that window rather than raising a noisy callback error. 

66 

67 :param _: The override snapshots on either side of the republish. Unused; the 

68 holder reads the live ``NOMAD`` config itself when reconciling. 

69 """ 

70 holder = getattr(tasks_app.state, "nomad_lifecycle", None) 

71 if holder is not None: 

72 await holder.reconcile() 

73 

74 

75#: Rebind callbacks for watched Tasks overrides, fired by both the background 

76#: refresher and the settings-API handlers; published on ``tasks_app.state`` below. 

77_OVERRIDE_REBIND_CALLBACKS: CallbackRegistry = { 

78 (SettingClassEnum.TASKS_SETTINGS, "NOMAD"): _reconcile_nomad, 

79}

82@asynccontextmanager 

83async def tasks_lifespan(app: FastAPI) -> AsyncGenerator[None, None]: 

84 """Manage the Tasks API's lifespan. 

85 

86 Initializes the Tasks database data and starts the settings-override 

87 refresher, the default lifespan, and the :class:`NomadLifecycle` holder that 

88 owns the live entered ``NomadExecutor``. 

89 

90 The holder is anchored to the module-level ``tasks_app`` rather than to 

91 ``app``: when this lifespan runs under the combined ``app.main:app``, 

92 ``app`` is the *parent* application, but requests routed to the mounted 

93 ``/api/tasks`` sub-app resolve ``request.app`` to ``tasks_app`` -- so the 

94 holder must live on ``tasks_app.state`` for ``get_request_executor`` to find 

95 it in both standalone and mounted deployments. 

96 

97 :param app: The FastAPI application instance whose lifespan this manages 

98 (``tasks_app`` standalone, the parent app when mounted). 

99 :type app: FastAPI 

100 :return: An async context manager yielding ``None`` for the lifespan duration. 

101 :rtype: AsyncGenerator[None, None] 

102 """ 

103 await init_tasks_db() 

104 await verify_taskhistory_execution_request_is_jsonb() 

105 async with ( 

106 settings_override_refresher( 

107 get_async_session_maker, 

108 # ALERT_SETTINGS is intentionally NOT wired here: ``alert_settings`` 

109 # is a single shared proxy owned by the PMM Extensions refresher. Wiring it here 

110 # too would, in the combined ``app.main:app`` process, have both 

111 # refreshers publish into it from their separate databases and clobber 

112 # each other every cycle. 

113 # ANONYMIZER_SETTINGS is safe to wire (unlike ALERT above): its proxy 

114 # is tasks-track-DB-owned, so no cross-refresher clobber. Without it 

115 # the API process serves stale defaults until an in-process PATCH. 

116 { 

117 SettingClassEnum.TASKS_SETTINGS: ProxyEntry( 

118 tasks_settings, TasksSettings 

119 ), 

120 SettingClassEnum.ANONYMIZER_SETTINGS: ProxyEntry( 

121 anonymizer_settings, AnonymizerSettings 

122 ), 

123 }, 

124 callbacks=_OVERRIDE_REBIND_CALLBACKS, 

125 ), 

126 default_lifespan(app), 

127 NomadLifecycle(tasks_app), 

128 ): 

129 yield 

130 

131 

132lifespan = tasks_lifespan 

133tasks_app = create_app( 

134 build_health_router(get_async_session_maker), 

135 tasks_router, 

136 periodic_router,

163def task_data_not_found_detail(exc: TaskDataNotFoundInExecutorError) -> dict[str, Any]: 

164 """Build structured detail for HTTP 410 from TaskDataNotFoundInExecutorError. 

165 

166 :param exc: The exception indicating task data was not found in the executor. 

167 :type exc: TaskDataNotFoundInExecutorError 

168 :return: A dictionary with message and optional resource_type, resource_id, 

169 executor_name, job_id, evaluation_id, and detail keys for use in an HTTP 410 

170 response body. 

171 :rtype: dict[str, Any] 

172 """ 

173 detail = { 

174 "message": "The requested task data is no longer available in the executor.", 

175 } 

176 if exc.resource_type is not None: 

177 detail["resource_type"] = exc.resource_type 

178 if exc.resource_id is not None: 

179 detail["resource_id"] = exc.resource_id 

180 if exc.executor_name is not None: 

181 detail["executor_name"] = exc.executor_name 

182 if exc.job_id is not None: 

183 detail["job_id"] = exc.job_id 

184 if exc.evaluation_id is not None: 

185 detail["evaluation_id"] = exc.evaluation_id 

186 if str(exc): 

187 detail["detail"] = str(exc) 

188 return detail

191@tasks_app.exception_handler(TaskDataNotFoundInExecutorError) 

192async def task_data_not_found_handler( 

193 _: Request, 

194 exc: TaskDataNotFoundInExecutorError, 

195) -> None: 

196 """Handle exceptions raised when task data is not found in the executor.""" 

197 payload = task_data_not_found_detail(exc) 

198 logger.debug( 

199 "Task data not found in executor; HTTP %s body: %s", 

200 status.HTTP_410_GONE, 

201 json.dumps(payload, default=str), 

202 ) 

203 raise HTTPGoneException(detail=payload)
