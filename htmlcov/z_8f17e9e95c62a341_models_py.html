16"""Define the ATW plugin's DB tables and their API request/response models. 

17 

18This module is loaded by the Alembic plugin-discovery loader at migration time, 

19so it must stay self-contained — importing only from ``app.core``, ``pydantic``, 

20``sqlalchemy``, and ``sqlmodel``, never from sibling plugins or other service 

21packages (that would register foreign tables in ``SQLModel.metadata`` and leak 

22them into the ``extensions`` autogenerate). The category taxonomy, which depends on 

23``app.inventory``, lives in :mod:`app.extensions.apps.atw.categories`. 

24 

25:mod:`app.tasks.task_status` is the one exception, and it is admitted on the 

26mechanism rather than by name: the restriction exists to keep foreign **tables** out 

27of ``SQLModel.metadata``, and that module defines none — it imports only the standard 

28library, and ``app/tasks/__init__.py`` is licence header only. It exists so 

29``terminal_status`` can carry the real status enum instead of a bare ``str`` whose 

30value set nothing constrains. Any other ``app.tasks`` import, or a SQLModel class 

31appearing in that module, is the leak this paragraph is about. 

32"""

116class AtwIncidentResponse(BaseModel): 

117 """Represent a persisted diagnostic incident. 

118 

119 Every stored field is always present, so — unlike returning the 

120 :class:`AtwIncident` table model directly — the generated client types them as 

121 required rather than optional. 

122 

123 The three run-aggregate fields are defaulted instead, which keeps them out of 

124 the published ``required`` set so a client generated against the previous 

125 payload still validates a response carrying them. Every route populates all 

126 three, ``last_activity_at`` included, so the defaults are never served on a 

127 real response; the generated client types that one as optional anyway, because 

128 a nullable field emits no schema default to mark it required. 

129 

130 :param id: The incident's UUID primary key. 

131 :param name: Human-readable incident label. 

132 :param case_ref: Support-case reference, if set. 

133 :param created_by: Username of the support engineer who created the incident. 

134 :param created_at: When the incident was created. 

135 :param updated_at: When the incident was last updated, if ever. 

136 :param closed_at: When the incident was closed, if ever; ``None`` means open. 

137 :param run_count: How many snippet executions are grouped under the incident. 

138 :param failed_run_count: How many of those runs reached a failed outcome. A 

139 run whose outcome is not yet known counts towards neither. 

140 :param last_activity_at: The most recent of the incident's own timestamps and 

141 its executions' dispatch or completion times. 

142 """ 

143 

144 model_config = ConfigDict(from_attributes=True) 

145 

146 id: UUID4 

147 name: NonEmptyStr 

148 case_ref: str | None 

149 created_by: str 

150 created_at: UTCDatetime 

151 updated_at: UTCDatetime | None 

152 closed_at: UTCDatetime | None 

153 run_count: NonNegativeInt = 0 

154 failed_run_count: NonNegativeInt = 0 

155 last_activity_at: UTCDatetime | None = None

158class AtwIncidentExecution(BaseUUIDSQLModel, table=True): 

159 """Link one diagnostic snippet execution to its incident grouping. 

160 

161 The four outcome columns denormalize what the tasks service knows about the 

162 run, because status lives behind that service's own database and the incident 

163 listing may not issue a per-row HTTP call to read it. Two writers keep them: 

164 :func:`app.extensions.apps.atw.recorder.record_atw_run`, which the tasks service calls 

165 at the terminal transitions it observes, and the ``reconcile_atw_executions`` 

166 sweep, which fills the rest — so a run the hook missed reads as unresolved for up 

167 to one reconcile interval, longer while a backlog drains. ``terminal_status`` carries 

168 the shared status enum and its CHECK constraint, so the column cannot hold a value 

169 the aggregate would silently skip. The enum is imported from 

170 :mod:`app.tasks.task_status`, a leaf module that defines no tables, so naming it 

171 here registers nothing of the tasks service's in ``SQLModel.metadata`` — the 

172 module docstring states why that import is admitted. 

173 

174 :param incident_id: Foreign key to the owning :class:`AtwIncident`. 

175 :param task_history_id: Logical reference to the tasks-service execution row. 

176 :param snippet_filename: Filename of the executed diagnostic snippet. 

177 :param terminal_status: The ``TaskHistoryStatusEnum`` value the run finished 

178 with, or ``None`` while its outcome is still unknown. 

179 :param finished_at: When the run finished, as the tasks service recorded it. 

180 Kept separate from the row's own ``updated_at`` so the reconciliation 

181 sweep's bookkeeping writes cannot move the incident's last-activity time. 

182 :param outcome_unrecoverable: Whether the upstream history row no longer 

183 resolves, so no outcome will ever be recorded and the sweep must stop 

184 re-querying it. Distinguishing this from ``terminal_status IS NULL`` is 

185 why the outcome needs two columns rather than one. 

186 :param reconcile_attempted_at: When the sweep last examined this row, 

187 whatever the outcome. Its selection orders by this so a row that is 

188 legitimately still running cannot re-occupy a batch slot forever and 

189 starve the rows behind it. 

190 :param incident: The incident this execution belongs to. 

191 """ 

192 

193 __tablename__ = "atw_incident_execution" 

194 __table_args__ = ( 

195 UniqueConstraint( 

196 "incident_id", 

197 "task_history_id", 

198 name="uq_atw_incident_execution_incident_task", 

199 ), 

200 ) 

201 

202 incident_id: UUID4 = SQLField( 

203 foreign_key="atw_incident.id", 

204 ondelete="CASCADE", 

205 ) 

206 task_history_id: int = SQLField(index=True) 

207 snippet_filename: str 

208 terminal_status: TaskHistoryStatusEnum | None = SQLField( 

209 default=None, 

210 sa_column=Column( 

211 EnumField(TaskHistoryStatusEnum, native_enum=False, create_constraint=True), 

212 nullable=True, 

213 index=True, 

214 ), 

215 ) 

216 finished_at: UTCDatetime | None = SQLField( 

217 default=None, sa_type=DateTimeWithTimezone 

218 ) 

219 outcome_unrecoverable: bool = SQLField(default=False, nullable=False) 

220 reconcile_attempted_at: UTCDatetime | None = SQLField( 

221 default=None, sa_type=DateTimeWithTimezone 

222 ) 

223 incident: AtwIncident = Relationship(back_populates="executions")

226class AtwSendStatusEnum(EnumFieldMixin, StrEnum): 

227 """Enumerate the lifecycle of one diagnostics send attempt. 

228 

229 The column stores member *names* (``PENDING``); the API serializes the 

230 ``StrEnum`` *values* (``pending``). 

231 """ 

232 

233 PENDING = "pending" 

234 RUNNING = "running" 

235 SUCCESS = "success" 

236 FAILED = "failed" 

237 

238 @classmethod 

239 def active_statuses(cls) -> frozenset["AtwSendStatusEnum"]: 

240 """Return the statuses a send can still leave on its own. 

241 

242 :return: The non-terminal statuses. 

243 """ 

244 return frozenset({cls.PENDING, cls.RUNNING}) 

245 

246 def is_terminal(self) -> bool: 

247 """Return whether this status is a finished outcome. 

248 

249 :return: ``True`` for a status no further transition leaves. 

250 """ 

251 return self not in self.active_statuses()

254class AtwSendLog(BaseUUIDSQLModel, table=True): 

255 """Record one attempt to deliver an incident's diagnostics to the receiver. 

256 

257 The row *is* the job: the create endpoint writes it as ``PENDING`` and 

258 enqueues the Celery task with its id, and the status endpoint reads it back. 

259 Because the receiver's credentials can create attachments but not read them, 

260 ``detail`` carries the full final-upload response and is the only evidence a 

261 send ever landed. 

262 

263 :param incident_id: Foreign key to the owning :class:`AtwIncident`. 

264 :param case_ref: The support-case reference this send targets, snapshotted at 

265 request time so a later edit of the incident does not rewrite history. 

266 :param requested_by: Username of the support engineer who started the send. 

267 :param status: The attempt's lifecycle status. 

268 :param started_at: When the worker picked the attempt up, if it did. 

269 :param finished_at: When the attempt reached a terminal status, if it did. 

270 :param detail: The attempt's recorded evidence -- selected executions, per-step 

271 outcomes, the full upload response, or the error that ended it. 

272 :param incident: The incident this attempt belongs to. 

273 """ 

274 

275 __tablename__ = "atw_send_log" 

276 

277 incident_id: UUID4 = SQLField( 

278 foreign_key="atw_incident.id", 

279 ondelete="CASCADE", 

280 index=True, 

281 ) 

282 case_ref: NonEmptyStr = SQLField(nullable=False) 

283 requested_by: str = SQLField(nullable=False) 

284 status: AtwSendStatusEnum = SQLField( 

285 default=AtwSendStatusEnum.PENDING, 

286 sa_column=Column( 

287 EnumField(AtwSendStatusEnum, native_enum=False, create_constraint=True), 

288 nullable=False, 

289 ), 

290 ) 

291 started_at: UTCDatetime | None = SQLField( 

292 default=None, sa_type=DateTimeWithTimezone 

293 ) 

294 finished_at: UTCDatetime | None = SQLField( 

295 default=None, sa_type=DateTimeWithTimezone 

296 ) 

297 detail: dict[str, Any] = SQLField(default_factory=dict, sa_column=Column(JSON)) 

298 incident: AtwIncident = Relationship(back_populates="send_logs")
