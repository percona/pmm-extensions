40#: The exceptions whose full message may be stored on ``last_sync_error``. 

41#: 

42#: An allowlist rather than a denylist, and matched by exact type rather than by 

43#: ``isinstance``: a ``SyncError`` subclass added later must be opted in 

44#: deliberately instead of inheriting persistence from its base — a subclass is 

45#: free to interpolate remote context its base never did. Every member here 

46#: interpolates only sync bookkeeping — the ``SyncItem`` an attempt was for. 

47#: ``ExecutorHostNotFoundError`` is deliberately absent: its message carries the 

48#: whole executor-host map from the Tasks API, which the inventory read routes 

49#: would then expose to any authenticated caller. 

50_MESSAGE_SAFE_ERRORS: Final = frozenset( 

51 { 

52 SyncFailError, 

53 SyncInstanceAlreadyInProgressError, 

54 SyncItemAlreadyInProgressError, 

55 } 

56)

59def _describe_sync_error(error: Exception) -> str: 

60 """Summarize an exception for storage on an entity's ``last_sync_error``. 

61 

62 The field is durable and readable through the inventory read routes, so the 

63 summary carries a message only for the exceptions whose *exact* type is 

64 listed in :data:`_MESSAGE_SAFE_ERRORS`; a subclass of one is not a member. 

65 ``HTTPException`` is matched by ``isinstance`` instead because that branch 

66 reduces rather than admits, and a subclass of it must reduce too. 

67 Everything else is reduced to its type name: a 

68 third-party exception's message is arbitrary text this code never inspected 

69 — a ``ValidationError`` echoes the offending input, a driver error can carry 

70 a DSN — and no redaction pass can be trusted over an open set. Nothing is 

71 lost by the reduction: only the description is persisted, and the exception 

72 itself is re-raised out of :meth:`SyncHealthReporter.record` to the boundary 

73 that logs it with its traceback. 

74 

75 The result is non-empty but not length-bounded; the inventory service 

76 truncates it to the column's contract when it stores it. 

77 

78 :param error: The exception that ended the sync attempt. 

79 :return: A non-empty description carrying no caller-supplied text. 

80 """ 

81 if isinstance(error, HTTPException): 

82 return f"{type(error).__name__}: HTTP {error.status_code}" 

83 if type(error) in _MESSAGE_SAFE_ERRORS: 

84 return f"{type(error).__name__}: {error}" 

85 return type(error).__name__

88class SyncHealthAttempt: 

89 """Track whether a sync attempt reached a real comparison against source. 

90 

91 Defaults to not-compared so a path that forgets to mark it records nothing, 

92 which is the safe direction: an unwritten column reads as stale, never as 

93 freshly confirmed. 

94 

95 :ivar compared: Whether the block held real source data and synced from it. 

96 """ 

97 

98 def __init__(self) -> None: 

99 self.compared = False 

100 

101 def mark_compared(self) -> None: 

102 """Record that source data was in hand and the entity was synced from it.""" 

103 self.compared = True

106class SyncHealthReporter: 

107 """Report per-entity sync outcomes to the inventory service. 

108 

109 Mechanism only: which entity levels a syncer mirrors is policy, declared on 

110 the syncer class (``BaseSyncer.mirrors_entity_levels``) and handed in here. 

111 

112 :param inventory_api: The inventory service client the outcomes post to. 

113 :param mirrored_levels: The entity levels whose attempts are recorded; 

114 every other level passes through unrecorded. 

115 """ 

116 

117 def __init__( 

118 self, 

119 inventory_api: RemoteAPI, 

120 mirrored_levels: frozenset[SyncInventoryEntityTypeEnum], 

121 ) -> None: 

122 self._inventory_api = inventory_api 

123 self._mirrored_levels = mirrored_levels 

124 

125 @asynccontextmanager 

126 async def record( 

127 self, 

128 entity_type: SyncInventoryEntityTypeEnum, 

129 created_entity: CreatedEntity, 

130 ) -> AsyncGenerator[SyncHealthAttempt, None]: 

131 """Write the entity's sync-health columns from this attempt's outcome. 

132 

133 Nests inside ``manage_sync_item`` so a failure is recorded before that 

134 boundary marks the SyncItem failed and raises under ``break_on_error``. 

135 A failure is recorded unconditionally — entering this block at a 

136 mirrored level *is* the genuine attempt, and a ``fetch_*`` that raises 

137 is exactly the failure worth reporting. A success is recorded only once 

138 the block marked itself compared, which is what excludes the 

139 filtered-out early return that leaves ``manage_sync_item``'s clean-exit 

140 path indistinguishable from a real sync. 

141 

142 A ``SyncFailError`` is the exception: it can only come from a *nested* 

143 level, because this level's own boundary raises it after this block has 

144 already exited. The syncers walk to children from inside the parent's 

145 ``perform_*_sync``, so a child's failure passes through here — and 

146 whether it does at all depends on ``break_on_error``, since otherwise 

147 the child's own boundary swallows it. Attributing it to this entity 

148 would make the columns describe an identical outcome differently in the 

149 two modes, and would report a parent as failing whose own mirrored 

150 fields were confirmed. The child records its own failure on its own row. 

151 

152 :param entity_type: The level being synced. 

153 :param created_entity: The entity being synced. 

154 :return: The marker the block flags once it holds real source data. 

155 """ 

156 attempt = SyncHealthAttempt() 

157 if entity_type not in self._mirrored_levels: 

158 yield attempt 

159 return 

160 attempted_at = datetime.now(UTC) 

161 try: 

162 yield attempt 

163 except SyncFailError as exc: 

164 await self._post( 

165 entity_type, 

166 created_entity, 

167 attempted_at, 

168 None if attempt.compared else exc, 

169 ) 

170 raise 

171 except Exception as exc: 

172 await self._post(entity_type, created_entity, attempted_at, exc) 

173 raise 

174 if attempt.compared: 

175 await self._post(entity_type, created_entity, attempted_at, None) 

176 

177 async def _post( 

178 self, 

179 entity_type: SyncInventoryEntityTypeEnum, 

180 created_entity: CreatedEntity, 

181 attempted_at: datetime, 

182 error: Exception | None, 

183 ) -> None: 

184 """Report one entity's sync outcome to the inventory service. 

185 

186 Best-effort: a bookkeeping write must not fail an otherwise-healthy 

187 sync item, and losing one leaves the entity looking stale rather than 

188 falsely fresh. 

189 

190 :param entity_type: The level being reported. 

191 :param created_entity: The entity being reported. 

192 :param attempted_at: When this attempt began, captured before the block 

193 ran so the inventory service orders by attempt, not by arrival. Read 

194 from the clock directly rather than through ``utc_now``, whose 

195 second truncation would hand two attempts within one second the same 

196 ordering key — and the service's guards admit an equal one, so the 

197 later arrival would win regardless of which attempt was newer. 

198 :param error: The exception that ended the attempt, or None on success. 

199 """ 

200 outcome = ( 

201 {"outcome": SyncOutcomeEnum.SUCCESS} 

202 if error is None 

203 else { 

204 "outcome": SyncOutcomeEnum.FAILURE, 

205 "error": _describe_sync_error(error), 

206 } 

207 ) 

208 payload = {"attempted_at": attempted_at.isoformat(), **outcome} 

209 try: 

210 await self._inventory_api.post( 

211 f"/{INVENTORY_PATH_SEGMENTS[entity_type]}/{created_entity.id}" 

212 "/sync-health", 

213 json=payload, 

214 ) 

215 except Exception: 

216 logger.exception( 

217 "Failed to record sync health for %s %s", 

218 entity_type.name, 

219 created_entity.id, 

220 )
