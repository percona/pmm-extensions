47class SettingResponse(BaseModel): 

48 """Represent a single setting's metadata and current value. 

49 

50 :param setting_class: The Pydantic class ``__name__`` the field belongs to. 

51 :param key: The field name on the settings class. 

52 :param key_path: Carry the canonical key segments for ``key`` such that 

53 ``"__".join(key_path) == key``. 

54 :param value: The current value visible through the proxy, dumped to a 

55 JSON-safe shape via the field's annotation. ``SecretStr`` fields are 

56 redacted to ``"**********"``. 

57 :param default_value: The field's declared default value, dumped via the 

58 same JSON serialiser. ``None`` when no default exists. 

59 :param type: A human-readable representation of the field's declared 

60 annotation (for operator visibility; validation uses the actual 

61 ``FieldInfo``). 

62 :param reload: The reload classification (HOT or NOT_OVERRIDABLE). 

63 :param description: The field's free-text description, or ``None``. 

64 :param is_secret: Whether the field's annotation contains a Pydantic secret 

65 (``SecretStr`` / ``SecretBytes``) at any depth. 

66 :param is_complex: Whether the field's annotation is or contains a Pydantic 

67 ``BaseModel`` subclass (true for nested submodels). 

68 :param has_override: Whether an **active** row in the ``settingoverride`` 

69 table applies to this ``(setting_class, key)`` pair. An inactive row is 

70 skipped by the cache loader, so the served value falls back to the 

71 declared default and reporting it as overridden would tell the UI a 

72 field is overridden while showing it that default. A nested row also 

73 marks every canonical prefix of its chain, so a parent reports ``True`` 

74 when only a deeper leaf carries a row. 

75 :param is_advanced: Whether the setting is flagged ``advanced`` so the UI can 

76 present it separately from everyday settings. Display-only: 

77 it does not affect PATCH/DELETE eligibility. 

78 :param is_applicable: Whether the setting applies under current runtime state 

79 (e.g. the active auth provider). ``False`` lets the UI present the field 

80 as inert. Display-only, like ``is_advanced``: it does not block 

81 PATCH/DELETE server-side; the runtime gate is the real enforcement. 

82 :param options: Selectable enum members for dropdown UIs, or ``None`` when 

83 the field is not an ``Enum`` annotation. Aliased members are excluded. 

84 :param updated_at: When the override applying to this key was last saved, 

85 falling back to the row's creation time for a row written before the 

86 stamp was recorded. ``None`` when ``has_override`` is ``False``. 

87 Timestamps carry second granularity. 

88 :param updated_by: The username that last saved that override, or ``None`` 

89 both when no override applies and when the row predates the actor 

90 column. A key can draw on several rows (a nested parent reporting on its 

91 leaves), in which case the pair comes from the row carrying the latest 

92 timestamp. Two writes landing within the same second are 

93 indistinguishable by timestamp, and the pair reported is then whichever 

94 contributing row was created later, which need not be the one written 

95 later. 

96 """ 

97 

98 model_config = ConfigDict(arbitrary_types_allowed=True) 

99 

100 setting_class: str 

101 key: str 

102 key_path: list[str] = Field(default_factory=list) 

103 value: Any 

104 default_value: Any 

105 type: str 

106 reload: ReloadClassification 

107 description: str | None 

108 is_secret: bool 

109 is_complex: bool 

110 has_override: bool 

111 is_advanced: bool = False 

112 is_applicable: bool = True 

113 options: list[SettingOption] | None = None 

114 updated_at: UTCDatetime | None = None 

115 updated_by: str | None = None

118class SettingsPatch(RootModel[dict[str, JsonValue]]): 

119 """Batch PATCH payload: a flat ``{field_name: new_value, ...}`` mapping. 

120 

121 An empty body is rejected as 422 because a no-op PATCH is a client bug, not 

122 a valid request. The server coerces each value to the field's declared 

123 type via :func:`coerce_field_value`; validation is all-or-nothing -- if any 

124 key fails, nothing is written. 

125 """ 

126 

127 root: dict[str, JsonValue] = Field(min_length=1)

149class SettingClassGroup(BaseModel): 

150 """Group one settings class's fields for the LIST response. 

151 

152 :param setting_class: The Pydantic class ``__name__`` this group represents. 

153 :param settings: The fields declared on the settings class, with their 

154 current values and metadata. 

155 :param is_app_owned: Whether this group belongs to a PMM Extensions app under 

156 ``app/extensions/apps/`` rather than core PMM Extensions wiring. 

157 :param app_id: The owning app's registry key when ``is_app_owned`` is 

158 ``True``; ``None`` for core groups. 

159 :param app_display_name: The owning app's human-facing label when 

160 ``is_app_owned`` is ``True``; ``None`` for core groups. 

161 :param app_enabled: Whether the owning app is currently enabled when 

162 ``is_app_owned`` is ``True``; ``None`` for core groups. Disabled 

163 apps remain listed so the frontend can hide them without a second 

164 lookup. 

165 """ 

166 

167 setting_class: str 

168 settings: list[SettingResponse] 

169 is_app_owned: bool = False 

170 app_id: str | None = None 

171 app_display_name: str | None = None 

172 app_enabled: bool | None = None
