16"""Wire PMM Extensions' DB-backed settings overrides into the web and worker processes. 

17 

18Holds the single composition of the PMM Extensions side proxy registry, so every PMM Extensions 

19process refreshes the same set and a class added here reaches all of them at 

20once rather than only the one whose wiring was remembered. 

21 

22This module is named directly in ``STATIC_CELERY_INCLUDE`` so its signal 

23receivers register at worker startup even in an image that ships no app with a 

24``celery.py``. It registers no Celery task, so it never appears in 

25``celery.tasks``. 

26"""

57def build_extensions_override_proxies() -> ProxyRegistry: 

58 """Compose the PMM Extensions side proxy registry: app-owned entries plus PMM Extensions' own. 

59 

60 PMM Extensions' own entries override whatever the activated apps declare. 

61 ``SETTINGS`` and ``ALERT_SETTINGS`` are the shared module-level proxies 

62 (``settings`` / ``alert_settings``), taken from 

63 :data:`~app.core.settings_override.extensions_owned.EXTENSIONS_OWNED_PROXIES`, 

64 so they keep exactly one refresher publishing them in each process even if 

65 an app were to declare them, and every publisher reads them from the PMM 

66 Extensions database. 

67 

68 ``collect_app_owned_settings_classes`` imports every activated app module, 

69 so this runs at call time, never at import: this module is itself named in 

70 ``STATIC_CELERY_INCLUDE``, so reaching the app tree at its module scope 

71 would import every activated app during include-import, alongside the app 

72 ``celery.py`` modules being imported from the same list. 

73 

74 :return: The wired proxy registry keyed by class identifier. 

75 :raises TypeError: Propagates from ``collect_app_owned_settings_classes`` 

76 when an app's ``APP_OWNED_SETTINGS_CLASSES`` declaration is malformed. 

77 :raises ValueError: Propagates from ``collect_app_owned_settings_classes`` 

78 when an activated app's declaration is invalid; that function 

79 enumerates the cases. 

80 """ 

81 proxies = { 

82 entry.setting_class: ProxyEntry(entry.proxy, entry.settings_cls) 

83 for entry in collect_app_owned_settings_classes() 

84 } 

85 proxies.update( 

86 { 

87 EXTENSIONS_SETTINGS: ProxyEntry( 

88 cast("OverridableSettingsProxy", extensions_settings), 

89 ExtensionsSettings, 

90 ), 

91 SNIPPETS_SETTINGS: ProxyEntry( 

92 cast("OverridableSettingsProxy", snippets_settings), SnippetsSettings 

93 ), 

94 **EXTENSIONS_OWNED_PROXIES, 

95 } 

96 ) 

97 return proxies

100async def republish_extensions_settings_snapshot(session: AsyncSession) -> None: 

101 """Republish the PMM Extensions settings snapshot from a session already in hand. 

102 

103 Let a Celery task decide against the overrides currently stored, not only 

104 those present when its child last refreshed: the worker's refresher advances 

105 only while something drives ``celery.loop``, so a child can hold a pre-write 

106 snapshot for an unbounded time, while an awaited republish inside a task 

107 body is driven by the same ``run_until_complete`` that drives the task. 

108 

109 Only ``EXTENSIONS_SETTINGS`` is republished, and no rebind callback fires: 

110 ``publish_snapshot`` has no callback channel. That is harmless for a worker 

111 caller, whose ``WORKER_OVERRIDE_CALLBACKS`` watches ``SETTINGS`` alone. It 

112 would not be for a web-process caller: ``extensions_overrides_lifespan`` registers 

113 ``EXTENSIONS_SETTINGS`` rebinders for ``INVENTORY_ENDPOINT``, ``TASKS_ENDPOINT`` 

114 and ``APP_DRAIN``, and republishing here leaves the periodic refresher an 

115 empty diff, so those rebinds would silently never fire. 

116 

117 :param session: A session bound to the PMM Extensions database, used to read the 

118 override rows. 

119 :raises Exception: Propagates whatever ``publish_snapshot`` raises, in 

120 practice ``SQLAlchemyError`` from the override-row query. There is no 

121 per-proxy handler here as there is in ``refresh_all``; the caller owns 

122 the failure. 

123 """ 

124 await publish_snapshot(extensions_settings, session, ExtensionsSettings) 

125 

126 

127#: The callbacks the worker refresher registers -- deliberately a strict subset 

128#: of the web lifespan's registry. The dropped entries either rebind ``app.state`` 

129#: clients or reseed beat-schedule rows, neither of which a worker child owns. 

130#: ``invalidate_pmm_clients`` is kept because ``ClientRegistry.IMMUTABLE_KEYS`` 

131#: excludes ``api_key``: a same-endpoint credential override would otherwise 

132#: refresh the snapshot while worker tasks keep the client with the old key. 

133#: ``apply_logging_dictconfig`` is kept so a HOT ``LOGGING`` override re-enters 

134#: ``dictConfig`` after Celery's ``setup_logging`` installed boot-time levels. 

135WORKER_OVERRIDE_CALLBACKS: CallbackRegistry = EXTENSIONS_OWNED_CALLBACKS

156@worker_process_init.connect 

157def start_extensions_settings_override_refresher(**_: Any) -> None: 

158 """Seed and arm the worker's PMM Extensions side DB-backed settings-override refresher. 

159 

160 A second refresher alongside the Tasks-side one in :mod:`app.tasks.celery`: 

161 the two proxy sets resolve against different databases and ``refresh_all`` 

162 opens one session per call, so one refresher serves exactly one database. 

163 After the inline seed, refreshes are pulled from ``task_prerun`` via 

164 :func:`refresh_extensions_overrides_if_due`. The seed is bounded by a fraction of 

165 ``celery.conf.worker_proc_alive_timeout`` so a hanging database cannot 

166 push the child past the prefork pool's liveness deadline. On seed expiry 

167 the child is still armed and may retain a possibly incomplete seed until 

168 the next due task boundary. 

169 

170 :param _: The ``worker_process_init`` signal keyword arguments (unused). 

171 :raises Exception: Propagates whatever composing the proxy registry or the 

172 initial inline refresh raises — a malformed app-owned declaration 

173 (``TypeError`` / ``ValueError``) or a session-maker failure — and is 

174 absorbed the same way. Per-proxy refresh failures and a bounded-seed 

175 expiry are caught and logged inside the refresher; the latter still 

176 arms the child for boundary refresh. 

177 """ 

178 _refresher.start( 

179 callbacks=WORKER_OVERRIDE_CALLBACKS, 

180 proc_alive_timeout=celery.conf.worker_proc_alive_timeout, 

181 )

184@task_prerun.connect 

185def refresh_extensions_overrides_if_due(**_: Any) -> None: 

186 """Refresh PMM Extensions side overrides at this task boundary when the interval is due. 

187 

188 Passes through to :meth:`WorkerRefresher.maybe_refresh`, which no-ops when 

189 disarmed or inside the interval. ``WORKER_OVERRIDE_CALLBACKS`` (PMM client 

190 eviction, LOGGING dictConfig) were registered at :meth:`WorkerRefresher.start`, 

191 so a changed endpoint or credential still invalidates cached clients before 

192 the task body runs. 

193 

194 :param _: The ``task_prerun`` signal keyword arguments (unused). 

195 """ 

196 _refresher.maybe_refresh()
