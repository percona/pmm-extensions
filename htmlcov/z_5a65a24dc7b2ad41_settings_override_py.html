16"""Wire PMM Extensions' DB-backed settings overrides into the web and worker processes. 

17 

18Holds the single composition of the PMM Extensions side proxy registry, so every PMM Extensions 

19process refreshes the same set and a class added here reaches all of them at 

20once rather than only the one whose wiring was remembered. 

21 

22This module is named directly in ``STATIC_CELERY_INCLUDE`` so its signal 

23receivers register at worker startup even in an image that ships no app with a 

24``celery.py``. It registers no Celery task, so it never appears in 

25``celery.tasks``. 

26"""

63def build_extensions_override_proxies() -> ProxyRegistry: 

64 """Compose the PMM Extensions side proxy registry: app-owned entries plus PMM Extensions' own. 

65 

66 PMM Extensions' own entries override whatever the activated apps declare, 

67 so ``SETTINGS`` and ``ALERT_SETTINGS`` -- shared module-level proxies 

68 (``settings`` / ``alert_settings``) -- keep the PMM Extensions refresher as their sole 

69 owner even if an app were to declare them: under the combined 

70 ``app.main:app`` the Tasks refresher must not also publish into them from 

71 the Tasks database. 

72 

73 ``collect_app_owned_settings_classes`` imports every activated app module, 

74 so this runs at call time, never at import: this module is itself named in 

75 ``STATIC_CELERY_INCLUDE``, so reaching the app tree at its module scope 

76 would import every activated app during include-import, alongside the app 

77 ``celery.py`` modules being imported from the same list. 

78 

79 :return: The wired proxy registry keyed by class identifier. 

80 :raises TypeError: Propagates from ``collect_app_owned_settings_classes`` 

81 when an app's ``APP_OWNED_SETTINGS_CLASSES`` declaration is malformed. 

82 :raises ValueError: Propagates from ``collect_app_owned_settings_classes`` 

83 when an activated app's declaration is invalid; that function 

84 enumerates the cases. 

85 """ 

86 proxies = { 

87 entry.setting_class: ProxyEntry(entry.proxy, entry.settings_cls) 

88 for entry in collect_app_owned_settings_classes() 

89 } 

90 proxies.update( 

91 { 

92 EXTENSIONS_SETTINGS: ProxyEntry(extensions_settings, ExtensionsSettings), 

93 SNIPPETS_SETTINGS: ProxyEntry(snippets_settings, SnippetsSettings), 

94 SETTINGS: ProxyEntry(settings, Settings), 

95 ALERT_SETTINGS: ProxyEntry(alert_settings, AlertSettings), 

96 } 

97 ) 

98 return proxies

101async def republish_extensions_settings_snapshot(session: AsyncSession) -> None: 

102 """Republish the PMM Extensions settings snapshot from a session already in hand. 

103 

104 Let a Celery task decide against the overrides currently stored, not only 

105 those present when its child last refreshed: the worker's refresher advances 

106 only while something drives ``celery.loop``, so a child can hold a pre-write 

107 snapshot for an unbounded time, while an awaited republish inside a task 

108 body is driven by the same ``run_until_complete`` that drives the task. 

109 

110 Only ``EXTENSIONS_SETTINGS`` is republished, and no rebind callback fires: 

111 ``publish_snapshot`` has no callback channel. That is harmless for a worker 

112 caller, whose ``WORKER_OVERRIDE_CALLBACKS`` watches ``SETTINGS`` alone. It 

113 would not be for a web-process caller: ``extensions_overrides_lifespan`` registers 

114 ``EXTENSIONS_SETTINGS`` rebinders for ``INVENTORY_ENDPOINT``, ``TASKS_ENDPOINT`` 

115 and ``APP_DRAIN``, and republishing here leaves the periodic refresher an 

116 empty diff, so those rebinds would silently never fire. 

117 

118 :param session: A session bound to the PMM Extensions database, used to read the 

119 override rows. 

120 :raises Exception: Propagates whatever ``publish_snapshot`` raises, in 

121 practice ``SQLAlchemyError`` from the override-row query. There is no 

122 per-proxy handler here as there is in ``refresh_all``; the caller owns 

123 the failure. 

124 """ 

125 await publish_snapshot(extensions_settings, session, ExtensionsSettings)

128async def invalidate_pmm_clients(change: SnapshotChange) -> None: 

129 """Evict cached PMM clients on the previous and current endpoints after a ``PMM`` override. 

130 

131 Evicts the ordered de-duplicated set of previous-and-current endpoints so a 

132 same-endpoint change (credentials, SSL) collapses to a single eviction, while 

133 an endpoint change also drops the client keyed by the endpoint no longer in 

134 use. The next :class:`PMMSyncer` key-misses to a fresh client via its 

135 ``default_factory`` PMM read. When ``PMM`` is absent from ``change.previous`` 

136 (override created), :func:`previous_or_base` supplies the YAML/env value. 

137 

138 :param change: The override snapshots on either side of the republish. 

139 """ 

140 previous_pmm = cast("PMMSettings", previous_or_base(change, settings, "PMM")) 

141 for endpoint in dict.fromkeys( 

142 str(pmm.endpoint) 

143 for pmm in (previous_pmm, settings.PMM) 

144 if pmm.endpoint is not None 

145 ): 

146 await settings.invalidate_client(endpoint)

149@fire_on_boot 

150async def apply_logging_dictconfig(_: SnapshotChange) -> None: 

151 """Re-apply ``logging.config.dictConfig`` after a global ``LOGGING`` override. 

152 

153 ``LOGGING`` is a HOT field, but ``LOGGING_CONFIG`` (the dict handed to 

154 ``dictConfig``) is not: the override snapshot replaces only the ``LOGGING`` 

155 key, so ``settings.LOGGING_CONFIG`` still carries the level baked in by the 

156 ``set_log_level`` model validator at construction time. This callback mirrors 

157 that validator -- inject the now-live ``settings.LOGGING`` into a copy of the 

158 config and re-apply it -- so a log-level change takes effect in the PMM Extensions web 

159 process and Celery worker children without a restart. Failures are logged and 

160 swallowed: a malformed config must not take the process down mid-request or 

161 mid-task. ``LOGGING_CONFIG`` sets ``disable_existing_loggers: False``, so 

162 re-entering ``dictConfig`` from a worker refresh cycle leaves the loggers 

163 Celery created at runtime enabled, and re-creates the ones the config names 

164 -- ``celery`` among them -- with their handlers. 

165 

166 Marked ``fire_on_boot`` because the boot-time ``dictConfig`` call reads the 

167 process-wide ``LOGGING_CONFIG``, which no snapshot carries: a process that 

168 starts with a ``LOGGING`` override already stored would otherwise keep the 

169 YAML/env level until the override next changed. 

170 

171 :param _: The snapshots on either side of the republish (unused; the 

172 level is re-read from the proxy). 

173 """ 

174 try: 

175 config = deepcopy(settings.LOGGING_CONFIG) 

176 config["loggers"][""]["level"] = settings.LOGGING 

177 config["loggers"]["app"]["level"] = settings.LOGGING 

178 logging.config.dictConfig(config) 

179 except Exception: 

180 logger.exception("Failed to re-apply logging config after LOGGING override") 

181 

182 

183#: The callbacks the worker refresher registers -- deliberately a strict subset 

184#: of the web lifespan's registry. The dropped entries either rebind ``app.state`` 

185#: clients or reseed beat-schedule rows, neither of which a worker child owns. 

186#: ``invalidate_pmm_clients`` is kept because ``ClientRegistry.IMMUTABLE_KEYS`` 

187#: excludes ``api_key``: a same-endpoint credential override would otherwise 

188#: refresh the snapshot while worker tasks keep the client with the old key. 

189#: ``apply_logging_dictconfig`` is kept so a HOT ``LOGGING`` override re-enters 

190#: ``dictConfig`` after Celery's ``setup_logging`` installed boot-time levels. 

191WORKER_OVERRIDE_CALLBACKS: CallbackRegistry = { 

192 (SETTINGS, "PMM"): invalidate_pmm_clients, 

193 (SETTINGS, "LOGGING"): apply_logging_dictconfig, 

194}

215@worker_process_init.connect 

216def start_extensions_settings_override_refresher(**_: Any) -> None: 

217 """Seed and arm the worker's PMM Extensions side DB-backed settings-override refresher. 

218 

219 A second refresher alongside the Tasks-side one in :mod:`app.tasks.celery`: 

220 the two proxy sets resolve against different databases and ``refresh_all`` 

221 opens one session per call, so one refresher serves exactly one database. 

222 After the inline seed, refreshes are pulled from ``task_prerun`` via 

223 :func:`refresh_extensions_overrides_if_due`. The seed is bounded by a fraction of 

224 ``celery.conf.worker_proc_alive_timeout`` so a hanging database cannot 

225 push the child past the prefork pool's liveness deadline. On seed expiry 

226 the child is still armed and may retain a possibly incomplete seed until 

227 the next due task boundary. 

228 

229 :param _: The ``worker_process_init`` signal keyword arguments (unused). 

230 :raises Exception: Propagates whatever composing the proxy registry or the 

231 initial inline refresh raises — a malformed app-owned declaration 

232 (``TypeError`` / ``ValueError``) or a session-maker failure — and is 

233 absorbed the same way. Per-proxy refresh failures and a bounded-seed 

234 expiry are caught and logged inside the refresher; the latter still 

235 arms the child for boundary refresh. 

236 """ 

237 _refresher.start( 

238 callbacks=WORKER_OVERRIDE_CALLBACKS, 

239 proc_alive_timeout=celery.conf.worker_proc_alive_timeout, 

240 )

243@task_prerun.connect 

244def refresh_extensions_overrides_if_due(**_: Any) -> None: 

245 """Refresh PMM Extensions side overrides at this task boundary when the interval is due. 

246 

247 Passes through to :meth:`WorkerRefresher.maybe_refresh`, which no-ops when 

248 disarmed or inside the interval. ``WORKER_OVERRIDE_CALLBACKS`` (PMM client 

249 eviction, LOGGING dictConfig) were registered at :meth:`WorkerRefresher.start`, 

250 so a changed endpoint or credential still invalidates cached clients before 

251 the task body runs. 

252 

253 :param _: The ``task_prerun`` signal keyword arguments (unused). 

254 """ 

255 _refresher.maybe_refresh()
