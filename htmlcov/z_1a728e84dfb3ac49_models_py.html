36class DipperExecuteWrite(BaseModel): 

37 """Define the JSON body for ``POST /api/apps/dipper/``. 

38 

39 :param service_id: Inventory ID of the database service to collect data from. 

40 :type service_id: int 

41 :param collector_type: Which collector script to run (environment or pmm). 

42 :type collector_type: CollectorTypeEnum 

43 :param executor_host: Nomad client hostname that will run the script. 

44 :type executor_host: NonEmptyStr 

45 :param sudo: Whether to invoke the script with ``sudo``. ``None`` defers 

46 to the script's own sudo policy (the default). 

47 :type sudo: bool | None 

48 :param args: Per-parameter arguments keyed by script parameter name. Validated 

49 server-side against the script's dynamic execution model. 

50 :type args: dict[str, Any] 

51 """ 

52 

53 service_id: int 

54 collector_type: CollectorTypeEnum = CollectorTypeEnum.ENVIRONMENT 

55 executor_host: NonEmptyStr = Field(title=EXECUTION_HOST_LABEL) 

56 sudo: bool | None = None 

57 args: dict[str, Any] = Field(
