87"""Cap a batch selection. 

88 

89The whole selection resolves in a single lookup, and each dispatched item then 

90adds an upstream call and a write, all in request order. The ceiling keeps one 

91request's cost bounded; it sits far above any realistic diagnostic selection. 

92"""

99"""Keep these fields per-snippet even when several snippets declare them identically. 

100 

101Unlike ``_SYNTHETIC_FIELD_NAMES`` (fields excluded entirely because a caller 

102re-synthesises them), these are ordinary per-snippet fields that must never be 

103promoted to the shared section: extra args are snippet-specific CLI flags, so 

104merging them would silently apply one snippet's flags to another's command.

108async def resolve_snippets(filenames: Sequence[str]) -> dict[str, SnippetScript]: 

109 """Resolve several snippet filenames against the shared snippet source at once. 

110 

111 The batch entry point the ATW call sites resolve their selections through, 

112 binding them to the framework's single resolution entry point so they inherit 

113 its traversal guard, order-preserving dedup, and one-query batch load. The 

114 result carries only the filenames that resolved, leaving each call site free to 

115 turn a missing filename into its own failure (a whole-request 404 for the 

116 schema form, a per-item error for the execute loop). 

117 

118 :param filenames: The requested snippet filenames, possibly with duplicates. 

119 :return: A mapping of each resolved filename to its :class:`SnippetScript`. 

120 :raises HTTPBadRequestException: When any filename is unsafe or malformed, 

121 failing the whole request before any item is dispatched. 

122 """ 

123 return await resolve_scripts(snippet_source, filenames)

266def parameter_fields(script: SnippetScript) -> list[AnyField]: 

267 """Return a snippet's parameter fields, without the synthetic execution ones. 

268 

269 The per-snippet schema appends an executor-host selector, a sudo toggle, a 

270 script-preview pane, and — for a snippet that opts in — an Extra Args input to 

271 the frontmatter parameters. The first three are batch-level or presentational, 

272 so a merged batch form owns them once (or not at all) rather than repeating 

273 them per snippet; Extra Args stays because it is per-snippet by nature. 

274 

275 :param script: The resolved snippet whose form schema is flattened. 

276 :return: Every parameter field the snippet declares, in schema order, plus the 

277 synthesized Extra Args field when the snippet opts into it. 

278 """ 

279 return [ 

280 field 

281 for section in snippet_source.build_form_schema(script).forms 

282 for field in section.fields 

283 if field.name not in _SYNTHETIC_FIELD_NAMES 

284 ]

287def batch_execution_fields(scripts: list[SnippetScript]) -> list[AnyField]: 

288 """Build the batch-level execution fields the whole selection shares. 

289 

290 These mirror :class:`ATWBatchExecuteWrite`'s own ``executor_host`` / ``sudo`` 

291 inputs, so they are shared by construction rather than by the merge rule. One 

292 toggle drives the whole batch, so it starts checked only when *every* 

293 optional-sudo snippet in the selection would start checked on its own form — 

294 a selection that disagrees falls back to unchecked rather than silently 

295 escalating the snippets that default to no sudo. 

296 

297 :param scripts: The resolved snippets the batch form covers. 

298 :return: The executor-host field, plus a sudo toggle when at least one 

299 selected snippet leaves sudo to the caller. 

300 """ 

301 fields = [ 

302 cast( 

303 AnyField, 

304 HostField( 

305 name=EXECUTOR_HOST_FIELD_NAME, 

306 label=EXECUTION_HOST_LABEL, 

307 required=True, 

308 ), 

309 ) 

310 ] 

311 optional_sudo = [script for script in scripts if script.snippet.sudo.is_optional] 

312 if optional_sudo: 

313 fields.append( 

314 cast( 

315 AnyField, 

316 BoolField( 

317 name=SUDO_FIELD_NAME, 

318 label="Run with sudo", 

319 default=all( 

320 script.snippet.sudo.sudo_default for script in optional_sudo 

321 ), 

322 description=( 

323 "Prepend sudo to the interpreter for every snippet in the " 

324 "batch that leaves sudo optional." 

325 ), 

326 ), 

327 ) 

328 ) 

329 return fields

332def shared_field_names(declarations: dict[str, list[AnyField]]) -> set[str]: 

333 """Return the parameter names that merge into the batch's shared section. 

334 

335 A parameter merges only when two or more selected snippets declare it *and* 

336 every declaration serialises identically — the wire form is what the renderer 

337 consumes, so byte-identity there is the sharing contract. Cosmetically similar 

338 but differing declarations (a per-product default, a required-vs-optional 

339 divergence) stay per-snippet, where they mean different things. A name in 

340 ``NON_SHAREABLE_FIELD_NAMES`` never merges, however unanimous its declarations. 

341 

342 :param declarations: Every declaration of each parameter name, keyed by name. 

343 :return: The names whose declarations are unanimous across two or more snippets. 

344 """ 

345 shared = set() 

346 for name, fields in declarations.items(): 

347 if name in NON_SHAREABLE_FIELD_NAMES or len(fields) < _MIN_SHARED_DECLARERS: 

348 continue 

349 dumps = [field.model_dump(by_alias=True) for field in fields] 

350 if all(dump == dumps[0] for dump in dumps[1:]): 

351 shared.add(name) 

352 return shared

355async def dispatch_batch_item( 

356 body: ATWBatchExecuteWrite, 

357 item: ATWBatchExecuteItemWrite, 

358 script: SnippetScript, 

359 tasks_api: RemoteAPI, 

360 *, 

361 execution_task_name: str | None = None, 

362) -> ScriptExecutionResponse: 

363 """Narrow the shared args to one already-resolved batch item and dispatch it. 

364 

365 Shared arguments are filtered to the parameters the snippet actually declares 

366 and excludes ``NON_SHAREABLE_FIELD_NAMES``, so a batch may offer a value no 

367 single snippet accepts (or one no snippet may share, like Extra Args), and the 

368 item's own ``args`` then override what remains. The snippet is resolved once 

369 for the whole batch by the caller and handed in, so a repeated filename costs 

370 one lookup. 

371 

372 :param body: The batch payload supplying the executor host, sudo choice, and 

373 shared arguments. 

374 :param item: The item naming its own argument overrides. 

375 :param script: The snippet resolved for ``item.snippet_filename``. 

376 :param tasks_api: The authenticated Tasks API client. 

377 :param execution_task_name: ATW's proxy for this snippet's interpreter, resolved 

378 once per batch by the caller. ``None`` dispatches under the interpreter 

379 unchanged, which is the documented degradation when the proxy cannot be used. 

380 :return: The dispatched task name, the created task-history id (``None`` when 

381 the Tasks API returned none), and the resolved snippet filename. 

382 :raises HTTPException: When the snippet's arguments fail validation, it is not 

383 executable, or the Tasks API returns an error status. 

384 :raises OSError: Propagated from ``execute_script`` when the Tasks API 

385 transport itself fails. 

386 """ 

387 declared = { 

388 field.name 

389 for field in parameter_fields(script) 

390 if field.name not in NON_SHAREABLE_FIELD_NAMES 

391 } 

392 args = {name: value for name, value in body.shared_args.items() if name in declared} 

393 args.update(item.args) 

394 return await execute_script( 

395 snippet_source, 

396 script, 

397 ScriptExecuteWrite(executor_host=body.executor_host, sudo=body.sudo, args=args), 

398 tasks_api, 

399 execution_task_name=execution_task_name, 

400 )

403async def fetch_task_history( 

404 tasks_api: RemoteAPI, task_history_id: int 

405) -> dict[str, Any]: 

406 """Fetch one task-history row, degrading an upstream failure to no data. 

407 

408 Mirrors the topology page's fan-out over the same endpoint, but keeps the page 

409 alive when a single row cannot be hydrated: a deleted or unreachable execution 

410 should blank that row's live fields, not fail the whole listing. 

411 

412 :param tasks_api: The authenticated Tasks API client. 

413 :param task_history_id: The execution to look up. 

414 :return: The upstream task-history payload, or an empty mapping on failure. 

415 """ 

416 try: 

417 return as_json_object(await tasks_api.get(f"/history/{task_history_id}")) 

418 except (HTTPException, OSError): 

419 logger.warning(ATW_HYDRATION_WARNING, task_history_id, exc_info=True) 

420 return {}
