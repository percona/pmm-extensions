111def build_derived_payload( 

112 parent_payload: dict[str, Any], derived_spec: DerivedTask 

113) -> dict[str, Any]: 

114 """Build the cascade payload for one derived task. 

115 

116 Deep-copies ``parent_payload``, drops the parent's ``RESERVED_FORM_KEY`` 

117 create-form stamp from the child's ``data`` (a derived task never prefills an 

118 edit form, so it must not carry the parent's stamp), suffixes ``name`` with 

119 ``derived_spec.name_suffix``, applies ``arg_substitutions`` literally to 

120 ``data["meta"]["args"]`` (when present and string-typed), applies 

121 ``payload_substitutions`` literally to ``data["payload"]`` (when present 

122 and string-typed), assigns each ``data_overrides`` entry directly onto 

123 ``data`` in iteration order, and sets ``data["parent"]`` when 

124 ``parent_link`` is true. The caller's ``parent_payload`` is never mutated. 

125 

126 :param parent_payload: The parent task's serialised payload (typically 

127 ``parent.model_dump()``). 

128 :param derived_spec: The declarative spec for this derived task. 

129 :return: A new dict ready to POST or PUT as a task payload. 

130 """ 

131 payload = copy.deepcopy(parent_payload) 

132 child_data = payload.get("data") 

133 if isinstance(child_data, dict): 133 ↛ 135line 133 didn't jump to line 135 because the condition on line 133 was always true

134 child_data.pop(RESERVED_FORM_KEY, None) 

135 parent_name = payload["name"] 

136 payload["name"] = f"{parent_name}{derived_spec.name_suffix}" 

137 if ( 

138 derived_spec.arg_substitutions 

139 or derived_spec.payload_substitutions 

140 or derived_spec.data_overrides 

141 or derived_spec.parent_link 

142 ): 

143 data = payload.setdefault("data", {}) 

144 _apply_arg_substitutions(data, derived_spec.arg_substitutions) 

145 _apply_payload_substitutions(data, derived_spec.payload_substitutions) 

146 if derived_spec.data_overrides: 

147 for key, value in derived_spec.data_overrides.items(): 

148 data[key] = value 

149 if derived_spec.parent_link: 149 ↛ 151line 149 didn't jump to line 151 because the condition on line 149 was always true

150 data["parent"] = parent_name 

151 return payload

154def _apply_arg_substitutions( 

155 data: dict[str, Any], substitutions: dict[str, str] | None 

156) -> None: 

157 """Apply literal ``str.replace`` substitutions to ``data["meta"]["args"]`` in place. 

158 

159 No-op when ``substitutions`` is falsy, when ``data["meta"]`` is not a 

160 dict, or when ``data["meta"]["args"]`` is not a string. 

161 

162 :param data: The derived payload's ``data`` mapping, mutated in place. 

163 :type data: dict[str, Any] 

164 :param substitutions: Ordered mapping of ``(old, new)`` literal 

165 substring pairs, applied in dict insertion order. 

166 :type substitutions: dict[str, str] | None 

167 """ 

168 if not substitutions: 

169 return 

170 meta = data.get("meta") 

171 if not isinstance(meta, dict) or not isinstance(meta.get("args"), str): 

172 return 

173 args = meta["args"] 

174 for old, new in substitutions.items(): 

175 args = args.replace(old, new) 

176 meta["args"] = args

179def _apply_payload_substitutions( 

180 data: dict[str, Any], substitutions: dict[str, str] | None 

181) -> None: 

182 """Apply literal ``str.replace`` substitutions to ``data["payload"]`` in place. 

183 

184 No-op when ``substitutions`` is falsy or when ``data["payload"]`` is 

185 not a string. 

186 

187 :param data: The derived payload's ``data`` mapping, mutated in place. 

188 :type data: dict[str, Any] 

189 :param substitutions: Ordered mapping of ``(old, new)`` literal 

190 substring pairs, applied in dict insertion order. 

191 :type substitutions: dict[str, str] | None 

192 """ 

193 if not substitutions: 

194 return 

195 payload_path = data.get("payload") 

196 if not isinstance(payload_path, str): 

197 return 

198 for old, new in substitutions.items(): 

199 payload_path = payload_path.replace(old, new) 

200 data["payload"] = payload_path

203def require_addressable_names(payloads: Iterable[Mapping[str, Any]]) -> None: 

204 """Refuse the whole cascade unless every planned name is a plain path segment. 

205 

206 The Tasks API takes a task's name in the request body, where nothing narrows 

207 it to one path segment, but every later request addressing that task composes 

208 the name into a path. A create checked only at its rollback DELETE would leave 

209 a task behind, because that DELETE's failure is logged rather than raised; an 

210 update carries the *existing* name in its PUT path, so the guard there never 

211 sees the new one, and each leg collects its own exception into the result 

212 rather than raising. Both make the check belong before the first request. 

213 

214 :param payloads: The serialised payloads of every task the cascade will write, 

215 parent first. 

216 :raises HTTPUnprocessableEntityException: If any planned name is not a single 

217 plain path segment. 

218 """ 

219 for payload in payloads: 

220 require_one_path_segment(payload["name"])

223async def cascade_create_tasks( 

224 tasks_api: RemoteAPI, 

225 parent_payload: dict[str, Any], 

226 derived_specs: Sequence[DerivedTask], 

227) -> None: 

228 """POST the parent then each derived task; roll back on any failure. 

229 

230 Every task is POSTed to the Tasks API root (``/``), matching the 

231 project-wide TaskAPI convention. On any POST failure, already-created 

232 tasks are deleted in reverse creation order via ``DELETE /{task_name}``. 

233 A rollback DELETE that itself fails is logged at WARNING and the 

234 rollback loop continues; the original POST exception is what surfaces 

235 to the caller. 

236 

237 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

238 :param parent_payload: The parent task's serialised payload. 

239 :param derived_specs: The list of derived-task specs to cascade. 

240 :raises HTTPUnprocessableEntityException: If any planned task name is 

241 not a single plain path segment. Raised before the first POST, so no 

242 task is created. 

243 :raises Exception: Re-raises whatever exception ``tasks_api.post`` 

244 produced (commonly :class:`fastapi.HTTPException` for non-2xx 

245 responses, but any transport-level error such as 

246 :class:`aiohttp.ClientError` or :class:`asyncio.TimeoutError` can 

247 propagate) after the rollback DELETEs complete. 

248 """ 

249 child_payloads = [ 

250 build_derived_payload(parent_payload, spec) for spec in derived_specs 

251 ] 

252 require_addressable_names([parent_payload, *child_payloads]) 

253 created_names = [] 

254 try: 

255 await tasks_api.post("/", json=parent_payload) 

256 created_names.append(parent_payload["name"]) 

257 for child_payload in child_payloads: 

258 await tasks_api.post("/", json=child_payload) 

259 created_names.append(child_payload["name"]) 

260 except Exception: 

261 for task_name in reversed(created_names): 

262 try: 

263 await tasks_api.delete(task_path(task_name)) 

264 except Exception as rollback_exc: # noqa: BLE001 

265 logger.warning( 

266 "Rollback DELETE failed for %r during cascade_create rollback: %s", 

267 task_name, 

268 rollback_exc, 

269 ) 

270 raise

273async def cascade_update_tasks( 

274 tasks_api: RemoteAPI, 

275 parent_existing_name: str, 

276 parent_updated: dict[str, Any], 

277 derived_existing_names: Sequence[str], 

278 derived_specs: Sequence[DerivedTask], 

279) -> CascadeResult: 

280 """PUT the parent and each derived task, best-effort. 

281 

282 Each leg is attempted independently when the parent payload preserves 

283 the parent's existing ``name`` — per-leg failures are collected into the 

284 returned :class:`CascadeResult`. The PUT URL path uses the *existing* 

285 task name (per the Tasks API contract); the updated payload, including 

286 any new ``name``, goes in the body. 

287 

288 When ``parent_updated["name"]`` differs from ``parent_existing_name`` 

289 (a rename) **and** the parent PUT fails, the derived loop is **skipped** 

290 rather than executed best-effort: each derived child payload built from 

291 ``parent_updated`` would carry the new ``data["parent"]`` link, so 

292 applying it would point a successfully-PUT child at a parent task that 

293 was never renamed and does not exist under that name. The skipped legs 

294 surface as :class:`CascadeFailure` entries so the caller can observe 

295 them. 

296 

297 The caller is responsible for ensuring ``derived_existing_names`` matches 

298 ``derived_specs`` in length and order — the function rejects a mismatch 

299 with :class:`ValueError` rather than silently zipping to the shorter list. 

300 

301 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

302 :param parent_existing_name: The current name of the parent task (used in 

303 the PUT URL; any rename goes in ``parent_updated["name"]``). 

304 :param parent_updated: The updated parent payload (may include a new 

305 ``name``). 

306 :param derived_existing_names: Current derived task names, aligned with 

307 ``derived_specs`` by index. 

308 :param derived_specs: The derived-task specs to cascade. 

309 :return: A :class:`CascadeResult` recording per-leg outcomes. 

310 :raises ValueError: When ``len(derived_existing_names) != len(derived_specs)``. 

311 :raises HTTPUnprocessableEntityException: If the rename, or a derived name 

312 built from it, is not a single plain path segment. Raised before the 

313 first PUT, so no leg is renamed. 

314 """ 

315 if len(derived_existing_names) != len(derived_specs): 

316 raise ValueError( 

317 f"derived_existing_names length {len(derived_existing_names)} " 

318 f"does not match derived_specs length {len(derived_specs)}" 

319 ) 

320 child_payloads = [ 

321 build_derived_payload(parent_updated, spec) for spec in derived_specs 

322 ] 

323 require_addressable_names([parent_updated, *child_payloads]) 

324 result = CascadeResult() 

325 parent_failed = False 

326 try: 

327 await tasks_api.put(task_path(parent_existing_name), json=parent_updated) 

328 result.successes.append(parent_updated["name"]) 

329 except Exception as exc: # noqa: BLE001 

330 result.failures.append(CascadeFailure(parent_existing_name, exc)) 

331 parent_failed = True 

332 parent_renamed = parent_updated["name"] != parent_existing_name 

333 if parent_failed and parent_renamed: 

334 for existing_name in derived_existing_names: 

335 result.failures.append( 

336 CascadeFailure( 

337 existing_name, 

338 RuntimeError( 

339 "Skipped derived PUT because the parent rename failed; " 

340 "applying it would orphan the derived task under a " 

341 "non-existent parent name." 

342 ), 

343 ) 

344 ) 

345 return result 

346 for existing_name, child_payload in zip( 

347 derived_existing_names, child_payloads, strict=True 

348 ): 

349 try: 

350 await tasks_api.put(task_path(existing_name), json=child_payload) 

351 result.successes.append(child_payload["name"]) 

352 except Exception as exc: # noqa: BLE001 

353 result.failures.append(CascadeFailure(existing_name, exc)) 

354 return result

357async def cascade_delete_tasks( 

358 tasks_api: RemoteAPI, 

359 parent_name: str, 

360 derived_names: Sequence[str], 

361) -> CascadeResult: 

362 """DELETE every derived task first, then the parent, best-effort. 

363 

364 The caller passes the *actual* derived task names rather than the 

365 declarative :class:`DerivedTask` specs: the cascade does not recompute 

366 names from ``parent_name`` + ``name_suffix`` because a previous partial 

367 rename could leave the stored derived names out of sync with the 

368 schema-suffix convention, and recomputing would silently 404 on 

369 orphaned children. The caller is expected to fetch the current children 

370 (for example via a ``parent`` foreign key lookup on the Tasks API) 

371 before invoking this helper. 

372 

373 HTTP 404 on any leg is tolerated as success — the desired end state 

374 (the task is absent) is already achieved. All other failures accumulate 

375 into the returned :class:`CascadeResult`. 

376 

377 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

378 :type tasks_api: RemoteAPI 

379 :param parent_name: The name of the parent task. 

380 :type parent_name: str 

381 :param derived_names: The actual stored names of the derived tasks to 

382 delete, fetched by the caller. 

383 :type derived_names: Sequence[str] 

384 :return: A :class:`CascadeResult` recording per-leg outcomes. 

385 :rtype: CascadeResult 

386 """ 

387 result = CascadeResult() 

388 for derived_name in derived_names: 

389 await _delete_one(tasks_api, derived_name, result) 

390 await _delete_one(tasks_api, parent_name, result) 

391 return result

394async def _delete_one( 

395 tasks_api: RemoteAPI, task_name: str, result: CascadeResult 

396) -> None: 

397 """Issue one DELETE, tolerate 404, and record other failures on ``result``. 

398 

399 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

400 :type tasks_api: RemoteAPI 

401 :param task_name: The task name to DELETE. 

402 :type task_name: str 

403 :param result: The :class:`CascadeResult` to update in place. 

404 :type result: CascadeResult 

405 """ 

406 try: 

407 await tasks_api.delete(task_path(task_name)) 

408 result.successes.append(task_name) 

409 except HTTPNotFoundException: 

410 result.successes.append(task_name) 

411 except Exception as exc: # noqa: BLE001 

412 result.failures.append(CascadeFailure(task_name, exc))

415def build_predecessor_payload( 

416 parent_payload: dict[str, Any], 

417 predecessor_payload: dict[str, Any], 

418 predecessor_spec: ChainedPredecessor, 

419) -> dict[str, Any]: 

420 """Build the cascade payload for one chained predecessor. 

421 

422 Deep-copies ``predecessor_payload``, overrides ``name`` with 

423 ``f"{parent_name}{spec.name_suffix}"`` so the schema-declared suffix is 

424 authoritative, and sets ``data["parent"]`` when ``parent_link`` is true. 

425 The caller's payloads are never mutated. 

426 

427 :param parent_payload: The parent task's serialised payload (read-only 

428 here; the parent's ``name`` is used to derive the predecessor name). 

429 :type parent_payload: dict[str, Any] 

430 :param predecessor_payload: The predecessor task's plugin-built payload. 

431 The consuming plugin's POST handler constructs this imperatively. 

432 :type predecessor_payload: dict[str, Any] 

433 :param predecessor_spec: The declarative spec for this predecessor. 

434 :type predecessor_spec: ChainedPredecessor 

435 :return: A new dict ready to POST or PUT as a task payload. 

436 :rtype: dict[str, Any] 

437 """ 

438 payload = copy.deepcopy(predecessor_payload) 

439 parent_name = parent_payload["name"] 

440 payload["name"] = f"{parent_name}{predecessor_spec.name_suffix}" 

441 if predecessor_spec.parent_link: 

442 payload.setdefault("data", {})["parent"] = parent_name 

443 return payload

446def build_predecessor_chain_execute_body( 

447 parent_name: str, 

448 predecessor_specs: Sequence[ChainedPredecessor], 

449) -> dict[str, Any]: 

450 """Build the JSON body for ``POST /execute/{first_predecessor_name}``. 

451 

452 The first predecessor in ``predecessor_specs`` is the execute target; 

453 ``chain_task_names`` lists any remaining predecessors followed by the 

454 parent. ``chain_on_failure`` is derived from the first spec's 

455 ``on_failure`` policy (``"continue"`` maps to ``True``, ``"halt"`` to 

456 ``False``). 

457 

458 :param parent_name: The parent task name. 

459 :param predecessor_specs: Ordered predecessor specs (same order as 

460 :func:`cascade_create_predecessors`). 

461 :return: A dict with ``chain_task_names`` and ``chain_on_failure`` ready 

462 for the tasks sub-app execute endpoint. 

463 :raises ValueError: When ``predecessor_specs`` is empty. 

464 """ 

465 if not predecessor_specs: 

466 raise ValueError( 

467 "build_predecessor_chain_execute_body requires at least one " 

468 "predecessor spec." 

469 ) 

470 built_names = [f"{parent_name}{spec.name_suffix}" for spec in predecessor_specs] 

471 return { 

472 "chain_task_names": [*built_names[1:], parent_name], 

473 "chain_on_failure": predecessor_specs[0].on_failure == "continue", 

474 }

477async def cascade_create_predecessors( 

478 tasks_api: RemoteAPI, 

479 parent_payload: dict[str, Any], 

480 predecessor_specs_with_payloads: Sequence[ 

481 tuple[ChainedPredecessor, dict[str, Any]] 

482 ], 

483) -> None: 

484 """POST the parent then every predecessor; roll back on any failure. 

485 

486 Cascade order (parent-first, matching :func:`cascade_create_tasks`): 

487 

488 1. POST the parent to ``/``. 

489 2. POST each predecessor in declared order to ``/``. The payload is 

490 built via :func:`build_predecessor_payload` (``parent_link`` 

491 applied, ``name`` suffixed from the parent's ``name``). 

492 

493 Chain execution is not fired here. When the user (or plugin handler) 

494 starts the first predecessor, build the execute body via 

495 :func:`build_predecessor_chain_execute_body` and call 

496 ``POST /execute/{first_predecessor_name}``. 

497 

498 Every task is POSTed to the Tasks API root (``/``). On any POST 

499 failure, already-created tasks are DELETEd in reverse creation order. 

500 A rollback DELETE that itself fails is logged at WARNING and the 

501 rollback loop continues; the original POST exception is what surfaces 

502 to the caller. 

503 

504 The empty-predecessor case is a programmer error: the consuming 

505 plugin should call :func:`cascade_create_tasks` (or POST the parent 

506 directly) instead. :class:`ValueError` is raised so the misuse 

507 surfaces at test time rather than silently downgrading to a 

508 single-task POST. 

509 

510 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

511 :param parent_payload: The parent task's serialised payload. The 

512 ``name`` key must be set before invocation — it is read to derive 

513 the predecessor names in step 2. 

514 :param predecessor_specs_with_payloads: Ordered list of 

515 ``(spec, predecessor_payload)`` tuples; the consuming plugin 

516 builds each predecessor payload. 

517 :raises ValueError: When ``predecessor_specs_with_payloads`` is empty. 

518 :raises HTTPUnprocessableEntityException: If any planned task name is 

519 not a single plain path segment. Raised before the first POST, so no 

520 task is created. 

521 :raises Exception: Re-raises whatever ``tasks_api.post`` produced 

522 (commonly :class:`fastapi.HTTPException`, but any transport-level 

523 error such as :class:`asyncio.TimeoutError` can propagate) after 

524 rollback DELETEs complete. 

525 """ 

526 if not predecessor_specs_with_payloads: 

527 raise ValueError( 

528 "cascade_create_predecessors requires at least one predecessor; " 

529 "callers must not invoke this helper for schemas without predecessors." 

530 ) 

531 built_payloads = [ 

532 build_predecessor_payload(parent_payload, pred_payload, spec) 

533 for spec, pred_payload in predecessor_specs_with_payloads 

534 ] 

535 require_addressable_names([parent_payload, *built_payloads]) 

536 created_names = [] 

537 try: 

538 await tasks_api.post("/", json=parent_payload) 

539 created_names.append(parent_payload["name"]) 

540 for built in built_payloads: 

541 await tasks_api.post("/", json=built) 

542 created_names.append(built["name"]) 

543 except Exception: 

544 for task_name in reversed(created_names): 

545 try: 

546 await tasks_api.delete(task_path(task_name)) 

547 except Exception as rollback_exc: # noqa: BLE001 

548 logger.warning( 

549 "Rollback DELETE failed for %r during " 

550 "cascade_create_predecessors rollback: %s", 

551 task_name, 

552 rollback_exc, 

553 ) 

554 raise

557async def cascade_create_independent_tasks( 

558 tasks_api: RemoteAPI, 

559 parent_payload: dict[str, Any], 

560 child_payloads: Sequence[dict[str, Any]], 

561) -> None: 

562 """POST the parent then each independent child; roll back on any failure. 

563 

564 Unlike :func:`cascade_create_tasks`, the children are not derived from 

565 the parent via substitutions — each ``child_payload`` is fully built 

566 by the caller. Unlike :func:`cascade_create_predecessors`, children 

567 are not named via :func:`build_predecessor_payload` or linked with 

568 ``data["parent"]``. 

569 

570 Every task is POSTed to the Tasks API root (``/``). On any POST 

571 failure, already-created tasks are DELETEd in reverse creation 

572 order. A rollback DELETE that itself fails is logged at WARNING 

573 and the rollback loop continues; the original POST exception is 

574 what surfaces to the caller. 

575 

576 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

577 :param parent_payload: The parent task's serialised payload. 

578 :param child_payloads: The list of independently-built child 

579 payloads, POSTed in declared order. 

580 :raises HTTPUnprocessableEntityException: If any planned task name is 

581 not a single plain path segment. Raised before the first POST, so no 

582 task is created. 

583 :raises Exception: Re-raises whatever exception ``tasks_api.post`` 

584 produced after the rollback DELETEs complete. 

585 """ 

586 require_addressable_names([parent_payload, *child_payloads]) 

587 created_names: list[str] = [] 

588 try: 

589 await tasks_api.post("/", json=parent_payload) 

590 created_names.append(parent_payload["name"]) 

591 for child in child_payloads: 

592 await tasks_api.post("/", json=child) 

593 created_names.append(child["name"]) 

594 except Exception: 

595 for task_name in reversed(created_names): 

596 try: 

597 await tasks_api.delete(task_path(task_name)) 

598 except Exception as rollback_exc: # noqa: BLE001 

599 logger.warning( 

600 "Rollback DELETE failed for %r during " 

601 "cascade_create_independent rollback: %s", 

602 task_name, 

603 rollback_exc, 

604 ) 

605 raise

608async def cascade_update_predecessors( 

609 tasks_api: RemoteAPI, 

610 parent_existing_name: str, 

611 parent_updated: dict[str, Any], 

612 predecessor_existing_names: Sequence[str], 

613 predecessor_specs_with_payloads: Sequence[ 

614 tuple[ChainedPredecessor, dict[str, Any]] 

615 ], 

616) -> CascadeResult: 

617 """PUT the parent and each predecessor, best-effort. 

618 

619 Each leg is attempted independently — per-leg failures are collected 

620 into the returned :class:`CascadeResult`. The PUT URL path uses the 

621 *existing* task name; the updated payload goes in the body. 

622 

623 Renames are rejected upfront with :class:`ValueError`. The chain 

624 wiring fired at create time stores the task names verbatim in celery's 

625 execution request, and this helper does NOT re-fire ``POST /execute``. 

626 Renaming the parent or any predecessor would leave the stored chain 

627 referencing names that no longer resolve, so the next chained 

628 execution would 404. Consumers that need to rename tasks in a chain 

629 must tear down and rebuild it (delete-then-create cascade) rather 

630 than relying on this helper. 

631 

632 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

633 :type tasks_api: RemoteAPI 

634 :param parent_existing_name: The current name of the parent task 

635 (used in the PUT URL). 

636 :type parent_existing_name: str 

637 :param parent_updated: The updated parent payload. ``parent_updated["name"]`` 

638 MUST equal ``parent_existing_name``. 

639 :type parent_updated: dict[str, Any] 

640 :param predecessor_existing_names: Current predecessor task names, 

641 aligned with ``predecessor_specs_with_payloads`` by index. Each 

642 name MUST equal the name produced by 

643 :func:`build_predecessor_payload` for the matching spec. 

644 :type predecessor_existing_names: Sequence[str] 

645 :param predecessor_specs_with_payloads: Ordered list of 

646 ``(spec, predecessor_payload)`` tuples. 

647 :type predecessor_specs_with_payloads: 

648 Sequence[tuple[ChainedPredecessor, dict[str, Any]]] 

649 :return: A :class:`CascadeResult` recording per-leg outcomes. 

650 :rtype: CascadeResult 

651 :raises ValueError: When 

652 ``len(predecessor_existing_names) != len(predecessor_specs_with_payloads)``, 

653 when ``parent_updated["name"] != parent_existing_name``, or when 

654 any built predecessor name differs from the matching existing name. 

655 """ 

656 if len(predecessor_existing_names) != len(predecessor_specs_with_payloads): 

657 raise ValueError( 

658 f"predecessor_existing_names length {len(predecessor_existing_names)} " 

659 f"does not match predecessor_specs_with_payloads length " 

660 f"{len(predecessor_specs_with_payloads)}" 

661 ) 

662 if parent_updated["name"] != parent_existing_name: 

663 raise ValueError( 

664 "cascade_update_predecessors does not support renaming the parent: " 

665 f"parent_updated['name']={parent_updated['name']!r} differs from " 

666 f"parent_existing_name={parent_existing_name!r}. The chain wired at " 

667 "create time stores task names verbatim and this helper does not " 

668 "re-fire POST /execute; rename via delete-then-create cascade instead." 

669 ) 

670 built_predecessors = [ 

671 (existing_name, build_predecessor_payload(parent_updated, pred_payload, spec)) 

672 for existing_name, (spec, pred_payload) in zip( 

673 predecessor_existing_names, 

674 predecessor_specs_with_payloads, 

675 strict=True, 

676 ) 

677 ] 

678 renamed_predecessors = [ 

679 (existing_name, built["name"]) 

680 for existing_name, built in built_predecessors 

681 if built["name"] != existing_name 

682 ] 

683 if renamed_predecessors: 

684 raise ValueError( 

685 "cascade_update_predecessors does not support renaming predecessors: " 

686 f"{renamed_predecessors!r}. The chain wired at create time stores " 

687 "task names verbatim and this helper does not re-fire POST /execute; " 

688 "rename via delete-then-create cascade instead." 

689 ) 

690 result = CascadeResult() 

691 try: 

692 await tasks_api.put(task_path(parent_existing_name), json=parent_updated) 

693 result.successes.append(parent_updated["name"]) 

694 except Exception as exc: # noqa: BLE001 

695 result.failures.append(CascadeFailure(parent_existing_name, exc)) 

696 for existing_name, built in built_predecessors: 

697 try: 

698 await tasks_api.put(task_path(existing_name), json=built) 

699 result.successes.append(built["name"]) 

700 except Exception as exc: # noqa: BLE001 

701 result.failures.append(CascadeFailure(existing_name, exc)) 

702 return result

705async def cascade_delete_predecessors( 

706 tasks_api: RemoteAPI, 

707 parent_name: str, 

708 predecessor_names: Sequence[str], 

709) -> CascadeResult: 

710 """DELETE every predecessor first, then the parent, best-effort. 

711 

712 Identical contract to :func:`cascade_delete_tasks` (just named for 

713 the predecessor schema field). The caller passes the *actual* stored 

714 predecessor names — the cascade does not recompute names from 

715 ``name_suffix`` because a previous partial rename could leave the 

716 stored names out of sync with the schema-suffix convention, and 

717 recomputing would silently 404 on orphaned children. 

718 

719 HTTP 404 on any leg is tolerated as success; all other failures 

720 accumulate into the returned :class:`CascadeResult`. Uses the shared 

721 :func:`_delete_one` helper. 

722 

723 :param tasks_api: The :class:`RemoteAPI` for the tasks sub-app. 

724 :type tasks_api: RemoteAPI 

725 :param parent_name: The name of the parent task. 

726 :type parent_name: str 

727 :param predecessor_names: The actual stored names of the predecessor 

728 tasks, fetched by the caller. 

729 :type predecessor_names: Sequence[str] 

730 :return: A :class:`CascadeResult` recording per-leg outcomes. 

731 :rtype: CascadeResult 

732 """ 

733 result = CascadeResult() 

734 for predecessor_name in predecessor_names: 

735 await _delete_one(tasks_api, predecessor_name, result) 

736 await _delete_one(tasks_api, parent_name, result) 

737 return result
