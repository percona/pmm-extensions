16"""Gate plugin-owned Celery periodic tasks by app state. 

17 

18A schedule's ``effective_enabled`` is 

19``(AppState.lifecycle_state == ENABLED) AND user_enabled``. This 

20module recomputes it for plugin-owned tasks and writes it through to the library 

21``sqlalchemy_celery_beat.PeriodicTask.enabled`` column, the single field the beat 

22scheduler filters on. The write is an ORM-instance mutation, so the library's 

23``after_update`` listener bumps ``PeriodicTaskChanged.last_update`` and the 

24running scheduler reloads without a restart. 

25 

26The work spans two databases — ``AppState`` and the wrapper rows live in the PMM Extensions 

27database, while ``PeriodicTask`` lives in the celery-beat database — so it cannot 

28ride on a single manager ``save()``. It is instead invoked from the two 

29enumerated writers of ``AppState.lifecycle_state``: startup seeding 

30(:func:`app.extensions.db.seed.init_extensions_db`) and the runtime toggle endpoint. 

31 

32The module carries a second, narrower concern with the same shape: a *user* 

33schedule whose task belongs to an app that does not offer scheduling at all. Such 

34a schedule predates the app withdrawing the capability, and the gateway guard only 

35refuses new writes, so :func:`disable_unschedulable_task_schedules` switches the 

36stored ones off once per PMM Extensions startup. It reads a third database — the Tasks one, 

37which owns the ``Task.owner`` those schedules resolve against. 

38"""

70async def seed_app_periodic_task_rows( 

71 session: AsyncSession, system_tasks: list[SystemPeriodicTaskSchedule] 

72) -> None: 

73 """Upsert one wrapper row per plugin-owned schedule and delete orphans. 

74 

75 Idempotent: ``get_or_create`` never overwrites an operator-set 

76 ``user_enabled``, so a disabled-by-the-user schedule stays disabled across 

77 restarts. 

78 

79 :param session: The PMM Extensions database session. 

80 :param system_tasks: The system periodic-task schedules to seed from. 

81 """ 

82 owned = [ 

83 task 

84 for schedule in system_tasks 

85 for task in schedule.tasks 

86 if task.owner_app_key 

87 ] 

88 for task in owned: 

89 await ExtensionsAppPeriodicTaskManager.get_or_create( 

90 session, 

91 ExtensionsAppPeriodicTaskBase( 

92 periodic_task_name=task.name, app_key=task.owner_app_key 

93 ), 

94 filter_include={"periodic_task_name"}, 

95 ) 

96 owned_names = [task.name for task in owned] 

97 await ExtensionsAppPeriodicTaskManager.delete_where( 

98 session, col(ExtensionsAppPeriodicTask.periodic_task_name).not_in(owned_names) 

99 )

102async def apply_effective_enabled( 

103 extensions_session: AsyncSession, 

104 celery_beat_session: AsyncSession, 

105 *, 

106 app_keys: Collection[str] | None = None, 

107) -> None: 

108 """Recompute ``effective_enabled`` and write it to ``PeriodicTask.enabled``. 

109 

110 ``effective_enabled`` is 

111 ``(AppState.lifecycle_state == ENABLED) AND user_enabled``. A missing 

112 ``AppState`` row is treated as enabled, mirroring 

113 :meth:`app.extensions.crud.AppStateManager.is_enabled`. The write is an ORM-instance 

114 mutation guarded by an equality check, so the library reload signal only 

115 fires when a value actually changes. 

116 

117 :param extensions_session: The PMM Extensions database session (``AppState`` + wrapper rows). 

118 :param celery_beat_session: The celery-beat database session (``PeriodicTask``). 

119 :param app_keys: Restrict the sweep to these app keys, or ``None`` for all. 

120 """ 

121 rows = await ExtensionsAppPeriodicTaskManager.for_app_keys( 

122 extensions_session, app_keys 

123 ) 

124 if not rows: 

125 return 

126 lifecycle_state = await AppStateManager.all_lifecycle_states(extensions_session) 

127 tasks = await BasePeriodicTaskManager.list( 

128 celery_beat_session, 

129 PeriodicTask.name.in_([row.periodic_task_name for row in rows]), 

130 ) 

131 tasks_by_name = {task.name: task for task in tasks} 

132 changed = False 

133 for row in rows: 

134 task = tasks_by_name.get(row.periodic_task_name) 

135 if task is None: 

136 continue 

137 effective = ( 

138 lifecycle_state.get(row.app_key, AppLifecycleEnum.ENABLED) 

139 == AppLifecycleEnum.ENABLED 

140 ) and row.user_enabled 

141 if task.enabled != effective: 

142 task.enabled = effective 

143 celery_beat_session.add(task) 

144 changed = True 

145 if changed: 

146 await celery_beat_session.commit()

149async def release_unowned_task_gating( 

150 celery_beat_session: AsyncSession, 

151 system_tasks: list[SystemPeriodicTaskSchedule], 

152) -> None: 

153 """Enable seeded ``PeriodicTask`` rows for schedules that carry no owner. 

154 

155 An unowned schedule gets no ``ExtensionsAppPeriodicTask`` wrapper row, so it has 

156 no user toggle and ``enabled`` is a column only app gating ever writes. 

157 :func:`apply_effective_enabled` iterates wrapper rows, which means a schedule 

158 that *loses* its owner keeps whatever bit the previous regime left behind and 

159 nothing ever writes it back — permanently off on an instance whose former 

160 owning app was disabled at the time. Re-assert the only correct value for a 

161 schedule nothing gates. 

162 

163 :param celery_beat_session: The celery-beat database session. 

164 :param system_tasks: The system periodic-task schedules to seed from. 

165 """ 

166 unowned = [ 

167 task.name 

168 for schedule in system_tasks 

169 for task in schedule.tasks 

170 if not task.owner_app_key 

171 ] 

172 if not unowned: 

173 return 

174 tasks = await BasePeriodicTaskManager.list( 

175 celery_beat_session, col(PeriodicTask.name).in_(unowned) 

176 ) 

177 changed = False 

178 for task in tasks: 

179 if not task.enabled: 

180 task.enabled = True 

181 celery_beat_session.add(task) 

182 changed = True 

183 if changed: 

184 await celery_beat_session.commit()

187async def sync_app_periodic_task_gating( 

188 system_tasks: list[SystemPeriodicTaskSchedule], 

189) -> None: 

190 """Seed wrapper rows and apply ``effective_enabled`` for every owned task. 

191 

192 Startup entry point: opens one PMM Extensions session and one celery-beat session, 

193 upserts the wrapper rows, releases any schedule that no longer has an owner, 

194 then writes ``effective_enabled`` through for all owned schedules. 

195 

196 :param system_tasks: The system periodic-task schedules to gate. 

197 """ 

198 extensions_session_maker = get_extensions_session_maker() 

199 celery_beat_session_maker = get_celery_beat_session_maker() 

200 async with ( 

201 extensions_session_maker() as extensions_session, 

202 celery_beat_session_maker() as celery_beat_session, 

203 ): 

204 await seed_app_periodic_task_rows(extensions_session, system_tasks) 

205 await release_unowned_task_gating(celery_beat_session, system_tasks) 

206 await apply_effective_enabled(extensions_session, celery_beat_session)

209async def _collect_owned_task_names( 

210 session: AsyncSession, owners: Collection[str] 

211) -> set[str]: 

212 """Collect the names of the active tasks owned by the given apps. 

213 

214 Loads only ``Task.name``, one keyset batch at a time, so no row's ``data`` 

215 payload is transferred: schedules are matched by name alone. 

216 

217 :param session: The Tasks database session. 

218 :param owners: The ``Task.owner`` values whose tasks to collect. 

219 :return: The names of every active task owned by ``owners``. 

220 """ 

221 owned_names: set[str] = set() 

222 batches = TaskManager.iter_active_batches( 

223 session, 

224 col(Task.owner).in_(owners), 

225 batch_size=ACTIVE_TASK_BATCH_SIZE, 

226 query_options=[load_only(Task.name)], # ty: ignore[invalid-argument-type] 

227 ) 

228 async for batch in batches: 

229 owned_names.update(task.name for task in batch) 

230 return owned_names

233async def _collect_owned_schedules( 

234 session: AsyncSession, owned_names: set[str] 

235) -> list[PeriodicTask]: 

236 """Collect the enabled schedules whose task is one of ``owned_names``. 

237 

238 Pages by primary key rather than by offset so the batches stay disjoint and 

239 exhaustive while other writers insert, delete or switch off schedules, and 

240 holds only the matches across batches. 

241 

242 :param session: The celery-beat database session. 

243 :param owned_names: The task names a schedule must resolve to. 

244 :return: The matching enabled schedules, in ascending id order. 

245 """ 

246 schedules: list[PeriodicTask] = [] 

247 last_schedule_id = 0 

248 while True: 

249 candidates = await PeriodicTaskManager.list( 

250 session, 

251 col(PeriodicTask.id) > last_schedule_id, 

252 enabled=True, 

253 order_by=[col(PeriodicTask.id)], 

254 limit=SCHEDULE_BATCH_SIZE, 

255 ) 

256 for candidate in candidates: 

257 task_name = resolve_schedule_task_name(candidate) 

258 if task_name is None: 

259 logger.warning( 

260 "Skipped periodic task %r: its args/kwargs do not name a task.", 

261 candidate.name, 

262 ) 

263 elif task_name in owned_names: 

264 schedules.append(candidate) 

265 if len(candidates) < SCHEDULE_BATCH_SIZE: 

266 break 

267 last_schedule_id = candidates[-1].id 

268 return schedules

271async def disable_schedules_for_owners( 

272 tasks_session: AsyncSession, 

273 celery_beat_session: AsyncSession, 

274 owners: Collection[str], 

275) -> list[str]: 

276 """Switch off every enabled schedule whose task is owned by one of ``owners``. 

277 

278 Resolve the task each enabled ``execute_task_by_name`` schedule runs from its 

279 ``args``/``kwargs`` the way the Tasks service reads them, and match it against 

280 the active tasks under ``owners``. A row whose arguments cannot be read is 

281 skipped with a warning, so one corrupt schedule cannot stop the rest from being 

282 switched off. The write is an ORM-instance mutation, so the library's 

283 ``after_update`` listener bumps ``PeriodicTaskChanged.last_update`` and a 

284 running scheduler reloads without a restart. Only enabled rows are selected, so 

285 a second run changes nothing. 

286 

287 :param tasks_session: The Tasks database session (``Task``). 

288 :param celery_beat_session: The celery-beat database session (``PeriodicTask``). 

289 :param owners: The ``Task.owner`` values whose schedules to switch off. 

290 :return: The names of the schedules switched off, empty when none were. 

291 """ 

292 if not owners: 

293 return [] 

294 owned_names = await _collect_owned_task_names(tasks_session, owners) 

295 if not owned_names: 

296 return [] 

297 schedules = await _collect_owned_schedules(celery_beat_session, owned_names) 

298 if not schedules: 

299 return [] 

300 names: list[str] = [ # ty: ignore[invalid-assignment] 

301 schedule.name for schedule in schedules 

302 ] 

303 for schedule in schedules: 

304 schedule.enabled = False # ty: ignore[invalid-assignment] 

305 celery_beat_session.add(schedule) 

306 await celery_beat_session.commit() 

307 for name in names: 

308 logger.warning( 

309 "Switched off periodic task %r: its task's app does not offer scheduling.", 

310 name, 

311 ) 

312 return names

315async def disable_unschedulable_task_schedules() -> None: 

316 """Switch off existing schedules of tasks whose app does not offer scheduling. 

317 

318 Startup entry point. The owners come from the app registry, so no app is named 

319 here and an app absent from the activation list is not swept. No session is 

320 opened when no owner qualifies. 

321 """ 

322 owners = get_app_registry().unschedulable_task_owners() 

323 if not owners: 

324 return 

325 tasks_session_maker = get_tasks_session_maker() 

326 celery_beat_session_maker = get_celery_beat_session_maker() 

327 async with ( 

328 tasks_session_maker() as tasks_session, 

329 celery_beat_session_maker() as celery_beat_session, 

330 ): 

331 await disable_schedules_for_owners(tasks_session, celery_beat_session, owners)
