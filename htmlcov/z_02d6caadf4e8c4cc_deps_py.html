73def validate_snippet_filename(filename: str) -> None: 

74 """Raise 400 if ``filename`` is not a safe snippet filename or path. 

75 

76 Accepts plain filenames (``check.sh``) and safe relative subdirectory 

77 paths (``team/check.sh``) that mirror what ``update_snippets()`` stores 

78 via ``path.relative_to(BASE_DIR)``. Rejects absolute paths, Windows 

79 separators, traversal components, and any component that does not meet 

80 the safe-name rules. 

81 

82 :param filename: The raw filename string to validate. 

83 :type filename: str 

84 :raises HTTPBadRequestException: If the filename is unsafe or malformed. 

85 """ 

86 # Reject any Windows separator — these never appear in POSIX-stored paths. 

87 if "\\" in filename: 

88 raise _invalid_snippet_filename(filename) 

89 

90 posix_path = PurePosixPath(filename) 

91 if posix_path.is_absolute(): 

92 raise _invalid_snippet_filename(filename) 

93 

94 parts = filename.split("/") 

95 # Reject traversal or hidden-file components, and empty parts (consecutive //). 

96 for part in parts: 

97 if part in ("", ".", "..") or not _is_safe_snippet_path_component(part): 

98 raise _invalid_snippet_filename(filename) 

99 

100 # The last component (the actual file) must have a lowercase alpha extension. 

101 if not _has_lowercase_alpha_suffix(parts[-1]): 

102 raise _invalid_snippet_filename(filename)

209async def check_snippet_batch_existence( 

210 session: AsyncSession, filenames: Iterable[str] 

211) -> SnippetBatchExistenceResult: 

212 """Verify that every filename has a DB row and an on-disk file. 

213 

214 Shared between the legacy Jinja2 batch route and the JSON API batch 

215 endpoint so both surfaces report the same hard-error categories. The 

216 helper deliberately does **not** check approval state — the JSON path 

217 treats already-approved as a soft-skip via the atomic 

218 ``update_where(... approved_at IS NULL)`` filter. 

219 

220 :param session: The active database session. 

221 :type session: AsyncSession 

222 :param filenames: The filenames the caller wants to act on. 

223 :type filenames: Iterable[str] 

224 :return: A populated :class:`SnippetBatchExistenceResult`. 

225 :rtype: SnippetBatchExistenceResult 

226 """ 

227 filenames_set = set(filenames) 

228 if not filenames_set: 228 ↛ 229line 228 didn't jump to line 229 because the condition on line 228 was never true

229 return SnippetBatchExistenceResult() 

230 snippets = await SnippetManager.list( 

231 session, col(Snippet.filename).in_(filenames_set) 

232 ) 

233 found = {snippet.filename for snippet in snippets} 

234 missing_in_db = sorted(filenames_set - found) 

235 missing_on_disk = sorted( 

236 snippet.filename for snippet in snippets if not Path(snippet).is_file() 

237 ) 

238 return SnippetBatchExistenceResult( 

239 snippets=snippets, 

240 missing_in_db=missing_in_db, 

241 missing_on_disk=missing_on_disk, 

242 )

245async def get_batch_existence( 

246 body: SnippetBatchApproveRequest, 

247 session: SessionDep, 

248) -> SnippetBatchExistenceResult: 

249 """Resolve and validate batch snippet existence, raising 400 on hard errors. 

250 

251 :param body: The parsed JSON body containing the filenames to approve. 

252 :type body: SnippetBatchApproveRequest 

253 :param session: The active database session. 

254 :type session: AsyncSession 

255 :return: Existence result guaranteed to have no hard errors. 

256 :rtype: SnippetBatchExistenceResult 

257 :raises HTTPBadRequestException: When any filename is missing from the DB 

258 or has no corresponding file on disk. 

259 """ 

260 existence = await check_snippet_batch_existence(session, body.filenames) 

261 if existence.has_errors: 

262 raise HTTPBadRequestException( 

263 detail=BatchApprovalErrorResponse( 

264 missing_in_db=existence.missing_in_db, 

265 missing_on_disk=existence.missing_on_disk, 

266 ).model_dump() 

267 ) 

268 return existence
