68 """Refresh the snippets cache from disk. 

69 

70 Admin-only and additionally gated by manual sync being enabled.

72 

73 :param user: The authenticated admin performing the refresh.

77 :rtype: RefreshResponse

161@approval_router.put("/snippet/approval", dependencies=[IsApiAuthenticated]) 

162async def snippets_api_approve( 

163 snippet: SnippetDep, user: ApiAdminUser, session: SessionDep 

164) -> SnippetResponse: 

165 """Approve a single snippet (idempotent). 

166 

167 Re-approving an already-approved snippet returns ``200`` with the 

168 current state — ``approved_at`` is *not* overwritten so the audit 

169 trail is preserved. 

170 

171 :param snippet: The snippet whose approval is being set. 

172 :type snippet: Snippet 

173 :param user: The authenticated admin performing the action. 

174 :type user: User 

175 :param session: The active database session. 

176 :type session: AsyncSession 

177 :return: The snippet entity after the call, including approval state. 

178 :rtype: SnippetResponse 

179 """ 

180 if not snippet.is_approved: 

181 snippet.approve(f"Approved by {user.username}", str(user.id)) 

182 await SnippetManager.save(session, snippet) 

183 logger.info("Snippet %r approved by %s", snippet.filename, user.username) 

184 return build_snippet_response(snippet)

219@approval_router.patch("/approvals", dependencies=[IsApiAuthenticated]) 

220async def snippets_api_batch_approve( 

221 existence: SnippetBatchExistenceDep, 

222 user: ApiAdminUser, 

223 session: SessionDep, 

224) -> BatchApprovalResponse: 

225 """Approve a set of snippets atomically (idempotent partial-update). 

226 

227 Reject the whole batch with ``400`` and a 

228 :class:`BatchApprovalErrorResponse` payload when any filename has no 

229 DB row or its file is missing on disk. Filenames that are already 

230 approved when the call starts are reported in 

231 ``skipped_already_approved`` — not as errors. The atomic write uses 

232 ``approved_at IS NULL`` as a CAS filter so two concurrent admins 

233 cannot double-approve. 

234 

235 :param existence: Pre-validated batch existence result from the dependency. 

236 :type existence: SnippetBatchExistenceResult 

237 :param user: The authenticated admin performing the action. 

238 :type user: User 

239 :param session: The active database session. 

240 :type session: AsyncSession 

241 :return: The success payload listing approved + skipped filenames. 

242 :rtype: BatchApprovalResponse 

243 :raises HTTPBadRequestException: On hard-error precheck failures with 

244 a :class:`BatchApprovalErrorResponse` body. 

245 """ 

246 pre_already_approved = [] 

247 to_approve = [] 

248 for snippet in existence.snippets: 

249 if snippet.is_approved: 

250 pre_already_approved.append(snippet.filename) 

251 else: 

252 to_approve.append(snippet.filename) 

253 pre_already_approved.sort() 

254 to_approve.sort() 

255 approved: list[str] = [] 

256 if to_approve: 

257 approved_rows = await SnippetManager.update_where( 

258 session, 

259 { 

260 "approved_at": utc_now(), 

261 "updated_by": str(user.id), 

262 "reason": f"Batch approved by {user.username}", 

263 }, 

264 col(Snippet.filename).in_(to_approve), 

265 col(Snippet.approved_at).is_(None), 

266 returning=["filename"], 

267 ) 

268 approved = sorted(approved_rows) 

269 

270 raced_out = sorted(set(to_approve) - set(approved)) 

271 skipped = sorted(pre_already_approved + raced_out) 

272 logger.info( 

273 "Batch-approve via JSON API by %s: approved=%s skipped=%s", 

274 user.username, 

275 approved, 

276 skipped, 

277 ) 

278 return BatchApprovalResponse( 

279 approved=approved, 

280 skipped_already_approved=skipped, 

281 )
