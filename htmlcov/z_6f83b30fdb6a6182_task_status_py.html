16"""Define the task-execution status vocabulary. 

17 

18Split out of :mod:`app.tasks.models` so a table in another service can denormalize 

19the status and type its column with the enum. Such a table cannot import 

20``app.tasks.models`` itself: plugin ``models.py`` modules are loaded by the Alembic 

21plugin-discovery loader, and pulling in another service's models would register its 

22tables in ``SQLModel.metadata`` and leak them into the ``extensions`` autogenerate. A fixed 

23value set is not a reason to widen the column to a bare ``str``, so the enum moves to 

24where both sides can reach it rather than the column losing its constraint. 

25 

26**Import-safety is the whole point of this module, and it is a property of what the 

27file does not contain.** Nothing here imports anything but the standard library, and 

28``app/tasks/__init__.py`` is licence header only, so importing this module defines no 

29table and executes no service wiring. Adding a SQLModel class, or an import that 

30reaches one, silently reintroduces the leak this split exists to avoid. 

31 

32:mod:`app.tasks.models` re-exports :class:`TaskHistoryStatusEnum`, so the established 

33import path keeps working for every existing consumer. 

34""" 

35 

36__all__ = ["TaskHistoryStatusEnum"] 

37 

38from enum import StrEnum 

39 

40 

41class TaskHistoryStatusEnum(StrEnum): 

42 """Define status codes for task executions. 

43 

44 :cvar FAILED: Enum value for failed tasks. 

45 :cvar PENDING: Enum value for pending tasks. 

46 :cvar RUNNING: Enum value for running tasks. 

47 :cvar SUCCESS: Enum value for successfully completed tasks. 

48 :cvar STOPPED: Enum value for stopped tasks. 

49 :cvar LOST: Enum value for tasks that are lost. 

50 :cvar STALE: Enum value for tasks skipped because executor placement 

51 exceeded the configured staleness threshold (for example a Nomad 

52 allocation that never left the queue). 

53 :cvar UNLAUNCHABLE: Enum value for tasks the executor node could not 

54 launch at all, because some command in the invocation does not 

55 resolve there. The payload never ran, so this is not a script 

56 failure. 

57 """ 

58 

59 FAILED = "failed" 

60 PENDING = "pending" 

61 RUNNING = "running" 

62 SUCCESS = "success" 

63 STOPPED = "stopped" 

64 LOST = "lost" 

65 STALE = "stale" 

66 UNLAUNCHABLE = "unlaunchable" 

67 

68 def is_finished(self) -> bool: 

69 """Check whether this status has an observed run outcome. 

70 

71 :return: ``True`` when output retrieval is meaningful for this status 

72 (FAILED, SUCCESS, STOPPED, STALE, or UNLAUNCHABLE); ``False`` 

73 otherwise. 

74 """ 

75 return self in [ 

76 TaskHistoryStatusEnum.FAILED, 

77 TaskHistoryStatusEnum.SUCCESS, 

78 TaskHistoryStatusEnum.STOPPED, 

79 TaskHistoryStatusEnum.STALE, 

80 TaskHistoryStatusEnum.UNLAUNCHABLE, 

81 ] 

82 

83 def is_terminal(self) -> bool: 

84 """Check if task execution has reached a terminal state. 

85 

86 :return: ``True`` if task execution will not transition again. 

87 """ 

88 return self.is_finished() or self == TaskHistoryStatusEnum.LOST 

89 

90 @classmethod 

91 def active_statuses(cls) -> frozenset["TaskHistoryStatusEnum"]: 

92 """Return the statuses whose executions are still in flight. 

93 

94 These are the non-terminal statuses (``PENDING`` / ``RUNNING``); a new 

95 non-terminal status only needs adding here. 

96 

97 :return: The frozen set of in-flight statuses. 

98 """ 

99 return frozenset({cls.PENDING, cls.RUNNING}) 

100 

101 def is_active(self) -> bool: 

102 """Check whether the task status indicates an in-flight execution. 

103 

104 :return: ``True`` if the status is ``PENDING`` or ``RUNNING``; 

105 ``False`` otherwise. 

106 """ 

107 return self in self.active_statuses() 

108 

109 def operator_summary(self) -> str | None: 

110 """Return the operator-facing prose for this status, if it has any. 

111 

112 The single source both :meth:`TaskHistory.alert_for_status` and the 

113 ``failure_reason`` composers read, so the alert summary and the stored 

114 reason cannot drift apart. Phrased as a sentence fragment because the 

115 alert interpolates it mid-sentence. 

116 

117 :return: The prose fragment, or ``None`` for a status carrying none. 

118 """ 

119 return { 

120 TaskHistoryStatusEnum.FAILED: "failed", 

121 TaskHistoryStatusEnum.LOST: "execution tracking lost", 

122 TaskHistoryStatusEnum.STALE: ( 

123 "skipped as stale (executor placement delayed past threshold)" 

124 ), 

125 TaskHistoryStatusEnum.UNLAUNCHABLE: ( 

126 "could not be launched (the executor node cannot run the " 

127 "requested command)" 

128 ), 

129 }.get(self)
