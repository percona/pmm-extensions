16"""Compute beat-faithful fire times for PMM Extensions' schedule models. 

17 

18Every time reported here comes from the schedule object Celery beat itself 

19consults: ``celery.schedules.schedule`` for an interval, 

20``sqlalchemy_celery_beat.tzcrontab.TzAwareCrontab`` for a crontab. The API's 

21account of a schedule and the scheduler's behaviour therefore cannot drift 

22apart on a cron dialect or a DST transition. 

23"""

43#: How far ``sqlalchemy_celery_beat.schedulers.ModelEntry.__init__`` back-dates 

44#: an unset ``last_run_at`` when ``start_time`` is set, so the first run lands 

45#: at ``start_time`` instead of one interval after it.

57def _clock_reading(schedule: BaseSchedule, moment: datetime) -> Callable[[], datetime]: 

58 """Return a celery ``nowfun`` reporting ``moment`` on ``schedule``'s clock. 

59 

60 Mirror ``TzAwareCrontab.nowfunc``, which reports the instant expressed in 

61 the schedule's own zone. ``crontab.remaining_delta`` compares calendar fields 

62 between that reading and a ``last_run_at`` that ``is_due`` has already 

63 converted, so a UTC-expressed reading makes the two disagree about the date 

64 whenever the zone's offset puts them on different days, and a sub-hourly 

65 expression then reports hourly runs. 

66 

67 An interval schedule uses the reading as an absolute instant rather than by 

68 calendar field, so converting it there changes nothing. 

69 

70 :param schedule: The schedule whose clock is being pinned. 

71 :param moment: The instant the returned callable reports as "now". 

72 :return: A zero-argument callable returning ``moment`` on that clock. 

73 """ 

74 return lambda: moment.astimezone(schedule.tz)

77def celery_schedule( 

78 interval: IntervalSchedule | None, crontab: CrontabSchedule | None 

79) -> BaseSchedule: 

80 """Return the celery schedule object beat consults for this schedule. 

81 

82 :param interval: The interval schedule, or ``None`` for a crontab schedule. 

83 :param crontab: The crontab schedule, or ``None`` for an interval schedule. 

84 :return: The ``celery.schedules`` object beat evaluates to decide when the 

85 task fires. 

86 :raises ValueError: If neither schedule is set. 

87 :raises OverflowError: If an interval's cadence exceeds 

88 :class:`~datetime.timedelta`'s range. 

89 """ 

90 if interval is not None: 

91 return interval.schedule 

92 if crontab is not None: 

93 return BaseCrontabSchedule.aware_crontab(crontab) 

94 raise ValueError("Either `interval` or `crontab` must be set.")

115def next_run_times( 

116 *, 

117 interval: IntervalSchedule | None, 

118 crontab: CrontabSchedule | None, 

119 start_time: datetime | None = None, 

120 last_run_at: datetime | None = None, 

121 enabled: bool = True, 

122 now: datetime | None = None, 

123 count: int = NEXT_RUNS_PREVIEW_COUNT, 

124) -> list[datetime]: 

125 """Return the next ``count`` UTC times beat would fire this schedule. 

126 

127 Mirror the two rules ``sqlalchemy_celery_beat.schedulers.ModelEntry`` 

128 applies around the schedule object: an unset ``last_run_at`` is back-dated 

129 by :data:`BEAT_BACKDATE` when ``start_time`` is set, so the first run lands 

130 at ``start_time``; and no run is reported before ``start_time``. 

131 

132 ``ModelEntry`` itself cannot produce the sequence. Iterating it re-anchors 

133 ``last_run_at`` to real now and yields the same fire time every time, and 

134 its failure paths need a ``Session``, so the rules are mirrored here rather 

135 than reused. ``ModelEntry``'s remaining behaviour is deliberately omitted: 

136 the ``one_off`` branch (PMM Extensions never sets the column), ``expires`` handling 

137 (PMM Extensions bounds dispatch, not the schedule), and disabling a schedule that fails 

138 to build (validation upstream makes it unreachable). 

139 

140 An already-due schedule fires at ``now``, so the first run it reports is 

141 ``now`` itself; every later run is the next fire time after the one before 

142 it. 

143 

144 :param interval: The interval schedule, or ``None`` for a crontab schedule. 

145 :param crontab: The crontab schedule, or ``None`` for an interval schedule. 

146 :param start_time: The earliest time the schedule may fire, if any. 

147 :param last_run_at: When the schedule last fired, if ever. 

148 :param enabled: Whether the schedule is enabled; a disabled schedule has no 

149 upcoming runs. 

150 :param now: The instant to evaluate from. Defaults to 

151 :func:`~app.core.utils.date_time.utc_now`. 

152 :param count: How many upcoming runs to return. 

153 :return: Up to ``count`` strictly increasing UTC datetimes, empty when the 

154 schedule is disabled. 

155 :raises ValueError: If neither ``interval`` nor ``crontab`` is set. 

156 :raises OverflowError: If the cadence or a computed run falls outside 

157 :class:`~datetime.datetime`'s range. The periodic-task write and preview 

158 models reject such a schedule at the request boundary. 

159 """ 

160 if not enabled: 

161 return [] 

162 schedule = celery_schedule(interval, crontab) 

163 now = now or utc_now() 

164 effective_last = last_run_at 

165 if effective_last is None: 

166 effective_last = now - BEAT_BACKDATE if start_time is not None else now 

167 cursor = start_time if start_time is not None and now < start_time else now 

168 runs: list[datetime] = [] 

169 for _ in range(count): 

170 schedule.nowfun = _clock_reading(schedule, cursor) 

171 due, remaining = schedule.is_due(effective_last) 

172 upcoming = ( 

173 cursor 

174 if due and not runs 

175 else cursor + timedelta(seconds=cast("float", remaining)) 

176 ) 

177 if runs and upcoming <= runs[-1]: 177 ↛ 178line 177 didn't jump to line 178 because the condition on line 177 was never true

178 break 

179 runs.append(upcoming) 

180 effective_last = cursor = upcoming 

181 return [run.astimezone(UTC) for run in runs]
