27#: MongoDB's port when a service carries none in inventory. 

28DEFAULT_MONGODB_PORT = 27017

31@dataclass(frozen=True, slots=True) 

32class InventoryService: 

33 """Carry one MongoDB service as inventory reports it. 

34 

35 Topology grouping — cluster identity, replica-set membership, environment — is 

36 PMM's to derive, not PMM Extensions'; this app only has to find the service and probe its 

37 host. No ``cluster``, ``replication_set`` or ``environment`` field lives here: 

38 carrying them with zero consumers would be a standing temptation to start 

39 grouping services in PMM Extensions again. 

40 

41 :param service_id: The inventory service id. 

42 :param external_id: **PMM's** service UUID, which inventory stores as 

43 ``external_id``. This is the join key against VictoriaMetrics, whose 

44 ``service_id`` label carries the same UUID — and it is the *only* safe one: 

45 ``service_name`` is reused across re-registrations while the superseded series 

46 live on until retention expires, so a name-keyed join silently mixes 

47 generations. ``None`` when inventory carries none, which makes the service 

48 invisible to the metrics source. 

49 :param name: The inventory service name. 

50 :param port: The service port, defaulted when inventory carries none. 

51 :param node_name: The node's registered name, if any. 

52 :param node_address: The node's registered address, if any. 

53 """ 

54 

55 service_id: int | None 

56 external_id: str | None 

57 name: str 

58 port: int 

59 node_name: str | None 

60 node_address: str | None 

61

63def _service_from_entry(entry: dict) -> InventoryService | None: 

64 """Build a service record from one raw ``GET /services/`` listing entry. 

65 

66 An entry whose node carries neither a name nor an address is skipped: there is 

67 nothing to match an executor host against, and nothing to connect to. 

68 

69 :param entry: One entry from the inventory services listing. 

70 :return: The service, or ``None`` when it carries no usable node identity. 

71 """ 

72 node = entry.get("node") or {} 

73 node_name = node.get("name") 

74 node_address = node.get("address") 

75 if not node_name and not node_address: 

76 logger.warning( 

77 "Skipping MongoDB service %r: inventory entry carries no node name " 

78 "or address", 

79 entry.get("name"), 

80 ) 

81 return None 

82 return InventoryService( 

83 service_id=entry.get("id"), 

84 external_id=entry.get("external_id") or None, 

85 name=entry.get("name") or node_name or node_address or "", 

86 port=entry.get("port") or DEFAULT_MONGODB_PORT, 

87 node_name=node_name, 

88 node_address=node_address, 

89 ) 

90

92async def list_mongodb_services(inventory_api: RemoteAPI) -> list[InventoryService]: 

93 """Return every MongoDB service in inventory, in inventory order. 

94 

95 :param inventory_api: The inventory API client. 

96 :return: The MongoDB services carrying a usable node identity. 

97 """ 

98 listed = await fetch_all_dict_items( 

99 lambda pagination: inventory_api.get( 

100 "/services/", 

101 params={ 

102 "service_type": ServiceTypeEnum.MONGODB.value, 

103 **pagination.model_dump(), 

104 }, 

105 ) 

106 ) 

107 services = [ 

108 service 

109 for entry in listed 

110 if (service := _service_from_entry(entry)) is not None 

111 ] 

112 logger.info( 

113 "OM inventory: inventory lists %d MongoDB service(s), %d usable", 

114 len(listed), 

115 len(services), 

116 ) 

117 return services
