58def _validate_hook_module_root(root: str) -> str: 

59 """Return ``root`` when it is a well-formed dotted module path, else raise. 

60 

61 :param root: A candidate ``HOOK_MODULE_ALLOWLIST`` entry. 

62 :return: The validated root, unchanged. 

63 :raises ValueError: When ``root`` is not a dotted module path, and so could 

64 never admit a hook. 

65 """ 

66 if not is_dotted_module_path(root): 

67 raise ValueError( 

68 f"{root!r} is not a dotted module path, so no hook path can match it" 

69 ) 

70 return root 

71 

72 

73#: An entry of :attr:`TasksSettings.HOOK_MODULE_ALLOWLIST`. It shares 

74#: :func:`app.tasks.hook_resolver.is_dotted_module_path` with the resolver's own 

75#: check, so a root that could never admit a hook is refused at load rather than 

76#: accepted and silently matched by nothing. 

77HookModuleRoot = Annotated[str, AfterValidator(_validate_hook_module_root)]

80def _validate_syncer_name(name: str) -> str: 

81 """Return ``name`` when it is a well-formed dotted syncer path, else raise. 

82 

83 :param name: A candidate :attr:`TasksSettings.INVENTORY_SYNC_SYNCER` value. 

84 :return: The validated name, unchanged. 

85 :raises ValueError: When ``name`` is blank or not a dotted path, in which 

86 case it can only be read as the sync-all default or fail every firing. 

87 """ 

88 if not is_dotted_module_path(name): 

89 raise ValueError( 

90 f"{name!r} is not a dotted 'module.ClassName' path, so it names no " 

91 "syncer; omit the setting to run every configured syncer" 

92 ) 

93 return name 

94 

95 

96#: The value of :attr:`TasksSettings.INVENTORY_SYNC_SYNCER`. A blank string 

97#: would otherwise be read as the sync-all default, silently widening a schedule 

98#: that was pinned on purpose, and a whitespace-only one would fail every firing. 

99SyncerName = Annotated[str, AfterValidator(_validate_syncer_name)] 

100 

101#: The longest syncer path an ``INVENTORY_SYNC_SCHEDULES`` entry may name. Its 

102#: seeded row name is the path appended to a fixed prefix, and the celery-beat 

103#: ``PeriodicTask.name`` column is bounded, so a longer path passes the dotted-path 

104#: check and then fails the insert at startup — on PostgreSQL only, since SQLite 

105#: does not enforce the width. Held against the real prefix and column by 

106#: ``tests/app/tasks/db/test_seed.py``. 

107MAX_SCHEDULED_SYNCER_LENGTH = 232

110class InventorySyncSchedule(BaseLowercaseModel): 

111 """Pair one syncer with the interval its own seeded schedule runs on. 

112 

113 :param syncer: The fully qualified syncer this schedule targets, in 

114 ``BaseSyncer.get_name()`` form. Unlike ``EXTENSIONS.SYNCERS[].SYNCER`` it is not 

115 auto-prefixed, so the two spellings are not interchangeable. Bounded 

116 because a well-formed path is not necessarily a schedulable one: a longer

143class TasksSettings(BaseYamlAppSettings): 

144 """Define settings for tasks configuration. 

145 

146 :cvar SETTINGS_PREFIXES: The prefixes for task-related settings in the 

147 configuration file. 

148 :param UVICORN_PORT: The port to be used by Uvicorn for running the server. 

149 Defaults to 8002. 

150 :param NOMAD: The Nomad executor used to integrate with Nomad. Per-leaf 

151 overrides are accepted via the settings API; the parent object itself 

152 is not patchable as a whole. 

153 :param DATABASE: The database configuration options. Defaults to an SQLite database 

154 with the name 'tasks.db'. 

155 :param SECURITY_HEADERS: Specific options for the SecurityHeadersMiddleware. 

156 Use ``False`` to disable the middleware completely. 

157 :param SYNC_LOCK_TTL: The timeout for the TaskHistory sync lock. Must be a 

158 positive duration. Defaults to 5 minutes. 

159 :param PRE_EXECUTION_CONNECTIVITY_CHECK: The mode for pre-execution connectivity 

160 checks. Defaults to ``PreExecutionCheckMode.DISABLED``. 

161 :param STALENESS_THRESHOLD_SECONDS: The maximum seconds allowed between a 

162 dispatch's scheduled time and its Nomad-side execution start before 

163 the allocation self-aborts as stale. Must be positive. Defaults to 3600. 

164 :param PENDING_ALLOCATION_TIMEOUT_SECONDS: The maximum seconds a RUNNING 

165 TaskHistory row may wait behind a Nomad allocation that has never 

166 produced TaskStates before sync escalates it to LOST. Measured from 

167 ``started_at``. Must be positive. Defaults to 3600. 

168 :param LOG_RETENTION_DAYS: The age in days beyond which finished task-execution 

169 logs (``taskhistory_log`` rows) are purged. Runtime-overridable; must be a 

170 positive integer no greater than 365. Defaults to 90. 

171 :param LOG_PURGE_BATCH_SIZE: The maximum number of ``taskhistory_log`` rows 

172 deleted per commit by the purge job. Runtime-overridable; must be positive. 

173 Defaults to 10,000. 

174 :param LOG_PURGE_INTERVAL: The schedule on which the log-purge periodic task 

175 runs. ``None`` disables seeding the task entirely. Read at startup (not 

176 runtime-overridable). Defaults to once per day. 

177 :param INVENTORY_SYNC_INTERVAL: The schedule on which the inventory-sync task 

178 runs. ``None`` disables seeding the schedule entirely, leaving the sync 

179 to whatever interval an operator attaches. Read at startup (not 

180 runtime-overridable). Defaults to ``None``. 

181 :param INVENTORY_SYNC_SYNCER: The fully qualified syncer 

182 (``"module.ClassName"``, matching ``BaseSyncer.get_name()``) the seeded 

183 schedule targets; ``None`` runs every configured syncer. Must be a 

184 dotted path, so a blank value is rejected rather than read as the 

185 sync-all default. A well-formed name that no configured syncer matches 

186 is still not rejected at seed time — the tasks service does not import 

187 the extensions syncers — so every firing of the schedule fails instead. Read at 

188 startup. Defaults to ``None``. 

189 :param INVENTORY_SYNC_SCHEDULES: Per-syncer schedules seeded beside the 

190 scalar default, for a syncer whose useful cadence differs from it. Each 

191 entry names a syncer in ``BaseSyncer.get_name()`` form and its own 

192 interval. When the pinned ``INVENTORY_SYNC_SYNCER`` default is seeded, 

193 each entry's seeded schedule is deferred until that syncer's first 

194 completed inventory sync, which then starts the entry once if it has 

195 never run; an entry an operator-managed schedule already covers is not 

196 seeded, so that schedule is not deferred. Read at startup. Defaults to 

197 no extra schedules. 

198 :param LOG_STREAM_CAP_BYTES: The maximum captured-log bytes retained per 

199 ``(task_history_id, source, stream)``. As a stream grows past the cap 

200 the writer drops the oldest chunks, keeping a bounded recent tail so a 

201 long-running execution's logs cannot grow without limit. Must be 

202 positive. Defaults to 104857600 (100 MiB). 

203 :param LOG_STREAM_EVICTION_MAX_ROWS: The maximum number of chunk rows the 

204 writer evicts per flush, bounding the per-append eviction work. Must be 

205 positive. Defaults to 1000. 

206 :param HOOK_MODULE_ALLOWLIST: The module roots a per-task hook path 

207 (``alert_detail_builder``, ``run_result_recorder``) may name. A path is 

208 admitted when its module equals a root or is a submodule of one, so each 

209 entry must itself be a dotted module path. Environment- and YAML-only, 

210 and deliberately not exposed as a runtime-overridable setting: widening 

211 the namespace live would itself be a privilege-escalation path, since the 

212 resolved callable is imported and invoked by the tasks service. Defaults 

213 to the namespace holding the shipped task apps. 

214 """

269 @model_validator(mode="after") 

270 def reject_double_scheduled_syncers(self) -> Self: 

271 """Reject the combinations that would fire one syncer's payload twice. 

272 

273 A per-node sync dispatches a task per node, so a doubled schedule is 

274 expensive rather than merely redundant, and nothing downstream notices: 

275 both rows are well-formed and fire independently. 

276 

277 Both scalar-facing checks are gated on the interval, since without one the 

278 scalar pair seeds nothing and there is no second firing to prevent. 

279 

280 ``UniqueList`` deduplicates whole entries, not syncers, so the same syncer 

281 at two intervals survives it and then collides on one seeded row name, 

282 leaving a schedule whose interval depends on configuration order. 

283 

284 :return: The validated instance. 

285 :raises ValueError: If an entry duplicates the scalar pin, sits beside the 

286 sync-all default, or repeats a syncer another entry already names. 

287 """ 

288 scalar_schedules = self.INVENTORY_SYNC_INTERVAL is not None 

289 seen: set[str] = set() 

290 for entry in self.INVENTORY_SYNC_SCHEDULES: 

291 if scalar_schedules and entry.syncer == self.INVENTORY_SYNC_SYNCER: 

292 raise ValueError( 

293 f"INVENTORY_SYNC_SCHEDULES names {entry.syncer!r}, which " 

294 "INVENTORY_SYNC_SYNCER already schedules; drop one of them" 

295 ) 

296 if entry.syncer in seen: 

297 raise ValueError( 

298 f"INVENTORY_SYNC_SCHEDULES names {entry.syncer!r} more than " 

299 "once; keep a single entry per syncer" 

300 ) 

301 seen.add(entry.syncer) 

302 if ( 

303 self.INVENTORY_SYNC_SCHEDULES 

304 and scalar_schedules 

305 and self.INVENTORY_SYNC_SYNCER is None 

306 ): 

307 raise ValueError( 

308 "INVENTORY_SYNC_SCHEDULES cannot be combined with the sync-all " 

309 "default, which already runs every configured syncer; pin " 

310 "INVENTORY_SYNC_SYNCER or clear INVENTORY_SYNC_INTERVAL" 

311 ) 

312 return self
