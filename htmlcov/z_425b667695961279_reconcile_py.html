16"""Reconcile dispatched steps against their Nomad dispatch's real status. 

17 

18Deliberately narrow: this module only ever *translates* a ``TaskHistory``'s 

19status onto the :class:`~app.extensions.apps.om_bootstrap.strategy.StepRecord` that 

20dispatched it — it never decides to dispatch the *next* step, retry a failed 

21one, or roll a run back. Those are PMM's ``om`` service's job, driving the state 

22machine as the HA-leader-only stepper: it reads a run's current state through 

23the API and decides what happens next. 

24``om_bootstrap`` only ever answers "is this step still running", mechanically, 

25so PMM's driver has something true to read. 

26 

27One exception, not a contradiction of the above: :func:`reconcile_run` also 

28flips a ``RUNNING`` run to 

29:attr:`~app.extensions.apps.om_bootstrap.models.BootstrapRunStatus.SUCCEEDED` once 

30every host and every run-level step has actually succeeded. That is not a 

31decision — there is nothing left to decide once everything succeeded, only a 

32fact to record — unlike 

33:attr:`~app.extensions.apps.om_bootstrap.models.BootstrapRunStatus.FAILED` (retries 

34exhausted) and 

35:attr:`~app.extensions.apps.om_bootstrap.models.BootstrapRunStatus.ROLLED_BACK` 

36(rollback finished), both real calls only the stepper makes, which reach 

37``run.status`` through ``api_routes.py``'s explicit ``:finish`` route instead. 

38 

39``finished_at`` is stamped with the reconciliation's own clock, not read back 

40from ``TaskHistory``: this runs on a polling interval, not a push, so the two 

41times differ by at most that interval — an acceptable approximation for a 

42receipt, not a claim of exact timing. 

43 

44``detail`` on a non-``SUCCESS`` terminal status is currently just the 

45``TaskHistory`` status name — it does not yet read the dispatch's stderr for a 

46real error message (the way ``om_inventory/dispatch.py``'s ``_read_stdout`` 

47does for its own purpose). A worthwhile follow-up, not done here to keep this 

48module to exactly the one thing its docstring claims. 

49 

50A Tasks API read that fails does not fail the reconcile: a ``404``/``410`` means 

51the step's ``TaskHistory`` record is gone and it can never finish, so the step 

52is recorded ``failed``; any other failure (an upstream error, a transport error, 

53a timeout) is transient, so the step stays ``running`` and the next poll tries 

54again. A readable answer that is not a ``TaskHistory`` — not a JSON object, or 

55without a string ``status`` — is a ``502``, as it would otherwise leave the step 

56looking in flight forever. 

57"""

102async def reconcile_step(tasks_api: RemoteAPI, step: StepRecord) -> StepRecord: 

103 """Check one step's dispatch and translate it if it reached a terminal status. 

104 

105 A no-op, returning ``step`` unchanged, for a step that is not 

106 :attr:`~app.extensions.apps.om_bootstrap.strategy.StepStatus.RUNNING`, carries no 

107 :attr:`~app.extensions.apps.om_bootstrap.strategy.StepRecord.task_history_id` (not 

108 yet dispatched), or whose dispatch is still in flight. 

109 

110 Also returns ``step`` unchanged when the Tasks API cannot be read right 

111 now; see the module docstring for which failures count as that. 

112 

113 :param tasks_api: The Tasks API client. 

114 :param step: The step to check. 

115 :return: ``step`` itself when nothing changed, or a new 

116 :class:`~app.extensions.apps.om_bootstrap.strategy.StepRecord` reflecting the 

117 dispatch's terminal status, or ``failed`` when its ``TaskHistory`` 

118 record no longer exists. 

119 :raises HTTPBadGatewayException: When the Tasks API answers with something 

120 that is not a ``TaskHistory`` — not a JSON object, or without a string 

121 ``status``. 

122 """ 

123 if step.status != StepStatus.RUNNING or step.task_history_id is None: 

124 return step 

125 try: 

126 payload = await tasks_api.get(f"/history/{step.task_history_id}") 

127 except (HTTPNotFoundException, HTTPGoneException): 

128 return step.model_copy( 

129 update={ 

130 "status": StepStatus.FAILED, 

131 "finished_at": utc_now(), 

132 "detail": f"Task history {step.task_history_id} no longer exists; " 

133 "the dispatch's outcome is unknown", 

134 } 

135 ) 

136 except (HTTPException, aiohttp.ClientError, OSError): 

137 logger.warning( 

138 "Could not read task history %s for step %r; leaving it running", 

139 step.task_history_id, 

140 step.name, 

141 exc_info=True, 

142 ) 

143 return step 

144 task_status = as_json_object(payload).get("status") 

145 if not isinstance(task_status, str): 

146 raise HTTPBadGatewayException( 

147 detail=f"Task history {step.task_history_id} has no status" 

148 ) 

149 if task_status in _IN_FLIGHT_STATUS_VALUES: 

150 return step 

151 if task_status == TaskHistoryStatusEnum.SUCCESS.value: 

152 return step.model_copy( 

153 update={"status": StepStatus.SUCCEEDED, "finished_at": utc_now()} 

154 ) 

155 return step.model_copy( 

156 update={ 

157 "status": StepStatus.FAILED, 

158 "finished_at": utc_now(), 

159 "detail": f"Task history {step.task_history_id} ended {task_status}", 

160 } 

161 )

164async def reconcile_run(tasks_api: RemoteAPI, run: BootstrapRun) -> bool: 

165 """Reconcile every running step across ``run`` — per-host, rollback, and run-level. 

166 

167 Mutates ``run.hosts``/``run.run_steps``/``run.status``/``run.finished_at`` 

168 directly when anything changed; the caller is responsible for committing the 

169 session. A finished step's scratch script 

170 (:func:`~app.extensions.apps.om_bootstrap.dispatch.cleanup_step_script`) is removed 

171 as part of reconciling it, win or lose — nothing downstream needs it once 

172 the dispatch that read it is done. The run-level steps' scripts are named 

173 under the run's first host regardless of which host actually ran them — 

174 see :func:`~app.extensions.apps.om_bootstrap.dispatch.step_script_filename`, and 

175 :meth:`~app.extensions.apps.om_bootstrap.strategy.InstallStrategy.build_run_step`'s 

176 own docstring for why that host is always the target. 

177 

178 Skips the SUCCEEDED inference once ``run.cancel_requested`` is set: without 

179 this, a cancel that lands just as the last step finishes — or whose 

180 best-effort stop failed — would flip to SUCCEEDED on the very next poll, 

181 permanently recording a run the operator aborted as one that finished 

182 normally, with no route back to ROLLED_BACK (``finish_run`` refuses to 

183 override a terminal run). 

184 

185 :param tasks_api: The Tasks API client. 

186 :param run: The run to reconcile. 

187 :return: Whether anything changed — callers use this to skip a write when 

188 every step was still in flight. 

189 """ 

190 states = parse_host_states(run) 

191 run_steps = parse_run_steps(run) 

192 seed_host = states[0].host if states else None 

193 changed, run_steps_changed = await asyncio.gather( 

194 _reconcile_step_list_per_host(tasks_api, run, states), 

195 _reconcile_step_list(tasks_api, run, seed_host, run_steps), 

196 ) 

197 if changed: 

198 run.hosts = dump_host_states(states) 

199 if run_steps_changed: 

200 run.run_steps = dump_run_steps(run_steps) 

201 

202 if ( 

203 run.status == BootstrapRunStatus.RUNNING 

204 and not run.cancel_requested 

205 and _fully_succeeded(states, run_steps) 

206 ): 

207 run.status = BootstrapRunStatus.SUCCEEDED 

208 run.finished_at = utc_now() 

209 changed = True 

210 

211 return changed or run_steps_changed

214async def _reconcile_step_list_per_host( 

215 tasks_api: RemoteAPI, run: BootstrapRun, states: list[HostBootstrapState] 

216) -> bool: 

217 """Reconcile every host's own ``steps`` and ``rollback_steps``, in place. 

218 

219 :param tasks_api: The Tasks API client. 

220 :param run: The run these hosts belong to — only read, for its id. 

221 :param states: The parsed host states to reconcile, mutated in place. 

222 :return: Whether anything changed. 

223 """ 

224 results = await asyncio.gather( 

225 *( 

226 _reconcile_step_list(tasks_api, run, state.host, step_list) 

227 for state in states 

228 for step_list in (state.steps, state.rollback_steps, state.finalize_steps) 

229 ) 

230 ) 

231 return any(results)

271def _fully_succeeded( 

272 states: list[HostBootstrapState], run_steps: list[StepRecord] 

273) -> bool: 

274 """Report whether every host, every run-level step, and every finalize step succeeded. 

275 

276 Rollback steps are deliberately excluded from this check, not reconciled 

277 into it: every host's ``rollback_steps`` are planned up front alongside its 

278 forward steps (see 

279 :class:`~app.extensions.apps.om_bootstrap.strategy.HostBootstrapState`'s own 

280 docstring) and stay 

281 :attr:`~app.extensions.apps.om_bootstrap.strategy.StepStatus.PENDING` for the 

282 entire life of a run that never needed rollback — counting them here 

283 would mean a normal, fully-succeeded run could never satisfy this check. 

284 

285 ``finalize_steps`` are checked explicitly, not folded into 

286 :attr:`~app.extensions.apps.om_bootstrap.strategy.HostBootstrapState.status`: that 

287 property derives purely from ``steps`` (see its own docstring), by design — 

288 a host isn't considered done finalizing until its finalize steps have too, 

289 but a host that hasn't started finalizing yet (every finalize step still 

290 ``pending``, correctly, until every run-level step succeeds) must not read as 

291 unfinished in the same way a genuinely stuck forward step would. 

292 

293 :param states: Every host's current state. 

294 :param run_steps: The run's current run-level steps. 

295 :return: Whether the run, as a whole, has nothing left to do but succeed. 

296 """ 

297 if not all(state.status == StepStatus.SUCCEEDED for state in states): 

298 return False

300 return False 

301 return all( 

302 step.status in _DONE_STATUSES 

303 for state in states 

304 for step in state.finalize_steps
