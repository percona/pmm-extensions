16"""Define the JSON API router for the Inventory plugin. 

17 

18Mounted at ``/api/apps/inventory/`` via ``apps_router`` in 

19``app/extensions/api/router.py``, which supplies the API authentication these routes 

20rely on. 

21 

22This is an operator API only — the plugin ships no browser surface, so nothing 

23here proxies entity reads. Read the catalog from the inventory service itself 

24at ``/api/inventory/*``, which remains its canonical CRUD surface and the one 

25the syncers write through. 

26 

27The router mounts the ad-hoc inventory-sync trigger (``POST /sync/``), the 

28running-state polling endpoint (``GET /sync/status/``), schedule discovery 

29(``GET /``), available-syncers (``GET /available-syncers/``), and the 

30per-service connectivity probe. Periodic-task CRUD remains delegated to 

31``/api/tasks/periodic/*`` as the single source of truth; this router does not 

32duplicate that surface. 

33"""

76@router.post("/sync/", status_code=status.HTTP_202_ACCEPTED, response_class=Response) 

77async def inventory_sync_trigger( 

78 syncers: SyncersDep, 

79 background_tasks: BackgroundTasks, 

80 internal_token: InternalTokenDep, 

81 body: InventorySyncTriggerWrite | None = _OPTIONAL_TRIGGER_BODY, 

82) -> Response: 

83 """Schedule an ad-hoc inventory sync as a background task. 

84 

85 Mirrors the Jinja2 ``POST /inventory/sync/`` handler but accepts an 

86 optional JSON body ``{"syncer": "<qualified_name>"}``. When ``syncer`` 

87 is absent, ``None``, or empty, every configured syncer runs in 

88 declaration order; otherwise only the named syncer is forwarded. An 

89 unknown or inapplicable syncer raises HTTP 400 — never a silent no-op. 

90 

91 Authentication is enforced by the parent ``api_router``'s 

92 ``IsApiAuthenticated`` dependency, so this handler needs no auth parameter. 

93 

94 :param syncers: Configured syncers from ``SyncersDep``. 

95 :type syncers: SyncersDep 

96 :param background_tasks: FastAPI's background task scheduler. 

97 :type background_tasks: BackgroundTasks 

98 :param internal_token: PMM Extensions internal service token injected by 

99 ``InternalTokenDep`` and forwarded to the background sync task. 

100 :param body: Optional trigger body. 

101 :type body: InventorySyncTriggerWrite | None 

102 :return: Empty 202 Accepted response. 

103 :rtype: Response 

104 :raises HTTPBadRequestException: When ``body.syncer`` is set but does 

105 not match any configured syncer that can sync inventory. 

106 """ 

107 syncer_name = body.syncer if body is not None else None 

108 try: 

109 selected = filter_syncers_by_name( 

110 syncers, 

111 syncer_name, 

112 lambda syncer: syncer.can_sync_inventory(), 

113 ) 

114 except ValueError as exc: 

115 raise HTTPBadRequestException(str(exc)) from exc 

116 background_tasks.add_task(run_inventory_sync, internal_token, *selected) 

117 return Response(status_code=status.HTTP_202_ACCEPTED)

120@router.get("/sync/status/") 

121async def inventory_sync_status(session: SessionDep) -> InventorySyncStatusResponse: 

122 """Return whether an inventory-wide sync is running, plus recent run outcomes. 

123 

124 Lets an operator poll a sync they triggered through ``POST /sync/`` 

125 without scraping any rendered page. 

126 

127 :param session: SQLModel async session. 

128 :return: The running flag and the most recent runs, newest first. 

129 """ 

130 is_running = await SyncItemManager.sync_is_running( 

131 session, 

132 SyncInventoryEntityTypeEnum.INVENTORY, 

133 ) 

134 runs = await SyncInstanceManager.list( 

135 session, 

136 order_by=[ 

137 col(SyncInstance.created_at).desc(), 

138 col(SyncInstance.id).desc(), 

139 ], 

140 limit=_SYNC_RUN_HISTORY_LIMIT, 

141 ) 

142 return InventorySyncStatusResponse( 

143 is_running=is_running, 

144 last_runs=[ 

145 SyncRunSummary( 

146 syncer=run.syncer, 

147 started_at=run.created_at, 

148 finished_at=run.updated_at, 

149 status=run.status, 

150 snapshot_complete=run.snapshot_complete, 

151 ) 

152 for run in runs 

153 ], 

154 )

191@router.post( 

192 "/services/{service_id:int}/check-connectivity/", 

193 dependencies=[IsApiAdmin], 

194) 

195async def inventory_service_check_connectivity( 

196 service: CreatedServiceDep, 

197 tasks_api: TaskAPI, 

198) -> ConnectivityCheckResponse: 

199 """Run a database connectivity probe for a service from its executor host. 

200 

201 A probe that ran but could not connect is reported as HTTP 200 with 

202 ``success=false`` and the upstream message in ``error``; only a probe that 

203 could not be attempted at all is an error status. 

204 

205 :param service: The service to probe, resolved from the path id. 

206 :param tasks_api: Authenticated Tasks ``RemoteAPI`` client. 

207 :return: The upstream probe result. 

208 :raises HTTPBadRequestException: When the service cannot be probed — 

209 unsupported type, missing node or port, or no executor registered for 

210 the node address. 

211 :raises HTTPBadGatewayException: When the Tasks API is unreachable or 

212 returns an unparseable body. 

213 """ 

214 return await probe_service_connectivity(service, tasks_api)
