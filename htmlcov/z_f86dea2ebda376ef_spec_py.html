16"""Build the ``run-python`` backup spec for the MySQL Backups app. 

17 

18:func:`build_backup_spec` is the pure ``(form, resolved) -> RunPythonSpec`` builder 

19fed to the framework's three-phase create path (and reused by the legacy Jinja 

20form path via ``deps.build_backup_task_payload``), so a backup task's Nomad 

21payload is byte-identical regardless of the call origin. The framework's 

22``assemble_envelope`` supplies the executor ``target``, ``_service_name``, and the 

23connectivity meta keys around this spec. 

24""" 

25 

26import yaml 

27from fastapi.encoders import jsonable_encoder 

28 

29from app.core.utils.path import payload_uri 

30from app.extensions.apps.framework.spec import ResolvedEntities, RunPythonSpec 

31from app.extensions.apps.mysql_backups.forms import ( 

32 BackupConfig, 

33 BackupConfigAll, 

34 BackupConfigServer, 

35 BackupCreate, 

36 CompressionAlgorithm, 

37 resolve_xtrabackup_compression, 

38 UploadProvider, 

39) 

40from app.extensions.apps.mysql_backups.models import BackupType 

41from app.extensions.apps.mysql_backups.payload_variants import variant_name 

42 

43_BASE_REQUIREMENTS = "packaging\nPyYAML\nPyMySQL[rsa,ed25519]" 

44

46def _xtrabackup_payload_name(upload: list[UploadProvider]) -> str: 

47 """Return the xtrabackup payload variant carrying exactly the selected providers. 

48 

49 The canonical payload ships all three upload-provider classes and sits within a 

50 few bytes of the 16 KiB Nomad dispatch limit, so every other selection gets a 

51 generated variant with the unreachable providers omitted. Keyed on the set of 

52 providers, so the form's ordering does not change the dispatched payload. 

53 

54 :param upload: The upload providers the form selected (possibly empty). 

55 :return: The payload filename beside this module. 

56 """ 

57 selected = set(upload) 

58 return variant_name(tuple(p.value for p in UploadProvider if p in selected)) 

59 

60 

61def _compression_override(form: BackupCreate) -> dict[str, CompressionAlgorithm]: 

62 """Return the compression key a blank XtraBackup algorithm needs, else nothing. 

63 

64 ``exclude_none`` drops a blank algorithm from the dispatched config, leaving the 

65 payload to default it alone — off the backup type, which cannot know which 

66 binary runs. Resolving it here keeps the dispatched config and the gated form 

67 telling one story. 

68 

69 :param form: The submitted create form. 

70 :return: A single-key mapping to merge into the config, or an empty one. 

71 """ 

72 if form.backup_type != BackupType.XTRABACKUP or form.compression_algorithm: 

73 return {} 

74 return { 

75 "compression_algorithm": resolve_xtrabackup_compression(form.xtrabackup_bin_cmd) 

76 }

79def build_backup_spec(form: BackupCreate, resolved: ResolvedEntities) -> RunPythonSpec: 

80 """Build the ``run-python`` backup spec from the validated form. 

81 

82 Select the per-``backup_type`` server-config host (XtraBackup → ``localhost``; 

83 Binlog → the alternative host or the service address; Mydumper → the service 

84 address), serialise the ``BackupConfig`` to the YAML ``config``, and select the 

85 ``file://`` payload and pip requirements by ``backup_type``. XtraBackup is keyed 

86 on ``form.upload`` as well: the dispatched payload is the variant carrying 

87 exactly the selected providers, and ``boto3`` is requested only when that 

88 variant can reach S3. The framework's ``assemble_envelope`` fills ``target`` 

89 (the executor ``HostRef``), ``_service_name``, and the connectivity keys around 

90 this spec. 

91 

92 :param form: The validated create form (a ``BackupCreate``). 

93 :param resolved: The entities resolved from the form's reference fields; its 

94 ``service`` is the ``ServiceRef`` selection (always present — the field is 

95 required). 

96 :return: The run-python spec consumed by ``assemble_envelope``. 

97 :raises ValueError: When ``form.backup_type`` is outside the known backup types. 

98 """ 

99 service = resolved.service 

100 

101 all_config = form.model_dump( 

102 exclude={ 

103 "task_name", 

104 "hostname", 

105 "service_id", 

106 "backup_type", 

107 "encryption_recipient", 

108 "alias", 

109 }, 

110 ) | _compression_override(form) 

111 

112 server_config = { 

113 "alias": form.alias or service.node.address, 

114 "backup_type": form.backup_type, 

115 # XtraBackup must run on the database host itself 

116 "host": ( 

117 "localhost" 

118 if form.backup_type == BackupType.XTRABACKUP 

119 else form.binlog_alternative_host 

120 if form.backup_type == BackupType.BINLOG and form.binlog_alternative_host 

121 else service.node.address 

122 ), 

123 "port": service.port, 

124 "upload": list(form.upload), 

125 } 

126 

127 # Keyed off the recipient (not the encrypt flags) so it survives whichever 

128 # mode is on — post-run with in-place off still needs it. Safe only because 

129 # BackupCreate's gates guarantee recipient <=> some encryption mode. 

130 if form.encryption_recipient: 

131 server_config["dir_encrypt_config"] = { 

132 "encryption_recipient": form.encryption_recipient 

133 } 

134 

135 backup_config = BackupConfig( 

136 all_servers=BackupConfigAll.model_validate(all_config), 

137 server_list=[BackupConfigServer.model_validate(server_config)], 

138 ) 

139 

140 requirements = _BASE_REQUIREMENTS 

141 if form.backup_type == BackupType.MYDUMPER: 

142 payload_name = "mydumper_payload" 

143 requirements += "\nboto3\nfilelock" 

144 elif form.backup_type == BackupType.XTRABACKUP: 

145 payload_name = _xtrabackup_payload_name(form.upload) 

146 # Only the variants carrying the S3 provider import boto3; asking for it 

147 # anyway would make the task install a dependency it never loads. 

148 if UploadProvider.S3 in form.upload: 

149 requirements += "\nboto3" 

150 requirements += "\nfilelock" 

151 elif form.backup_type == BackupType.BINLOG: 151 ↛ 155line 151 didn't jump to line 155 because the condition on line 151 was always true

152 payload_name = "binlog_payload" 

153 requirements += "\nboto3" 

154 else: 

155 raise ValueError(f"Invalid Backup Type {form.backup_type}") 

156 return RunPythonSpec( 

157 config=yaml.dump( 

158 jsonable_encoder(backup_config, by_alias=True, exclude_none=True) 

159 ), 

160 requirements=requirements, 

161 payload=payload_uri(__file__, payload_name), 

162 )
