82class RetirableSQLModel(RetiredAtBase, BaseSQLModel): 

83 """Store the retirement state of a tombstoned entity. 

84 

85 An entity that vanishes upstream is retired rather than deleted, so the 

86 references PMM Extensions persisted to it keep resolving. Because a replacement may 

87 reuse the retired row's unique key, ``retirement_key`` joins every unique 

88 index: it holds :data:`ACTIVE_RETIREMENT_KEY` while the row is active and 

89 the row's own primary key once retired, which keeps active rows mutually 

90 exclusive while letting any number of tombstones share their key. 

91 

92 :param retired_at: When the entity stopped being reported by its upstream 

93 source, or None while it is active. 

94 :param retirement_key: The discriminator carried inside every unique index. 

95 Excluded from serialization: the responses nest these table models, and a 

96 database-internal discriminator has no business in the published schema. 

97 """ 

98 

99 retirement_key: int = SQLField( 

100 default=ACTIVE_RETIREMENT_KEY, nullable=False, exclude=True 

101 )

278 @model_validator(mode="after") 

279 def validate_at_least_one_observation_field(self) -> Self: 

280 """Ensure at least one observed fact is set. 

281 

282 :return: The validated instance. 

283 :raises ValueError: If every observation field is unset. 

284 """ 

285 if all(getattr(self, name) is None for name in HOST_OBSERVATION_FIELD_NAMES): 

286 raise ValueError( 

287 f"At least one of {', '.join(sorted(HOST_OBSERVATION_FIELD_NAMES))} " 

288 "must be set", 

289 ) 

290 return self 

291 

292 

293#: Fields on a host observation that record a fact rather than identify or date one. 

294#: Derived from the declaring base so the model stays the single source of truth as 

295#: fields change. It must NOT be derived from a concrete subclass: the table and 

296#: response models also inherit ``BaseSQLModel``'s ``id`` / ``created_at`` / 

297#: ``updated_at``, and an always-populated ``created_at`` would satisfy the validator 

298#: for an otherwise empty observation. 

299HOST_OBSERVATION_FIELD_NAMES = frozenset(HostSystemObservationBase.model_fields) - { 

300 "node_id", 

301 "observed_at", 

302}
