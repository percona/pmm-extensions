33_BYTES_PER_KIB = 1024 

34_SIZE_UNITS = ("B", "KiB", "MiB", "GiB", "TiB") 

35_LABEL_PATH_MAX = 48 

36_LABEL_PATH_TAIL = (_LABEL_PATH_MAX - 1) // 2 

37 

38# Scan at most this many catalog pages when filling Choice options so a run of 

39# rows that yield no option — unusable (missing location / shell-unsafe) or 

40# collapsed into a value already offered — cannot starve the selector of older 

41# valid backups, while still bounding DB work for a free-text-friendly endpoint. 

42_MAX_CHOICE_SCAN_PAGES = 10

45def _format_size(size_bytes: int | None) -> str: 

46 """Render a byte count as a short binary unit label. 

47 

48 :param size_bytes: The size in bytes, or ``None`` when unknown. 

49 :return: A human-readable size such as ``1.0 GiB``, or ``unknown size``. 

50 """ 

51 if size_bytes is None: 

52 return "unknown size" 

53 size = float(size_bytes) 

54 unit_index = 0 

55 while size >= _BYTES_PER_KIB and unit_index < len(_SIZE_UNITS) - 1: 

56 size /= _BYTES_PER_KIB 

57 unit_index += 1 

58 if unit_index == 0: 

59 return f"{int(size)} {_SIZE_UNITS[unit_index]}" 

60 return f"{size:.1f} {_SIZE_UNITS[unit_index]}"

63def backup_source_label(run: MysqlBackupRun, *, value: str) -> str: 

64 """Build a human-readable selector label for a catalogued backup run. 

65 

66 :param run: A catalogued backup run. 

67 :param value: The Choice value (restore-valid location) shown in the label. 

68 :return: A label combining backup type, finish time, size, and location. 

69 """ 

70 type_label = BackupType.LABELS.get(run.backup_type, str(run.backup_type)) 

71 if run.finished_at is None: 

72 finished = "unknown time" 

73 else: 

74 finished = run.finished_at.astimezone(UTC).strftime("%Y-%m-%d %H:%M UTC") 

75 location = shorten_text( 

76 value, 

77 max_length=_LABEL_PATH_MAX, 

78 keep_last_chars=_LABEL_PATH_TAIL, 

79 ellipsis="…", 

80 ) 

81 return f"{type_label} · {finished} · {_format_size(run.size_bytes)} · {location}"

84def backup_run_to_choice(run: MysqlBackupRun) -> Choice | None: 

85 """Map one catalog row to a ``Choice``, or ``None`` when it has no usable value. 

86 

87 Skips locations the restore form's shell-safety validator would reject, so 

88 the selector never offers a value that 422s on submit. 

89 

90 :param run: A catalogued backup run. 

91 :return: A ``Choice`` whose ``value`` is restore-valid, or ``None``. 

92 """ 

93 value = restore_valid_backup_source(run.upload_destination, run.location) 

94 if value is None: 

95 return None 

96 return Choice(value=value, label=backup_source_label(run, value=value))

99async def choices_for_service( 

100 session: AsyncSession, key: CatalogServiceKey 

101) -> list[Choice]: 

102 """Return Choice options for a service's catalogued runs, newest first. 

103 

104 Offers each restore value once, under the label of the most recently finished 

105 run that resolves to it: a rerun that republishes into the directory an 

106 earlier run wrote to leaves both rows in the catalog, but only the newer 

107 run's dump is still there, so an older row's label would name data the 

108 restore would not write. The surviving row is the first the catalog yields 

109 for a value, so this rests on ``MysqlBackupRunManager.list_for_service`` 

110 ordering by ``finished_at`` descending. 

111 

112 Pages catalog rows until ``DEFAULT_PAGINATION_LIMIT`` distinct values are 

113 collected (or rows/pages are exhausted), so runs that are filtered out or 

114 collapsed do not shrink the usable option set below the intended cap. 

115 

116 :param session: The database session the catalog is queried on. 

117 :param key: The service whose catalog rows are mapped to options; a free-typed 

118 destination with no inventory row carries a name and no id. 

119 :return: Choice-compatible options; at most ``DEFAULT_PAGINATION_LIMIT`` items. 

120 """ 

121 by_value: dict[str, Choice] = {} 

122 offset = 0 

123 for _ in range(_MAX_CHOICE_SCAN_PAGES): 

124 page = await MysqlBackupRunManager.list_for_service( 

125 session, 

126 key, 

127 pagination=Pagination(offset=offset, limit=DEFAULT_PAGINATION_LIMIT), 

128 ) 

129 for run in page.items: 

130 choice = backup_run_to_choice(run) 

131 if choice is None or choice.value in by_value: 

132 continue 

133 by_value[choice.value] = choice 

134 if len(by_value) >= DEFAULT_PAGINATION_LIMIT: 

135 return list(by_value.values()) 

136 if not page.items or offset + len(page.items) >= page.total: 

137 break 

138 offset += len(page.items) 

139 return list(by_value.values())
