39async def ensure_task_schedulable(tasks_api: TaskAPI, task_name: str) -> None: 

40 """Refuse to schedule ``task_name`` unless an installed app offers scheduling. 

41 

42 :param tasks_api: The Tasks API client used to read the task. 

43 :param task_name: The name of the task a schedule would run. 

44 :raises HTTPUnprocessableEntityException: If ``task_name`` is not a single 

45 plain URL path segment. 

46 :raises HTTPBadRequestException: If no installed app offers scheduling for the 

47 task — because no registered app claims its owner, or because one does and 

48 withholds the capability. 

49 :raises HTTPException: Re-raised unchanged for an upstream client error 

50 (status < 500), including the ``404`` for an unknown task. 

51 :raises HTTPBadGatewayException: For an upstream server error (status >= 500) 

52 or a connection-level ``OSError``. 

53 """ 

54 with reraise_upstream_tasks_errors(): 

55 task = await get_task_by_name(tasks_api, task_name) 

56 if not get_app_registry().owner_offers_scheduling(task.owner): 

57 raise HTTPBadRequestException( 

58 f"Task {task_name!r} cannot be scheduled: " 

59 "no installed app offers scheduling for it." 

60 )

63async def require_schedulable_update( 

64 periodic_task_id: int, 

65 tasks_api: TaskAPI, 

66 body: Annotated[ArbitraryMapping, Body()], 

67) -> None: 

68 """Guard a schedule update on the task the schedule will run. 

69 

70 The Tasks service stores the body's ``task`` and keeps the existing schedule's 

71 task when that is empty, so the guard resolves the same name. A present 

72 ``task`` that is not a string is refused here: the Tasks service writes a 

73 non-string one into the schedule's ``kwargs.task_name``, and an explicit 

74 ``null`` is not the omission the fallback below reads it as. 

75 

76 :param periodic_task_id: The id of the schedule being replaced. 

77 :param tasks_api: The Tasks API client used to read the schedule and its task. 

78 :param body: The ``PeriodicTaskUpdate`` JSON body, read and left unaltered. 

79 :raises HTTPUnprocessableEntityException: If the body's ``task`` is present and 

80 not a string, or if the resolved name is not a single plain path segment. 

81 :raises HTTPBadRequestException: If no installed app offers scheduling for the 

82 resolved task. 

83 :raises HTTPException: Re-raised unchanged for an upstream client error 

84 (status < 500), including the ``404`` for an unknown schedule or task. 

85 :raises HTTPBadGatewayException: If the existing schedule carries no task name 

86 or is not a JSON object, and for an upstream server error (status >= 500) 

87 or a connection-level ``OSError``. 

88 """ 

89 if "task" in body and not isinstance(body["task"], str): 

90 raise HTTPUnprocessableEntityException( 

91 "The schedule's task must be a task name string." 

92 ) 

93 task_name = body.get("task") 

94 if not task_name: 

95 with reraise_upstream_tasks_errors(): 

96 schedule = as_json_object( 

97 await tasks_api.get(f"/periodic/{periodic_task_id}") 

98 ) 

99 task_name = schedule.get("task") 

100 if not isinstance(task_name, str) or not task_name: 

101 raise HTTPBadGatewayException( 

102 "The Tasks API returned a schedule with no task name." 

103 ) 

104 await ensure_task_schedulable(tasks_api, task_name)
