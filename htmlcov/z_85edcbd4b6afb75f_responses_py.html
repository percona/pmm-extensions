16"""Define the JSON API response models for the Snippets plugin. 

17 

18The shape is snippet-centric: the list endpoint returns snippet entities, 

19not tasks. Execution is a verb applied to a single snippet 

20(``POST /snippet/execute``) and produces a task in the tasks API via the 

21framework's :class:`~app.extensions.apps.framework.script_source.ScriptSource` seam; 

22the request/response bodies for that route are the framework's 

23``ScriptExecuteWrite`` / ``ScriptExecutionResponse``. 

24"""

46class SnippetResponse(BaseModel): 

47 """Represent a snippet entity as exposed by the JSON API. 

48 

49 :param filename: The snippet's filename on disk; doubles as its 

50 identifier in the API. 

51 :param title: The display title for the snippet (snippet metadata's 

52 ``title`` field, falling back to ``filename`` when unset). 

53 :param description: The snippet's free-text description, or an empty 

54 string when no description is set in metadata. 

55 :param service_type: The snippet's free-form service type 

56 (``service_type`` metadata field, for example ``"mysql"`` or 

57 ``"mongodb"``), or ``None`` when the snippet declares no service 

58 type. Distinct from the inventory ``ServiceTypeEnum``. 

59 :param size: Snippet file size in bytes. 

60 :param md5_digest: 32-character MD5 hex digest of the snippet file. 

61 :param is_approved: Whether the snippet has been approved for execution. 

62 :param approved_at: When the snippet was last approved, or ``None`` if 

63 unapproved. 

64 :param updated_by: User id that last toggled the approval state, or 

65 ``None`` if no toggle has occurred. 

66 :param reason: Free-form reason recorded the last time the snippet's 

67 approval state changed. 

68 :param requires_sudo: Whether the snippet requires sudo for execution 

69 (either always-sudo or sudo is user-toggleable). 

70 :param sudo_optional: Whether the user can toggle sudo at execution 

71 time. 

72 :param sudo_default: Default value for the sudo toggle when 

73 ``sudo_optional`` is ``True``. 

74 :param interpreter: The shell/interpreter command used to execute the 

75 snippet (for example, ``"bash"`` or ``"python3"``); ``None`` when 

76 no interpreter mapping resolves. 

77 :param created_at: When the snippet row was first inserted. 

78 :param updated_at: When the snippet row was last updated, or ``None`` 

79 if never updated since insert. 

80 """ 

81 

82 filename: NonEmptyStr 

83 title: NonEmptyStr 

84 description: str 

85 service_type: str | None = None 

86 size: int 

87 md5_digest: str 

88 is_approved: bool 

89 approved_at: UTCDatetime | None = None 

90 updated_by: str | None = None 

91 reason: str 

92 requires_sudo: bool 

93 sudo_optional: bool 

94 sudo_default: bool 

95 interpreter: str | None = None 

96 created_at: UTCDatetime 

97 updated_at: UTCDatetime | None = None

100def build_snippet_response(snippet: Snippet) -> SnippetResponse: 

101 """Project a :class:`Snippet` into its API response shape. 

102 

103 :param snippet: The snippet ORM row to transform. 

104 :return: The API response model mapped from ``snippet``. 

105 """ 

106 return SnippetResponse( 

107 filename=snippet.filename, 

108 title=snippet.title, 

109 description=snippet.description, 

110 service_type=snippet.service_type, 

111 size=snippet.size, 

112 md5_digest=snippet.md5_digest, 

113 is_approved=snippet.is_approved, 

114 approved_at=snippet.approved_at, 

115 updated_by=snippet.updated_by, 

116 reason=snippet.reason, 

117 requires_sudo=( 

118 snippet.sudo == SnippetSudoOption.ALWAYS or snippet.sudo.is_optional 

119 ), 

120 sudo_optional=snippet.sudo.is_optional, 

121 sudo_default=snippet.sudo.sudo_default, 

122 interpreter=snippet.execution_interpreter, 

123 created_at=snippet.created_at, 

124 updated_at=snippet.updated_at, 

125 )

128class SnippetBatchApproveRequest(BaseModel): 

129 """Represent the JSON body for ``PATCH /api/apps/snippets/approvals``. 

130 

131 Unlike the Form-bound :class:`~app.extensions.snippets.deps.SnippetBatchApproveForm` 

132 twin this is a plain Pydantic body — no ``Form()`` annotations — so FastAPI 

133 parses it as JSON. 

134 

135 :param filenames: Unique, non-empty list of snippet filenames to approve in a 

136 single atomic operation. Duplicates are silently deduplicated by 

137 ``UniqueList``. 

138 """ 

139 

140 filenames: UniqueList[NonEmptyStr] = Field(min_length=1)

143class BatchApprovalResponse(BaseModel): 

144 """Successful response for the batch-approve endpoint. 

145 

146 :param approved: Filenames whose approval state was toggled by this 

147 request (newly approved as a result of the call). 

148 :type approved: list[str] 

149 :param skipped_already_approved: Filenames that were already approved 

150 when the call started; the request is treated as a soft-skip 

151 (idempotent). 

152 :type skipped_already_approved: list[str] 

153 """ 

154 

155 approved: list[str] 

156 skipped_already_approved: list[str] 

157 

158 @computed_field 

159 @property 

160 def count(self) -> int: 

161 """Return the number of newly approved snippets. 

162 

163 :return: Length of ``approved``. 

164 :rtype: int 

165 """ 

166 return len(self.approved)

169class BatchApprovalErrorResponse(BaseModel): 

170 """Hard-error response payload for the batch-approve endpoint (400). 

171 

172 Returned only when the precheck rejects the whole request — either 

173 because some filenames have no DB row, or because the underlying 

174 files have been removed from disk. ``already_approved`` is *not* 

175 listed here; that is treated as a soft-skip in the success path. 

176 

177 :param missing_in_db: Filenames the request asked to approve that have 

178 no matching snippet row. 

179 :type missing_in_db: list[str] 

180 :param missing_on_disk: Filenames whose row exists but whose underlying 

181 file is no longer present. 

182 :type missing_on_disk: list[str] 

183 """ 

184 

185 missing_in_db: list[str] = Field(default_factory=list) 

186 missing_on_disk: list[str] = Field(default_factory=list)

214class SnippetsCapabilitiesResponse(AppDeploymentCapabilities): 

215 """Represent per-deployment capability flags for the Snippets plugin. 

216 

217 Exposes flags that gate the visibility of admin-only UI affordances 

218 (currently the manual refresh button) so the React shell can decide 

219 whether to render those controls without probing the gated endpoints. 

220 

221 Distinct from :class:`~app.extensions.apps.framework.schema.Capabilities`, 

222 which describes static UI feature flags on 

223 :attr:`~app.extensions.apps.framework.schema.AppSchema.capabilities` 

224 (chaining, scheduling, alert_on_fail). This model is the per- 

225 deployment runtime counterpart returned by ``GET /capabilities``. 

226 

227 :param manual_sync_enabled: Whether manual snippet refresh is enabled 

228 in this deployment (mirrors ``snippets_settings.ENABLE_MANUAL_SYNC``). 

229 :type manual_sync_enabled: bool 

230 """ 

231 

232 manual_sync_enabled: bool
