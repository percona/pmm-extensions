47#: on one event loop may consume together. Strictly below 1.0: 

48#: ``process_initializer`` does more than the seeds, and the child must still 

49#: send ``WORKER_UP`` inside the same window.

57def _remaining_seed_budget(loop: asyncio.AbstractEventLoop, window: float) -> float: 

58 """Return what is left of ``loop``'s shared inline-seed window. 

59 

60 The first bounded seed on ``loop`` opens a window of ``window`` seconds, and 

61 every later seed on the same loop is bounded by what remains, so the seeds of 

62 all refreshers in one prefork child together stay inside 

63 :data:`SEED_TIMEOUT_FRACTION` of the liveness deadline. Keyed by loop because 

64 each child gets a fresh loop at ``worker_process_init``; the entry is dropped 

65 with the loop. 

66 

67 :param loop: The event loop the seed runs on. 

68 :param window: The whole window in seconds, opened on the first call for 

69 ``loop``. 

70 :return: The seconds left in the window, never negative. 

71 """ 

72 now = time.monotonic() 

73 deadline = _seed_deadlines.setdefault(loop, now + window) 

74 return max(0.0, deadline - now)

77class WorkerRefresher: 

78 """Own one prefork child's settings-override boundary refresher. 

79 

80 Hold the lifecycle boilerplate every worker-side refresher needs — the 

81 enabled gate, the idempotent already-armed early-return, the initial 

82 inline seed, and disarm on shutdown — so each service module only 

83 supplies its own event loop, session maker and proxy set. After 

84 :meth:`start`, refreshes are pulled from ``task_prerun`` via 

85 :meth:`maybe_refresh` rather than driven by a background 

86 ``asyncio.sleep`` loop: a prefork child's loop only runs inside 

87 ``run_until_complete`` windows, so a timer-based cycle could starve. 

88 :meth:`prime` warms the proxies in the worker's main process before the 

89 fork, so each child's seed skips the settings build. 

90 

91 Every dependency is a zero-argument callable resolved at :meth:`start` 

92 time, never a value captured at construction: the instance is a 

93 module-level singleton built at import, while the event loop is recreated 

94 per prefork child and the session maker is rebound in tests. The 

95 last-refresh timestamp is likewise per-child after the fork. ``now`` is 

96 the same shape so tests can advance the due-check without writing 

97 ``_last_refresh``. 

98 

99 ``interval``, ``enabled`` and ``proc_alive_timeout`` are :meth:`start` 

100 parameters rather than reads of ``app.core.config.settings`` because the 

101 override substrate must not import that module at runtime — ``Settings`` 

102 is itself built from :mod:`app.core.settings_override.proxy`. 

103 

104 :param loop_getter: Returns the event loop that drives the seed and 

105 boundary refreshes. 

106 :param session_maker_factory: Returns the service-scoped 

107 ``async_sessionmaker`` the refresh cycle reads override rows through. 

108 :param proxies_factory: Composes the proxy registry to refresh, invoked 

109 once per effective :meth:`start` and once per effective :meth:`prime`. 

110 :param now: Returns the monotonic clock used for the due-check and 

111 last-refresh stamp. Defaults to :func:`time.monotonic`. 

112 """ 

113

132 def start( 

133 self, 

134 interval: timedelta | None = None, 

135 *, 

136 enabled: bool | None = None, 

137 callbacks: CallbackRegistry | None = None, 

138 proc_alive_timeout: float | None = None, 

139 ) -> None: 

140 """Seed this child's overrides and arm boundary refresh, unless disabled. 

141 

142 The initial refresh awaits inline so the snapshot is seeded before this 

143 method returns. No periodic ``asyncio.Task`` is created: subsequent 

144 refreshes are pulled from :meth:`maybe_refresh` at task boundaries. 

145 

146 When ``proc_alive_timeout`` is set, the inline seed is bounded via 

147 :func:`~app.core.settings_override.lifecycle.bounded_seed` by what 

148 remains of the loop's shared seed window: 

149 :data:`SEED_TIMEOUT_FRACTION` of that deadline, across every refresher 

150 seeding on the same loop. A hanging database therefore cannot push the 

151 child past the prefork pool's liveness window however many refreshers 

152 it runs. A seed that starts after the window is spent runs with a 0s 

153 budget, so it expires unless it completes without yielding. On 

154 seed-budget expiry the child is still armed — the next due 

155 ``task_prerun`` will refresh once any cancelled seed has finished 

156 unwinding — and may retain a possibly incomplete seed (proxies 

157 published before the hang keep DB overrides; others stay on their prior 

158 snapshots) until then. A cancelled seed that is still unwinding is 

159 retained on ``_pending_refresh`` so :meth:`maybe_refresh` does not open 

160 another session while the stuck cleanup holds a pool connection. 

161 

162 :param interval: Minimum delay between boundary refreshes. ``None``, 

163 the default, reads 

164 ``Settings.SETTINGS_OVERRIDE.REFRESH_INTERVAL``. 

165 :param enabled: Whether to arm a refresher at all. When ``False`` 

166 nothing is resolved (neither the proxy registry nor the session 

167 maker) and the child stays disarmed. ``None``, the default, reads 

168 ``Settings.SETTINGS_OVERRIDE.REFRESHER_ENABLED``. 

169 :param callbacks: Optional rebind callbacks fired by a boundary 

170 :func:`~app.core.settings_override.lifecycle.refresh_all` when a 

171 watched override changes value. The inline seed fires only those 

172 marked with :func:`~app.core.settings_override.lifecycle.fire_on_boot`, 

173 for keys it published an override for. 

174 :param proc_alive_timeout: The prefork pool's child-liveness deadline 

175 in seconds; the seed is bounded by what remains of the loop's 

176 shared window of :data:`SEED_TIMEOUT_FRACTION` of it. ``None`` (the 

177 default) leaves the inline seed unbounded and opens no window. 

178 :raises Exception: Re-raises whatever ``proxies_factory()`` raises — 

179 it is evaluated before the refresh starts — and whatever an 

180 unbounded seed propagates, in practice limited to 

181 ``session_maker_factory()`` failures. Per-proxy refresh failures 

182 are caught and logged inside ``refresh_all``. A bounded-seed 

183 expiry is caught inside :func:`~app.core.settings_override.lifecycle.bounded_seed` 

184 and does not propagate; the child is still armed. 

185 """ 

186 interval, enabled = resolve_refresher_options(interval, enabled=enabled) 

187 if not enabled: 

188 return 

189 if self._armed: 

190 return 

191 proxies = self._proxies_factory() 

192 loop = self._loop_getter() 

193 seed_timeout = ( 

194 None 

195 if proc_alive_timeout is None 

196 else _remaining_seed_budget( 

197 loop, proc_alive_timeout * SEED_TIMEOUT_FRACTION 

198 ) 

199 ) 

200 seeded, pending = loop.run_until_complete( 

201 bounded_seed( 

202 self._session_maker_factory, 

203 proxies, 

204 seed_timeout, 

205 callbacks=callbacks, 

206 ) 

207 ) 

208 now = self._now() 

209 self._last_refresh = now if seeded else now - interval.total_seconds() 

210 self._pending_refresh = ( 

211 pending if pending is not None and not pending.done() else None 

212 ) 

213 self._interval_seconds = interval.total_seconds() 

214 self._proxies = proxies 

215 self._callbacks = callbacks 

216 self._armed = True

218 def prime(self, *, enabled: bool | None = None) -> None: 

219 """Resolve every proxy's YAML/env settings before the prefork pool forks. 

220 

221 Intended for a ``worker_init`` receiver, which Celery dispatches in the 

222 worker's main process after the include modules are imported and before 

223 the pool forks. Each proxy caches its resolved instance, so the children 

224 inherit it and their inline seed does not rebuild the settings class, 

225 which re-reads the YAML profile, inside its seed budget. Nothing is 

226 armed, and neither the event loop nor the session maker is resolved. A 

227 proxy that fails to resolve stays unresolved, and the child retries it 

228 the next time it resolves that proxy. 

229 

230 :param enabled: Whether the refresher will arm at all. When ``False`` 

231 nothing is resolved. ``None``, the default, reads 

232 ``Settings.SETTINGS_OVERRIDE.REFRESHER_ENABLED``. 

233 :raises Exception: Re-raises whatever ``proxies_factory()`` raises; 

234 Celery logs an exception from a ``worker_init`` receiver and the 

235 worker continues. 

236 """ 

237 _, enabled = resolve_refresher_options(None, enabled=enabled) 

238 if not enabled: 

239 return 

240 for setting_class, entry in self._proxies_factory().items(): 

241 try: 

242 entry.proxy.get_base() 

243 except Exception: 

244 logger.debug( 

245 "Deferring %s settings resolution to the child seed", 

246 setting_class, 

247 exc_info=True, 

248 )

250 def maybe_refresh(self) -> None: 

251 """Refresh overrides when armed and the interval has elapsed. 

252 

253 Intended for ``task_prerun`` receivers. The due-check is a monotonic 

254 comparison with no I/O: tasks arriving inside the interval no-op. 

255 When a prior bounded refresh (seed or boundary) was cancelled and is 

256 still unwinding, this method also no-ops without awaiting that task — 

257 so a stuck ``AsyncSession.__aexit__`` cannot pile up further pool 

258 checkouts at later boundaries. 

259 When due, ``refresh_all`` runs to completion inside a single 

260 ``run_until_complete`` window, bounded by the refresh interval itself 

261 via :func:`~app.core.settings_override.lifecycle.bounded_refresh` 

262 (``asyncio.wait``, cancel without awaiting unwind — not 

263 ``wait_for``, which would still hang on a stuck 

264 ``AsyncSession.__aexit__``). Budget expiry logs once at WARNING and 

265 retains the cancelled task handle; exceptions raised from the 

266 ``run_until_complete`` call are logged and swallowed. A failure that 

267 surfaces only while a cancelled refresh is still unwinding is logged 

268 by the drain callback when that task finishes, not by this method. 

269 Neither case fails or aborts the task that triggered the refresh. 

270 Because proxies publish sequentially, a timed-out or failed cycle may 

271 leave a split registry (earlier proxies refreshed, later ones on their 

272 prior snapshots) rather than rolling everything back. 

273 The interval stamp advances on every attempted due refresh so a 

274 failing cycle cannot hammer the database on every subsequent 

275 dispatch. 

276 """ 

277 if not self._armed or self._proxies is None: 

278 return 

279 if self._pending_refresh is not None: 

280 if not self._pending_refresh.done(): 280 ↛ 282line 280 didn't jump to line 282 because the condition on line 280 was always true

281 return 

282 self._pending_refresh = None 

283 now = self._now() 

284 if now - self._last_refresh < self._interval_seconds: 

285 return 

286 try: 

287 completed, pending = self._loop_getter().run_until_complete( 

288 bounded_refresh( 

289 self._session_maker_factory, 

290 self._proxies, 

291 self._interval_seconds, 

292 self._callbacks, 

293 ) 

294 ) 

295 if not completed: 

296 self._pending_refresh = ( 

297 pending if pending is not None and not pending.done() else None 

298 ) 

299 logger.warning( 

300 "Settings-override boundary refresh exceeded its %.2fs budget; " 

301 "leaving a possibly incomplete refresh in place", 

302 self._interval_seconds, 

303 ) 

304 except Exception: 

305 logger.exception( 

306 "Settings-override boundary refresh failed; " 

307 "leaving a possibly incomplete refresh in place" 

308 ) 

309 finally: 

310 self._last_refresh = self._now()

312 def stop(self) -> None: 

313 """Disarm this child's refresher; a no-op when never started. 

314 

315 After disarm, :meth:`maybe_refresh` no-ops so a shut-down child 

316 performs no further boundary refresh. There is no periodic task to 

317 cancel. A cancelled refresh still unwinding is dropped from 

318 ``_pending_refresh`` without being awaited — the prefork child is 

319 exiting. 

320 """ 

321 self._armed = False 

322 self._proxies = None 

323 self._callbacks = None 

324 self._pending_refresh = None
