16"""Record a completed MySQL backup run into the catalog. 

17 

18This module owns the map from a terminal backup run onto one persisted 

19:class:`~app.extensions.apps.mysql_backups.models.MysqlBackupRun`. The tasks service 

20resolves it lazily through :func:`app.tasks.hook_resolver.resolve_hook`, 

21following the ``"module:function"`` path the backup task carries in 

22``Task.run_result_recorder`` (stamped at creation from 

23:data:`RUN_RESULT_RECORDER`), so this catalog knowledge stays inside the plugin 

24rather than leaking into ``app/tasks``. 

25 

26The recorder is invoked at every terminal status with the payload-reported 

27result (``None`` when a run reported nothing — an older payload, or a 

28non-success run). It records exactly one row per *successful* mydumper or 

29xtrabackup run and nothing for anything else: binlog runs, non-success 

30terminals, and unknown tools leave no record. A success that reported no result 

31(or a partial one) still records the run, leaving the unreported fields empty. 

32""" 

33

51T = TypeVar("T")

53#: Importable ``"module:function"`` path of this module's run-result recorder. 

54#: The backups app stamps it onto every backup task (``Task.run_result_recorder``) 

55#: at creation so the tasks service resolves the recorder lazily without 

56#: statically importing the plugin. 

57RUN_RESULT_RECORDER = f"{__name__}:record_backup_run" 

58 

59#: Tools that produce a per-run artifact worth cataloguing. Binlog runs 

60#: continuously and has no per-run completion to record, so it is excluded. 

61_CATALOGUED_TYPES = frozenset({BackupType.MYDUMPER, BackupType.XTRABACKUP})

64def _coerce(value: Any, expected: type[T], field: str) -> T | None: 

65 """Return ``value`` only when it is exactly ``expected``, else ``None``. 

66 

67 The result dict is written by a payload on a remote host, so every field is 

68 untrusted: a wrong-typed value is dropped rather than stored. ``bool`` is 

69 rejected for an ``int`` field, since ``True``/``False`` are ``int`` subclasses 

70 and never a valid size. A *present* value of the wrong type is a contract 

71 violation and is logged before being dropped; an absent field (``None``) 

72 passes through silently. 

73 

74 :param value: The candidate value pulled from the reported result. 

75 :param expected: The type the field must be to be kept. 

76 :param field: The field name, used only in the drop warning. 

77 :return: ``value`` when it matches ``expected``, otherwise ``None``. 

78 """ 

79 if expected is int and isinstance(value, bool): 

80 kept: T | None = None 

81 else: 

82 kept = value if isinstance(value, expected) else None 

83 if kept is None and value is not None: 

84 logger.warning( 

85 "Dropping malformed run-result field %r: expected %s, got %r.", 

86 field, 

87 expected.__name__, 

88 value, 

89 ) 

90 return kept

93def _positive_int(value: Any, field: str) -> int | None: 

94 """Return ``value`` only when it is a positive ``int``, else ``None``. 

95 

96 Layers a range guard over :func:`_coerce` for the inventory ids carried in 

97 task meta, which start at 1. A non-positive id would key the row to a service 

98 that cannot exist *and* drop it out of the ``service_id IS NULL`` name 

99 fallback the catalog query uses, leaving the row unreachable — recording it as 

100 absent keeps the name path open. 

101 

102 :param value: The candidate value pulled from the task meta. 

103 :param field: The meta key name, used only in the drop warning. 

104 :return: ``value`` when it is a positive ``int``, otherwise ``None``. 

105 """ 

106 coerced = _coerce(value, int, field) 

107 if coerced is not None and coerced <= 0: 

108 logger.warning( 

109 "Dropping out-of-range meta field %r: expected a positive id, got %r.", 

110 field, 

111 coerced, 

112 ) 

113 return None 

114 return coerced

117async def record_backup_run( 

118 session: AsyncSession, # noqa: ARG001 — seam contract; see below 

119 history: TaskHistory, 

120 result: dict[str, Any] | None, 

121) -> None: 

122 """Persist one catalog record for a successful mydumper/xtrabackup run. 

123 

124 No-ops for every case that must leave no record: a non-success terminal 

125 (failed, stopped, lost), a binlog or unknown-tool run, and a run already 

126 catalogued (idempotent on ``task_history_id``, so a re-invocation or a 

127 concurrent second call cannot create a duplicate). For a success worth 

128 recording, writes one row filling only the fields the run actually reported; 

129 a ``None`` or partial ``result`` leaves ``location``/``size_bytes``/ 

130 ``upload_destination`` empty rather than failing. 

131 

132 The ``mysql_backup_run`` table is owned by the **extensions** database, but the 

133 recorder seam opens and passes a *tasks*-database session — the two are 

134 distinct engines under SQLite (and only coincidentally the same shared 

135 database under Postgres). So the write is done on a fresh extensions session this 

136 function opens itself; the passed ``session`` is intentionally unused (all 

137 reads come off the already-loaded ``history``), kept only to satisfy the 

138 seam's ``(session, history, result)`` contract. 

139 

140 :param session: The tasks-database session opened by the recorder seam; 

141 unused — kept for the seam contract (see above). 

142 :param history: The terminal ``TaskHistory``, with ``task`` loaded. 

143 :param result: The payload-reported result, or ``None`` when none was read. 

144 """ 

145 if history.status != TaskHistoryStatusEnum.SUCCESS: 

146 return 

147 

148 task = history.task 

149 task_data = task.data if task else None 

150 marker = extract_backup_type_marker(task_data) 

151 try: 

152 backup_type = BackupType(marker) 

153 except ValueError: 

154 return 

155 if backup_type not in _CATALOGUED_TYPES: 

156 return 

157 

158 meta = task_data.get("meta") if isinstance(task_data, dict) else None 

159 meta = meta if isinstance(meta, dict) else {} 

160 reported = result if isinstance(result, dict) else {} 

161 

162 record = MysqlBackupRun( 

163 task_history_id=history.id, 

164 service_name=_coerce( 

165 meta.get(SERVICE_NAME_META_KEY), str, SERVICE_NAME_META_KEY 

166 ), 

167 service_id=_positive_int(meta.get(SERVICE_ID_META_KEY), SERVICE_ID_META_KEY), 

168 hostname=_coerce(meta.get("target"), str, "target"), 

169 backup_type=backup_type, 

170 location=_coerce(reported.get("backup_dir"), str, "backup_dir"), 

171 upload_destination=_coerce( 

172 reported.get("upload_destination"), str, "upload_destination" 

173 ), 

174 size_bytes=_coerce(reported.get("size_bytes"), int, "size_bytes"), 

175 started_at=history.started_at, 

176 finished_at=history.finished_at, 

177 ) 

178 

179 async with get_async_session_maker()() as extensions_session: 

180 await MysqlBackupRunManager.get_or_create( 

181 extensions_session, record, filter_include={"task_history_id"} 

182 )
