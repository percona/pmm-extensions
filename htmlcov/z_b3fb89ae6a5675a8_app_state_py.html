97async def list_apps(session: SessionDep) -> list[AppInfoResponse]: 

98 """List every configured app with its current enabled state. 

99 

100 Returns one entry per registered app, in declaration order (each parent is 

101 immediately followed by its child apps). Apps with no row default to 

102 ``ENABLED`` (``enabled=True``, ``toggleable=True``); protected apps 

103 (``inventory``) are forced to ``ENABLED`` and reported with 

104 ``toggleable=False``. A child app (``parent_key`` set) is not independently 

105 toggleable either (``toggleable=False``) and derives its lifecycle from the 

106 parent via :attr:`~app.extensions.apps.framework.base.BaseApp.state_key`. The list is 

107 non-paginated: app cardinality is bounded (<20). 

108 

109 :param session: The database session. 

110 :return: The per-app info list. 

111 """ 

112 states = await AppStateManager.all_lifecycle_states(session) 

113 return [

116 name=app.name, 

117 lifecycle_state=lifecycle, 

118 enabled=lifecycle == AppLifecycleEnum.ENABLED, 

119 toggleable=app.key not in PROTECTED_APP_KEYS and app.parent_key is None, 

120 uri_path=app.uri_path, 

121 css_class=app.css_class, 

122 sidebar=app.sidebar, 

123 has_api_router=app.api_router is not None, 

124 )

134@router.put("/{app_key:path}/state") 

135async def update_app_state( 

136 app_key: ToggleableAppKeyDep, 

137 body: AppStateWrite, 

138 session: SessionDep, 

139 celery_beat_session: CeleryBeatSessionDep, 

140) -> AppStateResponse: 

141 """Transition an app to a new lifecycle state. 

142 

143 Validates the requested edge against the allowed transitions 

144 (``ENABLED`` -> ``DISABLING``, ``DISABLED`` -> ``ENABLING``, 

145 ``DISABLING`` -> ``DISABLED``, ``ENABLING`` -> ``ENABLED``); an illegal edge 

146 returns 409. Returns 409 for protected apps (``inventory``) and 404 if the 

147 key does not match any configured plugin. A configured app with no row yet is 

148 treated as ``ENABLED`` for the gate and gets its row created with the 

149 requested state. Returns the updated row as an :class:`AppStateResponse` 

150 (the return annotation drives FastAPI's response schema). 

151 

152 After the ``AppState`` write commits, the app's owned periodic schedules are 

153 re-gated via :func:`app.extensions.periodic_tasks.apply_effective_enabled` so a 

154 non-``ENABLED`` app also stops its Celery beat tasks (and a return to 

155 ``ENABLED`` resumes them, subject to each schedule's ``user_enabled`` 

156 override). 

157 

158 A ``DISABLING`` transition on an app with no in-flight tasks drains 

159 immediately: :func:`app.extensions.app_drain.finalize_drain_if_complete` flips it 

160 straight to ``DISABLED`` (no ``task_postrun`` event will ever fire for an idle 

161 app), and the response reflects that resulting ``DISABLED`` state. 

162 

163 An ``ENABLING`` transition completes synchronously: there is no warm-up to 

164 wait for and nothing else ever advances ``ENABLING``, so it is flipped 

165 straight to ``ENABLED`` (before re-gating, so the app's schedules resume in 

166 the same request) and the response reflects that resulting ``ENABLED`` state. 

167 

168 :param app_key: The app key to transition. 

169 :type app_key: str 

170 :param body: The requested target lifecycle state. 

171 :type body: AppStateWrite 

172 :param session: The PMM Extensions database session. 

173 :type session: SessionDep 

174 :param celery_beat_session: The celery-beat database session. 

175 :return: The updated app-state response payload. 

176 :rtype: AppStateResponse 

177 :raises HTTPConflictException: When the requested transition edge is illegal. 

178 """ 

179 current = await AppStateManager.current_lifecycle(session, app_key) 

180 AppStateManager.assert_transition_allowed(current, body.lifecycle_state) 

181 state, created = await AppStateManager.get_or_create( 

182 session, 

183 AppStateBase(app_key=app_key, lifecycle_state=body.lifecycle_state), 

184 filter_include={"app_key"}, 

185 ) 

186 if not created: 

187 state = await AppStateManager.update(session, state, body) 

188 resulting_state = state.lifecycle_state 

189 # Enabling has no warm-up to wait for (only the disable path drains in-flight 

190 # work), so ENABLING completes synchronously: flip it straight to ENABLED. 

191 # This must happen *before* the re-gate below because 

192 # ``apply_effective_enabled`` only re-arms an app whose state is ENABLED, and 

193 # nothing else (no reconciler, no task signal) ever advances ENABLING — so 

194 # without this the app would sit in ENABLING forever. 

195 if resulting_state == AppLifecycleEnum.ENABLING: 

196 await AppStateManager.update_where( 

197 session, 

198 {"lifecycle_state": AppLifecycleEnum.ENABLED}, 

199 app_key=app_key, 

200 lifecycle_state=AppLifecycleEnum.ENABLING, 

201 ) 

202 resulting_state = AppLifecycleEnum.ENABLED 

203 await apply_effective_enabled(session, celery_beat_session, app_keys={app_key}) 

204 if ( 204 ↛ anywhereline 204 didn't jump anywhere: it always raised an exception.

205 resulting_state == AppLifecycleEnum.DISABLING 

206 and await finalize_drain_if_complete(session, app_key) 

207 ): 

208 resulting_state = AppLifecycleEnum.DISABLED 

209 return AppStateResponse( 

210 app_key=state.app_key, 

211 lifecycle_state=resulting_state, 

212 enabled=resulting_state == AppLifecycleEnum.ENABLED, 

213 )

216@router.post("/{app_key:path}/force-disable") 

217async def force_disable_app( 

218 app_key: ToggleableAppKeyDep, 

219 session: SessionDep, 

220) -> AppStateResponse: 

221 """Force a draining app straight to ``DISABLED``, terminating its tasks. 

222 

223 The emergency escape hatch when a cooperative drain stalls: it terminates 

224 each of the app's in-flight tasks with ``revoke(..., terminate=True)`` 

225 (SIGTERM), deletes their :class:`app.extensions.models.AppRunningTask` rows, and 

226 transitions the app to ``DISABLED``. Requires the app to already be 

227 ``DISABLING`` (else 409); the default toggle path (``PUT .../state``) never 

228 issues a terminating revoke. No periodic re-gating is needed — the owned 

229 schedules were already gated off at the ``DISABLING`` write. 

230 

231 :param app_key: The app to force-disable. 

232 :param session: The PMM Extensions database session. 

233 :return: The resulting ``DISABLED`` app-state response. 

234 :raises HTTPConflictException: When the app is not currently ``DISABLING``. 

235 """ 

236 current = await AppStateManager.current_lifecycle(session, app_key) 

237 if current != AppLifecycleEnum.DISABLING: 

238 raise HTTPConflictException( 

239 detail=f"App '{app_key}' is not draining (state {current}); " 

240 "force-disable only applies to a DISABLING app.", 

241 ) 

242 running = await AppRunningTaskManager.list(session, app_key=app_key) 

243 for task in running: 

244 celery.control.revoke(task.celery_task_id, terminate=True) 

245 await AppRunningTaskManager.delete_where(session, app_key=app_key) 

246 await AppStateManager.update_where( 

247 session, 

248 {"lifecycle_state": AppLifecycleEnum.DISABLED}, 

249 app_key=app_key, 

250 ) 

251 return AppStateResponse( 

252 app_key=app_key, 

253 lifecycle_state=AppLifecycleEnum.DISABLED, 

254 enabled=False, 

255 )
