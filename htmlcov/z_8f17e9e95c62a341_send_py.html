16"""Build and deliver an incident's diagnostics bundle, recording every attempt. 

17 

18The ``atw_send_log`` row *is* the job: the API creates it and enqueues the Celery 

19task with its id, and this module drives it to a terminal status. Because the 

20receiver's credentials can create attachments but not read them back, the row's 

21``detail`` is the only evidence a send ever landed -- so every write assigns a 

22fresh mapping and flags the JSON column, and a broad terminal guard makes sure no 

23failure family leaves a row stuck mid-flight. 

24"""

119class _CappedWriter: 

120 """Wrap a binary handle, refusing writes once the file outgrows a cap. 

121 

122 The cap is enforced *during* the build rather than on the finished file, so a 

123 runaway bundle stops mid-stream instead of after the whole thing is staged. 

124 """ 

125 

126 def __init__(self, handle: BinaryIO, cap_mb: int) -> None: 

127 """Bind the wrapper to the handle it guards. 

128 

129 :param handle: The open binary handle to write through. 

130 :param cap_mb: The largest the file may grow, in mebibytes. 

131 """ 

132 self._handle = handle 

133 self._cap_mb = cap_mb 

134 self._cap_bytes = cap_mb * _BYTES_PER_MIB 

135 

136 def write(self, data: bytes) -> int: 

137 """Write ``data`` through, then refuse if the file outgrew its cap. 

138 

139 :param data: The bytes to write. 

140 :return: The number of bytes written. 

141 :raises AtwBundleSizeError: When the file has grown past the cap. 

142 """ 

143 written = self._handle.write(data) 

144 if self._handle.tell() > self._cap_bytes: 

145 raise AtwBundleSizeError( 

146 f"Bundle exceeded the configured {self._cap_mb} MiB limit while " 

147 f"being built." 

148 ) 

149 return written 

150 

151 def tell(self) -> int: 

152 """Return the handle's current offset. 

153 

154 :return: The offset in bytes. 

155 """ 

156 return self._handle.tell() 

157 

158 def seek(self, offset: int, whence: int = 0) -> int: 

159 """Seek the underlying handle. 

160 

161 :param offset: The target offset. 

162 :param whence: How to interpret ``offset``. 

163 :return: The new absolute offset. 

164 """ 

165 return self._handle.seek(offset, whence) 

166 

167 def seekable(self) -> bool: 

168 """Report the handle as seekable so :mod:`zipfile` patches its headers. 

169 

170 :return: Whether the underlying handle supports seeking. 

171 """ 

172 return self._handle.seekable() 

173 

174 def flush(self) -> None: 

175 """Flush the underlying handle.""" 

176 self._handle.flush()

215def _entry_arcname(prefix: str, path: str, *, is_dir: bool) -> str: 

216 """Return the archive entry name one upstream file is written under. 

217 

218 A directory streams from the Tasks API as an on-the-fly tar.gz, so its entry 

219 is named for what it actually holds rather than for the directory. Upstream 

220 path components are filtered because :meth:`zipfile.ZipFile.open` writes 

221 whatever name it is handed, traversal and all -- unlike 

222 :meth:`zipfile.ZipFile.write`, which sanitizes. 

223 

224 :param prefix: The per-execution namespace. 

225 :param path: The file's path within that execution's output. 

226 :param is_dir: Whether the entry arrives as a tar.gz stream. 

227 :return: The entry name to write the member under. 

228 """ 

229 parts = [part for part in PurePosixPath(path).parts if part not in ("/", "..")] 

230 arcname = f"{prefix}/{'/'.join(parts)}" 

231 return f"{arcname}.tar.gz" if is_dir else arcname 

232

234def _log_arcname(prefix: str, step: str, stream: str) -> str: 

235 """Return the archive entry name one execution's log group is written under. 

236 

237 ``step`` is an upstream Nomad task name, so its components are filtered for 

238 the same reason :func:`_entry_arcname` filters an upstream file path. 

239 

240 :param prefix: The per-execution namespace. 

241 :param step: The step whose logs the member holds. 

242 :param stream: The stream the member holds, ``stdout`` or ``stderr``. 

243 :return: The entry name to write the member under. 

244 """ 

245 parts = [ 

246 part 

247 for part in PurePosixPath(f"{step}.{stream}.log").parts 

248 if part not in ("/", "..") 

249 ] 

250 return f"{prefix}/logs/{'/'.join(parts)}"

253async def _execution_status( 

254 tasks_api: RemoteAPI, execution: dict[str, Any] 

255) -> TaskHistoryStatusEnum | None: 

256 """Return one execution's upstream status. 

257 

258 The send log snapshots only an execution's identity, so whether its logs are 

259 safe to fetch has to be read back from the Tasks API. 

260 

261 :param tasks_api: The authenticated Tasks API client. 

262 :param execution: The selected execution descriptor. 

263 :return: The upstream status, or ``None`` when it is unrecognized. 

264 :raises AtwSendError: When the execution's status cannot be read. 

265 """ 

266 task_history_id = execution["task_history_id"] 

267 with _reraise_upstream_errors( 

268 f"Could not read the status of {_execution_label(execution)}" 

269 ): 

270 payload = await tasks_api.get(f"/history/{task_history_id}") or {} 

271 status = payload.get("status") 

272 try: 

273 return TaskHistoryStatusEnum(status) 

274 except ValueError: 

275 logger.warning( 

276 "Skipping the logs of execution %s: unrecognized status %r.", 

277 task_history_id, 

278 status, 

279 ) 

280 return None

283async def _add_execution_files( 

284 archive: zipfile.ZipFile, tasks_api: RemoteAPI, execution: dict[str, Any] 

285) -> list[dict[str, Any]]: 

286 """Stream one execution's output files into the archive. 

287 

288 Entries are written strictly one at a time: :mod:`zipfile` rejects a second 

289 open writable member while one is still open. 

290 

291 :param archive: The open archive to write into. 

292 :param tasks_api: The authenticated Tasks API client. 

293 :param execution: The selected execution descriptor. 

294 :return: One manifest entry per file written. 

295 :raises AtwSendError: When the execution's files cannot be listed or streamed 

296 -- most often because it has not finished, or its output has aged out of 

297 the executor node. 

298 :raises AtwBundleSizeError: Propagated from the archive's capped writer when 

299 this execution's files push the bundle past the plan's cap. 

300 """ 

301 task_history_id = execution["task_history_id"] 

302 with _reraise_upstream_errors( 

303 f"Could not list output files for {_execution_label(execution)}" 

304 ): 

305 listing = await tasks_api.get(f"/history/{task_history_id}/files/") or {} 

306 

307 prefix = _execution_prefix(execution) 

308 written: list[dict[str, Any]] = [] 

309 for path, metadata in listing.items(): 

310 is_dir = bool(metadata.get("is_dir")) 

311 arcname = _entry_arcname(prefix, path, is_dir=is_dir) 

312 with _reraise_upstream_errors( 

313 f"Could not read {path!r} from {_execution_label(execution)}" 

314 ): 

315 size = await _write_entry( 

316 archive, tasks_api, task_history_id, path, arcname 

317 ) 

318 written.append( 

319 {"path": path, "arcname": arcname, "size": size, "is_dir": is_dir} 

320 ) 

321 return written

324async def _write_entry( 

325 archive: zipfile.ZipFile, 

326 tasks_api: RemoteAPI, 

327 task_history_id: int, 

328 path: str, 

329 arcname: str, 

330) -> int: 

331 """Stream one upstream file into one archive member. 

332 

333 :param archive: The open archive to write into. 

334 :param tasks_api: The authenticated Tasks API client. 

335 :param task_history_id: The execution the file belongs to. 

336 :param path: The file's path within that execution's output. 

337 :param arcname: The entry name to write it under. 

338 :return: The number of bytes streamed. 

339 :raises AtwBundleSizeError: Propagated from the archive's capped writer when 

340 this file pushes the bundle past the plan's cap. 

341 :raises HTTPException: Propagates the Tasks API's error status for the stream. 

342 :raises OSError: Propagates a connection or timeout failure mid-stream. 

343 :raises ClientError: Propagates a truncated or malformed response mid-stream. 

344 """ 

345 size = 0 

346 with archive.open(arcname, "w", force_zip64=True) as entry: 

347 async for chunk in tasks_api.stream_chunks( 

348 f"/history/{task_history_id}/file/", params={"path": path} 

349 ): 

350 size += len(chunk) 

351 entry.write(chunk) 

352 return size

355def _decode_log_line(line: bytes, task_history_id: int) -> dict[str, Any] | None: 

356 """Decode one NDJSON line of an execution's log stream. 

357 

358 :param line: The raw line, carrying the trailing newline 

359 :meth:`RemoteAPI.stream` preserves. 

360 :param task_history_id: The execution the line came from. 

361 :return: The decoded record, or ``None`` when the line is blank, unparseable, 

362 or not a JSON object. 

363 """ 

364 if not (text := line.strip()): 

365 return None 

366 try: 

367 record = json.loads(text) 

368 except json.JSONDecodeError: 

369 logger.warning( 

370 "Skipping an unparseable log line from execution %s.", task_history_id 

371 ) 

372 return None 

373 if not isinstance(record, dict): 

374 logger.warning( 

375 "Skipping a non-object log line from execution %s.", task_history_id 

376 ) 

377 return None 

378 return record

381class _LogMember: 

382 """Hold the one open archive member a single log group is streamed into. 

383 

384 :mod:`zipfile` rejects a second open writable member while one is still open, 

385 so exactly one of these exists at a time and the caller replaces it when the 

386 group changes. 

387 """ 

388 

389 def __init__( 

390 self, archive: zipfile.ZipFile, prefix: str, group: tuple[str, str] 

391 ) -> None: 

392 """Open the member this group's messages are written into. 

393 

394 :param archive: The open archive to write into. 

395 :param prefix: The per-execution namespace. 

396 :param group: The group's ``(step, stream)`` pair. 

397 """ 

398 self.group = group 

399 self._arcname = _log_arcname(prefix, *group) 

400 self._size = 0 

401 self._handle = archive.open(self._arcname, "w", force_zip64=True) 

402 

403 def write(self, msg: str) -> int: 

404 """Append one log message to the member. 

405 

406 :param msg: The message to append. 

407 :return: The number of uncompressed bytes written. 

408 :raises AtwBundleSizeError: Propagated from the archive's capped writer 

409 when this message pushes the bundle past the plan's cap. 

410 """ 

411 written = self._handle.write(msg.encode()) 

412 self._size += written 

413 return written 

414 

415 def close(self) -> dict[str, Any]: 

416 """Finalize the member and describe it for the manifest. 

417 

418 Safe to call more than once, so a caller that closes on its normal path 

419 can still close unconditionally on the way out. 

420 

421 :return: The manifest entry describing the finished member. 

422 :raises AtwBundleSizeError: Propagated from the archive's capped writer 

423 when flushing the compressor's tail outgrows the plan's cap. 

424 """ 

425 self._handle.close() 

426 step, stream = self.group 

427 return { 

428 "step": step, 

429 "stream": stream, 

430 "arcname": self._arcname, 

431 "size": self._size, 

432 }

447async def _add_execution_logs( 

448 archive: zipfile.ZipFile, 

449 tasks_api: RemoteAPI, 

450 execution: dict[str, Any], 

451 step: str, 

452) -> tuple[list[dict[str, Any]], int]: 

453 """Stream one execution's captured logs for ``step`` into the archive. 

454 

455 A single step is fetched: for a run that happened, the prestart and poststop 

456 steps surrounding :data:`_MAIN_LOG_STEP` log setup machinery, which is noise 

457 on a support case. The one exception is a run that never started, whose only 

458 diagnostic lives in a prestart step — see :func:`_log_step_for`. 

459 

460 Members are keyed by a record's ``(step, stream)`` group and replaced when it 

461 changes -- the two upstream read paths both deliver contiguous runs per group 

462 but in opposite stream order, so keying on the group covers either without 

463 assuming a sequence. A record carrying no message is skipped before its group 

464 is considered, so a stream with nothing to say opens no member at all. 

465 

466 :param archive: The open archive to write into. 

467 :param tasks_api: The authenticated Tasks API client. 

468 :param execution: The selected execution descriptor. 

469 :param step: The Nomad step whose logs to stream. 

470 :return: One manifest entry per log group written, and the total number of 

471 uncompressed bytes they carry. 

472 :raises AtwSendError: When the execution's logs cannot be streamed. 

473 :raises AtwBundleSizeError: Propagated from the archive's capped writer when 

474 this execution's logs push the bundle past the plan's cap. 

475 :raises ValueError: Propagated from :meth:`RemoteAPI.stream` when one log line 

476 outgrows its line cap. 

477 """ 

478 task_history_id = execution["task_history_id"] 

479 prefix = _execution_prefix(execution) 

480 entries: list[dict[str, Any]] = [] 

481 total = 0 

482 member: _LogMember | None = None 

483 try: 

484 with _reraise_upstream_errors( 

485 f"Could not read logs for {_execution_label(execution)}" 

486 ): 

487 async for line in tasks_api.stream( 

488 f"/history/{task_history_id}/logs/", params={"step": step} 

489 ): 

490 record = _decode_log_line(line, task_history_id) 

491 if record is None: 

492 continue 

493 if not isinstance(msg := record.get("msg"), str) or not msg: 

494 continue 

495 group = (str(record.get("step", "")), str(record.get("type", ""))) 

496 if member is None or member.group != group: 

497 if member is not None: 

498 entries.append(member.close()) 

499 member = _LogMember(archive, prefix, group) 

500 total += member.write(msg) 

501 if member is not None: 

502 entries.append(member.close()) 

503 finally: 

504 if member is not None: 

505 member.close() 

506 return entries, total

529async def _stage_bundle( 

530 path: Path, 

531 row: AtwSendLog, 

532 incident_name: str, 

533 tasks_api: RemoteAPI, 

534 cap_mb: int, 

535) -> tuple[int, dict[str, Any]]: 

536 """Stage the incident's diagnostics as one zip on local disk. 

537 

538 :param path: Where to write the archive. 

539 :param row: The send log naming the selected executions. 

540 :param incident_name: The incident's human-readable label. 

541 :param tasks_api: The authenticated Tasks API client. 

542 :param cap_mb: The largest the bundle may grow, in mebibytes. 

543 :return: The number of collected output files and the bundle manifest. 

544 :raises AtwBundleSizeError: When the archive outgrows the plan's cap. 

545 :raises AtwNothingToSendError: When no execution produced a file or a log. 

546 :raises AtwSendError: When an execution's status, files, or logs cannot be 

547 collected. 

548 :raises ValueError: Propagated from :meth:`RemoteAPI.stream` when one log line 

549 outgrows its line cap. 

550 """ 

551 path.parent.mkdir(parents=True, exist_ok=True) 

552 manifest_executions: list[dict[str, Any]] = [] 

553 file_count = 0 

554 log_bytes = 0 

555 with ( 

556 path.open("wb") as handle, 

557 zipfile.ZipFile( 

558 _CappedWriter(handle, cap_mb), "w", zipfile.ZIP_DEFLATED 

559 ) as archive, 

560 ): 

561 for execution in row.detail.get("executions", []): 

562 status = await _execution_status(tasks_api, execution) 

563 files = await _add_execution_files(archive, tasks_api, execution) 

564 file_count += len(files) 

565 logs: list[dict[str, Any]] = [] 

566 if status is not None and status.is_finished(): 

567 logs, written = await _add_execution_logs( 

568 archive, tasks_api, execution, _log_step_for(status) 

569 ) 

570 log_bytes += written 

571 manifest_executions.append({**execution, "files": files, "logs": logs}) 

572 if not file_count and not log_bytes: 

573 raise AtwNothingToSendError(_NOTHING_TO_SEND_ERROR) 

574 manifest = _build_manifest(row, incident_name, manifest_executions) 

575 archive.writestr(_MANIFEST_ARCNAME, json_serializer(manifest)) 

576 return file_count, manifest

579async def _persist( 

580 session: AsyncSession, row: AtwSendLog, detail: dict[str, Any], **fields: Any 

581) -> None: 

582 """Write a fresh ``detail`` mapping and the given columns to the row. 

583 

584 A JSON column is not change-tracked through in-place mutation, so ``detail`` 

585 is always replaced wholesale and flagged -- a silently-dropped write here 

586 would lose the only record that a bundle reached the receiver. 

587 

588 :param session: The database session. 

589 :param row: The send log to update. 

590 :param detail: The evidence mapping to store. 

591 :param fields: Column values to set alongside it. 

592 :raises HTTPBadRequestException: Propagated from the manager when the row 

593 cannot be written. 

594 """ 

595 for name, value in fields.items(): 

596 setattr(row, name, value) 

597 row.detail = detail 

598 await AtwSendLogManager.save(session, row, flag_modified_fields=["detail"])

601async def run_send(send_log_id: UUID) -> None: 

602 """Build and deliver the bundle one send log describes. 

603 

604 Every failure family lands as a ``FAILED`` row carrying its reason: the row is 

605 the only place a support engineer can learn what happened, so the terminal 

606 write is guaranteed rather than best-effort — short of the database itself 

607 being unreachable, where the row is left to the stale sweep instead. 

608 

609 :param send_log_id: The send log driving this attempt. 

610 :raises HTTPNotFoundException: Propagated when the incident was deleted 

611 between the enqueue and this attempt. 

612 :raises HTTPBadRequestException: Propagated when a terminal row cannot be 

613 written. 

614 :raises Exception: Propagated when rolling back or refreshing after a failed 

615 settings re-read itself fails. Either the database is unreachable, so no 

616 terminal write was going to land either way and the row is left to the 

617 stale sweep, or the row was deleted under the send, so there is nothing 

618 left to write. 

619 """ 

620 async_session_maker = get_async_session_maker() 

621 async with async_session_maker() as session: 

622 row = await AtwSendLogManager.first(session, id=send_log_id) 

623 if row is None: 

624 logger.info( 

625 "Send log %s is gone; its incident was most likely deleted.", 

626 send_log_id, 

627 ) 

628 return 

629 await _run_send_for_row(session, row)

632async def _resolve_plan_after_refresh( 

633 session: AsyncSession, 

634 row: AtwSendLog, 

635) -> DeliveryPlanResolution: 

636 """Republish the PMM Extensions settings snapshot and resolve the delivery plan. 

637 

638 Every send resolves against a snapshot no older than itself, so a rotated 

639 secret, a repointed endpoint, or a first-time enabling write takes effect 

640 even when this worker child's task-boundary refresher has not yet run 

641 (boundary refresh does not advance mid-task or while a child is idle). 

642 

643 A failed republish must not escape and must not invent a fixed reason: this 

644 runs ahead of the broad terminal guard, so an escaping error would leave the 

645 row non-terminal until the stale sweep mislabels it as a lost worker, and a 

646 fixed unconfigured reason would turn a deliverable send — one whose in-hand 

647 snapshot already resolves a usable plan — into a terminal failure on a 

648 transient database error. Log, roll the session back so an aborted 

649 transaction does not also fail the terminal write that may still follow, 

650 refresh ``row`` so the rollback's expire does not break later attribute 

651 reads on the delivery path, and resolve against the snapshot this child 

652 already holds. 

653 

654 :param session: The database session. 

655 :param row: The send log being driven; refreshed after a failed republish 

656 so delivery can continue against the held snapshot. 

657 :return: The resolution taken against the fresh snapshot, or against the 

658 snapshot this child already holds when the republish failed. 

659 :raises Exception: Propagates a ``session.rollback()`` or 

660 ``session.refresh()`` that itself fails. Either the database is 

661 unreachable, so no terminal write was going to land either way and the 

662 row is left to the stale sweep, or ``row`` was deleted under the send — 

663 an incident cascades to its send logs — so there is nothing left to 

664 write. 

665 """ 

666 try: 

667 await republish_extensions_settings_snapshot(session) 

668 except Exception: 

669 logger.exception( 

670 "Could not re-read the diagnostics delivery settings; falling back " 

671 "to the snapshot this worker already holds." 

672 ) 

673 await session.rollback() 

674 await session.refresh(row) 

675 return resolve_delivery_plan()

678async def _run_send_for_row(session: AsyncSession, row: AtwSendLog) -> None: 

679 """Drive one send log from pending to a terminal status. 

680 

681 A row the stale sweep already failed is left alone: a task delivered late 

682 enough for that to happen would otherwise resurrect a terminal row and 

683 deliver a bundle the UI has already reported as failed — and, if the 

684 engineer re-sent in the meantime, attach it to the case twice. 

685 

686 The delivery settings are re-read once — a fresh snapshot is published — 

687 before the plan is resolved, so a send enqueued after an enabling write, a 

688 rotated secret, or a changed endpoint is not delivered against a snapshot 

689 older than it. A re-read that fails is the one exception: delivery then 

690 proceeds against the snapshot this child already holds. The failed row 

691 carries whichever reason the resolver gave, so an operator can tell inputs 

692 that stopped matching the plan from delivery nobody configured. 

693 

694 :param session: The database session. 

695 :param row: The send log to drive. 

696 :raises HTTPNotFoundException: Propagated from the manager when the incident 

697 was deleted between the enqueue and this attempt. 

698 :raises HTTPBadRequestException: Propagated from the manager when a terminal 

699 row cannot be written. 

700 :raises Exception: Propagated from ``_resolve_plan_after_refresh`` when 

701 rolling back or refreshing after a failed settings re-read itself 

702 fails. That call sits ahead of the broad terminal guard, so nothing 

703 here catches it. 

704 """ 

705 if row.status.is_terminal(): 

706 logger.info( 

707 "Send log %s is already %s; not delivering it again.", row.id, row.status 

708 ) 

709 return 

710 

711 detail = dict(row.detail) 

712 resolution = await _resolve_plan_after_refresh(session, row) 

713 if (reason := resolution.unavailable_reason) is not None: 

714 await _fail(session, row, detail, [], reason) 

715 return 

716 # A resolution refuses to hold neither member, so an empty reason is a plan. 

717 plan = cast(DeliveryPlan, resolution.plan) 

718 

719 incident = await AtwIncidentManager.get_or_404(session, id=row.incident_id) 

720 incident_name = incident.name 

721 await _persist( 

722 session, row, detail, status=AtwSendStatusEnum.RUNNING, started_at=utc_now() 

723 ) 

724 

725 records: list[StepRecord] = [] 

726 path = bundle_dir() / f"{row.id}-{uuid4().hex}{_BUNDLE_SUFFIX}" 

727 try: 

728 client = await get_tasks_api() 

729 with client.auth(get_internal_token()) as tasks_api: 

730 file_count, manifest = await _stage_bundle( 

731 path, row, incident_name, tasks_api, plan.max_bundle_size_mb 

732 ) 

733 size = path.stat().st_size 

734 async with get_delivery_executor( 

735 plan, step_observer=records.append 

736 ) as executor: 

737 with path.open("rb") as handle: 

738 result = await executor.upload_bundle( 

739 source_ref=f"atw-incident/{row.incident_id}", 

740 bundle=BundleSource(filename=path.name, content=handle, size=size), 

741 case_ref=row.case_ref, 

742 manifest=manifest, 

743 ) 

744 logger.info( 

745 "Diagnostics send %s delivered to case %s as %s: %s", 

746 row.id, 

747 row.case_ref, 

748 result.reference, 

749 result.detail, 

750 ) 

751 except Exception as exc: # noqa: BLE001 -- every family must land terminally 

752 logger.warning("Diagnostics send %s failed.", row.id, exc_info=True) 

753 await _fail(session, row, detail, records, _error_message(exc, records)) 

754 return 

755 finally: 

756 path.unlink(missing_ok=True) 

757 

758 await _persist( 

759 session, 

760 row, 

761 { 

762 **detail, 

763 "steps": [_step_detail(record) for record in records], 

764 "upload_response": None if result.detail is None else dict(result.detail), 

765 "upload_reference": result.reference, 

766 "bundle_size": size, 

767 "file_count": file_count, 

768 }, 

769 status=AtwSendStatusEnum.SUCCESS, 

770 finished_at=utc_now(), 

771 )

828def _reraise_upstream_errors(message: str) -> Iterator[None]: 

829 """Report an upstream Tasks-API failure raised in the block as a send failure. 

830 

831 A size-cap or decoding failure raised inside the block propagates untouched, 

832 so a local fault is not relabelled as an upstream one. ``OSError`` is the one 

833 blurred edge: a disk failure during a zip write inside the block is caught 

834 here too and reported against the Tasks API. 

835 

836 :param message: What the block was trying to do, named for a support engineer. 

837 :return: Control to the wrapped block. 

838 :raises AtwSendError: When the block raises an upstream failure, carrying 

839 ``message`` and the upstream's own detail. 

840 """

842 yield 

843 except (HTTPException, OSError, ClientError) as exc: 

844 raise AtwSendError(f"{message}: {_upstream_detail(exc)}") from exc

847def _error_message(exc: Exception, records: Sequence[StepRecord]) -> str: 

848 """Return the reason to record against a failed send. 

849 

850 A failure inside the delivery plan is attributed to the step that ended it 

851 and to the send inputs that step reads, ahead of the receiver's own message. 

852 A failure before any step reported one, such as bundle staging or the size 

853 cap, has no step to name and keeps the upstream detail alone. 

854 

855 :param exc: The exception that ended the attempt. 

856 :param records: The step transitions the executor observed. 

857 :return: A message a support engineer can act on. 

858 """ 

859 upstream = _upstream_detail(exc) 

860 failed = next( 

861 (record for record in reversed(records) if record.status == "failed"), None 

862 ) 

863 if failed is None: 

864 return upstream 

865 where = ( 

866 "the bundle upload" 

867 if failed.kind == "upload" 

868 else f"delivery step {failed.name!r}" 

869 ) 

870 reads = ( 

871 f", which reads {', '.join(failed.cited_inputs)}" if failed.cited_inputs else "" 

872 ) 

873 return f"The send failed in {where}{reads}: {upstream}"

876def purge_expired_bundles(ttl_seconds: int) -> int: 

877 """Delete staged bundles whose mtime is older than ``ttl_seconds``. 

878 

879 :param ttl_seconds: Maximum bundle age in seconds. 

880 :return: The number of bundles removed. 

881 """ 

882 staging = bundle_dir() 

883 if not staging.is_dir(): 

884 return 0 

885 cutoff = time.time() - ttl_seconds 

886 removed = 0 

887 for path in staging.glob(f"*{_BUNDLE_SUFFIX}"): 

888 try: 

889 if path.stat().st_mtime < cutoff: 

890 path.unlink(missing_ok=True) 

891 removed += 1 

892 except OSError: 

893 logger.exception("Failed to purge diagnostics bundle %s", path) 

894 return removed

897async def fail_stale_sends(stale_after: timedelta) -> int: 

898 """Drive sends abandoned by a lost worker to a terminal failed status. 

899 

900 Both non-terminal statuses need a driver: a worker killed between the enqueue 

901 and the terminal write leaves a row nothing else would ever advance, and the 

902 UI would poll it forever. Rows are aged from when they were last known to 

903 progress -- ``started_at`` once a worker picked one up, ``created_at`` while 

904 still queued. 

905 

906 :param stale_after: How long a send may stay non-terminal before it is failed. 

907 :return: The number of sends failed. 

908 :raises HTTPBadRequestException: Propagated from the manager when a swept row 

909 cannot be written. 

910 """ 

911 cutoff = utc_now() - stale_after 

912 async_session_maker = get_async_session_maker() 

913 async with async_session_maker() as session: 

914 rows = await AtwSendLogManager.list( 

915 session, 

916 col(AtwSendLog.status).in_(AtwSendStatusEnum.active_statuses()), 

917 func.coalesce(col(AtwSendLog.started_at), col(AtwSendLog.created_at)) 

918 < cutoff, 

919 ) 

920 for row in rows: 

921 await _persist( 

922 session, 

923 row, 

924 {**row.detail, "error": _WORKER_LOST_ERROR}, 

925 status=AtwSendStatusEnum.FAILED, 

926 finished_at=utc_now(), 

927 ) 

928 if rows: 

929 logger.info("Failed %d stale diagnostics send(s).", len(rows)) 

930 return len(rows)
