16"""Build archiver-specific detail blocks for task failure alerts. 

17 

18This module owns the archiver domain knowledge behind a failure alert: the 

19``PURGE_LIST`` -> (source, condition, target) mapping, the STDERR error-trace 

20extraction, the credential redaction applied before the detail egresses to an 

21external alerting provider, and the assembly of the owner-specific alert 

22additions. The generic tasks service consults it lazily through the 

23``app.tasks.alert_hooks`` resolver, following the ``"module:function"`` path the 

24archiver task carries in ``alert_detail_builder`` (stamped at creation from 

25:data:`ALERT_DETAIL_BUILDER`), so this knowledge stays inside the plugin package 

26rather than leaking into ``app/tasks``. 

27 

28The module depends only on ``app.tasks`` (the allowed direction) and is safe to 

29import standalone, so the lazy hook resolves identically in the Celery worker, 

30the PMM Extensions app, and the tasks API. The archiver config parser here is the single 

31source of truth for the purge-field mapping; the archives ``deps`` layer imports 

32it rather than duplicating the field extraction. 

33"""

51#: Importable ``"module:function"`` path of this module's failure-alert builder. 

52#: The archives ``deps`` layer stamps it onto the archiver task 

53#: (``TaskWrite.alert_detail_builder``) at creation so the tasks service can 

54#: resolve the builder lazily without statically importing the plugin. 

55ALERT_DETAIL_BUILDER = f"{__name__}:build_owner_alert_details" 

56 

57#: Substring marking an error line in the archiver's STDERR stream. Used by the 

58#: tail reader to ensure the last error block is fully captured. 

59STDERR_ERROR_MARKER = "ERROR" 

60 

61#: Minimum trailing STDERR bytes to accumulate before stopping the reverse scan 

62#: once an error marker has been seen. Covers the downstream trace cap 

63#: (:data:`MAX_TRACE_BYTES`) so an error block straddling a chunk boundary is 

64#: fully present in the reconstructed tail. 

65_STDERR_TAIL_MIN_BYTES = 4 * 1024 

66 

67#: Shown for the Error Details section when no trace can be extracted; the Error 

68#: Details block is never silently null/empty. 

69ARCHIVER_TRACE_PLACEHOLDER = "No error trace captured from STDERR." 

70 

71#: Shown for a configuration field that cannot be resolved from the task config. 

72ARCHIVER_FIELD_PLACEHOLDER = "unavailable" 

73 

74#: Hard cap on the emitted error trace, protecting the downstream alert payload. 

75MAX_TRACE_BYTES = 4096 

76 

77#: Replacement token substituted for any redacted credential. 

78_REDACTION_MASK = "***" 

79 

80#: ``scheme://user[:password]@`` userinfo prefix of a connection URI. 

81_URI_USERINFO_RE = re.compile(r"(?P<scheme>[a-zA-Z][a-zA-Z0-9+.\-]*://)[^/\s@]+@") 

82 

83#: Sensitive ``key=value`` / ``key: value`` credential pairs. Covers query-string 

84#: and config forms (``password``/``passwd``/``pwd``). 

85_KV_SECRET_RE = re.compile( 

86 r"(?i)(?P<key>password|passwd|pwd)\s*[=:]\s*(?P<val>[^&\s,;\"']+)" 

87) 

88 

89#: pt-archiver / Perl-DBI DSN password component ``p=<val>``. Case-sensitive: in a 

90#: DBI DSN lowercase ``p`` is the password while uppercase ``P`` is the port, so 

91#: only lowercase ``p`` is masked. The look-behind avoids matching the tail of a 

92#: longer token (e.g. ``...,p=`` matches, ``help=`` does not). 

93_DSN_PASSWORD_RE = re.compile(r"(?<![A-Za-z0-9_])(?P<key>p)=(?P<val>[^,\s\"']+)") 

94 

95#: CLI password flags: ``--password=val``, ``--password val``, ``-pVAL``. The 

96#: ``-p`` short flag is anchored to an argument boundary (look-behind) and 

97#: case-sensitive so it masks only a real ``mysql -pSECRET`` token, not the 

98#: ``-p`` inside ``--purge``/``--progress``/``--output-path`` nor the ``-P`` 

99#: (port) flag. ``--password`` stays case-insensitive. 

100_CLI_PASSWORD_RE = re.compile( 

101 r"(?P<flag>(?i:--password)(?:[=\s])|(?<![\w./-])-p(?=\S))(?P<val>[^\s\"']+)" 

102)

105def redact_secrets(text: str) -> str: 

106 """Mask credentials embedded in a free-text error trace. 

107 

108 Runs unconditionally (independent of the opt-in PII mask) so a connection 

109 string or password echoed by pt-archiver/MySQL into STDERR cannot leave the 

110 platform in a failure alert. Masks: URI userinfo (``scheme://user:pass@``), 

111 ``password``/``passwd``/``pwd`` key/value pairs, the Perl-DBI DSN ``p=`` 

112 component, and the ``--password``/``-p`` CLI flags. 

113 

114 :param text: The text that may contain credentials. 

115 :return: The text with any detected credential replaced by ``***``. 

116 """ 

117 text = _URI_USERINFO_RE.sub(rf"\g<scheme>{_REDACTION_MASK}@", text) 

118 text = _KV_SECRET_RE.sub(rf"\g<key>={_REDACTION_MASK}", text) 

119 text = _DSN_PASSWORD_RE.sub(rf"\g<key>={_REDACTION_MASK}", text) 

120 return _CLI_PASSWORD_RE.sub(rf"\g<flag>{_REDACTION_MASK}", text)

190def parse_archiver_purge_config(config_yaml: str | None) -> ArchiverPurgeFields | None: 

191 """Parse an archiver config YAML into its first-entry purge fields. 

192 

193 Only the first ``PURGE_LIST`` entry is represented. The function never 

194 raises: a missing, non-string, scalar, or unparseable config returns 

195 ``None`` so the failure-alert path can fall back to placeholders. 

196 

197 :param config_yaml: The serialized archiver config (``meta["config"]``). 

198 :return: The extracted fields, or ``None`` when no entry can be parsed. 

199 """ 

200 if not config_yaml: 

201 return None 

202 try: 

203 config = yaml.safe_load(config_yaml) 

204 except (yaml.YAMLError, TypeError): 

205 return None 

206 if not isinstance(config, dict): 

207 return None 

208 purge_list = config.get("PURGE_LIST") 

209 if not isinstance(purge_list, list) or not purge_list: 

210 return None 

211 purge_item = purge_list[0] 

212 if not isinstance(purge_item, dict): 212 ↛ 213line 212 didn't jump to line 213 because the condition on line 212 was never true

213 return None 

214 return ArchiverPurgeFields( 

215 source_db=purge_item.get("SOURCE_DB"), 

216 source_table=purge_item.get("SOURCE_TABLE"), 

217 where=purge_item.get("WHERE"), 

218 dest_db=purge_item.get("DEST_DB"), 

219 dest_table=purge_item.get("DEST_TABLE"), 

220 dest_file=purge_item.get("DEST_FILE"), 

221 source_query=purge_item.get("SOURCE_QUERY"),

225def extract_last_error_trace(stderr: str | None) -> str: 

226 """Extract the last error block from a failed execution's STDERR. 

227 

228 The trace is the contiguous run of lines from the last blank-line boundary 

229 preceding the final ``ERROR`` line through the end of the log, capped to 

230 :data:`MAX_TRACE_BYTES` (keeping the tail). When the log is empty or has no 

231 ``ERROR`` marker, :data:`ARCHIVER_TRACE_PLACEHOLDER` is returned so the 

232 Error Details section is never silently empty. 

233 

234 :param stderr: The decoded STDERR content, or ``None``. 

235 :return: The extracted error block, or the placeholder. 

236 """ 

237 if not stderr or not stderr.strip(): 

238 return ARCHIVER_TRACE_PLACEHOLDER 

239 

240 lines = stderr.splitlines() 

241 last_error_idx = None 

242 for idx, line in enumerate(lines): 

243 if STDERR_ERROR_MARKER in line: 

244 last_error_idx = idx 

245 if last_error_idx is None: 

246 return ARCHIVER_TRACE_PLACEHOLDER 

247 

248 start = 0 

249 for idx in range(last_error_idx, -1, -1): 

250 if not lines[idx].strip(): 

251 start = idx + 1 

252 break 

253 block = "\n".join(lines[start:]).strip() 

254 if not block: 254 ↛ 255line 254 didn't jump to line 255 because the condition on line 254 was never true

255 return ARCHIVER_TRACE_PLACEHOLDER 

256 

257 encoded = block.encode() 

258 if len(encoded) > MAX_TRACE_BYTES: 

259 block = encoded[-MAX_TRACE_BYTES:].decode(errors="ignore") 

260 return block

263def build_archiver_description( 

264 fields: ArchiverPurgeFields | None, 

265 error_trace: str, 

266 entities: set[PIIEntity], 

267) -> str: 

268 """Render the combined archiver failure description block. 

269 

270 The block always carries the four labeled lines (Error Details, Source, 

271 Condition, Target); unresolved config fields render 

272 :data:`ARCHIVER_FIELD_PLACEHOLDER`. The assembled text is first passed 

273 through :func:`redact_secrets` (always-on credential masking, independent of 

274 the PII mask) and then :func:`anonymize_text` so the configured PII entities 

275 are scrubbed before the detail leaves for the external alerting provider. An 

276 empty ``entities`` set skips only the PII pass; credential redaction still 

277 runs. 

278 

279 :param fields: The parsed archiver fields, or ``None`` when unavailable. 

280 :param error_trace: The extracted error trace or placeholder. 

281 :param entities: The PII entities to scrub from the assembled block. 

282 :return: The rendered (and anonymized) description block. 

283 """ 

284 source = (fields.source if fields else None) or ARCHIVER_FIELD_PLACEHOLDER 

285 condition = (fields.condition if fields else None) or ARCHIVER_FIELD_PLACEHOLDER 

286 target = (fields.target if fields else None) or ARCHIVER_FIELD_PLACEHOLDER 

287 

288 block = ( 

289 "=== ERROR DETAILS ===\n" 

290 f"{error_trace}\n" 

291 "\n" 

292 "=== ARCHIVER CONFIGURATION ===\n" 

293 f"Source: {source}\n" 

294 f"Condition: {condition}\n" 

295 f"Target: {target}" 

296 ) 

297 return anonymize_text(redact_secrets(block), entities)

300def _effective_entities(history: "TaskHistory") -> set[PIIEntity]: 

301 """Return the PII entities to scrub, preferring the history-level mask. 

302 

303 Falls back to the owning task's mask when the history has none, matching 

304 the documented :attr:`TaskHistory.anonymize_mask` semantics. A missing mask 

305 on both yields an empty set: PII scrubbing is opt-in (auto-masking the 

306 archiver DB/table/WHERE by default would gut the alert's usefulness). Note 

307 this only disables the *PII* pass — credential redaction in 

308 :func:`build_archiver_description` runs unconditionally regardless of mask. 

309 

310 :param history: The task execution history. 

311 :return: The set of PII entities to anonymize. 

312 """ 

313 mask = history.anonymize_mask 

314 if mask is None: 

315 mask = history.task.anonymize_mask 

316 return PIIEntity.decode_selection(mask or 0)

319async def _read_last_stderr(task_history_id: int) -> str | None: 

320 """Read and reconstruct the failed execution's trailing error STDERR. 

321 

322 Fetches the newest STDERR chunks (newest-first) via 

323 :meth:`TaskHistoryLogManager.get_stderr_tail_chunks` and reconstructs the 

324 chronological tail, scanning back far enough that an error block straddling a 

325 chunk boundary — including an ``ERROR`` marker split across the boundary 

326 itself — is fully captured. Opens its own session so the read works 

327 identically in the Celery worker and the tasks API, and swallows any error: 

328 a log-read failure must never prevent the failure alert itself from firing. 

329 

330 :param task_history_id: The ``TaskHistory`` identifier. 

331 :return: The trailing STDERR content, or ``None`` when unavailable. 

332 """ 

333 from app.tasks.crud import TaskHistoryLogManager 

334 from app.tasks.db import get_async_session_maker 

335 

336 try: 

337 async_session = get_async_session_maker() 

338 async with async_session() as session: 

339 chunks = await TaskHistoryLogManager.get_stderr_tail_chunks( 

340 session, task_history_id 

341 ) 

342 except Exception: 

343 logger.exception( 

344 "Failed to read STDERR for task history %s while building alert detail.", 

345 task_history_id, 

346 ) 

347 return None 

348 

349 return _reconstruct_error_tail(chunks)

352def _reconstruct_error_tail(chunks: list[str]) -> str | None: 

353 """Reconstruct the chronological error tail from newest-first chunks. 

354 

355 Walks the chunks newest-first, prepending each into the tail, and stops once 

356 an error marker has been seen and at least :data:`_STDERR_TAIL_MIN_BYTES` of 

357 trailing content has accumulated — enough that an error block straddling a 

358 chunk boundary is fully captured. Adjacent chunks are bridged with a short 

359 prefix so an ``ERROR`` marker split across the boundary is still detected. 

360 

361 :param chunks: The STDERR chunk contents ordered newest-first. 

362 :return: The reconstructed chronological tail, or ``None`` when empty. 

363 """ 

364 if not chunks: 

365 return None 

366 parts = [] 

367 total_bytes = 0 

368 marker_seen = False 

369 marker_overlap = len(STDERR_ERROR_MARKER) - 1 

370 for content in chunks: 

371 if not marker_seen: 371 ↛ 376line 371 didn't jump to line 376 because the condition on line 371 was always true

372 # Bridge the boundary with a prefix of the newer chunk so a marker 

373 # split across the two is still detected. 

374 boundary_prefix = parts[-1][:marker_overlap] if parts else "" 

375 marker_seen = STDERR_ERROR_MARKER in content + boundary_prefix 

376 parts.append(content) 

377 total_bytes += len(content.encode("utf-8")) 

378 if marker_seen and total_bytes >= _STDERR_TAIL_MIN_BYTES: 

379 break 

380 return "".join(reversed(parts))

383async def build_owner_alert_details( 

384 history: "TaskHistory", 

385) -> OwnerAlertDetails | None: 

386 """Build archiver failure-alert details for the given history. 

387 

388 Resolved lazily by ``app.tasks.alert_hooks`` from the ``alert_detail_builder`` 

389 path the archiver task carries (see :data:`ALERT_DETAIL_BUILDER`). Resolve the 

390 source database node and assemble the combined ``custom_details`` description 

391 block (error trace plus Source/Condition/Target). 

392 

393 The config is taken from the dispatch-time snapshot 

394 (``execution_request.meta``) so the alert describes what actually ran rather 

395 than a later edit to the mutable ``task.data["meta"]``; the task data is used 

396 only when the snapshot is empty (legacy histories). The source node 

397 (``meta["_pmm_node_name"]``, falling back to ``execution_request.target``) is 

398 the actionable identifier the summary exists to surface and is deliberately 

399 not PII-scrubbed; sensitive content lives in the redacted detail block. 

400 

401 :param history: The failed task execution history. 

402 :return: The owner-specific alert additions. 

403 """ 

404 meta = history.execution_request.meta or (history.task.data or {}).get("meta") or {} 

405 source_node = meta.get("_pmm_node_name") or history.execution_request.target 

406 

407 stderr = await _read_last_stderr(history.id) 

408 error_trace = extract_last_error_trace(stderr) 

409 fields = parse_archiver_purge_config(meta.get("config")) 

410 description = build_archiver_description( 

411 fields, error_trace, _effective_entities(history) 

412 ) 

413 

414 return OwnerAlertDetails( 

415 source_node=source_node, 

416 custom_details={"description": description}, 

417 )
