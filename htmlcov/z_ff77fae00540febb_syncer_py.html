16"""Implement the system facts inventory sync. 

17 

18The :class:`SystemFactsSyncer` enriches existing inventory entities with system facts 

19(OS version, installed packages, EOL-relevant config, and DB engine version) collected 

20by a task-based payload executed on the co-located executor host -- the same mechanism 

21as ``MySQLSyncer``. Facts are persisted as host/service *system observations* through the 

22inventory HTTP API; the syncer never creates or deletes nodes or services. 

23"""

83class SystemFactsSyncer(BaseTaskSyncer): 

84 """Collect host and service system facts and upsert them as inventory observations. 

85 

86 The base ``BaseSyncer`` lifecycle drives this syncer: ``perform_inventory_sync`` 

87 iterates existing inventory nodes, and for each node a single payload run collects 

88 host facts plus every co-located service's engine version. Results are cached so the 

89 per-service lifecycle steps do not trigger additional task runs. 

90 

91 :cvar SYNC_TO_LIMIT: The highest entity type synchronized. Set to 

92 ``SyncInventoryEntityTypeEnum.SERVICE`` (no schema/table recursion). 

93 :cvar mirrors_entity_levels: Empty, stated rather than inherited. This syncer 

94 walks nodes and services but writes a separate observation resource 

95 instead of their own fields, so refreshing either level's freshness would 

96 report a confirmation it never made — and would mask a failing PMM mirror 

97 of the same row. 

98 :cvar EOL_ENGINE_TYPES: Service types whose engine version is collected. 

99 """

200 async def resolve_task_target( 

201 self, host: str, name: str | None = None 

202 ) -> tuple[str, bool]: 

203 """Resolve the executor host and whether it is co-located with the node. 

204 

205 Mirrors ``get_task_target`` selection but additionally reports co-location: a 

206 node is co-located when an executor host matches it by name or address. When no 

207 executor matches (e.g. RDS / managed instances), the task still runs on a 

208 fallback host but host-level facts must be skipped. 

209 

210 :param host: The node's network address. 

211 :type host: str 

212 :param name: The node's name. Defaults to ``None``. 

213 :type name: str | None 

214 :return: A tuple of ``(target_host, is_colocated)``. 

215 :rtype: tuple[str, bool] 

216 :raises ExecutorHostNotFoundError: If ``strict_executor_matching`` is enabled and 

217 no executor host matches the node's name or address. 

218 """ 

219 # Co-location must reflect the *selected* target, not any matching host, so facts 

220 # from a forced/fallback executor aren't misattributed to the node. 

221 available_hosts = await self.get_available_hosts() 

222 target = await self.get_task_target(host, name) 

223 is_colocated = target == name or available_hosts.get(target) == host 

224 return target, is_colocated

231 async def fetch_node(self, created_node: CreatedNode) -> Node | None: 

232 """Collect host and service facts for a node via the task payload. 

233 

234 Runs the payload once on the resolved executor, caching the host facts (only when 

235 co-located) and every service's engine version for the per-service lifecycle. The 

236 node is always returned (never ``None``) so that service facts are still synced 

237 even when host facts are unavailable. 

238 

239 :param created_node: The node to collect facts for. 

240 :type created_node: CreatedNode 

241 :return: The node (unchanged); facts are stashed in the syncer caches. 

242 :rtype: Node | None 

243 """ 

244 services = [ 

245 service 

246 for service in created_node.services 

247 if self.can_sync_service(service) 

248 ] 

249 target, colocated = await self.resolve_task_target( 

250 created_node.address, created_node.name 

251 ) 

252 if not colocated and not services: 

253 # Nothing to collect: not the node's own host (no host facts) and no DB 

254 # engine service to probe. Skip the task run entirely. 

255 return created_node 

256 config = self.build_script_config(services, collect_host=colocated) 

257 meta = await self.build_meta(config, target) 

258 result = await self.wait_for_task_output(**meta) 

259 try: 

260 data = json.loads(result.stdout) 

261 except (json.JSONDecodeError, TypeError): 

262 logger.warning( 

263 "System facts payload returned non-JSON output for node %s", 

264 created_node.id, 

265 ) 

266 data = {} 

267 if not isinstance(data, dict): 267 ↛ 268line 267 didn't jump to line 268 because the condition on line 267 was never true

268 data = {} 

269 host_facts = data.get("host") 

270 if isinstance(host_facts, dict) and host_facts: 

271 self._host_facts_cache[created_node.id] = host_facts 

272 collected = data.get("services") 

273 if isinstance(collected, dict): 

274 # The payload keys facts by address; cache them by the unique service id so a 

275 # later sibling or failed collection cannot read another service's fact. 

276 service_by_address = {service.address: service for service in services} 

277 for address, fact in collected.items(): 

278 service = service_by_address.get(address) 

279 if service is not None and isinstance(fact, dict): 279 ↛ 277line 279 didn't jump to line 277 because the condition on line 279 was always true

280 self._service_facts_cache[service.id] = fact 

281 # Mark probed services (even versionless ones) so fetch_service won't re-collect. 

282 self._batch_attempted_services.update(service.id for service in services) 

283 return created_node

287 ) -> HostSystemObservationWrite | None: 

288 """Build a host observation, or ``None`` if no usable fact was collected. 

289

293 Dropping it would read back as never-observed, the inverse of what was

297 :return: A validated host observation, or ``None`` when nothing was collected. 

298 """

300 key: value 

301 for key in HOST_OBSERVATION_FIELD_NAMES

303 } 

304 if not fields:

306 return HostSystemObservationWrite( 

307 observed_at=host_facts.get("collected_at") or datetime.now(UTC),

388 async def fetch_service( 

389 self, created_service: CreatedService 

390 ) -> SystemFactsService | None: 

391 """Resolve the engine version for a service. 

392 

393 Uses the ``fetch_node`` cache when present (the scheduled-sync path). A standalone 

394 per-service sync (the service was never primed by a ``fetch_node`` batch) falls 

395 back to collecting just this service. A service that *was* batch-primed but whose 

396 version failed is skipped rather than re-collected, so a node with K dead services 

397 issues one task run, not ``1 + K``. 

398 

399 :param created_service: The service to fetch facts for. 

400 :type created_service: CreatedService 

401 :return: A carrier with the engine version, or ``None`` if none was collected 

402 (the service is then skipped, never written with an empty version). 

403 :rtype: SystemFactsService | None 

404 """ 

405 primed = created_service.id in self._batch_attempted_services 

406 self._batch_attempted_services.discard(created_service.id) 

407 fact = self._service_facts_cache.pop(created_service.id, None) 

408 if (not fact or not fact.get("db_engine_version")) and not primed: 

409 # Never primed by a batch run -> standalone per-service sync, collect directly. 

410 fact = await self._collect_single_service(created_service) 

411 if not fact or not fact.get("db_engine_version"): 

412 return None 

413 service = SystemFactsService.model_validate( 

414 created_service.model_dump(exclude={"node", "schemas"}) 

415 ) 

416 service.db_engine_version = fact["db_engine_version"] 

417 service.collected_at = fact.get("collected_at") 

418 return service

488

551 async def _due_by_ledger(self, nodes: list[CreatedNode]) -> list[CreatedNode]: 

552 """Return the nodes the retry policy lets this run attempt. 

553 

554 :param nodes: The unmeasured, measurable nodes. 

555 :return: Those due an attempt now, in the given order. 

556 """ 

557 node_ids = [node.id for node in nodes] 

558 own = await SyncItemManager.finished_entity_attempts( 

559 self._session, self.get_name(), SyncInventoryEntityTypeEnum.NODE, node_ids 

560 ) 

561 daily = await SyncItemManager.finished_entity_attempts( 

562 self._session, 

563 SystemFactsSyncer.get_name(), 

564 SyncInventoryEntityTypeEnum.NODE, 

565 node_ids, 

566 ) 

567 failures: Counter[int | None] = Counter() 

568 for item in own: 

569 if item.status == SyncStatusEnum.SUCCESS: 

570 failures[item.entity_id] = 0 

571 else: 

572 failures[item.entity_id] += 1 

573 latest_finish: dict[int | None, datetime] = {} 

574 for item in (*own, *daily): 

575 finished = item.finished_at 

576 latest_finish[item.entity_id] = max( 

577 finished, latest_finish.get(item.entity_id, finished) 

578 ) 

579 now = utc_now() 

580 return [ 

581 node 

582 for node in nodes 

583 if first_measurement_due( 

584 failures[node.id], 

585 latest_finish.get(node.id), 

586 now, 

587 retries=self.first_measurement_retries,
