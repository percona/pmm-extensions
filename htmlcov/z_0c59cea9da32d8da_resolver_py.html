16"""Resolve the effective delivery plan from the baked skeleton and its inputs. 

17 

18This module must stay out of ``app/extensions/bundle_upload/__init__.py``: 

19``app.extensions.config`` imports :class:`~app.extensions.bundle_upload.plan.DeliveryPlan` from 

20this package, so the package ``__init__`` executes while ``app.extensions.config`` is 

21still running its own module body, and importing ``extensions_settings`` from there 

22would fail. Import this module directly instead, as ``app.extensions.apps.atw.send`` 

23already does for ``factory``. 

24"""

46#: Reported when this deployment has no usable receiver at all: no baked plan, 

47#: or a declared secret still without a value. The operator has to configure 

48#: delivery. 

49UNCONFIGURED_REASON = "Diagnostics delivery is not configured" 

50 

51#: Reported when delivery was configured and the inputs have since stopped 

52#: fitting the plan, which an image upgrade renaming a declared secret produces. 

53#: The operator has to re-supply the inputs, so this may not read as the 

54#: never-configured state. It names no secret: those are the receiver's own 

55#: internal field names, which the operator cannot act on. 

56DRIFTED_INPUTS_REASON = ( 

57 "The stored diagnostics delivery inputs no longer match this deployment's " 

58 "delivery plan; re-supply them." 

59)

62class DeliveryUnavailableCode(StrEnum): 

63 """Name why delivery is unavailable, for a consumer that must branch on it. 

64 

65 The reason constants above are operator-facing prose and may be reworded; 

66 a caller choosing what to render or which action to offer branches on this 

67 instead, so no consumer has to match on the wording. 

68 """ 

69 

70 UNCONFIGURED = "unconfigured" 

71 DRIFTED_INPUTS = "drifted_inputs" 

72 

73 

74@dataclass(frozen=True, slots=True) 

75class DeliveryPlanResolution: 

76 """Report the effective delivery plan, or why there is not one. 

77 

78 Exactly one member is set. The reason exists because three distinct causes 

79 — no baked plan, a declared secret left empty, and inputs that no longer 

80 fit the plan — would otherwise collapse into one indistinguishable "not 

81 configured", and the last of them calls for a different action from the 

82 other two. 

83 

84 Build one through :meth:`resolved` or :meth:`unavailable` rather than the 

85 constructor, which states the same invariant but only as a runtime refusal. 

86 

87 :param plan: The plan to deliver with, or ``None`` when there is none. 

88 :param unavailable_reason: The operator-facing reason there is no plan, or 

89 ``None`` when there is one. Never names a secret the plan declares. 

90 :param code: The machine-readable form of ``unavailable_reason``, set 

91 whenever that is; ``None`` when there is a plan. 

92 """ 

93 

94 plan: DeliveryPlan | None 

95 unavailable_reason: str | None 

96 code: DeliveryUnavailableCode | None = None 

97 

98 def __post_init__(self) -> None: 

99 """Refuse a resolution that carries both outcomes, or neither. 

100 

101 Callers read one member off the other's emptiness, so an instance 

102 holding neither would report a deployment as able to deliver with no 

103 plan to deliver by. The code moves with the reason for the same reason: 

104 a consumer branching on it would otherwise have to fall back to the 

105 prose exactly when the prose is all there is. 

106 

107 :raises ValueError: When both outcomes are set, neither is, or the code 

108 and the reason do not accompany each other. 

109 """ 

110 if (self.plan is None) == (self.unavailable_reason is None): 

111 raise ValueError( 

112 "A delivery plan resolution carries either a plan or a reason " 

113 "for there being none, not both and not neither." 

114 ) 

115 if (self.unavailable_reason is None) != (self.code is None): 

116 raise ValueError( 

117 "A delivery plan resolution's unavailability code and reason " 

118 "are set together or not at all." 

119 ) 

120 

121 @classmethod 

122 def resolved(cls, plan: DeliveryPlan) -> "DeliveryPlanResolution": 

123 """Return the resolution for a deployment that can deliver. 

124 

125 :param plan: The plan to deliver with. 

126 :return: The resolution carrying it. 

127 """ 

128 return cls(plan=plan, unavailable_reason=None) 

129 

130 @classmethod 

131 def unavailable( 

132 cls, reason: str, code: DeliveryUnavailableCode 

133 ) -> "DeliveryPlanResolution": 

134 """Return the resolution for a deployment that cannot deliver. 

135 

136 :param reason: The operator-facing reason, from this module's constants. 

137 :param code: The machine-readable form of that reason. 

138 :return: The resolution carrying both. 

139 """ 

140 return cls(plan=None, unavailable_reason=reason, code=code) 

141 

142 @classmethod 

143 def unconfigured(cls) -> "DeliveryPlanResolution": 

144 """Return the resolution for a deployment with no usable receiver. 

145 

146 Names the outcome the three unconfigured paths below share, so each 

147 states which case it reached rather than restating the pairing of 

148 reason and code. 

149 

150 :return: The unconfigured resolution. 

151 """ 

152 return cls.unavailable( 

153 UNCONFIGURED_REASON, DeliveryUnavailableCode.UNCONFIGURED 

154 )

157def resolve_delivery_plan() -> DeliveryPlanResolution: 

158 """Return the effective delivery plan, or the reason there is none. 

159 

160 Merge ``DIAGNOSTICS_DELIVERY_INPUTS`` onto the baked 

161 ``DIAGNOSTICS_DELIVERY`` skeleton and rebuild the result through 

162 :class:`~app.extensions.bundle_upload.plan.DeliveryPlan` validation, so a merged 

163 plan has passed the same cross-reference checks a file-configured one does. 

164 Both settings are read on every call, since an override that landed after 

165 import must take effect. 

166 

167 Every unconfigured and every invalid case resolves to a reason rather than 

168 raising, so a caller rendering UI state from this value never fails merely 

169 because delivery is unconfigured or misconfigured. 

170 

171 Inputs naming a secret the skeleton does not declare resolve to the drift 

172 reason. Inputs merely omitting a declared name do not, because the merge 

173 below falls back to the skeleton's own value. The whole row is refused 

174 rather than the undeclared name dropped: ``DeliveryPlanInputs.endpoint`` is 

175 applied unconditionally in the same overlay, so a row whose secret names no 

176 longer fit can also point delivery at a receiver this plan never named. 

177 

178 Check order carries meaning. A deployment with no baked plan is 

179 unconfigured, never drifted — there is nothing to have drifted from. Drift 

180 is then decided before the empty-value check, because a rename produces both 

181 symptoms and only the rename is actionable. 

182 

183 :return: The resolved plan, or the reason delivery is unavailable. 

184 """ 

185 skeleton = extensions_settings.DIAGNOSTICS_DELIVERY 

186 if skeleton is None: 

187 return DeliveryPlanResolution.unconfigured() 

188 

189 inputs = extensions_settings.DIAGNOSTICS_DELIVERY_INPUTS 

190 secrets = dict(skeleton.secrets) 

191 if inputs is not None: 

192 if undeclared := set(inputs.secrets) - set(skeleton.secrets): 

193 logger.warning( 

194 "Diagnostics delivery inputs name secret(s) %s, which this " 

195 "deployment's plan does not declare; treating the stored inputs " 

196 "as drifted.", 

197 ", ".join(sorted(undeclared)), 

198 ) 

199 return DeliveryPlanResolution.unavailable( 

200 DRIFTED_INPUTS_REASON, DeliveryUnavailableCode.DRIFTED_INPUTS 

201 ) 

202 secrets.update(inputs.secrets) 

203 if empty := sorted( 

204 name for name, secret in secrets.items() if not secret.get_secret_value() 

205 ): 

206 logger.info( 

207 "Diagnostics delivery has no value for secret(s) %s; treating it as " 

208 "unconfigured.", 

209 ", ".join(empty), 

210 ) 

211 return DeliveryPlanResolution.unconfigured() 

212 if inputs is None: 

213 return DeliveryPlanResolution.resolved(skeleton) 

214 

215 data = skeleton.model_dump() 

216 overlay = ( 

217 {"secrets": secrets} 

218 if inputs.endpoint is None 

219 else {"secrets": secrets, "endpoint": inputs.endpoint} 

220 ) 

221 deep_dict_update(data, overlay) 

222 try: 

223 plan = DeliveryPlan.model_validate(data) 

224 except ValidationError as exc: 

225 logger.warning("Diagnostics delivery inputs produce an invalid plan: %s", exc) 

226 return DeliveryPlanResolution.unconfigured() 

227 return DeliveryPlanResolution.resolved(plan)
