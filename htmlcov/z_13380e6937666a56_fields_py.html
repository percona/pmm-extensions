16"""Define the constrained syncer fields and the values their config accepts. 

17 

18A syncer threshold is configurable three ways — a ``SYNCERS[]`` entry in 

19``settings.yaml``, JSON in ``EXTENSIONS__SYNCERS``, or an ``EXTENSIONS__SYNCER_EXTRA_KWARGS__<KEY>`` 

20env leaf — and ``SyncOptions`` forwards every one of them verbatim, untyped. The 

21syncer field and the load-time configuration check therefore have to agree on the 

22accepted values, so the annotated type, the field name and the spelling quoted back to 

23an operator all live here, one edit apart. This module deliberately imports nothing 

24from the PMM Extensions app so ``app.extensions.config`` can use it. 

25""" 

26 

27from datetime import timedelta 

28from typing import Annotated, Any, Final, NamedTuple 

29 

30from annotated_types import Ge, Gt 

31from pydantic import TypeAdapter 

32 

33StaleRunAfter = Annotated[timedelta, Gt(timedelta(0))] 

34"""Define the age beyond which an idle in-progress sync run is reclaimable. 

35 

36Positivity is an annotation constraint rather than a ``field_validator`` because 

37``SyncOptions`` carries ``extra="allow"`` and forwards every extra key verbatim, and 

38runtime-override coercion re-checks annotated-type constraints but does not re-run 

39field validators. 

40""" 

41 

42MissingGraceGenerations = Annotated[int, Ge(2)] 

43"""Define how many complete generations must report an entity absent before removal. 

44 

45The floor is 2, not 1: at 1 the grace counter collapses back to acting on a single 

46reported absence, which is the behaviour it exists to end. Expressed as an annotation 

47constraint for the reason :data:`StaleRunAfter` is. 

48""" 

49 

50TaskExecutionTimeout = Annotated[int, Gt(0)] 

51"""Define the number of seconds a task is allowed to take before it times out. 

52 

53At zero or below the poll loop's ``time_waiting < task_execution_timeout`` guard is 

54false on its first evaluation, so the loop never runs and the task is declared timed 

55out without having been polled once. Expressed as an annotation constraint for the 

56reason :data:`StaleRunAfter` is. 

57""" 

58 

59TasksExecutionWaitInterval = Annotated[int, Gt(0)] 

60"""Define the number of seconds between two task status checks. 

61 

62The interval is also what advances the poll loop's elapsed-time counter, so at zero 

63the counter never moves and the timeout above can never fire, and below zero it moves 

64backwards and the loop stops terminating at all. Expressed as an annotation constraint 

65for the reason :data:`StaleRunAfter` is. 

66""" 

67 

68FirstMeasurementRetries = Annotated[int, Ge(0)] 

69"""Define the retries a never-measured host gets after its failed first attempt. 

70 

71Zero is allowed and means a host whose first attempt fails is left to the daily 

72refresh; below zero no host would ever be attempted. Expressed as an annotation 

73constraint for the reason :data:`StaleRunAfter` is. 

74""" 

75 

76FirstMeasurementRetryInterval = Annotated[timedelta, Gt(timedelta(0))] 

77"""Define the least time between two first-measurement attempts on one host. 

78 

79At zero or below every pass is due to retry a failing host, so the retry budget is 

80spent within minutes instead of being spread over hours. Expressed as an annotation 

81constraint for the reason :data:`StaleRunAfter` is. 

82""" 

83 

84FirstMeasurementConcurrency = Annotated[int, Gt(0)] 

85"""Define the most first-measurement probes in flight at once. 

86 

87At zero the pass's semaphore admits no probe, so every candidate waits forever and 

88the run never ends. Expressed as an annotation constraint for the reason 

89:data:`StaleRunAfter` is. 

90""" 

91 

92 

93class SyncerFieldConstraint(NamedTuple): 

94 """Pair a constrained syncer field with the spelling of its accepted values. 

95 

96 :param adapter: Validator for the field's annotated type, constraints included. 

97 :param accepted: Human-readable list of the forms the field accepts, quoted back 

98 to the operator when their value is refused. 

99 """ 

100 

101 adapter: TypeAdapter[Any] 

102 accepted: str 

103 

104 

105CONSTRAINED_SYNCER_FIELDS: Final[dict[str, SyncerFieldConstraint]] = { 

106 "stale_run_after": SyncerFieldConstraint( 

107 TypeAdapter(StaleRunAfter), 

108 "integer or float seconds (3600), an ISO-8601 duration (PT1H), or HH:MM:SS", 

109 ), 

110 "missing_grace_generations": SyncerFieldConstraint( 

111 TypeAdapter(MissingGraceGenerations), 

112 "an integer of 2 or more", 

113 ), 

114 "task_execution_timeout": SyncerFieldConstraint( 

115 TypeAdapter(TaskExecutionTimeout), 

116 "a positive integer number of seconds", 

117 ), 

118 "tasks_execution_wait_interval": SyncerFieldConstraint( 

119 TypeAdapter(TasksExecutionWaitInterval), 

120 "a positive integer number of seconds", 

121 ), 

122 "first_measurement_retries": SyncerFieldConstraint( 

123 TypeAdapter(FirstMeasurementRetries), 

124 "an integer of 0 or more", 

125 ), 

126 "first_measurement_retry_interval": SyncerFieldConstraint( 

127 TypeAdapter(FirstMeasurementRetryInterval), 

128 "integer or float seconds (3600), an ISO-8601 duration (PT1H), or HH:MM:SS", 

129 ), 

130 "first_measurement_concurrency": SyncerFieldConstraint( 

131 TypeAdapter(FirstMeasurementConcurrency), 

132 "a positive integer", 

133 ), 

134} 

135"""Map each constrained syncer field to the validator its configured value must pass. 

136 

137``SyncOptions`` and ``SyncerExtraKwargs`` both allow untyped extras, so a value 

138configured for one of these fields is only checked against the field's real type once 

139a syncer is constructed — which, for the request-scoped ``get_syncers`` dependency, 

140means once per request. Checking it against this mapping at settings load turns a 

141misconfiguration into a single startup failure instead. Keys are the field names as 

142:class:`~app.core.models.BaseLowercaseModel` stores them, lowercased. 

143"""
