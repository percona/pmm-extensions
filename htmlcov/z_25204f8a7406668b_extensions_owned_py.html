16"""Refresh the PMM Extensions-owned shared settings classes in any process. 

17 

18``Settings`` and ``AlertSettings`` back the module-level ``settings`` and 

19``alert_settings`` proxies every service shares. Their override rows live in the 

20PMM Extensions database, whose settings API is their only writer, so every 

21process that honours them reads that database: the PMM Extensions web process 

22and worker children through their own refreshers, and the standalone Tasks and 

23Inventory processes through :func:`extensions_owned_overrides_refresher`. This 

24module holds the one definition of those classes and of the rebinds they need, 

25so the processes cannot drift apart, and lives under ``app/core`` so the 

26standalone services reach it without importing ``app/extensions``. 

27 

28Not re-exported from the package ``__init__``: ``app.core.config`` imports 

29``app.core.settings_override.registry``, which runs that ``__init__``, and this 

30module imports ``app.core.config``. 

31"""

66async def invalidate_pmm_clients(change: SnapshotChange) -> None: 

67 """Evict cached PMM clients on the previous and current endpoints after a ``PMM`` override. 

68 

69 Evicts the ordered de-duplicated set of previous-and-current endpoints so a 

70 same-endpoint change (credentials, SSL) collapses to a single eviction, while 

71 an endpoint change also drops the client keyed by the endpoint no longer in 

72 use. The next :class:`PMMSyncer` key-misses to a fresh client via its 

73 ``default_factory`` PMM read. When ``PMM`` is absent from ``change.previous`` 

74 (override created), :func:`previous_or_base` supplies the YAML/env value. 

75 

76 :param change: The override snapshots on either side of the republish. 

77 """ 

78 previous_pmm = cast( 

79 "PMMSettings", 

80 previous_or_base(change, cast("OverridableSettingsProxy", settings), "PMM"), 

81 ) 

82 for endpoint in dict.fromkeys( 

83 str(pmm.endpoint) 

84 for pmm in (previous_pmm, settings.PMM) 

85 if pmm.endpoint is not None 

86 ): 

87 await settings.invalidate_client(endpoint)

90@fire_on_boot 

91async def apply_logging_dictconfig(_: SnapshotChange) -> None: 

92 """Re-apply ``logging.config.dictConfig`` after a global ``LOGGING`` override. 

93 

94 ``LOGGING`` is a HOT field, but ``LOGGING_CONFIG`` (the dict handed to 

95 ``dictConfig``) is not: the override snapshot replaces only the ``LOGGING`` 

96 key, so ``settings.LOGGING_CONFIG`` still carries the level baked in by the 

97 ``set_log_level`` model validator at construction time. This callback mirrors 

98 that validator: it injects the now-live ``settings.LOGGING`` into a copy of 

99 the config and re-applies it, so a log-level change takes effect without a 

100 restart in every process that registers it: the PMM Extensions web process, 

101 the Celery worker children, and the standalone Tasks and Inventory 

102 processes. Failures are logged and swallowed: a malformed config must not 

103 take the process down mid-request or mid-task. ``LOGGING_CONFIG`` sets 

104 ``disable_existing_loggers: False``, so re-entering ``dictConfig`` from a 

105 worker refresh cycle leaves the loggers Celery created at runtime enabled, 

106 and re-creates the ones the config names, ``celery`` among them, with 

107 their handlers. 

108 

109 Marked ``fire_on_boot`` because the boot-time ``dictConfig`` call reads the 

110 process-wide ``LOGGING_CONFIG``, which no snapshot carries: a process that 

111 starts with a ``LOGGING`` override already stored would otherwise keep the 

112 YAML/env level until the override next changed. 

113 

114 :param _: The snapshots on either side of the republish (unused; the 

115 level is re-read from the proxy). 

116 """ 

117 try: 

118 config = deepcopy(settings.LOGGING_CONFIG) 

119 config["loggers"][""]["level"] = settings.LOGGING 

120 config["loggers"]["app"]["level"] = settings.LOGGING 

121 logging.config.dictConfig(config) 

122 except Exception: 

123 logger.exception("Failed to re-apply logging config after LOGGING override")

126#: The PMM Extensions-owned shared classes, keyed by class name, bound to the 

127#: module-level proxies every service reads. 

128EXTENSIONS_OWNED_PROXIES: ProxyRegistry = { 

129 SETTINGS: ProxyEntry(cast("OverridableSettingsProxy", settings), Settings), 

130 ALERT_SETTINGS: ProxyEntry( 

131 cast("OverridableSettingsProxy", alert_settings), AlertSettings 

132 ), 

133} 

134 

135#: The rebinds the shared classes need in any process that publishes them. 

136#: ``invalidate_pmm_clients`` is registered because ``ClientRegistry.IMMUTABLE_KEYS`` 

137#: excludes ``api_key``: a same-endpoint credential override would otherwise 

138#: refresh the snapshot while cached clients keep the old key. 

139#: ``apply_logging_dictconfig`` re-enters ``dictConfig`` so a HOT ``LOGGING`` 

140#: override replaces the boot-time levels. 

141EXTENSIONS_OWNED_CALLBACKS: CallbackRegistry = { 

142 (SETTINGS, "PMM"): invalidate_pmm_clients, 

143 (SETTINGS, "LOGGING"): apply_logging_dictconfig, 

144}

147def extensions_override_database() -> DatabaseOptions: 

148 """Return the PMM Extensions database options, capped at one pooled connection. 

149 

150 The refresher holds one session per cycle, so a single connection is all it 

151 needs, and the cap keeps the extra engine inside a shared connection budget 

152 whatever ``EXTENSIONS__DATABASE__POOL_SIZE`` sizes the service's own pool to. 

153 

154 :return: The PMM Extensions database options with ``POOL_SIZE`` 1 and 

155 ``MAX_OVERFLOW`` 0. 

156 :raises ValidationError: When the resolved ``EXTENSIONS__DATABASE__*`` 

157 values do not validate. 

158 """ 

159 return ExtensionsDatabaseSettings().DATABASE.model_copy( 

160 update={"POOL_SIZE": 1, "MAX_OVERFLOW": 0} 

161 )

164@asynccontextmanager 

165async def extensions_owned_overrides_refresher( 

166 *setting_classes: str, 

167) -> AsyncGenerator[None, None]: 

168 """Refresh the named PMM Extensions-owned classes from the PMM Extensions database. 

169 

170 Builds an engine of its own for the lifespan and disposes it on exit, so a 

171 process that never enters this (the combined ``app.main:app``, whose PMM 

172 Extensions refresher already publishes these classes) never builds it. The 

173 seed is bounded by :data:`EXTENSIONS_OWNED_SEED_TIMEOUT`: on expiry the 

174 lifespan yields on whatever the seed published, YAML/env when the database 

175 is unreachable, and the periodic loop applies the overrides once it answers. 

176 

177 :param setting_classes: Keys of :data:`EXTENSIONS_OWNED_PROXIES` to refresh. 

178 :return: An async context manager yielding ``None`` for the lifespan duration. 

179 :raises KeyError: When a name is not a PMM Extensions-owned class. 

180 :raises ValidationError: When the resolved ``EXTENSIONS__DATABASE__*`` 

181 values do not validate. 

182 :raises Exception: Propagates whatever building the engine raises (an 

183 unknown or missing database driver, for example) and whatever 

184 :func:`~app.core.settings_override.lifecycle.settings_override_refresher` 

185 re-raises from its initial refresh. 

186 """ 

187 proxies = {name: EXTENSIONS_OWNED_PROXIES[name] for name in setting_classes} 

188 engine = create_app_async_engine(extensions_override_database()) 

189 try: 

190 session_maker = get_async_session_maker_from_engine(engine) 

191 async with settings_override_refresher( 

192 lambda: session_maker, 

193 proxies, 

194 callbacks=EXTENSIONS_OWNED_CALLBACKS, 

195 seed_timeout=EXTENSIONS_OWNED_SEED_TIMEOUT, 

196 ): 

197 yield 

198 finally: 

199 await engine.dispose()
