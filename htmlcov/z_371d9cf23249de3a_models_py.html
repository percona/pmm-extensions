104class RetirableSQLModel(RetiredAtBase, BaseSQLModel): 

105 """Store the retirement state of a tombstoned entity. 

106 

107 An entity that vanishes upstream is retired rather than deleted, so the 

108 references PMM Extensions persisted to it keep resolving. Because a replacement may 

109 reuse the retired row's unique key, ``retirement_key`` joins every unique 

110 index: it holds :data:`ACTIVE_RETIREMENT_KEY` while the row is active and 

111 the row's own primary key once retired, which keeps active rows mutually 

112 exclusive while letting any number of tombstones share their key. 

113 

114 :param retired_at: When the entity stopped being reported by its upstream 

115 source, or None while it is active. 

116 :param retirement_key: The discriminator carried inside every unique index. 

117 Excluded from serialization: the responses nest these table models, and a 

118 database-internal discriminator has no business in the published schema. 

119 """ 

120 

121 retirement_key: int = SQLField( 

122 default=ACTIVE_RETIREMENT_KEY, nullable=False, exclude=True 

123 )

164class LinkageMethodEnum(StrEnum): 

165 """Enumerate how an external-identity binding came to be recorded. 

166 

167 Every member names an operator action: nothing here records a binding the 

168 syncer made on its own, because ordinary sync creation writes no alias row. 

169 

170 Values are spelled out rather than derived with ``auto()``. The column 

171 persists the member *name*, but the API serializes the *value*, so under 

172 ``auto()`` the two move in opposite ways when a member is renamed: the 

173 stored form follows the rename while the published one silently changes 

174 with it. Pinning the value holds the wire contract — this enum reaches the 

175 generated API client — and leaves a rename a database concern alone. 

176 """ 

177 

178 OPERATOR_CONFIRMATION = "operator_confirmation" 

179 OPERATOR_UNLINK = "operator_unlink"

211class SyncHealthWrite(SQLModel): 

212 """Define the body reporting one entity's sync outcome. 

213 

214 :param outcome: Whether the attempt succeeded or failed. 

215 :param error: The failure's message, never empty. Required on FAILURE, 

216 absent on SUCCESS. 

217 :param attempted_at: When the syncer began this attempt. Stamped as 

218 ``last_synced_at`` on success and as ``newest_attempt_at`` when newer, 

219 and used to order reports so a late-arriving one from an older attempt 

220 cannot overwrite a newer one. Refused when it sits further ahead of 

221 this service's clock than the tolerated skew, since nothing later could 

222 then supersede it. 

223 """ 

224 

225 outcome: SyncOutcomeEnum 

226 error: NonEmptyStr | None = None 

227 attempted_at: UTCDatetime 

228 

229 @model_validator(mode="after") 

230 def _validate_error_matches_outcome(self) -> Self: 

231 if self.outcome is SyncOutcomeEnum.FAILURE and self.error is None: 

232 raise ValueError("error is required when outcome is failure") 

233 if self.outcome is SyncOutcomeEnum.SUCCESS and self.error is not None: 

234 raise ValueError("error must be omitted when outcome is success") 

235 return self 

236 

237 @field_validator("attempted_at") 

238 @classmethod 

239 def _validate_attempted_at_is_not_ahead(cls, value: datetime) -> datetime: 

240 if value > utc_now() + SYNC_ATTEMPT_MAX_CLOCK_SKEW: 

241 raise ValueError( 

242 "attempted_at is further ahead of server time than the " 

243 "tolerated clock skew" 

244 ) 

245 return value

1082#: Derived from the declaring base so the model stays the single source of truth as 

1083#: fields change. It must NOT be derived from a concrete subclass: the table and 

1084#: response models also inherit ``BaseSQLModel``'s ``id`` / ``created_at`` / 

1085#: ``updated_at``, and an always-populated ``created_at`` would satisfy the validator

1097def host_observation_min_content_check() -> CheckConstraint: 

1098 """Build the table-level guard mirroring the at-least-one-fact validator. 

1099 

1100 The columns come from :data:`HOST_OBSERVATION_FIELD_NAMES`, so this 

1101 declaration and the Pydantic validator stay in lockstep as fields are added. 

1102 A migrated database does not: the revision that creates the CHECK spells its 

1103 columns out, so a new fact column reaches an existing deployment only 

1104 through a revision of its own. Sorting makes the rendered expression 

1105 deterministic across interpreter runs, which a frozenset's iteration order 

1106 is not. 

1107 

1108 A plain ``IS NOT NULL`` disjunction covers the JSON-typed facts only because 

1109 they are declared ``JSON(none_as_null=True)``. SQLAlchemy's default persists 

1110 an unset JSON value as the JSON text ``null``, which is not SQL NULL and so 

1111 would satisfy this CHECK on a row carrying no fact at all. 

1112 

1113 :return: A CHECK requiring at least one observed fact to be non-NULL. 

1114 """ 

1115 return CheckConstraint( 

1116 or_( 

1117 *( 

1118 column(name).is_not(None) 

1119 for name in sorted(HOST_OBSERVATION_FIELD_NAMES) 

1120 ) 

1121 ), 

1122 name=HOST_OBSERVATION_MIN_CONTENT_CONSTRAINT, 

1123 )
