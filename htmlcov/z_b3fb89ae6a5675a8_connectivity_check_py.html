16"""Define the admin-only ``/api/extensions/admin/connectivity-check`` endpoint. 

17 

18Expose a single generic ``POST`` that probes the caller-specified external / 

19inter-service endpoints (PMM, Inventory, Tasks, Nomad, and the diagnostics 

20delivery receiver) on demand and reports normalized per-endpoint connectivity 

21status, so an admin can confirm an endpoint or credential change is reachable 

22and valid before relying on it. The request must name which services to probe 

23(``targets``, required, no default), so the settings flow can validate only the 

24endpoint being edited while a full sweep still names them all. 

25 

26The probes are driven by the overridable ``RemoteAPI.check_connectivity`` 

27capability and fan out concurrently. A failure for one endpoint is captured and 

28classified independently, so a single outage never fails the whole response. 

29PMM, Inventory, and Tasks are built through the existing 

30``settings.get_remote_api`` / ``ClientRegistry`` path (honoring SSL config and 

31hot-reloaded overrides); Nomad reuses the Tasks ``/hosts/`` proxy -- a healthy 

32response proves both Tasks and Nomad are reachable, so no new Tasks-app endpoint 

33is added. Because Tasks and Nomad share that single probe, requesting either (or 

34both) runs ``/hosts/`` exactly once. 

35"""

93async def _probe_pmm(pmm_api: PMMAPIDep) -> ConnectivityResult: 

94 """Probe PMM, or report "not configured" when no endpoint/key is set. 

95 

96 :param pmm_api: The PMM client, or ``None`` when PMM is unconfigured. 

97 :return: The PMM connectivity result. 

98 """ 

99 if pmm_api is None: 

100 return build_connectivity_result( 

101 ServiceEnum.PMM, 

102 ConnectivityStatusEnum.NOT_CONFIGURED, 

103 detail="PMM is not configured.", 

104 ) 

105 return await pmm_api.check_connectivity(ServiceEnum.PMM)

108async def _probe_tasks_and_nomad( 

109 tasks_api: TaskAPI, 

110) -> tuple[ConnectivityResult, ConnectivityResult]: 

111 """Derive both Tasks and Nomad results from a single ``/hosts/`` probe. 

112 

113 A ``200`` (including an empty cluster) means both are reachable. Only an 

114 *app-level* ``502 Bad Gateway`` -- the Tasks app answered with a JSON error 

115 because its executor backend is unreachable -- means Tasks is reachable 

116 while Nomad is not. A ``502`` with a non-JSON body is nginx answering 

117 because the whole Tasks app is down; it carries the 

118 :data:`UPSTREAM_NON_JSON_HEADER` marker and is classified as Tasks 

119 unreachable. Any other failure (other HTTP errors such as a ``500`` 

120 indicating an unhealthy Tasks API itself, plus connection-level / SSL / 

121 timeout / auth failures) applies to both, since a healthy Tasks API cannot 

122 be confirmed and Nomad therefore cannot be either. 

123 

124 :param tasks_api: The authenticated Tasks API client. 

125 :return: A ``(tasks_result, nomad_result)`` pair. 

126 """ 

127 try: 

128 async with asyncio.timeout(PROBE_TIMEOUT_SECONDS): 

129 await tasks_api.get(_HOSTS_PROBE_PATH) 

130 except Exception as exc: # noqa: BLE001 -- classified, never re-raised 

131 if isinstance(exc, HTTPBadGatewayException) and not (exc.headers or {}).get( 

132 UPSTREAM_NON_JSON_HEADER 

133 ): 

134 # The Tasks app answered with a JSON 502 but its executor backend is 

135 # down, so Tasks is reachable but Nomad is not. A bare non-JSON 502 

136 # (nginx, Tasks app down) carries the marker header and falls through. 

137 return ( 

138 build_connectivity_result( 

139 ServiceEnum.TASKS, ConnectivityStatusEnum.REACHABLE 

140 ), 

141 build_connectivity_result( 

142 ServiceEnum.NOMAD, 

143 ConnectivityStatusEnum.UNREACHABLE, 

144 detail="Executor backend unreachable.", 

145 ), 

146 ) 

147 # The Tasks endpoint itself failed (HTTP 5xx other than 502, connection, 

148 # SSL, timeout, or auth); Nomad cannot be confirmed either. 

149 probe_status = classify_connectivity_error(exc) 

150 return ( 

151 build_connectivity_result(ServiceEnum.TASKS, probe_status), 

152 build_connectivity_result(ServiceEnum.NOMAD, probe_status), 

153 ) 

154 return ( 

155 build_connectivity_result(ServiceEnum.TASKS, ConnectivityStatusEnum.REACHABLE), 

156 build_connectivity_result(ServiceEnum.NOMAD, ConnectivityStatusEnum.REACHABLE), 

157 ) 

158 

159 

160#: The connectivity status each unavailability code reports as. Delivery being 

161#: unconfigured and its stored inputs having drifted are separate outcomes 

162#: because only the second is fixed by re-supplying the inputs. 

163_DELIVERY_UNAVAILABLE_STATUS: dict[DeliveryUnavailableCode, ConnectivityStatusEnum] = { 

164 DeliveryUnavailableCode.UNCONFIGURED: ConnectivityStatusEnum.NOT_CONFIGURED, 

165 DeliveryUnavailableCode.DRIFTED_INPUTS: ConnectivityStatusEnum.INPUTS_DRIFTED, 

166}

169async def _probe_delivery() -> ConnectivityResult: 

170 """Report whether the configured diagnostics-delivery receiver answers. 

171 

172 Issues the delivery plan's own declared probe request rather than replaying 

173 its send steps: a resolution step may mutate state, and its values may cite 

174 inputs that exist only during a real send. A plan that declares no probe is 

175 reported as such instead of being guessed at. 

176 

177 The resolver's prose reason is passed through verbatim as the detail, while 

178 the status carries the same distinction machine-readably. 

179 

180 Branches on the resolution's code rather than its plan, so the unavailable 

181 outcomes map through one table instead of being re-derived from the prose. 

182 That branch narrows nothing for a type checker, so the ``plan is None`` arm 

183 below is what narrows the optional in place of an assertion; the 

184 resolution's own invariant is what makes that arm unreachable. 

185 

186 :return: The delivery connectivity result. 

187 """ 

188 resolution = resolve_delivery_plan() 

189 if (code := resolution.code) is not None: 

190 return build_connectivity_result( 

191 ServiceEnum.DELIVERY, 

192 _DELIVERY_UNAVAILABLE_STATUS[code], 

193 detail=resolution.unavailable_reason, 

194 ) 

195 plan = resolution.plan 

196 if plan is None or plan.probe is None: 

197 return build_connectivity_result( 

198 ServiceEnum.DELIVERY, ConnectivityStatusEnum.PROBE_UNDECLARED 

199 ) 

200 try: 

201 async with asyncio.timeout(EXTERNAL_PROBE_TIMEOUT_SECONDS): 

202 async with get_delivery_executor(plan) as executor: 

203 await executor.probe() 

204 except Exception as exc: # noqa: BLE001 -- classified, never re-raised 

205 return build_connectivity_result( 

206 ServiceEnum.DELIVERY, classify_connectivity_error(exc) 

207 ) 

208 return build_connectivity_result( 

209 ServiceEnum.DELIVERY, ConnectivityStatusEnum.REACHABLE 

210 )

213@router.post("/") 

214async def check_connectivity( 

215 body: ConnectivityCheckRequest, 

216 pmm_api: PMMAPIDep, 

217 inventory_api: InventoryAPI, 

218 tasks_api: TaskAPI, 

219) -> list[ConnectivityResult]: 

220 """Probe the requested external / inter-service endpoints and report status. 

221 

222 Probe only the services named in ``body.targets``, running the selected 

223 probes concurrently. Each is isolated: one endpoint's failure is classified 

224 independently and never fails the whole response, which always returns 

225 ``200`` with one entry per requested service, in request order. Tasks and 

226 Nomad share a single ``/hosts/`` probe, so requesting either (or both) runs 

227 it once. 

228 

229 :param body: The request naming which services to probe. 

230 :param pmm_api: The PMM client dependency, or ``None`` when unconfigured. 

231 :param inventory_api: The authenticated Inventory API client. 

232 :param tasks_api: The authenticated Tasks API client. 

233 :return: One normalized connectivity result per requested service. 

234 """ 

235 # Preserve request order while collapsing duplicates. 

236 targets = list(dict.fromkeys(body.targets)) 

237 want_tasks = ServiceEnum.TASKS in targets 

238 want_nomad = ServiceEnum.NOMAD in targets 

239 

240 # Register each requested probe as a (service, coroutine) pair. Tasks/Nomad 

241 # share one coroutine, so it is registered once under a sentinel key. 

242 probes: dict[str, object] = {} 

243 if ServiceEnum.PMM in targets: 

244 probes[ServiceEnum.PMM] = _probe_pmm(pmm_api) 

245 if ServiceEnum.INVENTORY in targets: 

246 probes[ServiceEnum.INVENTORY] = inventory_api.check_connectivity( 

247 ServiceEnum.INVENTORY, path=_INVENTORY_PROBE_PATH 

248 ) 

249 if want_tasks or want_nomad: 

250 probes["_tasks_nomad"] = _probe_tasks_and_nomad(tasks_api) 

251 if ServiceEnum.DELIVERY in targets: 

252 probes[ServiceEnum.DELIVERY] = _probe_delivery() 

253 

254 completed = dict(zip(probes, await asyncio.gather(*probes.values()), strict=False)) 

255 

256 # Expand the shared Tasks/Nomad result into the requested halves. 

257 by_service: dict[str, ConnectivityResult] = {} 

258 for key, result in completed.items(): 

259 if key == "_tasks_nomad": 

260 tasks_result, nomad_result = result 

261 if want_tasks: 

262 by_service[ServiceEnum.TASKS] = tasks_result 

263 if want_nomad: 

264 by_service[ServiceEnum.NOMAD] = nomad_result 

265 else: 

266 by_service[key] = result 

267 

268 return [by_service[service] for service in targets]
