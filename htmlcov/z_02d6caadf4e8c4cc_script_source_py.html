16"""Back the snippets plugin's JSON surface with the framework ``ScriptSource`` seam. 

17 

18This adapter is the first real adoption of 

19:class:`~app.extensions.apps.framework.script_source.ScriptSource`: it wires the hooks 

20``derive_script_routes`` needs around the surrounding snippets engine 

21(``app/extensions/snippets/``), so :data:`~app.extensions.apps.snippets.app.app` can declare 

22``script_source=`` instead of hand-wiring the listing / per-script schema / 

23execute / history routes. Being DB-backed, its ``list_scripts`` hook pushes the 

24list route's search/sort/filter/paging down to SQL (via 

25:meth:`~app.extensions.snippets.crud.SnippetManager.snippet_list_page` over 

26:attr:`~app.extensions.snippets.crud.SnippetManager.list_query_spec`) instead of fetching 

27every row and slicing in-process. Its ``list_query_dep`` composes the Core sort/search 

28dependency with the snippets approval and service-type filters, so the spec stays the 

29sole sort allowlist while the route keeps its filter surface. 

30 

31Three real couplings the synthetic kit deferred are bridged here: 

32 

33* **The deferred ``-hostname-`` coupling.** ``BaseSnippet.get_execution_model`` 

34 bakes a required ``executor_host`` (validation alias ``-hostname-``) into the 

35 execution model, but the framework validates ``body.args`` (which never carries 

36 ``-hostname-``). :class:`SnippetScript.get_execution_model` returns an *args-only* 

37 subclass that makes ``executor_host`` optional so that validation succeeds; the 

38 execute hook re-attaches the real host via ``model_construct``. 

39* **The request-less listing.** ``load_script`` / ``list_scripts`` / 

40 ``load_scripts`` are plain callables with no request or session, so they open 

41 their own request-less session (mirroring ``app/extensions/snippets/celery.py``). 

42 ``load_scripts`` resolves a whole selection in one ``IN`` query. Only loaded 

43 columns and file/meta ``cached_property`` values are read after it closes — 

44 never a lazy relationship. 

45* **The request-less artifact URL.** :func:`build_snippet_source` reads 

46 ``SNIPPETS_BASE_URL`` / ``BASE_URL`` rather than the legacy request-derived 

47 fallback, so an unconfigured deployment fails loudly instead of emitting an 

48 executor-unreachable localhost URL. 

49"""

184def _detach(session: AsyncSession, snippet: Snippet) -> Snippet: 

185 """Expunge a snippet from its session so it stays usable after the session closes. 

186 

187 The request-less loaders below each read only loaded columns and file/meta 

188 ``cached_property`` values once their session closes — never a lazy 

189 relationship — so detaching the row keeps it usable without a live session. 

190 

191 :param session: The open session the snippet was loaded in. 

192 :param snippet: The loaded snippet row. 

193 :return: The now-detached snippet. 

194 """ 

195 session.expunge(snippet) 

196 return snippet

199async def _load_script(filename: str) -> SnippetScript: 

200 """Resolve a snippet filename to a detached :class:`SnippetScript`. 

201 

202 Opens a request-less session, fetches the row, and detaches it via 

203 :func:`_detach` so the script stays usable after the session closes. 

204 

205 :param filename: The snippet filename. 

206 :return: The wrapped, detached snippet. 

207 :raises HTTPBadRequestException: When the filename is unsafe or malformed 

208 (the snippets-specific strictness the generic seam guard does not enforce). 

209 :raises HTTPNotFoundException: When the row is absent or its file is missing. 

210 """ 

211 validate_snippet_filename(filename) 

212 async_session = get_async_session_maker() 

213 async with async_session() as session: 

214 snippet = await SnippetManager.get_or_404(session, filename=filename) 

215 if not snippet.path.is_file(): 

216 raise HTTPNotFoundException(detail=snippet_not_found_detail(filename)) 

217 return SnippetScript(_detach(session, snippet))

220async def _list_scripts( 

221 list_query: SnippetListQuery | None, pagination: Pagination | None 

222) -> tuple[list[SnippetScript], int]: 

223 """Return a filtered, sorted, paginated page of snippets and the filtered total. 

224 

225 The Core sort/search, the snippets approval and service-type filters, and the 

226 filtered total are all pushed down to 

227 :meth:`~app.extensions.snippets.crud.SnippetManager.snippet_list_page`, so the count and 

228 data queries share predicates and the ``total`` matches the visible page. The rows 

229 go through :func:`_detach` so they stay usable after the request-less session 

230 closes, mirroring :func:`_load_script`. The snippets list route always derives a 

231 query, so ``list_query`` and ``pagination`` are always supplied; the ``None`` 

232 branches keep the hook honest against the framework's non-query call shapes, where an 

233 absent ``pagination`` means the whole set rather than a default-sized first page. A 

234 query with no pagination is answered unsliced but still filtered, searched, and 

235 ordered — the shape 

236 :meth:`~app.extensions.apps.framework.list_query.InMemoryListQueryApplier.list_scripts` 

237 honours for a disk-backed source, so the two implementations of the one contract 

238 agree. 

239 

240 :param list_query: The resolved sort/search/filter selections, or ``None`` for the 

241 manager's default ordering with no filters. 

242 :param pagination: The validated offset/limit window, or ``None`` for every 

243 matching snippet. 

244 :return: The wrapped snippets and the filtered total across all pages. 

245 """ 

246 async_session = get_async_session_maker() 

247 async with async_session() as session: 

248 if pagination is None: 

249 rows = await ( 

250 SnippetManager.list(session) 

251 if list_query is None 

252 else SnippetManager.snippet_list_all(session, list_query=list_query) 

253 ) 

254 total = len(rows) 

255 else: 

256 page = await ( 

257 SnippetManager.list_paginated(session, pagination=pagination) 

258 if list_query is None 

259 else SnippetManager.snippet_list_page( 

260 session, list_query=list_query, pagination=pagination 

261 ) 

262 ) 

263 rows, total = page.items, page.total 

264 return [SnippetScript(_detach(session, snippet)) for snippet in rows], total

267async def _load_scripts(filenames: Sequence[str]) -> dict[str, SnippetScript]: 

268 """Resolve several snippet filenames in one query, keyed by filename. 

269 

270 The batch counterpart of :func:`_load_script`: one request-less session and a 

271 single ``filename IN (...)`` query answer the whole selection instead of one 

272 session and one query per filename. Each returned row is detached via 

273 :func:`_detach` before the session closes so the scripts stay usable 

274 afterwards. A filename whose row is absent, or whose file is missing on disk, 

275 is simply left out of the result — the framework's caller decides what an 

276 unresolved filename means. 

277 

278 Each row is keyed by every requested spelling that matched it, not the stored 

279 ``filename``: the ``IN (...)`` match honours the column's collation, so on a 

280 case-insensitive collation a request for ``Check.sh`` matches row ``check.sh``. 

281 Folding the returned row back to each requested string keeps this hook's keys 

282 aligned with :func:`_load_script`, whose ``get_or_404`` compared inside the 

283 database, and means a selection carrying two case variants of one file 

284 (``Check.sh`` and ``check.sh``) resolves both rather than dropping one. 

285 Accent-insensitive collations remain a known boundary — casefolding covers 

286 case, not accents. 

287 

288 :param filenames: The snippet filenames to resolve. Callers routing through 

289 :func:`~app.extensions.apps.framework.script_source.resolve_scripts` pass a 

290 deduplicated, traversal-checked selection, but the hook is independently 

291 callable, so it re-validates each filename and tolerates duplicates. 

292 :return: A mapping of each resolved filename to its detached 

293 :class:`SnippetScript`; unresolved filenames are absent. 

294 :raises HTTPBadRequestException: When any filename is unsafe or malformed (the 

295 snippets-specific strictness the generic seam guard does not enforce). 

296 """ 

297 for filename in filenames: 

298 validate_snippet_filename(filename) 

299 requested_by_casefold: dict[str, list[str]] = defaultdict(list) 

300 for filename in filenames: 

301 requested_by_casefold[filename.casefold()].append(filename) 

302 async_session = get_async_session_maker() 

303 async with async_session() as session: 

304 snippets = await SnippetManager.list( 

305 session, col(Snippet.filename).in_(filenames) 

306 ) 

307 resolved: dict[str, Snippet] = {} 

308 for snippet in snippets: 

309 if not snippet.path.is_file(): 

310 continue 

311 detached = _detach(session, snippet) 

312 requested = requested_by_casefold.get( 

313 snippet.filename.casefold(), [snippet.filename] 

314 ) 

315 for spelling in requested: 

316 resolved[spelling] = detached 

317 return {filename: SnippetScript(snippet) for filename, snippet in resolved.items()}
