52class PMMSyncer(BaseSyncer): 

53 """Manage synchronization of PMM inventory entities. 

54 

55 This class extends `BaseSyncer` to handle synchronization operations specific to PMM 

56 entities such as nodes and services. It interacts with the PMM remote API to 

57 retrieve, update, and retire inventory data, ensuring that the local inventory is 

58 consistent with the remote source. 

59 

60 :cvar SYNC_TO_LIMIT: The highest entity type that can be synchronized. 

61 Set to `SyncInventoryEntityTypeEnum.SERVICE`. 

62 :cvar mirrors_entity_levels: The node and service levels, whose own fields 

63 this syncer compares against PMM and writes back through ``update_node`` 

64 and ``update_service``. 

65 :cvar reads_retired_entities: The node and service levels, whose incoming 

66 reports this syncer matches against the local inventory by external id. 

67 :param inventory_api: The remote API interface for interacting with the inventory 

68 system. 

69 :param access_token: The access token used for authenticating with the inventory 

70 API. 

71 :param sync_instance: The synchronization instance used to track sync processes. 

72 :param sync_items: A dictionary mapping tuples of entity type and ID to SyncItem 

73 objects. 

74 :param sync_id: The unique identifier for this synchronization. 

75 :param keepalive_api: Whether to keep the PMMRemoteAPI instance alive after

236 async def _retire_absent( 

237 self, 

238 entity_type: SyncInventoryEntityTypeEnum, 

239 absent_entities: Iterable[_Retirable], 

240 retire: Callable[[_Retirable], Awaitable[None]], 

241 *, 

242 permitted: bool, 

243 filtered_external_ids: set[str], 

244 ) -> None: 

245 """Advance the missing-grace counter for absent entities and retire the spent. 

246 

247 An entity excluded by the caller's filter is held without its counter moving: 

248 an operator exclusion is evidence in neither direction, exactly like an 

249 incomplete generation. An entity that is *already* retired is skipped 

250 outright, neither held nor counted: it is in the state this method exists to 

251 reach, and it has no ``SyncItem`` in this run to close. 

252 

253 :param entity_type: The type of the absent entities. 

254 :param absent_entities: The local entities this generation did not report. 

255 :param retire: The retirement call to make once grace is spent. 

256 :param permitted: Whether this generation may retire anything. 

257 :param filtered_external_ids: External IDs excluded by the fetch filter. 

258 :raises SyncFailError: If holding or retiring an entity fails and 

259 ``break_on_error`` is set. 

260 :raises HTTPBadRequestException: If a ledger write hits a database error. 

261 """ 

262 for created_entity in absent_entities: 

263 if created_entity.is_retired: 

264 continue 

265 excluded = created_entity.external_id in filtered_external_ids 

266 if not permitted or excluded: 

267 await self.hold_entity(entity_type, created_entity) 

268 continue 

269 missing = await SyncEntityAbsenceManager.record_missing( 

270 self._session, 

271 self.get_name(), 

272 entity_type, 

273 created_entity.id, 

274 ) 

275 if missing < self.missing_grace_generations: 

276 await self.hold_entity(entity_type, created_entity) 

277 continue 

278 await retire(created_entity) 

279 await SyncEntityAbsenceManager.clear( 

280 self._session, 

281 self.get_name(), 

282 entity_type, 

283 created_entity.id, 

284 )

286 async def perform_inventory_sync(self) -> None: 

287 """Perform the inventory synchronization process. 

288 

289 Synchronize the entire inventory by fetching nodes from the PMM API and 

290 creating or updating corresponding nodes in the local inventory. A node that 

291 the fetch did not report is retired only once a complete generation has 

292 reported it absent ``missing_grace_generations`` times in a row. A partial 

293 read is not evidence that anything disappeared from PMM. 

294 

295 :raises ValidationError: If an entity fails validation and ``break_on_error`` 

296 is set. 

297 :raises SyncFailError: If synchronizing an entity fails and ``break_on_error`` 

298 is set. 

299 :raises HTTPBadRequestException: If a ledger write hits a database error. 

300 """ 

301 # Keyed by primary key, not external id: a tombstone and the replacement 

302 # that took its external id both come back from a retired-inclusive read, 

303 # and an external-id key would let whichever came last shadow the other. 

304 syncable_nodes: dict[int | None, CreatedNode] = {} 

305 external_id_to_id: dict[str, int | None] = {} 

306 for node in await self.get_inventory_nodes(): 

307 syncable_nodes[node.id] = node 

308 claim_identity(external_id_to_id, node.external_id, node, syncable_nodes) 

309 logger.debug("Syncable nodes: %s", syncable_nodes) 

310 snapshot = await self._fetch_snapshot() 

311 self._generation = snapshot 

312 self._snapshot_complete = snapshot.diagnostics.is_complete 

313 try: 

314 present_ids: list[int | None] = [] 

315 for node in snapshot.nodes: 

316 matched_id = external_id_to_id.get(node.external_id) 

317 if (created_node := syncable_nodes.pop(matched_id, None)) is None: 

318 logger.debug("Creating new node: %r", node) 

319 created_node = CreatedNode.model_validate( 

320 await self.inventory_api.post( 

321 "/nodes/", 

322 json=node.model_dump(exclude={"services"}), 

323 ), 

324 ) 

325 else: 

326 await self._revive_if_retired( 

327 SyncInventoryEntityTypeEnum.NODE, created_node 

328 ) 

329 present_ids.append(created_node.id) 

330 await self.sync_node(created_node, node) 

331 logger.debug("Nodes absent from PMM: %s", syncable_nodes) 

332 owns_run = await self._owns_run() 

333 if owns_run: 

334 await SyncEntityAbsenceManager.clear( 

335 self._session, 

336 self.get_name(), 

337 SyncInventoryEntityTypeEnum.NODE, 

338 *present_ids, 

339 ) 

340 await self._retire_absent( 

341 SyncInventoryEntityTypeEnum.NODE, 

342 syncable_nodes.values(), 

343 self.retire_node, 

344 permitted=owns_run and self._generation_is_complete(), 

345 filtered_external_ids=snapshot.diagnostics.filtered_node_ids, 

346 ) 

347 finally: 

348 self._generation = None
