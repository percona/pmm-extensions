60def _infer_encryption_format(all_servers: dict[str, Any]) -> EncryptionFormat: 

61 """Return the encryption format a task stored before the selector was running. 

62 

63 Derived from the fields that used to be the only signal, so a task keeps the 

64 encryption it already ran when its edit form reloads. A key file on any engine 

65 infers AES-256; GPG follows ``ENCRYPT`` / ``POST_RUN_ENCRYPT``. 

66 

67 An absent ``ENCRYPT`` reads as disabled. The payload treats the same absence as 

68 *enabled*, but that fail-safe guards a standalone run against hand-authored 

69 config, which never reaches this function: every config PMM Extensions itself writes names 

70 ``ENCRYPT`` explicitly, an invariant its own contract test pins. 

71 

72 :param all_servers: The stored ``ALL_SERVERS`` config block. 

73 :return: The inferred format. 

74 """ 

75 return encryption_format_for_passes( 

76 aes256=bool(all_servers.get("XTRABACKUP_AES256_KEYFILE")), 

77 gpg=bool(all_servers.get("ENCRYPT") or all_servers.get("POST_RUN_ENCRYPT")), 

78 ) 

79 

80 

81# The create path writes the recipient block through ``BackupConfigServer``, whose 

82# case-insensitive aliases uppercase the outer key, and through ``DirEncryptConfig``, 

83# which renames the recipient to the space-separated spelling the directory 

84# encryptor reads. The models' own attribute names are accepted as fallbacks so a 

85# config written against them rather than against the serialized shape still 

86# reloads. 

87_DIR_ENCRYPT_CONFIG_KEYS = ("DIR_ENCRYPT_CONFIG", "dir_encrypt_config") 

88_ENCRYPTION_RECIPIENT_KEYS = ("encryption recipient", "encryption_recipient")

91def _extract_encryption_recipient(server_config: dict[str, Any]) -> str | None: 

92 """Return the GPG recipient a stored ``SERVER_LIST`` entry names, if any. 

93 

94 Losing the recipient is not a cosmetic gap: a GPG ``encryption_format`` 

95 requires one, so a reconstruction that drops it cannot validate and the task 

96 keeps whatever it already had instead of being re-stamped. 

97 

98 :param server_config: The first ``SERVER_LIST`` entry. 

99 :return: The recipient, or ``None`` when the entry names none. 

100 """ 

101 for config_key in _DIR_ENCRYPT_CONFIG_KEYS: 

102 block = server_config.get(config_key) 

103 if not isinstance(block, dict): 

104 continue 

105 for recipient_key in _ENCRYPTION_RECIPIENT_KEYS: 

106 recipient = block.get(recipient_key) 

107 if recipient: 

108 return recipient 

109 return None

112def parse_backup_task_data(task: dict[str, Any]) -> dict[str, Any]: 

113 """Parse backup task data for editing. 

114 

115 Extracts configuration from an existing backup task to populate the edit form. 

116 

117 Delegates the shared ``SERVER_LIST`` parsing to 

118 :func:`~app.extensions.apps.shared.backups.edit_form.parse_server_list_config`, layering on the 

119 mysql-specific alias, encryption recipient, and the mydumper / xtrabackup / 

120 binlog / upload-quiet keys. A task stored before ``ENCRYPTION_FORMAT`` existed 

121 has its format inferred by :func:`_infer_encryption_format`. 

122 

123 :param task: The task data retrieved from the Tasks API. 

124 :return: A dictionary containing parsed backup configuration. 

125 """ 

126 task_config = yaml.safe_load(task["data"]["meta"]["config"]) 

127 server_config = task_config["SERVER_LIST"][0] 

128 all_servers_config = task_config.get("ALL_SERVERS", {}) 

129 

130 extra_fields = { 

131 "port": server_config.get("PORT"), 

132 "alias": server_config.get("ALIAS"), 

133 } 

134 recipient = _extract_encryption_recipient(server_config) 

135 if recipient is not None: 

136 extra_fields["encryption_recipient"] = recipient 

137 if "ENCRYPTION_FORMAT" not in all_servers_config: 

138 extra_fields["encryption_format"] = _infer_encryption_format(all_servers_config) 

139 extra_fields["binlog_alternative_host"] = all_servers_config.get( 

140 "BINLOG_ALTERNATIVE_HOST" 

141 ) 

142 extra_fields["mydumper_verbose"] = all_servers_config.get("MYDUMPER_VERBOSE") 

143 extra_fields["xtrabackup_quiet"] = all_servers_config.get("XTRABACKUP_QUIET") 

144 extra_fields["upload_quiet"] = all_servers_config.get("UPLOAD_QUIET") 

145 

146 return parse_server_list_config( 

147 task, server_config, all_servers_config, extra_fields 

148 )

151async def resolve_mysql_service( 

152 service_id: int, inventory_api: InventoryAPI 

153) -> CreatedService: 

154 """Resolve an inventory service by id for the backup-catalog query route. 

155 

156 Lets the Inventory API's ``404`` propagate unchanged: an unknown ``service_id`` 

157 is a real client error, not an empty catalog. The catalog query distinguishes 

158 the two — this raises for a service that does not exist, while a service that 

159 exists but has no recorded runs yields an empty list. A resolvable service of 

160 the wrong type is treated the same as an unknown one: the catalog holds only 

161 MySQL runs and falls back to matching on ``service_name`` for rows carrying no 

162 id, so serving a non-MySQL service would let it leak the runs of a MySQL 

163 service that happens to share its name. 

164 

165 Retired services resolve too: the catalog is a historical record, and a service 

166 the inventory stopped seeing upstream is exactly the one whose past runs are 

167 still wanted. 

168 

169 :param service_id: The inventory id of the service to resolve. 

170 :param inventory_api: The Inventory API client used to resolve the service. 

171 :return: The resolved service. 

172 :raises HTTPNotFoundException: When the resolved service is not a MySQL service. 

173 """ 

174 service_data = await inventory_api.get( 

175 f"/services/{service_id}", params={"include_retired": "true"} 

176 ) 

177 service = CreatedService.model_validate(service_data) 

178 if service.type is not ServiceTypeEnum.MYSQL: 

179 raise HTTPNotFoundException(detail="Service not found") 

180 return service

186async def resolve_optional_catalog_service_key( 

187 inventory_api: InventoryAPI, 

188 service_id: str | None = Query( 

189 None, 

190 description=( 

191 "Cascade parent from the restore form. Inventory numeric ids are " 

192 "resolved to a MySQL service, keying the catalog query on its id; " 

193 "custom names query the catalog by name directly. Omitted, blank, " 

194 "sentinel, or unknown values yield an empty list so free-text entry " 

195 "is never blocked by a failed options fetch." 

196 ), 

197 ), 

198) -> CatalogServiceKey | None: 

199 """Resolve the cascade parent to the catalog query keys, or ``None``. 

200 

201 Numeric ids go through :func:`resolve_mysql_service` (MySQL-typed only) and 

202 yield both keys, so a rename between recording and querying cannot detach the 

203 rows; unknown ids degrade to ``None``. Non-numeric values yield the raw value 

204 as the name and no id, so a free-typed restore destination can still list 

205 catalog rows by that name — deliberately unguarded by Inventory type checks, 

206 matching the restore form's ``ServiceRef(allow_custom=True)`` escape hatch. 

207 Omitted, blank, and sentinel parents also yield ``None``. 

208 

209 The numeric test is ``str.isdecimal``, not ``str.isdigit``: the latter also 

210 accepts digits ``int`` cannot parse (superscripts such as ``"²"``), which would 

211 take the numeric branch and degrade to ``None`` rather than reaching the name 

212 branch the free-text escape hatch exists to serve. A decimal string ``int`` 

213 still cannot parse — one longer than ``sys.get_int_max_str_digits()`` — has no 

214 usable name reading either, so it degrades to ``None``. The parse is guarded on 

215 its own so that a ``pydantic.ValidationError`` from resolving the service, being 

216 a ``ValueError`` subclass, is not swallowed as an unparsable id. 

217 

218 :param inventory_api: The Inventory API client used to resolve numeric ids. 

219 :param service_id: The cascade parent's submitted value, or ``None`` when 

220 omitted. 

221 :return: The keys to query the catalog with, or ``None`` when the parent is 

222 unusable. 

223 :raises HTTPException: When the Inventory lookup fails with a status other 

224 than 404. 

225 """ 

226 if service_id is None: 

227 return None 

228 trimmed = service_id.strip() 

229 if not trimmed or trimmed == UNKNOWN_SERVICE_SENTINEL: 

230 return None 

231 if not trimmed.isdecimal(): 

232 return CatalogServiceKey(service_name=trimmed, service_id=None) 

233 try: 

234 parsed = int(trimmed) 

235 except ValueError: 

236 return None 

237 try: 

238 service = await resolve_mysql_service(parsed, inventory_api) 

239 except HTTPNotFoundException: 

240 return None 

241 return CatalogServiceKey(service_name=service.name, service_id=service.id)

253async def resolve_catalogued_history_ids( 

254 task: MysqlBackupsTask, tasks_api: TaskAPI 

255) -> list[int]: 

256 """Return the ids of a task's most recent successful runs, newest first. 

257 

258 Walks at most :data:`MAX_TASK_RUN_SCAN` history rows. Only a ``SUCCESS`` 

259 history can have produced a catalog row, so filtering on it narrows the walk 

260 without hiding a catalogued run. 

261 

262 Reads the window strictly: an upstream body that is not a usable history 

263 page raises rather than reporting an empty history, so a broken Tasks service 

264 cannot masquerade as a task that has never produced a backup. Individual rows are 

265 still skipped rather than fatal — a row without an integer ``id`` cannot be 

266 joined to a catalog record, and one odd row should not fail the whole page. 

267 

268 :param task: The resolved backup task, already checked for existence and 

269 ownership. 

270 :param tasks_api: The Tasks API client used to read the task's history. 

271 :return: Task-history ids, newest first, capped at the scan limit. 

272 :raises HTTPBadGatewayException: If the Tasks API answers with a body that 

273 cannot be read as a history page. 

274 :raises HTTPException: The error the Tasks API itself answered with, mapped by 

275 the remote client and propagated rather than read as an empty history. 

276 """ 

277 window = await fetch_task_history_window( 

278 tasks_api, 

279 task.name, 

280 window_size=MAX_TASK_RUN_SCAN, 

281 status=TaskHistoryStatusEnum.SUCCESS, 

282 sort=_NEWEST_HISTORY_FIRST, 

283 strict=True, 

284 ) 

285 return [ 

286 item["id"] 

287 for item in window["items"] 

288 if isinstance(item, dict) and isinstance(item.get("id"), int) 

289 ]
