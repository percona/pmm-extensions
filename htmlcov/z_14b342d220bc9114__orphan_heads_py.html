16"""Filter ``alembic_version_extensions`` rows whose migration scripts are not loaded. 

17 

18Kept in a separate module (not ``env.py``) for the same reason as 

19``_discovery.py``: importing ``env.py`` runs migrations as a side effect, so 

20helpers that want unit tests live here. Unlike ``_discovery.py``'s helpers, 

21which run standalone, ``skip_unresolvable_heads`` needs a live 

22``MigrationContext`` and so is called after ``context.configure(...)`` rather 

23than at import time. 

24 

25Each app that owns migrations is an independent branch rooted at ``base`` and 

26recorded in the shared ``alembic_version_extensions`` table. An image that strips an 

27app removes its ``versions/`` directory; Alembic skips the absent 

28``version_locations`` entry but still has to resolve the recorded revision, and 

29one unresolvable row makes every upgrade on the track fail. Dropping such rows 

30from the heads Alembic is handed — while leaving them in the table — lets the 

31remaining branches migrate and lets a returning app resume from its preserved 

32revision instead of re-running its branch from ``base``. 

33""" 

34 

35import logging 

36from pathlib import Path 

37from typing import TYPE_CHECKING 

38 

39from alembic.script.revision import ResolutionError 

40 

41if TYPE_CHECKING: 

42 from alembic.runtime.environment import EnvironmentContext 

43 from alembic.script import ScriptDirectory 

44 

45logger = logging.getLogger(__name__) 

46 

47 

48def partition_heads( 

49 script: "ScriptDirectory", heads: tuple[str, ...] 

50) -> tuple[tuple[str, ...], tuple[str, ...]]: 

51 """Split recorded heads by whether the loaded revision map resolves them. 

52 

53 :param script: The Alembic script directory holding the revision map. 

54 :param heads: Revision ids read from the version table. 

55 :return: A tuple of (resolvable, unresolvable) revision ids, each in the 

56 order they appeared in ``heads``. 

57 """ 

58 resolvable: list[str] = [] 

59 unresolvable: list[str] = [] 

60 for revision in heads: 

61 try: 

62 script.revision_map.get_revision(revision) 

63 except ResolutionError: 

64 unresolvable.append(revision) 

65 else: 

66 resolvable.append(revision) 

67 return tuple(resolvable), tuple(unresolvable)

70def empty_version_locations(script: "ScriptDirectory") -> tuple[str, ...]: 

71 """Return configured ``version_locations`` entries that contribute no revisions. 

72 

73 A location contributes no revisions when it is absent from disk or present 

74 but contains no migration scripts the loaded revision map resolved. Both 

75 conditions are evidence of a stripped (or scaffolded-but-empty) app and 

76 license skipping unresolvable heads. 

77 

78 :param script: The Alembic script directory holding the configured paths 

79 and the loaded revision map. 

80 :return: Each configured location that contributed no revisions, in 

81 configuration order. 

82 """ 

83 locations = script.version_locations or () 

84 if not locations: 

85 return () 

86 # Resolve both sides so configured locations match Alembic's rev.path parents. 

87 populated = { 

88 str(Path(rev.path).resolve().parent) for rev in script.walk_revisions() 

89 } 

90 return tuple( 

91 str(location) 

92 for location in locations 

93 if str(Path(location).resolve()) not in populated 

94 )

97def skip_unresolvable_heads(env_context: "EnvironmentContext") -> None: 

98 """Make the migration context ignore heads whose migration scripts are gone. 

99 

100 Wrap the context's ``get_current_heads`` so both the migrations function and 

101 the ``HeadMaintainer`` built alongside it see the same filtered set. Rows for 

102 the dropped revisions stay in the version table: Alembic only ever targets a 

103 specific ``version_num`` when it writes. 

104 

105 Fail-closed: heads are only dropped when a configured ``version_locations`` 

106 entry contributes no resolvable revisions — absent from disk or present and 

107 empty — which is what a stripped app looks like. With every location present 

108 and populated the evidence is absent — the cause may be version skew, a 

109 squashed revision, or an entry pruned from ``version_locations`` for an app 

110 that is gone — so the heads pass through unfiltered and Alembic raises as it 

111 does today. 

112 

113 Offline (``--sql``) mode needs no hook: there ``get_current_heads`` returns 

114 the ``starting_rev`` argument instead of reading the version table. 

115 

116 Call after ``context.configure(...)`` and before ``context.run_migrations()``. 

117 

118 :param env_context: The Alembic environment context from ``env.py``. 

119 """ 

120 migration_context = env_context.get_context() 

121 script = env_context.script 

122 read_heads = migration_context.get_current_heads 

123 

124 def get_current_heads() -> tuple[str, ...]: 

125 heads = read_heads() 

126 resolvable, unresolvable = partition_heads(script, heads) 

127 if not unresolvable: 

128 return heads 

129 

130 empty = empty_version_locations(script) 

131 if not empty: 

132 logger.error( 

133 "%d revision(s) recorded in %s do not resolve (%s) while every " 

134 "configured version_locations entry is present on disk and " 

135 "contributes at least one revision. Likely causes: version skew, " 

136 "a squashed revision, or a version_locations entry that was " 

137 "pruned or went stale for an app that is no longer installed. " 

138 "They are left in place for Alembic to reject.", 

139 len(unresolvable), 

140 migration_context.version_table, 

141 ", ".join(unresolvable), 

142 ) 

143 return heads 

144 

145 logger.warning( 

146 "Skipping %d revision(s) recorded in %s with no migration script: " 

147 "%s. Configured version_locations that contributed no revisions: %s.", 

148 len(unresolvable), 

149 migration_context.version_table, 

150 ", ".join(unresolvable), 

151 ", ".join(empty), 

152 ) 

153 return resolvable 

154 

155 migration_context.get_current_heads = get_current_heads
