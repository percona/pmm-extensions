31class TaskWrite(TaskBase): 

32 """Define the model for creating new tasks. 

33 

34 :param name: The name of the task. 

35 :param data: The task data stored in JSON format. 

36 :param backend: The backend used for task execution. Defaults to Nomad. 

37 :param owner: The owner of the task. Defaults to ``"ANY"``. 

38 :param is_template: Whether the task is a template. Defaults to False. 

39 :param protected: Whether the task is protected from deletion. Defaults to False. 

40 :param alert_on_fail: Whether to trigger an alert on task failure and 

41 auto-resolve it on subsequent success. Defaults to False. 

42 :param alert_detail_builder: The ``"module:function"`` path of a plugin 

43 callable that enriches this task's failure alert, or None. Must name a 

44 public callable in a module under one of the roots 

45 :attr:`app.tasks.config.TasksSettings.HOOK_MODULE_ALLOWLIST` lists; 

46 anything else is rejected. 

47 :param run_result_recorder: The ``"module:function"`` path of a plugin 

48 callable that records this task's structured run result at terminal 

49 status, or None. Constrained to the same allow-listed namespace as 

50 ``alert_detail_builder``. 

51 :param output_files_path: The path, relative to the executor's working 

52 directory, where output files generated by the task are expected, or 

53 None. 

54 :param anonymize_mask: The bitmask representing PII entities to be anonymized in 

55 logs and files generated by the task. Defaults to 0 (no anonymization). 

56 """ 

57 

58 @field_validator("alert_detail_builder", "run_result_recorder") 

59 @classmethod 

60 def validate_hook_path_allow_listed( 

61 cls, value: str | None, info: ValidationInfo 

62 ) -> str | None: 

63 """Reject a hook path outside the allow-listed module namespace. 

64 

65 A hook path is imported and invoked by the tasks service, so an 

66 unconstrained value would let any caller who can write a task name an 

67 arbitrary callable. The check lives here rather than on 

68 :class:`TaskBase` so reading back a row whose stored path predates the 

69 allow-list keeps working. 

70 

71 :param value: The candidate ``"module:function"`` path, or ``None``. 

72 :param info: The validation context, carrying the field name. 

73 :return: The validated path, or ``None`` when the field is unset. 

74 :raises HookPathNotAllowedError: When the path is malformed or names a 

75 module outside the allow-listed namespace. 

76 """ 

77 if value is None: 

78 return None 

79 return validate_hook_path(value, field=info.field_name or "hook path")

82class TaskExecuteRequest(BaseModel): 

83 """Represent a request to execute a task with additional metadata and payload. 

84 

85 :param meta: A dictionary of meta variables for the task execution. 

86 Defaults to an empty dictionary. ``requirements`` may be a 

87 newline-separated string or a list of non-empty requirement strings; a 

88 list is joined with newlines. 

89 :param payload: Optional payload data or file path for the task execution. 

90 Defaults to ``None``. 

91 :param eta: The earliest time the task can be executed. Defaults to ``None``, 

92 meaning it will be executed as soon as possible. 

93 :param anonymize_mask: Bitmask of PII entities to anonymize. Defaults to 

94 ``None``. 

95 :param chain_task_names: Ordered list of task names to execute sequentially after 

96 this one completes. Defaults to ``None``. 

97 :param chain_on_failure: Whether the chain should continue when a task fails, 

98 stops, or is lost. Defaults to ``False`` (chain only on success). 

99 """ 

100 

101 meta: dict[str, Any] = Field( 

102 default={}, 

103 description=( 

104 "Meta variables for the task execution. For `run-python`, " 

105 "`requirements` lists the pip requirements to install, either as one " 

106 'newline-separated string (`"PyMySQL\\nfilelock"`) or as a list of ' 

107 'requirement strings (`["PyMySQL", "filelock"]`); a list is joined ' 

108 "into the string form before the task is stored and dispatched." 

109 ), 

110 examples=[
