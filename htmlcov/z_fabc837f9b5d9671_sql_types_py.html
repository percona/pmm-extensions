16"""Define custom SQLAlchemy column types."""

18import zlib 

19from typing import Any 

20 

21from sqlalchemy import JSON, LargeBinary, TypeDecorator 

22from sqlalchemy.dialects.postgresql import JSONB 

23from sqlalchemy.engine.interfaces import Dialect 

24from sqlalchemy.sql.type_api import TypeEngine

26MAYBE_COMPRESSED_TEXT_RAW_PREFIX = b"\x00" 

27MAYBE_COMPRESSED_TEXT_COMPRESSED_PREFIX = b"\x01" 

28MAYBE_COMPRESSED_TEXT_MIN_BYTES = 256 

29MAYBE_COMPRESSED_TEXT_MIN_SAVINGS_RATIO = 0.1

32class AutoJSON(TypeDecorator): 

33 """Resolve to JSONB on PostgreSQL and JSON on other dialects. 

34 

35 :cvar impl: The base implementation type. 

36 :vartype impl: type[JSON] 

37 :cvar cache_ok: Allow SQLAlchemy to cache compiled statements using this type. 

38 :vartype cache_ok: bool 

39 """ 

40 

41 impl = JSON 

42 cache_ok = True 

43 

44 def load_dialect_impl(self, dialect: Dialect) -> TypeEngine[Any]: 

45 """Return the dialect-specific type implementation. 

46 

47 :param dialect: The SQLAlchemy dialect in use. 

48 :type dialect: Dialect 

49 :return: JSONB for PostgreSQL, JSON for all other dialects. 

50 :rtype: TypeEngine[Any] 

51 """ 

52 if dialect.name == "postgresql": 

53 return dialect.type_descriptor(JSONB()) 

54 return super().load_dialect_impl(dialect)

57class MaybeCompressedText(TypeDecorator): 

58 """Store UTF-8 text as bytes, optionally zlib-compressed, with a 1-byte prefix. 

59 

60 The prefix distinguishes raw and compressed payloads so each row can pick the 

61 smaller representation without needing a separate column. Compression is only 

62 applied when the raw payload is at least ``MAYBE_COMPRESSED_TEXT_MIN_BYTES`` and 

63 the compressed form saves at least ``MAYBE_COMPRESSED_TEXT_MIN_SAVINGS_RATIO`` 

64 of the original size. 

65 

66 :cvar impl: The underlying SQLAlchemy column type. 

67 :vartype impl: type[LargeBinary] 

68 :cvar cache_ok: Allow SQLAlchemy to cache compiled statements using this type. 

69 :vartype cache_ok: bool 

70 """ 

71 

72 impl = LargeBinary 

73 cache_ok = True 

74 

75 def process_bind_param(self, value: Any, dialect: Dialect) -> Any: # noqa: ARG002 

76 """Serialize UTF-8 text into a prefixed byte string for storage. 

77 

78 :param value: The text value to store. 

79 :type value: Any 

80 :param dialect: The SQLAlchemy dialect in use. 

81 :type dialect: Dialect 

82 :return: The prefixed bytes ready for the underlying ``LargeBinary`` column, 

83 or ``None`` when the input is ``None``. 

84 :rtype: Any 

85 """ 

86 if value is None: 

87 return None 

88 encoded = value.encode("utf-8") 

89 if len(encoded) < MAYBE_COMPRESSED_TEXT_MIN_BYTES: 

90 return MAYBE_COMPRESSED_TEXT_RAW_PREFIX + encoded 

91 compressed = zlib.compress(encoded) 

92 if len(compressed) >= len(encoded) * ( 

93 1 - MAYBE_COMPRESSED_TEXT_MIN_SAVINGS_RATIO 

94 ): 

95 return MAYBE_COMPRESSED_TEXT_RAW_PREFIX + encoded 

96 return MAYBE_COMPRESSED_TEXT_COMPRESSED_PREFIX + compressed 

97 

98 def process_result_value(self, value: Any, dialect: Dialect) -> Any: # noqa: ARG002 

99 """Deserialize a prefixed byte string back into UTF-8 text. 

100 

101 :param value: The raw bytes from the database. 

102 :type value: Any 

103 :param dialect: The SQLAlchemy dialect in use. 

104 :type dialect: Dialect 

105 :return: The decoded UTF-8 text, or ``None`` when the input is ``None``. 

106 :rtype: Any 

107 """ 

108 if value is None: 

109 return None 

110 prefix, payload = value[:1], value[1:] 

111 if prefix == MAYBE_COMPRESSED_TEXT_COMPRESSED_PREFIX: 

112 return zlib.decompress(payload).decode("utf-8") 

113 return payload.decode("utf-8")
