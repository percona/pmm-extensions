16"""Define the MySQL Backups plugin's self-contained catalog model and API response. 

17 

18This module is intentionally **self-contained** — it imports only from 

19``app.core``, ``pydantic``, ``sqlalchemy``, ``sqlmodel``, and ``yaml`` (for the 

20shared config parser), never from ``app.inventory`` / ``app.tasks`` / the app 

21framework's form DSL, and **never from a sibling in its own package**. The extensions 

22Alembic discovery loads this file *by path* to register the table in the 

23migration metadata, so a package-qualified sibling import executes 

24``mysql_backups/__init__.py``, which imports the whole app and re-enters this 

25module mid-initialisation — an ``ImportError`` that only ``make 

26checkmigrations`` reproduces, however import-free the sibling itself is. 

27Pulling in the heavier modules would additionally bleed their tables into the 

28extensions autogenerate comparison. 

29 

30That constraint is also what decides where a piece two siblings both need goes 

31— the ``backup_source`` resolution below, and the vocabulary enums the backup 

32and restore forms share: this module is the one both the catalog response and 

33:mod:`app.extensions.apps.mysql_backups.restore.models` can depend on, so anything 

34either would otherwise import from the other lands here instead. 

35 

36The split mirrors ``app.extensions.apps.atw``, in the direction that matters: there, 

37``atw.models`` is the self-contained module and the one inventory-dependent 

38piece (the category taxonomy) lives apart, in ``atw.categories``. Here, this 

39module plays the role of ``atw.models``, and the plugin's heavy form/DSL 

40surface (:class:`~app.extensions.apps.mysql_backups.forms.BackupCreate` and friends) 

41plays the role of ``atw.categories`` — the piece split *out* because it, not 

42the table, needed the heavier imports. 

43"""

110def restore_valid_backup_source( 

111 upload_destination: str | None, location: str | None 

112) -> str | None: 

113 """Return a restore-form-valid ``backup_source``, or ``None``. 

114 

115 Judge shell-safety on the single candidate :func:`preferred_backup_source` 

116 produced, never on the raw fields independently: a rejected candidate yields 

117 ``None`` rather than falling back to the other field, so a caller is never 

118 handed a source pointing at a different artifact than the one the run's own 

119 preference names. 

120 

121 :param upload_destination: The run's recorded upload destination. 

122 :param location: The run's recorded on-disk location. 

123 :return: The restore-valid source, or ``None`` when absent, blank, or 

124 rejected by :func:`ensure_backup_source_shell_safe`. 

125 """ 

126 value = preferred_backup_source(upload_destination, location) 

127 if value is None: 

128 return None 

129 try: 

130 ensure_backup_source_shell_safe(value) 

131 except ValueError: 

132 return None 

133 return value

166class MysqlBackupRun(BaseSQLModel, table=True): 

167 """Persist one completed MySQL backup run's produced output. 

168 

169 Written by the run-result recorder on a successful mydumper or xtrabackup 

170 run — one row per run, keyed by :attr:`task_history_id`. The record is 

171 tool-agnostic: the xtrabackup incremental layout is captured as a different 

172 :attr:`location` string, not a different shape. Binlog runs and non-success 

173 terminals leave no row. 

174 

175 :param task_history_id: The id of the ``TaskHistory`` this run belongs to; 

176 unique, so one run maps to exactly one record. 

177 :param service_name: The inventory service name the backup was taken from 

178 (the task's ``_service_name`` meta), kept as the fallback query key for 

179 rows carrying no :attr:`service_id` and as the only usable key for a 

180 free-typed restore destination that has no inventory row at all. A row 

181 reached this way can still be confused with another service's: two MySQL 

182 services may share a name, since ``Service.name`` carries no uniqueness 

183 constraint. 

184 :param service_id: The inventory id of the service the backup was taken from 

185 (the task's ``_service_id`` meta), preferred as the per-service query key 

186 because it survives a rename, which the name does not. Not a foreign key 

187 — the catalog is extensions-owned and inventory lives in a separate database, so 

188 nothing enforces that the id still resolves. Empty on rows recorded 

189 before the id was stamped, which is what keeps the name fallback needed. 

190 :param hostname: The backup target host (the task's ``target`` meta). 

191 :param backup_type: The backup tool the run used, mydumper or xtrabackup. 

192 :param location: The resolved on-disk directory the run produced, stored 

193 exactly as the payload reported it. 

194 :param upload_destination: The upload destination when one was configured, 

195 else ``None``. 

196 :param size_bytes: The backup size in bytes, when the run reported it. 

197 :param started_at: When the run started. 

198 :param finished_at: When the run finished. 

199 """ 

200 

201 __tablename__ = "mysql_backup_run"

303def extract_backup_type_marker(task_data: dict[str, Any] | None) -> str | None: 

304 """Return the ``BACKUP_TYPE`` marker from a task's YAML config, or ``None``. 

305 

306 Reads the value defensively: a missing ``meta``, unparseable YAML, or an 

307 absent key all resolve to ``None`` rather than raising. Returns the raw 

308 single-letter marker (``"M"``/``"X"``/``"B"``); callers that need the typed 

309 :class:`BackupType` coerce it themselves (see ``deps.py``'s 

310 ``_extract_backup_type_from_task``). 

311 

312 :param task_data: The task's ``data`` dict (``meta`` carries the YAML config). 

313 :return: The raw backup-type marker, or ``None``. 

314 """ 

315 match task_data: 

316 case {"meta": {"config": str(raw_config)}} if raw_config: 

317 pass 

318 case _: 

319 return None 

320 

321 try: 

322 config = yaml.safe_load(raw_config) 

323 except yaml.YAMLError: 

324 return None 

325 

326 match config: 

327 case {"SERVER_LIST": [{"BACKUP_TYPE": str(backup_type)}, *_]}: 

328 return backup_type 

329 case _: 

330 return None
