102def _sanitize_client_kwargs(kwargs: dict[str, Any]) -> dict[str, Any]: 

103 """Return a copy of remote-API client kwargs safe to log. 

104 

105 Any value carrying an embedded URL password (e.g. ``endpoint``) is 

106 redacted so credentials never reach the debug log. ``SecretStr`` values 

107 already mask themselves on ``repr`` and are left untouched. 

108 

109 :param kwargs: The client construction kwargs about to be logged. 

110 :type kwargs: dict[str, Any] 

111 :return: A copy with credential URLs redacted. 

112 :rtype: dict[str, Any] 

113 """ 

114 safe: dict[str, Any] = {} 

115 for key, value in kwargs.items(): 

116 raw = str(value) 

117 redacted = redact_credential_url(raw) 

118 safe[key] = redacted if redacted != raw else value 

119 return safe

437def _encryption_key_error() -> str: 

438 """Return the remediation text for a missing or malformed key. 

439 

440 Names the absolute dotenv path rather than the default spelling. 

441 ``ENV_FILE`` is a supported indirection and this repository's own checkouts 

442 use it, so prose that hardcodes the default sends the reader to a file the 

443 loader never reads: the append succeeds, the next start fails identically, 

444 and the natural reading is a malformed key rather than an unread file. It 

445 is resolved because the setting is a relative path by default, and a 

446 process whose working directory is not the one the reader is standing in 

447 would otherwise be told to edit ``.env`` without being told which. 

448 ``openssl rand -hex 32``, which ``SECRET_KEY``'s own message offers, 

449 produces 64 characters Fernet rejects, so the two remediations are 

450 deliberately different. 

451 

452 :return: The remediation sentence, naming the dotenv file in use. 

453 """ 

454 return ( 

455 "ENCRYPTION_KEY must be set to a valid Fernet key (32 url-safe " 

456 "base64-encoded bytes). Generate one with `make encryption-key` or " 

457 "`openssl rand -base64 32`, then add it as ENCRYPTION_KEY=<key> to " 

458 f"{pre_env_settings.ENV_FILE.resolve()} (the file ENV_FILE names), " 

459 "export it, or mount it as a file named ENCRYPTION_KEY under " 

460 "SECRETS_DIR. It has no default and is never derived from SECRET_KEY."

467class SettingsOverrideOptions(BaseCaseInsensitiveModel): 

468 """Configure the DB-backed settings-override layer. 

469 

470 Left unmarked on :class:`Settings` so the parent classifies 

471 ``NOT_OVERRIDABLE`` (matching ``SECURITY_HEADERS``). Marking it 

472 ``NESTED_ONLY`` would expose ``REFRESH_INTERVAL`` and ``REFRESHER_ENABLED`` 

473 to the override API; ``ALLOWED_KEYS`` would stay refused by its own 

474 ``not_overridable_field`` marker, which is why that marker is kept. 

475 

476 :param REFRESH_INTERVAL: How often each service refreshes its DB-backed 

477 setting overrides. Defaults to 30 seconds, and must be strictly 

478 positive. In a web process this is the wall-clock delay between 

479 periodic refresh cycles (``start_refresh_task`` hands 

480 ``interval.total_seconds()`` to ``asyncio.sleep``). In a prefork 

481 worker child it is checked at task boundaries — at most one refresh 

482 per interval per child per refresher — rather than a free-running 

483 timer; the same value is also the hang budget passed to 

484 ``bounded_refresh``, so lowering it for fresher overrides also 

485 tightens how long a due task may stall. A child running both the 

486 PMM Extensions side and Tasks-side refreshers can pay that budget twice when 

487 both are due at the same boundary. A non-positive value is rejected 

488 so neither path can hammer the database every iteration. 

489 :param REFRESHER_ENABLED: Master kill-switch for the DB-override 

490 background refresher. Tests set this to ``False`` to keep 

491 ``TestClient`` lifespans hermetic; production leaves it ``True``. 

492 :param ALLOWED_KEYS: The exhaustive set of settings keys the override API 

493 may write, each spelled ``"<SettingsClassName>.<KEY>"`` (the key being 

494 a top-level field name or a ``__``-delimited nested path). ``None``, 

495 the default, places no restriction. A set activates a default-locked 

496 allowlist: any pair it does not name is refused. The allowlist only 

497 ever *restricts*: a field that is already not overridable stays that 

498 way even when listed. Whether the named class and field exist is not 

499 checked at load time, so a typo'd entry allows nothing rather than 

500 raising; whitespace anywhere in an entry is rejected outright, since 

501 such an entry reads correct but matches nothing. 

502 """ 

503 

504 REFRESH_INTERVAL: Annotated[TimedeltaSeconds, Field(gt=timedelta(0))] = timedelta( 

505 seconds=30 

506 ) 

507 REFRESHER_ENABLED: bool = True 

508 ALLOWED_KEYS: set[SettingsOverrideKey] | None = ( # ty: ignore[invalid-assignment]
