16"""Resolve a plugin callable named by a ``"module:function"`` path, with caching. 

17 

18Several per-task hooks let the owning plugin *declare* a callable by 

19``"module:function"`` string on the task and have core resolve it lazily the 

20first time it is needed — keeping the tasks service free of static ``app.extensions`` 

21imports. Those hooks share this resolver so the ``importlib`` 

22resolve-and-cache boilerplate lives in one place. 

23This process-local cache is correct because the import path deterministically 

24resolves to the same callable in every worker and needs no invalidation. 

25 

26A hook path names a callable that this service imports and invokes, so the 

27module it names is constrained to 

28:attr:`app.tasks.config.TasksSettings.HOOK_MODULE_ALLOWLIST`. The same check 

29runs at the write boundary (:class:`app.tasks.models.TaskWrite`) and here, so a 

30path that reached the database by any other route fails closed at invoke time 

31instead of being imported. 

32""" 

33 

34import importlib 

35import logging 

36from collections.abc import Callable 

37from typing import Any 

38 

39logger = logging.getLogger(__name__) 

40 

41#: Cache of resolved callables, keyed by ``"module:function"`` path. 

42_RESOLVED: dict[str, Callable[..., Any]] = {} 

43 

44 

45class HookPathNotAllowedError(ValueError): 

46 """Define exception for a hook path that is malformed or not allow-listed. 

47 

48 Subclasses :class:`ValueError` so the hook call sites, which already treat a 

49 bad path as a skipped enrichment rather than a failure, keep degrading 

50 gracefully. 

51 """ 

52 

53 

54def is_dotted_module_path(module_path: str) -> bool: 

55 """Return whether every dot-separated segment is a Python identifier. 

56 

57 :param module_path: A dotted module name, or the module part of a hook path. 

58 :return: ``True`` when the module part is a well-formed dotted name. 

59 """ 

60 return bool(module_path) and all( 

61 part.isidentifier() for part in module_path.split(".") 

62 ) 

63 

64 

65def validate_hook_path(path: str, field: str = "hook path") -> str: 

66 """Return ``path`` when it names an allow-listed callable, else raise. 

67 

68 A hook path is admitted only when it is a well-formed ``"module:function"`` 

69 pair naming a public function in a module under one of the configured 

70 allow-listed roots. Everything else is rejected — a malformed pair, a dunder 

71 or private attribute, a module outside the namespace — because the resolved 

72 callable is imported and invoked by the tasks service. 

73 

74 :param path: The candidate callable path in ``"module:function"`` form. 

75 :param field: The name of the field carrying the path, quoted in the 

76 rejection message. Defaults to a generic label. 

77 :return: The validated path, unchanged. 

78 :raises HookPathNotAllowedError: When the path is malformed or names a 

79 module outside the allow-listed namespace. 

80 """ 

81 # Deferred: app.tasks.config imports the Nomad executor, which imports 

82 # app.tasks.models, which imports this module, so tasks_settings does not 

83 # exist yet while that chain is still initialising. 

84 from app.tasks.config import tasks_settings 

85 

86 allowed = tasks_settings.HOOK_MODULE_ALLOWLIST 

87 module_path, _, func_name = path.partition(":") 

88 if ( 

89 not is_dotted_module_path(module_path) 

90 or not func_name.isidentifier() 

91 or func_name.startswith("_") 

92 ): 

93 reason = 'is not a "module:function" path naming a public callable' 

94 elif not any( 

95 module_path == root or module_path.startswith(f"{root}.") for root in allowed 

96 ): 

97 reason = "names a module outside the allow-listed namespace" 

98 else: 

99 return path 

100 

101 logger.warning("Rejected %s %r: it %s.", field, path, reason) 

102 raise HookPathNotAllowedError( 

103 f"{field} {path!r} {reason}; allow-listed module roots: {', '.join(allowed)}" 

104 ) 

105 

106 

107def resolve_hook(path: str) -> Callable[..., Any]: 

108 """Import and return the callable named by a ``"module:function"`` path. 

109 

110 Validate the path against the allow-list before anything else, so neither a 

111 denied module nor a cache entry poisoned before the allow-list narrowed can 

112 be served. Cache the resolved callable so a repeated lookup for the same 

113 path skips the import. 

114 

115 :param path: The callable path in ``"module:function"`` form. 

116 :return: The resolved callable. 

117 :raises HookPathNotAllowedError: When the path is malformed or names a 

118 module outside the allow-listed namespace. 

119 :raises ImportError: When the named module cannot be imported. 

120 :raises AttributeError: When the module has no attribute ``function``. 

121 """ 

122 validate_hook_path(path) 

123 cached = _RESOLVED.get(path) 

124 if cached is not None: 

125 return cached 

126 module_path, func_name = path.split(":", 1) 

127 func = getattr(importlib.import_module(module_path), func_name) 

128 _RESOLVED[path] = func 

129 return func
