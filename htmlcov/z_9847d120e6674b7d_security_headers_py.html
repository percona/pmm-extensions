77class PermissionsPolicyOptions(BaseCaseInsensitiveModel): 

78 """Configure the Permissions Policy header settings. 

79 

80 :param allow_self: Set of directives that allow access from the same origin. 

81 Defaults to an empty set. 

82 :type allow_self: set[PermissionsPolicyDirective] 

83 :param allow_all: Set of directives that allow access from any origin. Defaults to 

84 an empty set. 

85 :type allow_all: set[PermissionsPolicyDirective] 

86 """ 

87 

88 allow_self: set[PermissionsPolicyDirective] = set() 

89 allow_all: set[PermissionsPolicyDirective] = set() 

90 

91 @model_validator(mode="after") 

92 def validate_no_intersection(self) -> Self: 

93 """Ensure that no directives are present in both `allow_self` and `allow_all`. 

94 

95 :return: The validated `PermissionsPolicyOptions` instance. 

96 :rtype: PermissionsPolicyOptions 

97 :raises ValueError: If any directives are found in both `allow_self` and 

98 `allow_all`. 

99 """ 

100 if intersection := self.allow_self & self.allow_all: 

101 raise ValueError( 

102 f"Directives cannot be in both allow_self and allow_all: {intersection}" 

103 ) 

104 return self 

105 

106 @property 

107 def header(self) -> str: 

108 """Generate the header string based on the configured directives. 

109 

110 :return: The formatted Permissions Policy header. 

111 :rtype: str 

112 """ 

113 policy = ", ".join(f"{directive}=(self)" for directive in self.allow_self) 

114 policy += ", ".join(f"{directive}=*" for directive in self.allow_all) 

115 return policy + ", ".join( 

116 f"{directive}=()" 

117 for directive in set(PermissionsPolicyDirective) 

118 - self.allow_self 

119 - self.allow_all 

120 )

124 """Configure the Strict Transport Security (HSTS) header settings. 

125 

126 :param max_age: Maximum age (in seconds) for the HSTS policy. 

127 :type max_age: int 

128 :param include_sub_domains: Whether to apply the HSTS policy to all subdomains.

130 :type include_sub_domains: bool 

131 :param preload: Whether to include the `preload` directive for HSTS. Defaults to 

132 False. 

133 :type preload: bool 

134 """ 

135 

136 max_age: Annotated[int, Gt(0)] 

137 include_sub_domains: bool = False 

138 preload: bool = False 

139 

140 @property 

141 def header(self) -> str: 

142 """Generate the HSTS header string based on the configured options. 

143 

144 :return: The formatted Strict Transport Security header. 

145 :rtype: str 

146 """ 

147 hsts = f"max-age={self.max_age}" 

148 if self.include_sub_domains: 

149 hsts += "; includeSubDomains" 

150 if self.preload: 

151 hsts += "; preload" 

152 return hsts

226 async def dispatch( 

227 self, request: Request, call_next: RequestResponseEndpoint 

228 ) -> Response: 

229 """Process the incoming request and add security headers to the response. 

230 

231 If `Content-Security-Policy` is enabled, generate a nonce for inline scripts and 

232 include it in the policy. 

233 

234 :param request: The incoming HTTP request. 

235 :type request: Request 

236 :param call_next: The next middleware or endpoint in the ASGI application. 

237 :type call_next: RequestResponseEndpoint 

238 :return: The HTTP response with added security headers. 

239 :rtype: Response 

240 """ 

241 extra_headers = {} 

242 if ( 

243 self.options.content_security_policy_strict 

244 and request.url.path 

245 not in self.options.content_security_policy_exclude_paths 

246 ): 

247 nonce = token_urlsafe(32) 

248 request.state.csp_nonce = nonce 

249 extra_headers["Content-Security-Policy"] = ( 

250 f"script-src 'nonce-{nonce}' 'strict-dynamic'; object-src 'none'; " 

251 f"base-uri 'none'; frame-ancestors 'none'; form-action 'self'" 

252 ) 

253 response = await call_next(request) 

254 for header, value in (self.security_headers | extra_headers).items(): 

255 response.headers.setdefault(header, value) 

256 return response
