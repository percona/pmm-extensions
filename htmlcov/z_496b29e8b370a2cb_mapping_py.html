16"""Map MongoDB services to the executor host their probe must run on. 

17 

18This is the question the worker exists to answer: given ``rsc-node02``, which host does 

19a script run on to reach that mongod? 

20 

21The matching order mirrors 

22:meth:`~app.extensions.sync.models.BaseTaskSyncer.get_task_target` — node name first, then 

23node address — against the executor list the Tasks API serves from 

24:meth:`~app.tasks.execution.executors.nomad.models.NomadExecutor.get_hosts`, which 

25already filters server-side to ready clients with a healthy ``raw_exec`` driver. 

26 

27**What this module deliberately does not copy** is that method's fallback. With 

28``strict_executor_matching`` off, an unmatched node resolves to 

29``next(iter(available_hosts))`` — an arbitrary unrelated host — and the probe runs 

30there and reports facts about a mongod that is not on that box. Here an unmatched 

31service is :attr:`~app.extensions.apps.om_inventory.models.NodeResolution.ORPHANED` and is not 

32probed at all. 

33 

34That case is the norm, not an edge: an inventory row routinely outlives the executor 

35that served it. 

36"""

97async def get_executor_states(tasks_api: RemoteAPI) -> dict[str, ExecutorState]: 

98 """Return every executor host the backend knows about, usable or not. 

99 

100 ``GET /hosts/`` would be the smaller call, but it answers only "where can a job 

101 be placed": a host that is down, or up with an unhealthy driver, or never 

102 registered are all equally *absent* from it. OM has to describe hosts it cannot 

103 probe, so an absence it cannot explain is the one answer it must not give. 

104 

105 Duplicate registrations of one name collapse, which is what makes name matching 

106 safe. On a backend that predates ``/hosts/states/`` this raises rather than 

107 silently degrading - a OM that quietly reported every host as fine would be 

108 worse than one that failed the sweep. 

109 

110 :param tasks_api: The tasks API client. 

111 :return: The executor hosts keyed by name. 

112 """ 

113 entries = await tasks_api.get("/hosts/states/") 

114 if not isinstance(entries, list): 114 ↛ 115line 114 didn't jump to line 115 because the condition on line 114 was never true

115 raise TypeError( 

116 f"/hosts/states/ answered {type(entries).__name__}, not a list of hosts" 

117 ) 

118 states: dict[str, ExecutorState] = {} 

119 for entry in entries: 

120 state = ExecutorState( 

121 name=entry["name"], 

122 address=entry["address"], 

123 reachable=entry["reachable"], 

124 driver_healthy=entry["driver_healthy"], 

125 detail=entry.get("detail"), 

126 ) 

127 # One name, several registrations: restarting a host's agent leaves the old 

128 # registration behind as ``down`` beside the new one, so a plain dict 

129 # comprehension would keep whichever came last and call a running machine 

130 # unreachable. Measured in this workspace's sandbox — ``pmm-client-node00`` 

131 # was registered once ready and twice down, and the sweep refused to dispatch 

132 # to a host that was up. ``get_hosts`` never had to care, because everything 

133 # in it was usable by construction. 

134 existing = states.get(entry["name"]) 

135 if existing is None or (state.usable and not existing.usable): 

136 states[entry["name"]] = state 

137 usable = sum(1 for state in states.values() if state.usable) 

138 logger.info( 

139 "OM inventory: %d executor host(s) known, %d usable", 

140 len(states), 

141 usable, 

142 ) 

143 return states

146def usable_executor_hosts(states: dict[str, ExecutorState]) -> dict[str, str]: 

147 """Narrow the states to the ``{name: address}`` mapping dispatch works from. 

148 

149 :param states: Every known executor host. 

150 :return: Only the ones a job can be placed on. 

151 """ 

152 return {name: state.address for name, state in states.items() if state.usable}

155def map_service( 

156 service: InventoryService, executor_hosts: dict[str, str] 

157) -> MappedService: 

158 """Resolve one service to an executor host, or mark it orphaned. 

159 

160 :param service: The inventory service to map. 

161 :param executor_hosts: The available executor hosts, ``{name: address}``. 

162 :return: The mapping outcome for this service. 

163 """ 

164 if service.node_name and service.node_name in executor_hosts: 

165 return MappedService(service, service.node_name, NodeResolution.NAME) 

166 

167 if service.node_address: 

168 for host, address in executor_hosts.items(): 

169 if address == service.node_address: 

170 return MappedService(service, host, NodeResolution.ADDRESS) 

171 

172 # Deliberately not falling back to an arbitrary host — see the module docstring. 

173 logger.info( 

174 "OM inventory: service %r is orphaned (node name=%r address=%r matches no " 

175 "executor host); it will not be probed", 

176 service.name, 

177 service.node_name, 

178 service.node_address, 

179 ) 

180 return MappedService(service, None, NodeResolution.ORPHANED)

183def map_services( 

184 services: list[InventoryService], executor_hosts: dict[str, str] 

185) -> list[MappedService]: 

186 """Resolve every service to an executor host, preserving inventory order. 

187 

188 :param services: The MongoDB services from inventory. 

189 :param executor_hosts: The available executor hosts, ``{name: address}``. 

190 :return: One mapping per service, resolved or orphaned. 

191 """ 

192 mapped = [map_service(service, executor_hosts) for service in services] 

193 resolved = sum(1 for entry in mapped if entry.is_resolved) 

194 logger.info( 

195 "OM inventory: mapped %d service(s) -> %d resolved, %d orphaned", 

196 len(mapped), 

197 resolved, 

198 len(mapped) - resolved, 

199 ) 

200 return mapped
